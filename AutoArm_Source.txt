// AutoArm: auto-discovered mixed mechanical arm controller.
// Change only this value in code. All detected group participation settings are
// generated in this programmable block's Custom Data.

// C# 6 programmable-block fragment. Paste inside Program.
class AutoMember
{
    public IMyMechanicalConnectionBlock B;
    public int Sign;
    public double Q0, Last, Home;
    public long U, V;
}

class AutoGroup
{
    public int Index;
    public bool Rotary;
    public string Sig, Label;
    public double MoveW = 1, TurnW = 1;
    public readonly List<AutoMember> M = new List<AutoMember>();
    public Vector3D Lin, Ang;
}

class AutoEdge
{
    public IMyMechanicalConnectionBlock B;
    public long A, Z;
    public int K;
    public bool BaseMark, HeadMark;
}

class AutoLink
{
    public AutoEdge E;
    public long U, V;
    public AutoLink(AutoEdge e, long u, long v) { E = e; U = u; V = v; }
}

    bool HeadMark(IMyTerminalBlock b)
    {
        return EndpointOverride == null ? Mark(b.CustomName, "Head") :
            b.EntityId == EndpointOverride.EntityId && b.CustomName.Equals(EndpointName, StringComparison.OrdinalIgnoreCase);
    }
    public string Fingerprint = "", Report = "";
    public bool Ready;

    int StableCount;

    public AutoTopology(Program owner, string arm)
    {
        P = owner; Arm = arm == null ? "" : arm.Trim();
    }

    int Kind(IMyMechanicalConnectionBlock b)
    {
        if (b is IMyPistonBase) return 2;
        if (b is IMyMotorStator && !(b is IMyMotorSuspension)) return 1;
        return 0;
    }

    bool Attached(IMyMechanicalConnectionBlock b)
    {
        return b != null && !b.Closed && b.IsAttached && b.Top != null && b.TopGrid != null;
    }

    bool Mark(string name, string role)
    {
        return name != null && name.StartsWith(Arm + " - " + role, StringComparison.OrdinalIgnoreCase);
    }

    public bool Discover(out string why)
    {
        ZeroOwned(); Building = true;
        why = ""; Ready = false; StableCount = 0; Groups.Clear(); Rigid.Clear();
        Edges.Clear(); Used.Clear(); Corridor.Clear(); Watch.Clear(); Report = Fingerprint = "";
        if (Arm.Length == 0) return Bad("ArmName is empty.", out why);
        if (!Guard("pre-inventory reserve", out why)) return false;
        All.Clear(); Inventory.Clear(); P.GridTerminalSystem.GetBlocks(Inventory);
        if (Inventory.Count > MaxInventory) return Bad("Terminal inventory exceeds the " + MaxInventory + "-block discovery bound.", out why);
        if (!Guard("terminal inventory", out why)) return false;
        Base = null; Head = null; int bases = 0, heads = 0;
        for (int i = 0; i < Inventory.Count; i++)
        {
            if ((i & 63) == 0 && !Guard("terminal inventory filter", out why)) return false;
            var b = Inventory[i]; if (!b.IsSameConstructAs(P.Me)) continue;
            if (HeadMark(b)) { Head = b; heads++; }
            var m = b as IMyMechanicalConnectionBlock; if (m == null) continue;
            if (m.EntityId == ExcludedMechanical) continue;
            All.Add(m); if (Mark(m.CustomName, "Base")) { Base = m; bases++; }
        }
        if (bases != 1) return Bad("Need exactly one mechanical block named '" + Arm + " - Base*'; found " + bases + ".", out why);
        if (!Attached(Base) || Kind(Base) == 0) return Bad("The Base marker must be an attached piston, rotor, or hinge.", out why);
        if (heads != 1) return Bad("Need exactly one terminal block named '" + Arm + " - Head*'; found " + heads + ".", out why);
        BaseId = Base.EntityId; HeadId = Head.EntityId; HeadGrid = Head.CubeGrid.EntityId;
        End = EndOverride ?? Head;
        if (End.Closed || End.CubeGrid == null) return Bad("Endpoint unavailable.", out why);
        EndId = End.EntityId; EndGrid = End.CubeGrid.EntityId;
        Hull = Base.CubeGrid.EntityId; Start = Base.TopGrid.EntityId;
        if (Hull == Start || EndGrid == Hull) return Bad("Base top/Head must be beyond the hull.", out why);
        for (int i = 0; i < All.Count; i++)
        {
            if ((i & 63) == 0 && !Guard("mechanical edge inventory", out why)) return false;
            if (!Attached(All[i])) continue;
            var e = new AutoEdge { B = All[i], A = All[i].CubeGrid.EntityId, Z = All[i].TopGrid.EntityId,
                K = Kind(All[i]), BaseMark = Mark(All[i].CustomName, "Base"), HeadMark = HeadMark(All[i]) };
            if (e.A == e.Z) return Bad("Self-loop mechanical connection: " + e.B.CustomName, out why);
            Edges.Add(e);
        }
        if (!ScopeEdges(out why)) return false;
        var root = Between(Hull, Start);
        if (!Has(root, Base.EntityId)) return Bad("Base marker attachment changed while scanning.", out why);
        if (!AddGroup(root, out why)) return false;
        Corridor.Add(Start); Corridor.Add(Hull);
        var dist = Distances(EndGrid, out why); if (dist == null) return false;
        int d;
        if (!dist.TryGetValue(Start, out d)) return Bad("Head route returns through hull or is unreachable.", out why);
        long at = Start;
        while (at != EndGrid)
        {
            if (!Guard("corridor recognition", out why)) return false;
            if (Groups.Count >= MaxGroups) return Bad("Active arm exceeds " + MaxGroups + " generalized groups.", out why);
            int here = dist[at];
            var nextEdges = new List<AutoLink>();
            var destinations = new List<long>();
            for (int i = 0; i < Edges.Count; i++)
            {
                var e = Edges[i]; if (Used.Contains(e.B.EntityId)) continue;
                long n = e.A == at ? e.Z : e.Z == at ? e.A : 0; int nd;
                if (n == 0 || !dist.TryGetValue(n, out nd) || nd != here - 1) continue;
                nextEdges.Add(new AutoLink(e, at, n)); if (!destinations.Contains(n)) destinations.Add(n);
            }
            if (destinations.Count == 0) return Bad("No unambiguous next stage from rigid grid " + at + ".", out why);
            if (destinations.Count == 1)
            {
                if (!AddGroup(nextEdges, out why)) return false;
                at = destinations[0]; Corridor.Add(at);
                continue;
            }
            var paths = new List<List<AutoLink>>();
            var candidate = new HashSet<long>();
            for (int i = 0; i < destinations.Count; i++)
            {
                long n = destinations[i]; var path = new List<AutoLink>(); int count = 0;
                for (int j = 0; j < nextEdges.Count; j++) if (nextEdges[j].V == n) { path.Add(nextEdges[j]); count++; }
                if (count != 1) return Bad("Unsupported nested/parallel bundle at split grid " + at + ".", out why);
                candidate.Add(path[0].E.B.EntityId);
                paths.Add(path);
            }
            if (OwnedCount() + candidate.Count > MaxMembers) return Bad("Split/rejoin candidate exceeds " + MaxMembers + " owned actuators.", out why);
            int guard = 0;
            while (true)
            {
                if (!Guard("split/rejoin recognition", out why)) return false;
                long meet = paths[0][paths[0].Count - 1].V; bool same = true;
                for (int i = 1; i < paths.Count; i++) if (paths[i][paths[i].Count - 1].V != meet) same = false;
                if (same) { at = meet; break; }
                if (++guard > MaxMembers) return Bad("Parallel paths do not reconverge within the supported bound.", out why);
                var occupied = new HashSet<long>();
                for (int i = 0; i < paths.Count; i++)
                {
                    if (!Guard("split/rejoin path expansion", out why)) return false;
                    var path = paths[i]; long u = path[path.Count - 1].V;
                    if (!occupied.Add(u)) return Bad("Parallel paths merge or cross before a common stage boundary.", out why);
                    AutoLink chosen = null; int count = 0; int ud = dist[u];
                    for (int j = 0; j < Edges.Count; j++)
                    {
                        if ((j & 31) == 0 && !Guard("split/rejoin edge search", out why)) return false;
                        var e = Edges[j]; if (Used.Contains(e.B.EntityId) || candidate.Contains(e.B.EntityId)) continue;
                        long v = e.A == u ? e.Z : e.Z == u ? e.A : 0; int vd;
                        if (v != 0 && dist.TryGetValue(v, out vd) && vd == ud - 1) { chosen = new AutoLink(e, u, v); count++; }
                    }
                    if (count != 1) return Bad("Unsupported branch/cross-link inside split path at grid " + u + ".", out why);
                    candidate.Add(chosen.E.B.EntityId);
                    if (OwnedCount() + candidate.Count > MaxMembers) return Bad("Split/rejoin candidate exceeds " + MaxMembers + " owned actuators.", out why);
                    path.Add(chosen);
                }
            }
            int stages = paths[0].Count;
            for (int i = 1; i < paths.Count; i++) if (paths[i].Count != stages)
                return Bad("Parallel arms have unequal stage counts; variable-ratio closure is unsupported.", out why);
            for (int s = 0; s < stages; s++)
            {
                var stage = new List<AutoLink>();
                for (int i = 0; i < paths.Count; i++) { stage.Add(paths[i][s]); Corridor.Add(paths[i][s].U); Corridor.Add(paths[i][s].V); }
                if (!AddGroup(stage, out why)) return false;
            }
        }
        if (!ClassifyUnused(out why)) return false;
        if (!BuildWatch(out why)) return false;
        if (!SnapshotFromInventory(out Snapshot, out why)) return Bad(why, out why);
        var fp = new System.Text.StringBuilder();
        for (int i = 0; i < Groups.Count; i++) { Groups[i].Index = i; if (i > 0) fp.Append('|'); fp.Append(Groups[i].Sig); }
        Fingerprint = fp.ToString();
        Rigid.AddRange(Corridor); Rigid.Sort();
        Report += Groups.Count + " groups / " + OwnedCount() + " actuators / " + Rigid.Count + " rigid segments detected.";
        for (int i = 0; i < Rigid.Count; i++) Report += "\nR" + (i + 1).ToString("00") + " = grid " + Rigid[i];
        for (int i = 0; i < Groups.Count; i++) if (Groups[i].M.Count > 1)
        { Report += " Parallel 1:1 copies: rechecked live."; break; }
        Report += "\nCached checks each pass; full scan every 30.";
        FullScanIn = 30;
        Building = false; ZeroOwned();
        return true;
    }

    bool ScopeEdges(out string why)
    {
        why = "";
        var byGrid = new Dictionary<long, List<AutoEdge>>();
        for (int i = 0; i < Edges.Count; i++)
        {
            if ((i & 63) == 0 && !Guard("moving-component adjacency", out why)) return false;
            var e = Edges[i]; List<AutoEdge> a, z;
            if (!byGrid.TryGetValue(e.A, out a)) byGrid[e.A] = a = new List<AutoEdge>();
            if (!byGrid.TryGetValue(e.Z, out z)) byGrid[e.Z] = z = new List<AutoEdge>();
            a.Add(e); z.Add(e);
        }
        var keep = new HashSet<long>(); var nodes = new HashSet<long>(); var q = new Queue<long>();
        nodes.Add(Start); q.Enqueue(Start);
        while (q.Count > 0)
        {
            if (!Guard("moving-component scope", out why)) return false;
            long u = q.Dequeue(); List<AutoEdge> list;
            if (!byGrid.TryGetValue(u, out list)) continue;
            for (int i = 0; i < list.Count; i++)
            {
                if ((i & 31) == 0 && !Guard("moving-component expansion", out why)) return false;
                var e = list[i]; keep.Add(e.B.EntityId);
                if (keep.Count > MaxWorkEdges) return Bad("Moving mechanical component exceeds the " + MaxWorkEdges + "-edge discovery bound.", out why);
                long v = e.A == u ? e.Z : e.A;
                if (v != Hull && nodes.Add(v)) q.Enqueue(v);
            }
        }
        var scoped = new List<AutoEdge>();
        for (int i = 0; i < Edges.Count; i++)
        {
            if ((i & 63) == 0 && !Guard("moving-component filter", out why)) return false;
            if (keep.Contains(Edges[i].B.EntityId)) scoped.Add(Edges[i]);
        }
        Edges.Clear(); Edges.AddRange(scoped);
        return true;
    }

    bool Has(List<AutoLink> links, long id)
    {
        for (int i = 0; i < links.Count; i++) if (links[i].E.B.EntityId == id) return true;
        return false;
    }

    List<AutoLink> Between(long u, long v)
    {
        var r = new List<AutoLink>();
        for (int i = 0; i < Edges.Count; i++)
        {
            var e = Edges[i];
            if ((e.A == u && e.Z == v) || (e.A == v && e.Z == u)) r.Add(new AutoLink(e, u, v));
        }
        return r;
    }

    Dictionary<long, int> Distances(long target, out string why)
    {
        why = ""; var d = new Dictionary<long, int>(); var q = new Queue<long>(); d[target] = 0; q.Enqueue(target);
        while (q.Count > 0)
        {
            if (!Guard("Head-distance graph", out why)) return null;
            long u = q.Dequeue(); int nd = d[u] + 1;
            for (int i = 0; i < Edges.Count; i++)
            {
                var e = Edges[i]; long v = e.A == u ? e.Z : e.Z == u ? e.A : 0;
                if (v == 0 || v == Hull || d.ContainsKey(v)) continue;
                d[v] = nd; q.Enqueue(v);
            }
        }
        return d;
    }

    bool AddGroup(List<AutoLink> links, out string why)
    {
        why = ""; if (links.Count == 0) return Bad("Empty mechanical stage.", out why);
        if (Groups.Count >= MaxGroups) return Bad("Active Base-to-Head corridor exceeds " + MaxGroups + " generalized groups.", out why);
        if (OwnedCount() + links.Count > MaxMembers) return Bad("Active Base-to-Head corridor exceeds " + MaxMembers + " owned actuators.", out why);
        links.Sort((x, y) => x.E.B.EntityId.CompareTo(y.E.B.EntityId));
        int kind = Kind(links[0].E.B);
        if (kind == 0) return Bad("Unsupported connection on active route: " + links[0].E.B.CustomName, out why);
        var g = new AutoGroup { Rotary = kind == 1 };
        Vector3D baseL = Vector3D.Zero, baseA = Vector3D.Zero;
        for (int i = 0; i < links.Count; i++)
        {
            var l = links[i];
            if (Kind(l.E.B) != kind) return Bad("Parallel stage mixes piston and rotary actuators.", out why);
            var m = new AutoMember { B = l.E.B, U = l.U, V = l.V, Sign = 1 };
            Vector3D lin, ang;
            if (!Raw(m, EndPosition, out lin, out ang, out why)) return Bad(why, out why);
            if (i == 0) { baseL = lin; baseA = ang; }
            else
            {
                double dot = kind == 1 ? Vector3D.Dot(baseA, ang) : Vector3D.Dot(baseL, lin);
                m.Sign = dot < 0 ? -1 : 1; lin *= m.Sign; ang *= m.Sign;
                bool compatible = kind == 1
                    ? Vector3D.Dot(baseA, ang) >= AxisCos && (baseL - lin).Length() <= RotaryLineM
                    : Vector3D.Dot(baseL, lin) >= AxisCos;
                if (!compatible) return Bad("Parallel " + (kind == 1 ? "rotors/hinges lack one coaxial full twist" : "pistons are not collinear") +
                    " at stage containing '" + l.E.B.CustomName + "'. General linkage ratios are unsupported.", out why);
            }
            m.Q0 = m.Last = Position(m);
            if (!Finite(m.Q0)) return Bad("Non-finite actuator position: " + m.B.CustomName, out why);
            g.M.Add(m); Used.Add(m.B.EntityId);
        }
        var sig = new System.Text.StringBuilder(kind == 1 ? "R" : "P");
        var label = new System.Text.StringBuilder(kind == 1 ? "Rotary: " : "Piston: ");
        for (int i = 0; i < g.M.Count; i++)
        {
            var m = g.M[i]; sig.Append(':').Append(m.B.EntityId).Append(m.Sign > 0 ? '+' : '-').Append('@').Append(m.U).Append('>').Append(m.V);
            if (i > 0) label.Append(" | "); label.Append(m.B.CustomName);
        }
        g.Sig = sig.ToString(); g.Label = label.ToString(); Groups.Add(g);
        Report += "G" + Groups.Count + " " + g.Label + "\n";
        return true;
    }

    bool Raw(AutoMember m, Vector3D head, out Vector3D lin, out Vector3D ang, out string why)
    {
        lin = ang = Vector3D.Zero; why = ""; var b = m.B;
        if (!Attached(b)) { why = "Detached actuator: " + b.CustomName; return false; }
        long a = b.CubeGrid.EntityId, z = b.TopGrid.EntityId; int dir;
        if (a == m.U && z == m.V) dir = 1; else if (a == m.V && z == m.U) dir = -1;
        else { why = "Actuator endpoints changed: " + b.CustomName; return false; }
        Vector3D axis = b.WorldMatrix.Up; double n = axis.Length();
        if (!Finite(axis.X) || !Finite(axis.Y) || !Finite(axis.Z) || !Finite(head.X) || !Finite(head.Y) || !Finite(head.Z) ||
            !Finite(n) || n < .9) { why = "Invalid actuator axis or Head position: " + b.CustomName; return false; }
        axis /= n;
        if (b is IMyPistonBase) lin = axis * dir;
        else { ang = axis * -dir; lin = Vector3D.Cross(ang, head - b.GetPosition()); }
        return true;
    }

    public double Position(AutoMember m)
    {
        var p = m.B as IMyPistonBase; return p != null ? p.CurrentPosition : ((IMyMotorStator)m.B).Angle;
    }



    public bool CheckSync(AutoGroup g, out string why)
    {
        why = "";
        double lo = double.PositiveInfinity, hi = double.NegativeInfinity, spread = 0;
        for (int i = 0; i < g.M.Count; i++)
        {
            var m = g.M[i]; double q = Position(m);
            if (!Finite(q) || !Finite(m.Q0)) { why = "Non-finite actuator position in G" + (g.Index + 1) + ": " + m.B.CustomName; return false; }
            double d = q - m.Q0; if (g.Rotary) d = Wrap(d); d *= m.Sign;
            if (g.Rotary) for (int j = 0; j < i; j++)
            {
                var n = g.M[j]; double nd = Wrap(Position(n) - n.Q0) * n.Sign;
                spread = Math.Max(spread, Math.Abs(Wrap(d - nd)));
            }
            lo = Math.Min(lo, d); hi = Math.Max(hi, d);
        }
        if (g.M.Count < 2) return true;
        double tol = g.Rotary ? .25 * Math.PI / 180 : .02;
        if (!g.Rotary) spread = hi - lo;
        if (spread <= tol) return true;
        why = "Parallel actuator desynchronization in G" + (g.Index + 1) + ": " + spread.ToString("0.0000") + (g.Rotary ? " rad." : " m.");
        return false;
    }

    public bool Refresh(AutoGroup g, Vector3D headWorld, MatrixD reference, out double lo, out double hi, out string why)
    {
        lo = -1e6; hi = 1e6; why = ""; Vector3D sl = Vector3D.Zero, sa = Vector3D.Zero, firstL = Vector3D.Zero, firstA = Vector3D.Zero;
        if (!CheckSync(g, out why)) return Fail(why);
        for (int i = 0; i < g.M.Count; i++)
        {
            var m = g.M[i]; var b = m.B;
            if (!Attached(b) || !b.Enabled || !b.IsWorking) { why = "Unavailable owned actuator: " + b.CustomName; return Fail(why); }
            Vector3D l, a; if (!Raw(m, headWorld, out l, out a, out why)) return Fail(why);
            l *= m.Sign; a *= m.Sign;
            if (i == 0) { firstL = l; firstA = a; }
            else if (g.Rotary ? Vector3D.Dot(firstA, a) < AxisCos || (firstL - l).Length() > RotaryLineM : Vector3D.Dot(firstL, l) < AxisCos)
            { why = "Parallel screw mismatch G" + (g.Index + 1) + "; not 1:1."; return Fail(why); }
            sl += l; sa += a;
            double ml, mh; if (!Bounds(m, out ml, out mh, out why)) return Fail(why); lo = Math.Max(lo, ml); hi = Math.Min(hi, mh);
        }
        if (lo > hi) { why = "No common signed velocity remains in G" + (g.Index + 1) + "."; return Fail(why); }
        sl /= g.M.Count; sa /= g.M.Count;
        g.Lin = InFrame(sl, reference); g.Ang = InFrame(sa, reference);
        return true;
    }

    bool Bounds(AutoMember m, out double lo, out double hi, out string why)
    {
        lo = -1e6; hi = 1e6; why = ""; double q = Position(m), mn, mx;
        var p = m.B as IMyPistonBase;
        if (p != null) { lo = -p.MaxVelocity; hi = p.MaxVelocity; mn = p.MinLimit; mx = p.MaxLimit; }
        else
        {
            var r = (IMyMotorStator)m.B; mn = r.LowerLimitRad; mx = r.UpperLimitRad;
            if (r.RotorLock) lo = hi = 0;
        }
        if (!Finite(q) || !Finite(mn) || !Finite(mx) || !Finite(lo) || !Finite(hi) || mn > mx || lo > hi)
        { why = "Invalid position, limits, or maximum velocity: " + m.B.CustomName; return false; }
        if (p != null && p.MaxVelocity <= 0) { why = "Piston reports no positive maximum velocity: " + m.B.CustomName; return false; }
        double accel = p != null ? P.PistonAccelerationMps2 : P.JointAccelerationRpmPerSec * Math.PI / 30;
        double margin = p != null ? .001 : .1 * Math.PI / 180;
        if (!Finite(accel) || accel <= 0) { why = "Acceleration must be finite and positive for travel braking."; return false; }
        if (mn > -1e6) lo = Math.Max(lo, -Math.Sqrt(2 * accel * Math.Max(0, q - mn - margin)));
        if (mx < 1e6) hi = Math.Min(hi, Math.Sqrt(2 * accel * Math.Max(0, mx - q - margin)));
        if (m.Sign < 0) { double x = lo; lo = -hi; hi = -x; }
        return true;
    }



    public void Set(AutoGroup g, double velocity)
    {
        for (int i = 0; i < g.M.Count; i++)
        {
            var m = g.M[i]; float v = (float)(m.Sign * velocity);
            var p = m.B as IMyPistonBase; if (p != null) p.Velocity = v; else ((IMyMotorStator)m.B).TargetVelocityRad = v;
        }
    }

    public void ZeroOwned()
    {
        for (int i = 0; i < Groups.Count; i++) for (int j = 0; j < Groups[i].M.Count; j++)
        {
            var b = Groups[i].M[j].B; if (b == null || b.Closed) continue;
            try { var p = b as IMyPistonBase; if (p != null) p.Velocity = 0; else { var r = b as IMyMotorStator; if (r != null) r.TargetVelocityRad = 0; } }
            catch (Exception) { }
        }
    }

    public bool StableSetup(out string why)
    {
        why = ""; if (!Validate(out why, true)) return false;
        bool moved = false;
        for (int i = 0; i < Groups.Count; i++) for (int j = 0; j < Groups[i].M.Count; j++)
        {
            var m = Groups[i].M[j]; double target, q = Position(m);
            if (!Finite(q)) { why = "Non-finite actuator position while settling: " + m.B.CustomName; StableCount = 0; return false; }
            var p = m.B as IMyPistonBase; target = p != null ? p.Velocity : ((IMyMotorStator)m.B).TargetVelocityRad;
            if (Math.Abs(target) > 1e-4) { why = "Waiting for owned actuator targets to reach zero."; StableCount = 0; return false; }
            double delta = q - m.Last; if (Groups[i].Rotary) delta = Wrap(delta);
            if (Math.Abs(delta) > 1e-4) moved = true; m.Last = q;
        }
        StableCount = moved ? 0 : StableCount + 1;
        if (StableCount < 2) { why = "Waiting for two stable stopped topology scans."; return false; }
        for (int i = 0; i < Groups.Count; i++) for (int j = 0; j < Groups[i].M.Count; j++) Groups[i].M[j].Q0 = Position(Groups[i].M[j]);
        Ready = true; return true;
    }

    public bool Validate(out string why, bool full = false)
    {
        why = "";
        if (!BudgetAvailable()) { why = "Budget low; motion refused."; return Fail(why); }
        if (Base == null || Head == null || Base.Closed || Head.Closed || Base.EntityId != BaseId || Head.EntityId != HeadId ||
            !Base.IsSameConstructAs(P.Me) || !Head.IsSameConstructAs(P.Me) || !Mark(Base.CustomName, "Base") || !HeadMark(Head))
        { why = "Base/Head marker disappeared or was renamed."; return Fail(why); }
        if (Head.CubeGrid == null || Head.CubeGrid.EntityId != HeadGrid) { why = "Head marker moved to another rigid grid; Rescan is required."; return Fail(why); }
        var marker = Head as IMyMechanicalConnectionBlock;
        if (End == null || End.Closed || End.EntityId != EndId || End.CubeGrid == null || End.CubeGrid.EntityId != EndGrid ||
            Focus != null && (Focus.Closed || Focus.CubeGrid.EntityId != EndGrid || Focus.Marker.CubeGrid.EntityId != EndGrid) ||
            Focus == null && End != Head && marker != null && (!Attached(marker) || marker.Top.EntityId != EndId))
        { why = "Endpoint changed; Rescan is required."; return Fail(why); }
        for (int i = 0; i < Edges.Count; i++)
        {
            var e = Edges[i]; var b = e.B;
            if (!Attached(b) || b.CubeGrid.EntityId != e.A || b.TopGrid.EntityId != e.Z || Kind(b) != e.K ||
                Mark(b.CustomName, "Base") != e.BaseMark || HeadMark(b) != e.HeadMark)
            { why = "Known mechanical connection changed: " + (b == null ? "missing block" : b.CustomName) + ". Rescan is required."; return Fail(why); }
        }
        for (int i = 0; i < Groups.Count; i++) if (!CheckSync(Groups[i], out why)) return Fail(why);
        if (!full && --FullScanIn > 0) return true;
        string current;
        if (!CurrentSnapshot(out current, out why)) return Fail(why);
        if (ScanBases != 1 || ScanHeads != 1) { why = "Base/Head marker is no longer unique; Rescan is required."; return Fail(why); }
        if (current != Snapshot) { why = "Mechanical topology changed; Rescan is required."; return Fail(why); }
        FullScanIn = 30;
        return true;
    }

    bool ClassifyUnused(out string why)
    {
        why = ""; var seen = new HashSet<long>(); int ignored = 0;
        for (int i = 0; i < Edges.Count; i++)
        {
            var seed = Edges[i]; if (Used.Contains(seed.B.EntityId) || seen.Contains(seed.B.EntityId)) continue;
            var q = new Queue<AutoEdge>(); var boundary = new HashSet<long>(); q.Enqueue(seed); seen.Add(seed.B.EntityId); int count = 0;
            while (q.Count > 0)
            {
                if (!Guard("off-corridor classification", out why)) return false;
                var e = q.Dequeue(); count++;
                if ((e.A == Hull && Corridor.Contains(e.Z)) || (e.Z == Hull && Corridor.Contains(e.A)))
                    return Bad("A second mechanical anchor connects the base grid to the active corridor: " + e.B.CustomName, out why);
                if (Corridor.Contains(e.A)) boundary.Add(e.A); if (Corridor.Contains(e.Z)) boundary.Add(e.Z);
                for (int j = 0; j < Edges.Count; j++)
                {
                    var n = Edges[j]; if (Used.Contains(n.B.EntityId) || seen.Contains(n.B.EntityId)) continue;
                    if (n.A == e.A || n.A == e.Z || n.Z == e.A || n.Z == e.Z) { seen.Add(n.B.EntityId); q.Enqueue(n); }
                }
            }
            if (boundary.Count > 1) return Bad("Unsupported closure: cross-linked segments.", out why);
            ignored += count;
        }
        if (ignored > 0) Report += "Ignored " + ignored + " off-corridor actuators; they are never commanded.\n";
        return true;
    }

    bool BuildWatch(out string why)
    {
        why = "";
        var q = new Queue<long>(); Watch.Add(Start); q.Enqueue(Start);
        while (q.Count > 0)
        {
            if (!Guard("topology watch-set", out why)) return false;
            long u = q.Dequeue();
            for (int i = 0; i < Edges.Count; i++)
            {
                var e = Edges[i]; long v = e.A == u ? e.Z : e.Z == u ? e.A : 0;
                if (v == 0 || v == Hull || !Watch.Add(v)) continue; q.Enqueue(v);
            }
        }
        return true;
    }

    bool CurrentSnapshot(out string snapshot, out string why)
    {
        snapshot = ""; why = "";
        if (!BudgetAvailable()) { why = "Budget low; scan refused."; return false; }
        Inventory.Clear(); P.GridTerminalSystem.GetBlocks(Inventory);
        return SnapshotFromInventory(out snapshot, out why);
    }

    bool SnapshotFromInventory(out string snapshot, out string why)
    {
        snapshot = ""; why = "";
        if (Inventory.Count > MaxInventory) { why = "Terminal inventory exceeds the " + MaxInventory + "-block validation bound."; return false; }
        SnapshotRows.Clear(); ScanBases = ScanHeads = 0;
        for (int i = 0; i < Inventory.Count; i++)
        {
            if ((i & 31) == 0 && !BudgetAvailable()) { why = "Scan budget exhausted."; return false; }
            var terminal = Inventory[i]; if (!terminal.IsSameConstructAs(P.Me)) continue;
            if (HeadMark(terminal)) ScanHeads++;
            var b = terminal as IMyMechanicalConnectionBlock; if (b == null) continue;
            if (b.EntityId == ExcludedMechanical) continue;
            long a = b.CubeGrid.EntityId, z = Attached(b) ? b.TopGrid.EntityId : 0;
            if (Mark(b.CustomName, "Base")) ScanBases++;
            if (Watch.Contains(a) || Watch.Contains(z)) SnapshotRows.Add(b.EntityId + ":" + a + ":" + z + ":" + Kind(b));
        }
        if (!BudgetAvailable()) { why = "Post-scan budget exhausted."; return false; }
        SnapshotRows.Sort(StringComparer.Ordinal); SnapshotText.Clear();
        for (int i = 0; i < SnapshotRows.Count; i++) { if (i > 0) SnapshotText.Append('|'); SnapshotText.Append(SnapshotRows[i]); }
        snapshot = SnapshotText.ToString(); return true;
    }

    int OwnedCount() { int n = 0; for (int i = 0; i < Groups.Count; i++) n += Groups[i].M.Count; return n; }


    bool BudgetAvailable()
    {
        return P.Runtime.MaxInstructionCount <= 0 || P.Runtime.CurrentInstructionCount < P.Runtime.MaxInstructionCount * .55;
    }

    bool Guard(string phase, out string why)
    {
        why = ""; if (BudgetAvailable()) return true;
        return Bad("Budget: " + phase + "; Off. Reduce topology/Rescan.", out why);
    }

    bool Fail(string why) { Ready = false; if (Building) { Groups.Clear(); Used.Clear(); } else ZeroOwned(); return false; }
    bool Bad(string text, out string why) { why = text; Fail(text); return false; }
}

                !long.TryParse(part.Substring(to + 1), out b) || id == 0 || a == 0 || b == 0 || a == b || !ids.Add(id)) return false;
        }
        return true;
    }

public Program()
{
    Runtime.UpdateFrequency = UpdateFrequency.Update10;
    Topology = new AutoTopology(this, ArmName);
    Tools = new ToolSwapController(this);
    try { ZeroSavedDrives(); Discover(false); }
    catch (Exception e)
    {
        Enabled = GoingHome = PendingOn = ConfigValid = false;
        Status = LastFault = "Startup discovery exception: " + e.Message;
        try { Topology.ZeroOwned(); } catch { }
    }
    Say(Banner() + "\n" + Status);
    Echo(DisplayText);
}

// Reloading the PB does not clear native actuator velocities. Recover only
// identities scoped to this PB/arm, before configuration can fail discovery.
void ZeroSavedDrives()
{
    var ini = AutoConfig.Saved(Storage); var ids = new HashSet<long>();
    string section = "AutoArm Config";
    if (ini.Get(section, "PB").ToInt64() == Me.EntityId && ini.Get(section, "Arm").ToString() == ArmName && ini.Get(section, "Format").ToInt32() == 1)
        foreach (string row in (ini.Get(section, "Rows").ToString() + "\n" + ini.Get(ToolWeightsSection, "Rows").ToString()).Split('\n'))
        {
            string[] cells = row.Split('|'); string key = cells.Length == 2 ? cells[1] : cells.Length == 3 ? cells[0] : "";
            foreach (string part in key.Split(':'))
            {
                string token = part.Split('@')[0]; long id;
                if (token.Length > 1 && (token.EndsWith("+") || token.EndsWith("-")) && long.TryParse(token.Substring(0, token.Length - 1), out id)) ids.Add(id);
            }
        }
    var config = new MyIni(); MyIniParseResult error;
    config.TryParse(Me.CustomData, out error);
    var mounts = new List<string>();
    if (config.Get(ArmSection, "Arm").ToString() == ArmName && config.Get(ToolsSection, EnabledKey).ToBoolean(false))
        for (int i = 1; i <= 16; i++)
        { string name = config.Get("Tool" + i.ToString("00"), "Mount").ToString().Trim(); if (name.Length > 0 && !mounts.Contains(name)) mounts.Add(name); }
    var blocks = new List<IMyTerminalBlock>(); GridTerminalSystem.GetBlocks(blocks);
    var unique = new HashSet<long>();
    foreach (string name in mounts)
    { IMyTerminalBlock found = null; int count = 0; foreach (var b in blocks) if (b.CustomName == name) { found = b; count++; } if (count == 1) unique.Add(found.EntityId); }
    foreach (var b in blocks)
    {
        if (b.Closed || !b.IsSameConstructAs(Me) || (!ids.Contains(b.EntityId) && !unique.Contains(b.EntityId))) continue;
        try { var piston = b as IMyPistonBase; var rotor = b as IMyMotorStator;
            if (piston != null) piston.Velocity = 0; else if (rotor != null && !(rotor is IMyMotorSuspension)) rotor.TargetVelocityRad = 0;
        } catch { }
    }
}

public void Main(string argument, UpdateType source)
{
    DisplayWritten = false;
    double elapsed = Runtime.TimeSinceLastRun.TotalSeconds;
    bool timed = (source & (UpdateType.Update1 | UpdateType.Update10 | UpdateType.Update100 | UpdateType.Once)) != 0;
    if (Finite(elapsed) && elapsed > 0)
    {
        if (Enabled || (Tools != null && Tools.Busy)) ControlElapsed += elapsed;
        if (!Topology.Ready) SetupElapsed = Math.Min(0.5, SetupElapsed + elapsed);
        StatusElapsed = Math.Min(StatusRefreshSeconds, StatusElapsed + elapsed);
    }
    try
    {
        if (!string.IsNullOrWhiteSpace(argument)) RunCommands(argument);
        if (Tools != null && Tools.Busy)
        {
            if (!ConfigValid) { Tools.Cancel("Invalid configuration; Reload required."); return; }
            if (Tools.Automatic && HasPilotInput()) { Tools.Cancel("Pilot canceled swap."); return; }
            if (ControlElapsed > 0.5) { Fault("Tool operation update gap exceeded 0.5 seconds."); return; }
            if (!timed) return;
            if (!Finite(elapsed) || elapsed < 0) { Fault("Invalid tool operation interval."); return; }
            double toolDt = ControlElapsed;
            if (toolDt <= 0.000001) return;
            ControlElapsed = 0;
            ToolGeometry.Owner = this;
            Tools.Tick(Math.Min(toolDt, .1), toolDt);
            return;
        }
        if (!Topology.Ready)
        {
            // A validation failure must require a new explicit On. Stable setup may
            // recover diagnostics while Off, but it must never resume old commands.
            if (Enabled || GoingHome)
            {
                StopMotion("Topology lost readiness; stopped. Run Check/Rescan/Reload, then On.");
                Say(Status);
                return;
            }
            if (!timed || !Finite(elapsed) || elapsed < 0 || SetupElapsed <= 0.000001) return;
            SetupElapsed = 0;
            MaintainSetup();
            if (!Topology.Ready) return;
        }
        if (PendingOn)
        {
            PendingOn = false;
            EnableControl();
        }
        if (!Enabled) return;
        if (ControlElapsed > 0.5) { Fault("Control update gap exceeded 0.5 seconds."); return; }
        // Commands can run in the same simulation frame as the previous call.
        // They change state immediately, but only timed updates advance motion.
        if (!timed) return;
        if (!Finite(elapsed) || elapsed < 0) { Fault("Invalid negative/non-finite control interval."); return; }
        double dt = ControlElapsed;
        if (dt <= 0.000001) return;
        ControlElapsed = 0;
        if (dt > 0.5) { Fault("Control update gap exceeded 0.5 seconds."); return; }
        if (GoingHome) UpdateHome(Math.Min(dt, 0.1));
        else UpdateControl(Math.Min(dt, 0.1), dt);
    }
    catch (Exception e)
    {
        Fault("Controller exception: " + e.Message);
    }
    finally
    {
        // SE clears the PB display on every invocation. Replay cached output;
        // routine numeric formatting happens at 4 Hz, not every control tick.
        Echo(DisplayText);
    }
}

void Say(string text)
{
    DisplayText = DisplayWritten ? DisplayText + "\n" + text : text;
    if (DisplayText.Length > 8000) DisplayText = DisplayText.Substring(DisplayText.Length - 8000);
    DisplayWritten = true;
}

string Banner()
{
    return "AutoArm " + ControllerBuildVersion + " | " + ArmName;
}

void Discover(bool explicitRequest)
{
    StopMotion("");
    PivotProbe = null;
    SetupElapsed = 0;
    ConfigValid = false;
    string why;
    AutoConfig.EnsureDefaults(this);
    MyIni toolConfig; var options = new AutoSettings();
    if (!AutoConfig.ReadCurrent(Me.CustomData, out toolConfig, out why) || !options.Read(toolConfig, out why)) throw new Exception(why);
    bool toolEnabled;
    if (toolConfig.ContainsKey(ToolsSection, EnabledKey) && !toolConfig.Get(ToolsSection, EnabledKey).TryGetBoolean(out toolEnabled)) throw new Exception("Tools.Enabled must be true/false.");
    if (toolConfig.Get(ToolsSection, EnabledKey).ToBoolean(false) && Topology.Groups.Count > 0)
    {
        string preserved; bool differs;
        if (!AutoConfig.LoadOrGenerate(this, Topology, Me.CustomData, out preserved, out differs, out why, true)) throw new Exception(why);
    }
    if (Tools != null && !Tools.Load(out why)) throw new Exception(why);
    Topology.SetEndpointOverride(Tools == null ? null : Tools.Endpoint, Tools == null ? 0 : Tools.IgnoredMount, Tools == null ? null : Tools.EndFrame, Tools == null ? null : Tools.Focus);
    if (!Topology.Discover(out why))
    {
        Status = "Discovery failed: " + why;
        if (explicitRequest) Say(Status);
        return;
    }
    Topology.ZeroOwned();
    string updated;
    bool changed;
    if (!AutoConfig.LoadOrGenerate(this, Topology, Me.CustomData, out updated, out changed, out why))
    {
        Status = "Custom Data refused: " + why;
        if (explicitRequest) Say(Status);
        return;
    }
    ConfigValid = true;
    Allocate(Topology.Groups.Count);
    LoadHome();
    Status = (changed ? "Configuration generated/updated; " : "Configuration loaded; ") +
        Topology.Groups.Count + " groups found. Waiting for stable stopped scans.";
    Runtime.UpdateFrequency = Tools != null && Tools.Busy ? UpdateFrequency.Update1 : UpdateFrequency.Update10;
    if (explicitRequest) Say(Status + "\n" + Topology.Report);
}

void MaintainSetup()
{
    if (!ConfigValid) return;
    string why;
    Topology.ZeroOwned();
    if (!Topology.StableSetup(out why))
    {
        Status = "Setup waiting: " + why;
        return;
    }
    Status = "Ready and OFF. Run On.";
    Say(Banner() + "\n" + Status);
}

void Allocate(int n)
{
    J = new double[6, n];
    B = new double[6, n];
    Gram = new double[6, 6];
    Chol = new double[6, 6];
    Scale = new double[n]; Lo = new double[n]; Hi = new double[n];
    X = new double[n]; Trial = new double[n]; LastQdot = new double[n];
    BoundState = new int[n];
    Task = new double[6]; Residual = new double[6]; Rhs = new double[6];
    Temp = new double[6]; Solution = new double[6];
}

void EnableControl()
{
    string toolWhy = "";
    if (SwapBatchLock || (Tools != null && !Tools.AllowManual(out toolWhy)))
    { PendingOn = false; Say(SwapBatchLock ? "On refused in tool command batch." : "On refused: " + toolWhy); return; }
    if (!ConfigValid)
    {
        PendingOn = false;
        Status = "On refused: Custom Data configuration is invalid. Fix it, then Reload.";
        Say(Status);
        return;
    }
    if (!Topology.Ready)
    {
        PendingOn = true;
        Status = "On pending: two stopped scans required.";
        Runtime.UpdateFrequency = UpdateFrequency.Update10;
        Say(Status);
        return;
    }
    string why;
    if (!Topology.Validate(out why, true)) { Fault("On refused: " + why); return; }
    Vector3D p, f, u;
    if (!ReadPose(out p, out f, out u)) { Fault("On refused: Base/Head markers unavailable."); return; }
    CapturePose(p, f, u);
    PivotProbe = null;
    LastPilotLinear = LastPilotAngular = LastFeedbackLinear = LastFeedbackAngular = Vector3D.Zero;
    LastRequestedLinear = LastRequestedAngular = LastAchievedLinear = LastAchievedAngular = Vector3D.Zero;
    LastLinearAdvance = LastAngularAdvance = 1;
    LastPositionError = LastOrientationErrorDeg = 0;
    LastSolverPasses = LastActiveBounds = 0;
    LastSolverConverged = LastSolverBudgetLimited = false;
    LastFault = "";
    Enabled = true;
    GoingHome = false;
    ControlElapsed = 0;
    StatusElapsed = StatusRefreshSeconds;
    DisplayPinned = false;
    InputController = null;
    InputScan = 10;
    Runtime.UpdateFrequency = UpdateFrequency.Update1;
    Status = "ON: pilot input plus pose feedback";
    Say(Banner() + "\n" + Status);
}

void StartControl()
{
    if (SwapBatchLock) { Say("Tool busy; Stop first."); return; }
    Discover(false);
    if (!ConfigValid) { Say(Status); return; }
    if (Tools != null && Tools.Busy) Tools.StartOn();
    else EnableControl();
}

void StopMotion(string reason)
{
    if (Topology != null) Topology.ZeroOwned();
    if (Tools != null) Tools.ZeroMount();
    Enabled = GoingHome = PendingOn = false;
    ControlElapsed = 0;
    DisplayPinned = false;
    TargetValid = ObservationValid = false;
    AimingTarget = false;
    PositionI = OrientationI = Vector3D.Zero;
    if (LastQdot != null) Array.Clear(LastQdot, 0, LastQdot.Length);
    if (BoundState != null) Array.Clear(BoundState, 0, BoundState.Length);
    Runtime.UpdateFrequency = UpdateFrequency.Update10;
    if (!string.IsNullOrWhiteSpace(reason))
    {
        Status = reason;
        LastFault = reason;
    }
}

void Fault(string reason)
{
    if (Tools != null && Tools.Busy) Tools.Cancel(reason);
    StopMotion(reason);
    Say("STOPPED: " + reason);
}

bool ReadPose(out Vector3D p, out Vector3D f, out Vector3D u)
{
    p = f = u = Vector3D.Zero;
    if (Topology == null || Topology.Base == null || Topology.End == null || Topology.Base.Closed || Topology.End.Closed) return false;
    MatrixD r = Topology.Base.WorldMatrix;
    p = InFrame(Topology.EndPosition - Topology.Base.GetPosition(), r);
    var pose = Topology.EndPose;
    f = InFrame(pose.Forward, r);
    u = InFrame(pose.Up, r);
    if (!Finite(p.LengthSquared())) return false;
    return Orthonormalise(ref f, ref u);
}

void UpdateControl(double dt, double observationDt)
{
    string why;
    if (!ToolDriving && Tools != null && !Tools.AllowManual(out why)) { Fault(why); return; }
    if (!Topology.Validate(out why)) { Fault("Topology changed: " + why); return; }
    Vector3D p, f, u;
    if (!ReadPose(out p, out f, out u)) { Fault("Base/Head marker became unavailable."); return; }
    if (!TargetValid) { Fault("No held pose. Run On."); return; }
    Vector3D measuredV = Vector3D.Zero, measuredW = Vector3D.Zero;
    if (ObservationValid)
    {
        measuredV = (p - PreviousP) / observationDt;
        measuredW = RotationError(PreviousF, PreviousU, f, u) / observationDt;
        double blend = observationDt / (0.15 + observationDt);
        FilteredV += (measuredV - FilteredV) * blend;
        FilteredW += (measuredW - FilteredW) * blend;
    }
    PreviousP = p; PreviousF = f; PreviousU = u; ObservationValid = true;

    Vector3D pe = TargetP - p;
    Vector3D oe = RotationError(f, u, TargetF, TargetU);
    LastPositionError = pe.Length();
    LastOrientationErrorDeg = oe.Length() * 180 / Math.PI;
    if (!Finite(LastPositionError) || !Finite(LastOrientationErrorDeg)) { Fault("Non-finite pose feedback."); return; }

    if (ToolDriving) LastPilotLinear = LastPilotAngular = Vector3D.Zero;
    else ReadPilot(f, u, out LastPilotLinear, out LastPilotAngular);
    bool manualTurn = LastPilotAngular.LengthSquared() > 1e-12;
    if (manualTurn) AimingTarget = false;
    if (AimingTarget && LastOrientationErrorDeg <= AimToleranceDeg) AimingTarget = false;
    bool aiming = ToolDriving || manualTurn || AimingTarget;
    double orientationBand = ToolDriving ? 0 : (aiming ? 0.5 * AimToleranceDeg : OrientationToleranceDeg) * Math.PI / 180;
    Vector3D positionDemand = Outside(pe, ToolDriving ? 0 : PositionDeadbandM) * PositionKp;
    Vector3D orientationDemand = Outside(oe, orientationBand) * OrientationKp;
    if (ToolDriving)
    {
        positionDemand = ClampLength(positionDemand, ToolDriveMoveCap);
        orientationDemand = ClampLength(orientationDemand, ToolDriveTurnCap);
    }
    Vector3D rawPositionFeedback = positionDemand + PositionI - FilteredV * 0.0;
    Vector3D rawOrientationFeedback = orientationDemand + OrientationI - FilteredW * 0.0;
    Vector3D boundedPositionFeedback = ClampLength(rawPositionFeedback, PositionCorrectionCap);
    LastFeedbackLinear = BlendManualCorrection(LastPilotLinear, boundedPositionFeedback,
        ManualCounterCorrectionFraction, out LastCounterCorrectionLimited);
    bool angularLimited;
    LastFeedbackAngular = BlendManualCorrection(LastPilotAngular, ClampLength(rawOrientationFeedback,
        Math.Min(TurnDeg, OrientationCorrectionCapDeg) * Math.PI / 180),
        ManualCounterCorrectionFraction, out angularLimited);
    LastCounterCorrectionLimited |= angularLimited;

    // Core invariant: tracking correction is added to live pilot demand. Pose error,
    // recovery state and target advancement never multiply or suppress pilot input.
    LastRequestedLinear = LastPilotLinear + LastFeedbackLinear;
    LastRequestedAngular = LastPilotAngular + LastFeedbackAngular;
    if (!Solve(p, dt, LastRequestedLinear, LastRequestedAngular, out LastAchievedLinear,
        out LastAchievedAngular, out why))
    {
        Fault("Velocity allocation failed: " + why);
        return;
    }
    if (!Commit()) return;

    Vector3D positionClipResidual = rawPositionFeedback - LastFeedbackLinear;
    Vector3D orientationClipResidual = rawOrientationFeedback - LastFeedbackAngular;
    // Only loss caused by command bounds inhibits bias learning. Damping and
    // unreachable task components exist even in the unconstrained reference.
    Vector3D positionAllocationResidual = LastUnconstrainedLinear - LastAchievedLinear;
    Vector3D orientationAllocationResidual = LastUnconstrainedAngular - LastAchievedAngular;
    UpdateIntegral(ref PositionI, ToolDriving ? positionDemand - FilteredV : pe, ToolDriving ? 0 : PositionDeadbandM, positionClipResidual, positionAllocationResidual,
        ToolDriving ? PositionKi / PositionKp : PositionKi, Math.Min(PositionIntegralCap, PositionCorrectionCap), ToolDriving ? double.MaxValue : 0.10, dt, LastSolverConverged && !LastSolverBudgetLimited);
    UpdateIntegral(ref OrientationI, ToolDriving ? orientationDemand - FilteredW : oe, ToolDriving ? 0 : orientationBand, orientationClipResidual, orientationAllocationResidual, ToolDriving ? OrientationKi / OrientationKp : OrientationKi,
        Math.Min(OrientationIntegralCapDeg, Math.Min(TurnDeg, OrientationCorrectionCapDeg)) * Math.PI / 180, ToolDriving ? double.MaxValue : 10.0 * Math.PI / 180,
        dt, LastSolverConverged && !LastSolverBudgetLimited);

    LastLinearAdvance = PilotAdvance(LastPilotLinear, LastAchievedLinear - LastFeedbackLinear);
    LastAngularAdvance = PilotAdvance(LastPilotAngular, LastAchievedAngular - LastFeedbackAngular);
    // Allocation is a kinematic prediction, not measured delivery. Bound newly
    // requested target lead even when physical contact prevents predicted motion.
    LastLinearAdvance *= TargetLeadAdvance(pe, LastPilotLinear * (dt * LastLinearAdvance), MaximumPilotTargetLeadM);
    TargetP += LastPilotLinear * (dt * LastLinearAdvance);
    if (LastPilotAngular.LengthSquared() > 1e-20 && LastAngularAdvance > 0)
    {
        Vector3D step = LastPilotAngular * (dt * LastAngularAdvance);
        TargetF = Rotate(TargetF, step); TargetU = Rotate(TargetU, step);
        Orthonormalise(ref TargetF, ref TargetU);
    }
    RenderRoutineStatus();
}

void RenderRoutineStatus()
{
    if (DisplayWritten || DisplayPinned || StatusElapsed < StatusRefreshSeconds) return;
    StatusElapsed = 0;
    RoutineStatusBuilds++;
    Status = (ToolDriving ? "TOOL motion | error " : "ON | input+feedback | error ") + LastPositionError.ToString(ThreeDecimals) + " m / " +
        LastOrientationErrorDeg.ToString("0.00") + " deg";
    if (LastCounterCorrectionLimited) Status += " | manual priority";
    Say(Status + "\nPilot " + LastPilotLinear.Length().ToString(ThreeDecimals) + " + feedback " +
        LastFeedbackLinear.Length().ToString(ThreeDecimals) + " m/s; advance " +
        (100 * LastLinearAdvance).ToString("0") + "%\nSolver " + LastSolverPasses +
        " passes; " + LastActiveBounds + " bounds. Stop zeros drives.");
}

bool Solve(Vector3D headReferencePosition, double dt, Vector3D linear, Vector3D angular,
    out Vector3D achievedLinear, out Vector3D achievedAngular, out string why)
{
    achievedLinear = achievedAngular = Vector3D.Zero;
    why = "";
    LastSolverPasses = LastActiveBounds = 0;
    LastSolverConverged = LastSolverBudgetLimited = false;
    int n = Topology.Groups.Count;
    double moveDemand = linear.Length(), turnDemand = angular.Length();
    double demand = moveDemand + turnDemand;
    MatrixD frame = Topology.Base.WorldMatrix;
    Vector3D headWorld = Topology.EndPosition;
    for (int i = 0; i < n; i++)
    {
        AutoGroup g = Topology.Groups[i];
        double physicalLo, physicalHi;
        if (!Topology.Refresh(g, headWorld, frame, out physicalLo, out physicalHi, out why)) return false;
        double cap = g.Rotary ? JointRpm * Math.PI / 30 : PistonMps;
        double lo = Math.Max(physicalLo, -cap), hi = Math.Min(physicalHi, cap);
        double accel = g.Rotary ? JointAccelerationRpmPerSec * Math.PI / 30 : PistonAccelerationMps2;
        if (accel > 0)
        {
            // A changed limit/lock may make the old command infeasible. Centre the
            // acceleration window on the nearest currently feasible command.
            double centre = Clamp(LastQdot[i], lo, hi);
            lo = Math.Max(lo, centre - accel * dt);
            hi = Math.Min(hi, centre + accel * dt);
        }
        if (lo > hi + 1e-10) { why = "No common command interval for " + g.Label; return false; }
        double preference = demand > 1e-9 ? (moveDemand * g.MoveW + turnDemand * g.TurnW) / demand :
            0.5 * (g.MoveW + g.TurnW);
        double s = Scale[i] = preference > 0 ? cap * Math.Sqrt(preference) : 0;
        Lo[i] = s > 1e-12 ? lo / s : 0;
        Hi[i] = s > 1e-12 ? hi / s : 0;
        X[i] = s > 1e-12 ? Clamp(LastQdot[i] / s, Lo[i], Hi[i]) : 0;
        int old = BoundState[i];
        BoundState[i] = Hi[i] - Lo[i] <= 1e-12 ? 2 :
            old == -1 && Math.Abs(X[i] - Lo[i]) <= 1e-9 ? -1 :
            old == 1 && Math.Abs(X[i] - Hi[i]) <= 1e-9 ? 1 : 0;
        for (int r = 0; r < 6; r++)
        {
            double weight = r < 3 ? PositionTaskWeight : OrientationTaskWeight;
            J[r, i] = Component(r < 3 ? g.Lin : g.Ang, r % 3);
            B[r, i] = weight * J[r, i] * s;
        }
    }
    for (int r = 0; r < 6; r++)
    {
        double weight = r < 3 ? PositionTaskWeight : OrientationTaskWeight;
        Task[r] = weight * Component(r < 3 ? linear : angular, r % 3);
    }
    double ridge = SolverDamping * SolverDamping;
    int maxPasses = Math.Min(64, Math.Max(8, 2 * n + 4));
    for (int pass = 0; pass < maxPasses; pass++)
    {
        if (Runtime.CurrentInstructionCount > Runtime.MaxInstructionCount * 0.68)
        {
            LastSolverBudgetLimited = true;
            break;
        }
        LastSolverPasses = pass + 1;
        BuildTaskMatrix(n, ridge, true);
        if (!Cholesky(Gram, Rhs, Chol, Temp, Solution)) { why = "singular/non-finite damped task matrix"; return false; }
        double alpha = 1;
        int hit = -1, side = 0;
        for (int i = 0; i < n; i++)
        {
            Trial[i] = X[i];
            if (BoundState[i] != 0) continue;
            double value = 0;
            for (int r = 0; r < 6; r++) value += B[r, i] * Solution[r];
            Trial[i] = value;
            double delta = value - X[i];
            int bound = delta < -1e-14 && value < Lo[i] - 1e-12 ? -1 :
                delta > 1e-14 && value > Hi[i] + 1e-12 ? 1 : 0;
            if (bound != 0)
            {
                double a = Math.Max(0, ((bound < 0 ? Lo[i] : Hi[i]) - X[i]) / delta);
                if (a < alpha) { alpha = a; hit = i; side = bound; }
            }
        }
        for (int i = 0; i < n; i++) if (BoundState[i] == 0)
            X[i] = Clamp(X[i] + alpha * (Trial[i] - X[i]), Lo[i], Hi[i]);
        if (hit >= 0)
        {
            X[hit] = side < 0 ? Lo[hit] : Hi[hit];
            BoundState[hit] = side;
            continue;
        }
        for (int r = 0; r < 6; r++)
        {
            double value = -Task[r];
            for (int i = 0; i < n; i++) value += B[r, i] * X[i];
            Residual[r] = value;
        }
        int release = -1;
        double worst = 1e-9;
        for (int i = 0; i < n; i++)
        {
            if (BoundState[i] == 0 || BoundState[i] == 2) continue;
            double gradient = ridge * X[i];
            for (int r = 0; r < 6; r++) gradient += B[r, i] * Residual[r];
            double violation = BoundState[i] < 0 ? -gradient : gradient;
            if (violation > worst) { worst = violation; release = i; }
        }
        if (release >= 0) { BoundState[release] = 0; continue; }
        LastSolverConverged = true;
        break;
    }
    if (LastSolverPasses == 0) { why = "no instruction budget for a solver pass"; return false; }
    for (int i = 0; i < n; i++)
    {
        double velocity = X[i] * Scale[i];
        if (!Finite(velocity)) { why = "non-finite group velocity"; return false; }
        LastQdot[i] = velocity;
        if (BoundState[i] != 0) LastActiveBounds++;
        achievedLinear += Topology.Groups[i].Lin * velocity;
        achievedAngular += Topology.Groups[i].Ang * velocity;
    }
    LastUnconstrainedLinear = achievedLinear; LastUnconstrainedAngular = achievedAngular;
    if (LastActiveBounds > 0 && LastSolverConverged && !LastSolverBudgetLimited)
    {
        if (Runtime.CurrentInstructionCount > Runtime.MaxInstructionCount * 0.68) LastSolverBudgetLimited = true;
        else if (!UnconstrainedTask(n, ridge, out LastUnconstrainedLinear, out LastUnconstrainedAngular))
        { why = "non-finite unconstrained task reference"; return false; }
    }
    return Finite(achievedLinear.LengthSquared()) && Finite(achievedAngular.LengthSquared());
}

static double Component(Vector3D v, int axis)
{
    return axis == 0 ? v.X : axis == 1 ? v.Y : v.Z;
}

void BuildTaskMatrix(int n, double ridge, bool bounded)
{
    for (int r = 0; r < 6; r++)
    {
        Rhs[r] = Task[r];
        if (bounded) for (int i = 0; i < n; i++) if (BoundState[i] != 0) Rhs[r] -= B[r, i] * X[i];
        for (int c = 0; c <= r; c++)
        {
            double total = r == c ? ridge : 0;
            for (int i = 0; i < n; i++) if (!bounded || BoundState[i] == 0) total += B[r, i] * B[c, i];
            Gram[r, c] = total;
        }
    }
}

bool UnconstrainedTask(int n, double ridge, out Vector3D linear, out Vector3D angular)
{
    linear = angular = Vector3D.Zero;
    BuildTaskMatrix(n, ridge, false);
    if (!Cholesky(Gram, Rhs, Chol, Temp, Solution)) return false;
    for (int i = 0; i < n; i++)
    {
        double rate = 0;
        for (int r = 0; r < 6; r++) rate += B[r, i] * Solution[r];
        rate *= Scale[i];
        linear += Topology.Groups[i].Lin * rate; angular += Topology.Groups[i].Ang * rate;
    }
    return Finite(linear.LengthSquared()) && Finite(angular.LengthSquared());
}

bool Commit()
{
    if (Runtime.CurrentInstructionCount > Runtime.MaxInstructionCount * 0.88)
    {
        Fault("Instruction budget exhausted before output commit.");
        return false;
    }
    for (int i = 0; i < Topology.Groups.Count; i++) Topology.Set(Topology.Groups[i], LastQdot[i]);
    // Every member of a proven parallel group receives its signed group command.
    return true;
}

void UpdateIntegral(ref Vector3D integral, Vector3D error, double deadband,
    Vector3D clipResidual, Vector3D allocationResidual, double ki,
    double cap, double capture, double dt, bool permit)
{
    Vector3D outside = Outside(error, deadband);
    if (ki <= 0 || !permit || error.Length() > capture)
    {
        if (IntegralUnwindSeconds <= 0) integral = Vector3D.Zero;
        else integral /= 1 + dt / IntegralUnwindSeconds;
        return;
    }
    // A held pose still needs the learned load bias. Reaching the deadband is
    // success, not a reason to leak gravity compensation away.
    integral = ClampLength(integral, cap);
    if (outside.LengthSquared() < 1e-20) return;
    Vector3D delta = outside * (ki * dt);
    // Treat output/limiter clipping and actuator allocation as independent
    // constraints: their residuals may oppose and numerically cancel.
    double scale = Math.Min(IntegralGrowthScale(delta, integral, clipResidual),
        IntegralGrowthScale(delta, integral, allocationResidual));
    integral = ClampLength(integral + delta * scale, cap);
}

static double IntegralGrowthScale(Vector3D delta, Vector3D integral, Vector3D residual)
{
    double growth = Vector3D.Dot(delta, residual);
    if (growth <= 1e-12) return 1;
    // Scale the original increment jointly for both constraints. Sequential
    // vector projections can rotate it back into a previously blocked direction.
    return Clamp(-Vector3D.Dot(integral, residual) / growth, 0, 1);
}

static Vector3D BlendManualCorrection(Vector3D pilotVelocity, Vector3D correction,
    double maximumOppositionFraction, out bool limited)
{
    limited = false;
    double speed = pilotVelocity.Length();
    if (speed <= 1e-12) return correction;
    Vector3D direction = pilotVelocity / speed;
    double component = Vector3D.Dot(correction, direction);
    double minimum = -speed * maximumOppositionFraction;
    if (component >= minimum) return correction;
    limited = true;
    return correction + direction * (minimum - component);
}

double PilotAdvance(Vector3D pilot, Vector3D achievedMinusFeedback)
{
    double d = pilot.LengthSquared();
    if (d < 1e-12) return 1;
    return Clamp(Vector3D.Dot(achievedMinusFeedback, pilot) / d, 0, 1);
}

static double TargetLeadAdvance(Vector3D error, Vector3D step, double limit)
{
    double a = step.LengthSquared();
    if (a < 1e-20) return 1;
    double e = error.LengthSquared(), radius = Math.Max(limit * limit, e);
    if ((error + step).LengthSquared() <= radius) return 1;
    double b = Vector3D.Dot(error, step);
    double root = Math.Sqrt(Math.Max(0, b * b + a * (radius - e)));
    // The alternate root avoids subtracting near-equal positive values.
    return Clamp(b >= 0 ? (root + b > 0 ? (radius - e) / (root + b) : 0) : (root - b) / a, 0, 1);
}

static Vector3D Outside(Vector3D error, double deadband)
{
    double n = error.Length();
    return n > deadband && n > 1e-12 ? error * ((n - deadband) / n) : Vector3D.Zero;
}

void ReadPilot(Vector3D headForward, Vector3D headUp, out Vector3D linear, out Vector3D angular)
{
    linear = angular = Vector3D.Zero;
    Vector3D input; Vector2 mouse; double roll;
    IMyShipController c = PilotInputs(out input, out mouse, out roll);
    if (c == null) return;
    Vector3D forward, left, up;
    if (MovementMode == PilotMode.HEAD)
    {
        forward = headForward; up = headUp;
        if (!Orthonormalise(ref forward, ref up)) throw new Exception("Invalid Head control frame.");
        left = Unit(Vector3D.Cross(up, forward));
    }
    else ControllerAxes(c, out forward, out left, out up);
    Vector3D moveForward = forward, moveUp = up;
    if (MovementMode == PilotMode.VRT) { moveForward = up; moveUp = -forward; }
    Vector3D local = new Vector3D(-input.Z, -input.X, input.Y);
    if (local.LengthSquared() > 1) local = Unit(local);
    linear = (moveForward * local.X + left * local.Y + moveUp * local.Z) * MoveMps;
    // Preserve the established Q/E convention: positive raw roll rotates +Head Forward.
    angular = headForward * roll;
    if (ReadMouse) angular += (left * mouse.X - up * mouse.Y) * MouseSensitivity;
    angular = ClampLength(angular, 1) * (TurnDeg * Math.PI / 180);
}

IMyShipController PilotInputs(out Vector3D move, out Vector2 mouse, out double roll)
{
    move = Vector3D.Zero; mouse = Vector2.Zero; roll = 0;
    var c = FindInputController(); if (c == null) return null;
    if (ReadKeyboard) move = c.MoveIndicator;
    if (ReadMouse) mouse = c.RotationIndicator;
    if (ReadRoll) roll = c.RollIndicator;
    if (!Finite(move.LengthSquared()) || !Finite(mouse.X) || !Finite(mouse.Y) || !Finite(roll))
        throw new Exception("Non-finite pilot input.");
    roll = Clamp(roll, -1, 1); return c;
}

bool HasPilotInput()
{
    Vector3D move; Vector2 mouse; double roll;
    if (PilotInputs(out move, out mouse, out roll) == null) return false;
    // Detect the operator's enabled raw demand, even at Speed=0 or when
    // simultaneous roll/mouse vectors cancel in the selected view frame.
    return move.LengthSquared() > 1e-8 || mouse.LengthSquared() > 1e-8 || Math.Abs(roll) > 1e-4;
}

bool InputScope(IMyShipController c)
{
    return c != null && !c.Closed && c.CubeGrid != null && c.IsSameConstructAs(Me);
}

IMyShipController FindInputController(bool force = false)
{
    if (!force && InputScope(InputController) && InputController.IsUnderControl)
        return InputController;
    if (!force && ++InputScan < 10) return null;
    InputScan = 0;
    InputScratch.Clear();
    GridTerminalSystem.GetBlocksOfType<IMyShipController>(InputScratch, b => InputScope(b) && b.IsUnderControl);
    InputController = null;
    foreach (var b in InputScratch) if (InputController == null || b.EntityId < InputController.EntityId) InputController = b;
    return InputController;
}

void ControllerAxes(IMyShipController c, out Vector3D forward, out Vector3D left, out Vector3D up)
{
    if (!InputScope(c) || !c.IsUnderControl) throw new Exception("HRZ/VRT requires an active cockpit.");
    MatrixD frame = Topology.Base.WorldMatrix;
    forward = InFrame(c.WorldMatrix.Forward, frame); up = InFrame(c.WorldMatrix.Up, frame);
    if (!Orthonormalise(ref forward, ref up)) throw new Exception("Invalid active cockpit orientation axes.");
    left = Unit(Vector3D.Cross(up, forward));
}

void SetMovementMode(string text)
{
    PilotMode mode;
    if (!AutoSettings.ReadMode(text, out mode)) throw new Exception("MovementMode expects HEAD, HRZ or VRT.");
    MovementMode = mode; InputController = null; InputScan = 10;
    // A view change changes input axes, never the held position/orientation.
    Say("Mode " + mode + "; target held. Reload restores default.");
}

void HoldCurrent()
{
    if (!Enabled) { EnableControl(); return; }
    Vector3D p, f, u;
    if (!ReadPose(out p, out f, out u)) { Fault("Hold failed: markers unavailable."); return; }
    CapturePose(p, f, u);
    GoingHome = false;
    ControlElapsed = 0;
    StatusElapsed = StatusRefreshSeconds;
    DisplayPinned = false;
    Topology.ZeroOwned();
    if (LastQdot != null) Array.Clear(LastQdot, 0, LastQdot.Length);
    if (BoundState != null) Array.Clear(BoundState, 0, BoundState.Length);
    Status = "ON: holding recaptured Head pose";
    Say("Hold: pose recaptured; ON.");
}

void CapturePose(Vector3D p, Vector3D f, Vector3D u)
{
    SetPoseTarget(p, f, u, 3);
    PreviousP = p; PreviousF = f; PreviousU = u;
    ObservationValid = true; AimingTarget = false;
    FilteredV = FilteredW = PositionI = OrientationI = Vector3D.Zero;
}

bool SetPoseTarget(Vector3D p, Vector3D f, Vector3D u, int parts)
{
    if ((parts & 1) != 0 && !Finite(p.LengthSquared()) || (parts & 2) != 0 && !Orthonormalise(ref f, ref u)) return false;
    if ((parts & 1) != 0) TargetP = p;
    if ((parts & 2) != 0) { TargetF = f; TargetU = u; }
    TargetValid = true;
    return true;
}

void PoseCommand(string[] w, int parts)
{
    if (!Enabled) throw new Exception("Run On before pose commands.");
    string why;
    if (SwapBatchLock || Tools != null && !Tools.AllowManual(out why)) throw new Exception("Pose command refused: tool lock.");
    int n = parts == 3 ? 9 : parts == 1 ? 3 : 6;
    if (w.Length != n + 1) throw new Exception("Pose command has wrong arity.");
    var v = new double[n];
    for (int i = 0; i < n; i++) v[i] = Parse(w[i + 1]);
    Vector3D p = parts == 2 ? TargetP : new Vector3D(v[0], v[1], v[2]);
    int j = parts == 3 ? 3 : 0;
    Vector3D f = parts == 1 ? TargetF : new Vector3D(v[j], v[j + 1], v[j + 2]);
    Vector3D u = parts == 1 ? TargetU : new Vector3D(v[j + 3], v[j + 4], v[j + 5]);
    if ((parts & 1) != 0 && !Finite(p.LengthSquared()) || (parts & 2) != 0 && !Orthonormalise(ref f, ref u)) throw new Exception("Invalid pose target.");
    if (GoingHome) HoldCurrent();
    if (!Enabled) return;
    SetPoseTarget(p, f, u, parts);
    if ((parts & 2) != 0) AimingTarget = true;
}

void Face(string direction)
{
    if (!Enabled) throw new Exception("Run On before Face.");
    Vector3D forward = new Vector3D(1, 0, 0), left = new Vector3D(0, 1, 0), up = new Vector3D(0, 0, 1);
    if (MovementMode != PilotMode.HEAD) ControllerAxes(FindInputController(true), out forward, out left, out up);
    Vector3D targetForward, targetUp = up;
    switch (direction.ToLowerInvariant())
    {
        case "forward": targetForward = forward; break;
        case "back": targetForward = -forward; break;
        case "left": targetForward = left; break;
        case "right": targetForward = -left; break;
        case "down": targetForward = -up; targetUp = forward; break;
        case "up": targetForward = up; targetUp = -forward; break;
        default: throw new Exception("Face expects Forward, Back, Left, Right, Up or Down.");
    }
    if (GoingHome) HoldCurrent();
    if (!Enabled) return;
    SetPoseTarget(TargetP, targetForward, targetUp, 2);
    AimingTarget = true;
}

void Step(string axis, double degrees)
{
    if (!Enabled) throw new Exception("Run On before a rotation step.");
    string kind = axis.ToLowerInvariant();
    if (kind != "yaw" && kind != "pitch" && kind != "roll") throw new Exception("Step expects Yaw, Pitch or Roll.");
    Vector3D viewForward = Vector3D.Zero, viewLeft = Vector3D.Zero, viewUp = Vector3D.Zero;
    if (MovementMode != PilotMode.HEAD && kind != "roll")
        ControllerAxes(FindInputController(true), out viewForward, out viewLeft, out viewUp);
    if (GoingHome) HoldCurrent();
    if (!Enabled) return;
    Vector3D a;
    switch (kind)
    {
        case "yaw": a = MovementMode == PilotMode.HEAD ? new Vector3D(0, 0, 1) : viewUp; break;
        case "pitch": a = MovementMode == PilotMode.HEAD ? -Unit(Vector3D.Cross(TargetU, TargetF)) : -viewLeft; break;
        default: a = TargetF; break;
    }
    Vector3D step = Unit(a) * (degrees * Math.PI / 180);
    SetPoseTarget(TargetP, Rotate(TargetF, step), Rotate(TargetU, step), 2);
    AimingTarget = true;
}

void RunCommands(string argument)
{
    SwapBatchLock = Tools != null && Tools.Automatic;
    string[] commands = argument.Replace("\r", "").Split(new char[] { ';', '\n' }, StringSplitOptions.RemoveEmptyEntries);
    try { foreach (string raw in commands) RunCommand(raw.Trim()); }
    finally { SwapBatchLock = false; }
}

void RunCommand(string command)
{
    if (command.StartsWith("MiningArm ", StringComparison.OrdinalIgnoreCase)) command = command.Substring(10).Trim();
    string[] w = command.Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
    if (w.Length == 0) return;
    string c = w[0].ToLowerInvariant();
    ToolGeometry.Owner = this;
    if (Tools != null && Tools.Handle(command)) { SwapBatchLock = true; return; }
    if (Tools != null && Tools.Busy && c != "on" && c != "toggle" && c != "onoff" && c != "-oo" && c != "off" && c != "stop" && c != "info" && c != "status" && c != "version")
        throw new Exception("Tool busy; SwapCancel/Stop first.");
    if (c.Length == 1 && c[0] >= '1' && c[0] <= '9') { NumericAction(c[0] - '0'); return; }
    switch (c)
    {
        case "version": Say(Banner()); break;
        case "on": StartControl(); break;
        case "off": case "stop": if (Tools != null && Tools.Busy) Tools.Cancel(c); StopMotion(c == "stop" ? "Stop command." : "Off command."); Say(Status); break;
        case "toggle": if (Enabled) { StopMotion("Toggle OFF."); Say(Status); } else StartControl(); break;
        case "onoff": case "-oo":
            c = w.Length < 2 ? "" : w[1].ToLowerInvariant();
            if (c != "on" && c != "off" && c != "toggle") throw new Exception("OnOff expects On, Off or Toggle.");
            RunCommand(w[1]);
            break;
        case "hold": HoldCurrent(); break;
        case "check": Check(); break;
        case "rescan": Discover(true); break;
        case "reload": Reload(); break;
        case "info": case "headinfo": case "help": PrintInfo(); break;
        case "pivotprobe":
            DisplayPinned = true;
            if (PivotProbe == null) PivotProbe = new PivotProbeTool(this, Topology);
            PivotProbe.Run(w);
            break;
        case "status": DisplayPinned = false; StatusElapsed = StatusRefreshSeconds; Say("Live status display resumed."); break;
        case "settings": PrintInfo(); break;
        case "joints": PrintJoints(); break;
        case "bindings": PrintBindings(); break;
        case "sethome": case "-sh": CaptureHome(); break;
        case "gohome": case "-gh": BeginHome(); break;
        case "homeinfo": PrintHome(); break;
        case "speed": case "-s": MoveMps = Number(w, 1, 0, 5, "Speed"); break;
        case "readkeyboard": case "-rk": ReadKeyboard = InputToggle(w, ReadKeyboard); break;
        case "readroll": ReadRoll = InputToggle(w, ReadRoll); break;
        case "readmouse": case "-rm": ReadMouse = InputToggle(w, ReadMouse); break;
        case "mode": case "movementmode": case "controlmode":
            if (w.Length != 2) throw new Exception("MovementMode expects HEAD, HRZ or VRT.");
            SetMovementMode(w[1]); break;
        case "orientationtolerance":
            double tolerance = Number(w, 1, 0, 30.0, OrientationToleranceKey);
            StopMotion("Stopped: OrientationTolerance changed; run On to recapture.");
            OrientationToleranceDeg = tolerance;
            Say(Status);
            break;
        case "turnspeed": TurnDeg = Number(w, 1, 0.01, 30, "TurnSpeed"); break;
        case "stepsize": case "stepangle": StepDeg = Number(w, 1, 0.01, 180, "StepSize"); break;
        case "jointspeed": JointRpm = Number(w, 1, 0.01, 5, "JointSpeed"); break;
        case "pistonspeed": PistonMps = Number(w, 1, 0.01, 2, "PistonSpeed"); break;
        case "homespeed": HomeJointRpm = Number(w, 1, 0.01, 5, "Home joint RPM"); HomePistonMps = Number(w, 2, 0.01, 2, "Home piston m/s"); break;
        case "face": if (w.Length < 2) throw new Exception("Face expects a direction."); Face(w[1]); break;
        case "forward": case "back": case "left": case "right": case "up": case "down": Face(c); break;
        case "step": if (w.Length < 2) throw new Exception("Step expects an axis."); Step(w[1], w.Length > 2 ? Parse(w[2]) : StepDeg); break;
        case "yaw": Step("yaw", Number(w, 1, -180, 180, "Yaw")); break;
        case "pitch": Step("pitch", Number(w, 1, -180, 180, "Pitch")); break;
        case "roll": Step("roll", Number(w, 1, -180, 180, "Roll")); break;
        case "yawleft": case "yawright": Step("yaw", c == "yawleft" ? StepDeg : -StepDeg); break;
        case "pitchup": case "pitchdown": Step("pitch", c == "pitchup" ? StepDeg : -StepDeg); break;
        case "rollleft": case "rollright": Step("roll", c == "rollright" ? StepDeg : -StepDeg); break;
        case "dumptrace": PrintInfo(); Say("See Info; Custom Data unchanged."); break;
        case "traceclear": Say("No trace buffer."); break;
        case "moveto": PoseCommand(w, 1); break;
        case "orientto": PoseCommand(w, 2); break;
        case "movetoposori": PoseCommand(w, 3); break;
        case "move": case "relmove": case "runfrompanel":
            throw new Exception(c + " is not supported by AutoArm's discovered pose control.");
        default: throw new Exception("Unknown command '" + w[0] + "'. Use Info.");
    }
}

static bool InputToggle(string[] words, bool current)
{
    string value = words.Length == 2 ? words[1].ToLowerInvariant() : "";
    if (value == "on") return true;
    if (value == "off") return false;
    if (value == "toggle") return !current;
    throw new Exception(words[0] + " expects On, Off or Toggle.");
}

void Reload()
{
    StopMotion("Reload command.");
    PivotProbe = null;
    SetupElapsed = 0;
    string updated, why;
    bool changed;
    ConfigValid = false;
    MyIni configIni;
    if (!AutoConfig.ReadCurrent(Me.CustomData, out configIni, out why)) throw new Exception(why);
    if (Topology.Groups.Count == 0 || (Tools != null && Tools.Configured) || configIni.Get(ToolsSection, EnabledKey).ToBoolean(false)) { Discover(true); return; }
    if (Topology.Groups.Count > 0 && !Topology.Validate(out why, true))
    {
        ConfigValid = false;
        Say("Reload refused: " + why + " Run Rescan after repairing topology.");
        return;
    }
    if (!AutoConfig.LoadOrGenerate(this, Topology, Me.CustomData, out updated, out changed, out why))
    {
        ConfigValid = false;
        Status = "Reload refused without changing Custom Data: " + why;
        Say(Status);
        return;
    }
    ConfigValid = true;
    Allocate(Topology.Groups.Count);
    LoadHome();
    Status = "Configuration reloaded; controller remains OFF.";
    Say(Status);
}

void Check()
{
    string why;
    if (!ConfigValid) { Say("CHECK FAILED: configuration is not valid. Run Reload or Rescan."); return; }
    if (!Topology.Validate(out why, true)) { Say("CHECK FAILED: " + why); return; }
    Vector3D p, f, u;
    if (!ReadPose(out p, out f, out u)) { Say("CHECK FAILED: markers unavailable."); return; }
    foreach (AutoGroup g in Topology.Groups)
    {
        double lo, hi;
        if (!Topology.Refresh(g, Topology.EndPosition, Topology.Base.WorldMatrix, out lo, out hi, out why))
        { Say("CHECK FAILED: " + why); return; }
    }
    Say("CHECK PASS | " + Topology.Groups.Count + " groups | " + Topology.Fingerprint + "\n" + Topology.Report);
}

void CaptureHome()
{
    StopMotion("SetHome command.");
    if (!ConfigValid) { Say("SetHome refused: fix Custom Data, then Reload."); return; }
    string why = "topology is not ready";
    if (!Topology.Ready || !Topology.Validate(out why, true)) { Say("SetHome refused: " + why); return; }
    Home.Clear();
    foreach (AutoGroup g in Topology.Groups) foreach (AutoMember m in g.M) Home[m.B.EntityId] = MemberPosition(m);
    HomeFingerprint = Topology.Fingerprint;
    WriteHome();
    Say("Home saved for " + Home.Count + " actuators. Controller remains OFF.");
}

void LoadHome()
{
    Home.Clear(); HomeFingerprint = "";
    if (string.IsNullOrWhiteSpace(Storage)) return;
    var ini = new MyIni();
    MyIniParseResult result;
    if (!ini.TryParse(Storage, out result)) return;
    string section = HomeSection;
    if (!ini.ContainsSection(section))
    {
        var sections = new List<string>(); ini.GetSections(sections);
        foreach (string candidate in sections)
            if (candidate.EndsWith(" Home", StringComparison.Ordinal) && ini.Get(candidate, "Fingerprint").ToString() == Topology.Fingerprint)
            {
                if (section != HomeSection) return;
                section = candidate;
            }
    }
    string fingerprint = ini.Get(section, "Fingerprint").ToString("");
    if (fingerprint != Topology.Fingerprint) return;
    foreach (AutoGroup g in Topology.Groups) foreach (AutoMember m in g.M)
    {
        string key = "E" + m.B.EntityId;
        double value = ini.Get(section, key).ToDouble(double.NaN);
        if (!Finite(value)) { Home.Clear(); return; }
        Home[m.B.EntityId] = value;
    }
    HomeFingerprint = fingerprint;
    if (section != HomeSection)
    {
        ini.DeleteSection(section); Storage = ini.ToString(); WriteHome();
    }
}

void WriteHome()
{
    var ini = AutoConfig.Saved(Storage);
    ini.DeleteSection(HomeSection);
    ini.Set(HomeSection, "Fingerprint", HomeFingerprint);

}

void BeginHome()
{
    string permission;
    if (SwapBatchLock || (Tools != null && !Tools.AllowManual(out permission))) { Say("GoHome refused: tool lock."); return; }
    StopMotion("");
    if (!ConfigValid) { Say("GoHome refused: fix Custom Data, then Reload."); return; }
    if (!Topology.Ready || HomeFingerprint != Topology.Fingerprint)
    { Say("GoHome refused: use SetHome for the current topology."); return; }

    if (!Topology.Validate(out why, true)) { Say("GoHome refused: " + why); return; }
    foreach (AutoGroup g in Topology.Groups) foreach (AutoMember m in g.M)
    {

        { Say("GoHome refused: saved target is outside current limits at " + m.B.CustomName + "."); return; }
        m.Home = resolved;
    }

    PivotProbe = null;
    ControlElapsed = 0;
    Runtime.UpdateFrequency = UpdateFrequency.Update1;
    Status = "GoHome in progress; live pilot input cancels.";
    Say(Status);
}

void UpdateHome(double dt)
{
    if (HasPilotInput()) { Fault("GoHome cancelled by pilot input."); return; }
    string why;
    if (Tools != null && !Tools.AllowManual(out why)) { Fault(why); return; }
    if (!Topology.Validate(out why)) { Fault("GoHome topology fault: " + why); return; }
    bool complete = true;
    for (int i = 0; i < Topology.Groups.Count; i++)
    {
        AutoGroup g = Topology.Groups[i];
        double error = 0, branch = 0;
        bool haveBranch = false;
        double agreement = g.Rotary ? 0.5 * Math.PI / 180 : 0.005;
        // A limited member fixes the legal native-angle branch. Unlimited peers
        // may then select the equivalent circular error nearest that branch.
        if (g.Rotary) foreach (AutoMember m in g.M)
        {
            if (Unlimited((IMyMotorStator)m.B)) continue;
            double e = m.Sign * MemberHomeError(m, true);
            if (!haveBranch) { branch = e; haveBranch = true; }
            else if (Math.Abs(e - branch) > agreement) { Fault("GoHome limited-member routes disagree at " + g.Label); return; }
        }
        foreach (AutoMember m in g.M)
        {
            double e = m.Sign * MemberHomeError(m, g.Rotary);
            bool unlimited = g.Rotary && Unlimited((IMyMotorStator)m.B);
            if (!haveBranch) { branch = e; haveBranch = true; }
            else if (unlimited) e = branch + Wrap(e - branch);
            if (Math.Abs(e - branch) > agreement) { Fault("GoHome member errors disagree at " + g.Label); return; }
            error += e;
        }
        error /= g.M.Count;
        double tolerance = g.Rotary ? 0.25 * Math.PI / 180 : 0.005;
        if (Math.Abs(error) > tolerance) complete = false;
        double lo, hi;
        if (!Topology.Refresh(g, Topology.EndPosition, Topology.Base.WorldMatrix, out lo, out hi, out why))
        { Fault("GoHome: " + why); return; }
        double cap = g.Rotary ? HomeJointRpm * Math.PI / 30 : HomePistonMps;
        double lower = Math.Max(lo, -cap), upper = Math.Min(hi, cap);
        if (lower > upper) { Fault("GoHome has no common velocity interval at " + g.Label); return; }
        double velocity = Math.Abs(error) <= tolerance ? 0 : Clamp(error * 0.5, lower, upper);
        double accel = g.Rotary ? JointAccelerationRpmPerSec * Math.PI / 30 : PistonAccelerationMps2;
        double centre = Clamp(LastQdot[i], lower, upper);
        velocity = Clamp(velocity, Math.Max(lower, centre - accel * dt), Math.Min(upper, centre + accel * dt));
        LastQdot[i] = velocity;
    }
    if (complete)
    {
        StopMotion("GoHome complete.");
        Say(Status);
    }
    else Commit();
}

static bool Unlimited(IMyMotorStator r)
{
    return r.LowerLimitRad < -1000000 && r.UpperLimitRad > 1000000;
}

bool ResolveHome(AutoMember m, double requested, out double resolved)
{
    resolved = requested;
    var p = m.B as IMyPistonBase;
    if (p != null)
    {
        if (!Finite(resolved) || resolved < p.MinLimit - 0.005 || resolved > p.MaxLimit + 0.005) return false;
        resolved = Clamp(resolved, p.MinLimit, p.MaxLimit);
        return true;
    }
    var r = (IMyMotorStator)m.B;
    if (!Finite(resolved)) return false;
    double lo = r.LowerLimitRad, hi = r.UpperLimitRad, turn = 2 * Math.PI;
    double k = Math.Round((r.Angle - requested) / turn);
    double kMin = lo < -1000000 ? double.NegativeInfinity : Math.Ceiling((lo - requested - 0.25 * Math.PI / 180) / turn);
    double kMax = hi > 1000000 ? double.PositiveInfinity : Math.Floor((hi - requested + 0.25 * Math.PI / 180) / turn);
    if (kMin > kMax) return false;
    resolved = requested + Clamp(k, kMin, kMax) * turn;
    if (!Finite(resolved) || resolved < lo - 0.25 * Math.PI / 180 || resolved > hi + 0.25 * Math.PI / 180) return false;
    resolved = Clamp(resolved, lo, hi);
    return true;
}

double MemberHomeError(AutoMember m, bool rotary)
{
    double error = m.Home - MemberPosition(m);
    if (!rotary) return error;
    return Unlimited((IMyMotorStator)m.B) ? Wrap(error) : error;
}

double MemberPosition(AutoMember m)
{
    var rotor = m.B as IMyMotorStator;
    if (rotor != null) return rotor.Angle;
    var piston = m.B as IMyPistonBase;
    if (piston != null) return piston.CurrentPosition;
    throw new Exception("Unsupported owned actuator type.");
}

double Number(string[] words, int index, double min, double max, string label)
{
    if (words.Length <= index) throw new Exception(label + " needs a value.");
    double value = Parse(words[index]);
    if (value < min || value > max) throw new Exception(label + " must be " + min + ".." + max + ".");
    Say(label + " = " + value.ToString("0.###") + ".");
    return value;
}

double Parse(string text)
{
    double value;
    if (!ReadDouble(text, out value))
        throw new Exception("Expected a finite number with decimal point.");
    return value;
}
static bool ReadDouble(string text, out double value) => double.TryParse(text, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out value) && Finite(value);

static bool Cholesky(double[,] a, double[] b, double[,] l, double[] temp, double[] answer)
{
    for (int i = 0; i < 6; i++) for (int j = 0; j <= i; j++)
    {
        double value = a[i, j];
        for (int k = 0; k < j; k++) value -= l[i, k] * l[j, k];
        if (i == j)
        {
            if (!Finite(value) || value <= 1e-15) return false;
            l[i, j] = Math.Sqrt(value);
        }
        else l[i, j] = value / l[j, j];
    }
    for (int i = 0; i < 6; i++)
    {
        double value = b[i];
        for (int j = 0; j < i; j++) value -= l[i, j] * temp[j];
        temp[i] = value / l[i, i];
    }
    for (int i = 5; i >= 0; i--)
    {
        double value = temp[i];
        for (int j = i + 1; j < 6; j++) value -= l[j, i] * answer[j];
        answer[i] = value / l[i, i];
        if (!Finite(answer[i])) return false;
    }
    return true;
}

static Vector3D RotationError(Vector3D f, Vector3D u, Vector3D tf, Vector3D tu)
{
    if (!Orthonormalise(ref f, ref u) || !Orthonormalise(ref tf, ref tu)) return new Vector3D(double.NaN);
    Vector3D l = Vector3D.Cross(u, f), tl = Vector3D.Cross(tu, tf);
    double d00 = tf.X * f.X + tl.X * l.X + tu.X * u.X,
        d01 = tf.X * f.Y + tl.X * l.Y + tu.X * u.Y,
        d02 = tf.X * f.Z + tl.X * l.Z + tu.X * u.Z;
    double d10 = tf.Y * f.X + tl.Y * l.X + tu.Y * u.X,
        d11 = tf.Y * f.Y + tl.Y * l.Y + tu.Y * u.Y,
        d12 = tf.Y * f.Z + tl.Y * l.Z + tu.Y * u.Z;
    double d20 = tf.Z * f.X + tl.Z * l.X + tu.Z * u.X,
        d21 = tf.Z * f.Y + tl.Z * l.Y + tu.Z * u.Y,
        d22 = tf.Z * f.Z + tl.Z * l.Z + tu.Z * u.Z;
    var skew = new Vector3D(d21 - d12, d02 - d20, d10 - d01) * 0.5;
    double angle = Math.Acos(Clamp((d00 + d11 + d22 - 1) * 0.5, -1, 1));
    if (angle < 1e-7) return skew;
    if (Math.PI - angle > 1e-5) return skew * (angle / Math.Sin(angle));
    // Near pi, the skew is ill-conditioned. Extract an eigenaxis from the full
    // relative rotation, selecting the largest diagonal for numeric stability.
    double x, y, z;
    if (d00 >= d11 && d00 >= d22)
    {
        x = Math.Sqrt(Math.Max(0, (d00 + 1) * 0.5));
        y = (d01 + d10) / (4 * Math.Max(x, 1e-10));
        z = (d02 + d20) / (4 * Math.Max(x, 1e-10));
    }
    else if (d11 >= d22)
    {
        y = Math.Sqrt(Math.Max(0, (d11 + 1) * 0.5));
        x = (d01 + d10) / (4 * Math.Max(y, 1e-10));
        z = (d12 + d21) / (4 * Math.Max(y, 1e-10));
    }
    else
    {
        z = Math.Sqrt(Math.Max(0, (d22 + 1) * 0.5));
        x = (d02 + d20) / (4 * Math.Max(z, 1e-10));
        y = (d12 + d21) / (4 * Math.Max(z, 1e-10));
    }
    Vector3D axis = Unit(new Vector3D(x, y, z));
    if (skew.LengthSquared() > 1e-16 && Vector3D.Dot(axis, skew) < 0) axis = -axis;
    return Unit(axis) * angle;
}

static Vector3D Rotate(Vector3D value, Vector3D rotation)
{
    double angle = rotation.Length();
    if (angle < 1e-12) return value;
    Vector3D axis = rotation / angle;
    return value * Math.Cos(angle) + Vector3D.Cross(axis, value) * Math.Sin(angle) +
        axis * Vector3D.Dot(axis, value) * (1 - Math.Cos(angle));
}

static bool Orthonormalise(ref Vector3D f, ref Vector3D u)
{
    double fn = f.Length();
    if (!Finite(fn) || fn < 1e-10) return false;
    f /= fn;
    u -= f * Vector3D.Dot(u, f);
    double un = u.Length();
    if (!Finite(un) || un < 1e-10) return false;
    u /= un;
    return true;
}

static Vector3D Unit(Vector3D value)
{
    double n = value.Length();
    if (!Finite(n) || n < 1e-12) throw new Exception("Invalid zero/non-finite vector.");
    return value / n;
}

static Vector3D ClampLength(Vector3D value, double cap)
{
    double n = value.Length();
    if (!Finite(n)) return new Vector3D(double.NaN);
    return n > cap && n > 1e-12 ? value * (cap / n) : value;
}

static Vector3D InFrame(Vector3D value, MatrixD frame) => new Vector3D(Vector3D.Dot(value, frame.Forward), Vector3D.Dot(value, frame.Left), Vector3D.Dot(value, frame.Up));

static double Clamp(double value, double lo, double hi) => Math.Max(lo, Math.Min(hi, value));

static double Wrap(double angle) => angle - Math.Floor((angle + Math.PI) / (2 * Math.PI)) * (2 * Math.PI);

static bool Finite(double value) => !double.IsNaN(value) && !double.IsInfinity(value);

static bool Finite(Vector3D value) => Finite(value.X) && Finite(value.Y) && Finite(value.Z);


// Passive RAM-only pivot probe. The player supplies all motion. Configuration
// and Storage are untouched; existing topology-fault handling can stop drives.
class PivotProbeTool
{
    const double MinA=Math.PI/180,MaxA=15*Math.PI/180,OtherA=.1*Math.PI/180;
    const double PairA=.1*Math.PI/180,OtherP=.001,MinTravel=.01,AxisCos=.999993908;
    readonly Program P; readonly AutoTopology T; readonly List<Result> R=new List<Result>();
    Shot A; int G=-1; string Fingerprint="";
    class Shot { public Vector3D P,F,U; public double[][] Q; public Vector3D[] Axis,Base,Top; }
    public class Result
    {
        public int Group; public double Delta,Travel,Orientation;
        public readonly double[] Error=new double[3],Spread=new double[3]; public string Verdict;
    }
    public IList<Result> Trials => R.AsReadOnly();
    public PivotProbeTool(Program owner,AutoTopology topology){P=owner;T=topology;}

    public void Run(string[] w)
    {
        int n=w.Length>0&&w[0].Equals("pivotprobe",StringComparison.OrdinalIgnoreCase)?1:0;
        if(w.Length<=n){Help();return;} string c=w[n].ToLowerInvariant();
        if(c=="start"&&w.Length==n+2){int g;if(!int.TryParse(w[n+1],out g)){Say("Group must be an integer.");return;}Start(g-1);return;}
        if(c=="end"&&w.Length==n+1){End();return;} if(c=="report"&&w.Length==n+1){Report();return;}
        if(c=="reset"&&w.Length==n+1){Clear();R.Clear();Say("PivotProbe RAM cleared; no blocks changed.");return;} Help();
    }
    void Start(int g)
    {
        string why;if(!Ready(out why)){Say("PivotProbe refused: "+why);return;}
        if(A!=null){Say("A Start snapshot exists; run End or Reset.");return;}
        if(g<0||g>=T.Groups.Count){Say("Group must be 1.."+T.Groups.Count+".");return;}
        if(!T.Groups[g].Rotary){Say("Only rotary/hinge groups have a pivot.");return;}
        Shot s;if(!Capture(g,out s,out why)){Say("PivotProbe refused: "+why);return;}
        A=s;G=g;Fingerprint=T.Fingerprint;Say("Captured G"+(g+1)+". Manually jog ONLY this group 1..15 deg, zero all targets, settle, then run PivotProbe End. Parallel members must follow discovered signs.");
    }
    void End()
    {
        if(A==null){Say("Run PivotProbe Start first.");return;}string why;
        if(!Ready(out why)){Say("End refused: "+why);return;}
        if(Fingerprint!=T.Fingerprint){Clear();Say("Inconclusive: topology changed.");return;}
        Shot b;if(!Capture(G,out b,out why)){Clear();Say("Inconclusive: "+why);return;}
        Result r;if(!Analyse(A,b,G,out r,out why)){Clear();Say("Inconclusive: "+why);return;}
        if(R.Count==12)R.RemoveAt(0);R.Add(r);Clear();Say(Format(r));
    }
    bool Ready(out string why)
    {
        why="";if(P.Enabled||P.GoingHome){why="Controller must be Off.";return false;}
        if(!T.Ready){why="topology is not ready.";return false;}if(!T.Validate(out why,true))return false;
        foreach(AutoGroup g in T.Groups)foreach(AutoMember m in g.M){var p=m.B as IMyPistonBase;double v=p!=null?p.Velocity:((IMyMotorStator)m.B).TargetVelocityRad;if(!Finite(v)||Math.Abs(v)>1e-4){why="owned target not zero: "+m.B.CustomName;return false;}}
        return true;
    }
    bool Capture(int selected,out Shot s,out string why)
    {
        s=new Shot();why="";MatrixD f=T.Base.WorldMatrix;Vector3D o=T.Base.GetPosition();
        s.P=Point(T.EndPosition,o,f);s.F=InFrame(T.EndPose.Forward,f);s.U=InFrame(T.EndPose.Up,f);
        if(!Finite(s.P)||!Unit(ref s.F)||!Unit(ref s.U)){why="invalid Base/Head pose.";return false;}
        s.Q=new double[T.Groups.Count][];
        for(int i=0;i<T.Groups.Count;i++){s.Q[i]=new double[T.Groups[i].M.Count];for(int j=0;j<s.Q[i].Length;j++){s.Q[i][j]=T.Position(T.Groups[i].M[j]);if(!Finite(s.Q[i][j])){why="non-finite joint position.";return false;}}}
        AutoGroup g=T.Groups[selected];int count=g.M.Count;s.Axis=new Vector3D[count];s.Base=new Vector3D[count];s.Top=new Vector3D[count];
        for(int i=0;i<count;i++)
        {
            AutoMember m=g.M[i];var r=m.B as IMyMotorStator;
            if(r==null||r.RotorLock||m.B.Top==null){why="selected member unavailable/locked.";return false;}
            double q=s.Q[selected][i],margin=.25*Math.PI/180;
            if(!Finite(r.LowerLimitRad)||!Finite(r.UpperLimitRad)||r.LowerLimitRad>r.UpperLimitRad){why="selected limits are invalid.";return false;}
            if((r.LowerLimitRad>-1e6&&q<=r.LowerLimitRad+margin)||(r.UpperLimitRad<1e6&&q>=r.UpperLimitRad-margin)){why="selected member near limit: "+m.B.CustomName;return false;}
            int dir=m.B.CubeGrid.EntityId==m.U&&m.B.TopGrid.EntityId==m.V?1:m.B.CubeGrid.EntityId==m.V&&m.B.TopGrid.EntityId==m.U?-1:0;
            if(dir==0){why="selected endpoints changed.";return false;}
            s.Axis[i]=InFrame(m.B.WorldMatrix.Up*(-dir*m.Sign),f);s.Base[i]=Point(m.B.GetPosition(),o,f);s.Top[i]=Point(m.B.Top.GetPosition(),o,f);
            if(!Unit(ref s.Axis[i])||!Finite(s.Base[i])||!Finite(s.Top[i])){why="invalid axis/pivot candidate.";return false;}
        }return true;
    }
    bool Analyse(Shot a,Shot b,int group,out Result result,out string why)
    {
        result=null;why="";
        for(int i=0;i<T.Groups.Count;i++)if(i!=group)for(int j=0;j<T.Groups[i].M.Count;j++){double d=b.Q[i][j]-a.Q[i][j];if(T.Groups[i].Rotary)d=Wrap(d);if(Math.Abs(d)>(T.Groups[i].Rotary?OtherA:OtherP)){why="another group moved: G"+(i+1);return false;}}
        AutoGroup g=T.Groups[group];double first=0,total=0,pair=0;
        for(int i=0;i<g.M.Count;i++)
        {
            double d=Wrap(b.Q[group][i]-a.Q[group][i])*g.M[i].Sign;if(i==0)first=d;else{d=first+Wrap(d-first);pair=Math.Max(pair,Math.Abs(d-first));}total+=d;
            if(Vector3D.Dot(a.Axis[0],a.Axis[i])<AxisCos||Vector3D.Dot(a.Axis[i],b.Axis[i])<AxisCos){why="Axis drift >0.2 deg.";return false;}
        }
        double delta=total/g.M.Count;if(pair>PairA){why="Pair delta >0.1 deg.";return false;}
        if(Math.Abs(delta)<MinA||Math.Abs(delta)>MaxA){why="motion must be 1..15 deg; measured "+Deg(delta).ToString(ThreeDecimals);return false;}
        double travel=(b.P-a.P).Length();if(travel<MinTravel){why="Head travel below 0.01 m.";return false;}
        double ori=Math.Max(Angle(Rot(a.F,a.Axis[0],delta),b.F),Angle(Rot(a.U,a.Axis[0],delta),b.U));
        if(!Finite(ori)||ori>1){why="orientation residual "+ori.ToString(ThreeDecimals)+" deg: slip/other motion.";return false;}
        result=new Result{Group=group,Delta=delta,Travel=travel,Orientation=ori};
        for(int k=0;k<3;k++)
        {
            Vector3D mean=Vector3D.Zero;var predicted=new Vector3D[g.M.Count];
            for(int i=0;i<g.M.Count;i++){Vector3D c=Candidate(a,i,k);predicted[i]=c+Rot(a.P-c,a.Axis[i],delta);mean+=predicted[i];}mean/=g.M.Count;
            for(int i=0;i<predicted.Length;i++)result.Spread[k]=Math.Max(result.Spread[k],(predicted[i]-mean).Length());
            result.Error[k]=(b.P-mean).Length();if(!Finite(result.Error[k])||!Finite(result.Spread[k])){why="non-finite residual.";return false;}
        }
        int best=0;for(int k=1;k<3;k++)if(result.Error[k]<result.Error[best])best=k;double next=double.PositiveInfinity;
        for(int k=0;k<3;k++)if(k!=best)next=Math.Min(next,result.Error[k]);
        result.Verdict=result.Spread[best]>.01?"inconclusive: pair mismatch":next-result.Error[best]<.001?"axis-equivalent/inconclusive":Name(best)+" smallest (not a calibration)";return true;
    }
    void Report()
    {
        if(R.Count==0){Say("No accepted trials. Capture positive and negative manual motions.");return;}
        var s=new System.Text.StringBuilder("PIVOT PROBE RAM REPORT - no pivot changed\n");for(int i=0;i<R.Count;i++)s.Append('#').Append(i+1).Append(' ').Append(Format(R[i])).Append('\n');
        s.Append("Collect both directions; residuals are evidence, not proof. Flex/load/slip can dominate.");Say(s.ToString());
    }
    string Format(Result r){return "G"+(r.Group+1)+" d="+Deg(r.Delta).ToString(ThreeDecimals)+"deg travel="+r.Travel.ToString("0.0000")+"m ori="+r.Orientation.ToString(ThreeDecimals)+"deg\n Base e/spread="+Row(r,0)+" Top="+Row(r,1)+" Mid="+Row(r,2)+" RESULT="+r.Verdict;}
    string Row(Result r,int k){return r.Error[k].ToString("0.0000")+"/"+r.Spread[k].ToString("0.0000")+"m";}
    Vector3D Candidate(Shot s,int i,int k){return k==0?s.Base[i]:k==1?s.Top[i]:(s.Base[i]+s.Top[i])*.5;}
    string Name(int k){return k==0?"Base center":k==1?"Top center":"Midpoint";}
    void Clear(){A=null;G=-1;Fingerprint="";}void Help(){Say("PivotProbe Start <group> | End | Report | Reset");}void Say(string s){P.Say(s);}
    static double Deg(double a) => a*180/Math.PI;

    static bool Unit(ref Vector3D v){double n=v.Length();if(!Finite(v)||!Finite(n)||n<1e-10)return false;v/=n;return true;}
    static Vector3D Point(Vector3D p,Vector3D o,MatrixD f) => InFrame(p-o,f);
    static Vector3D Rot(Vector3D v,Vector3D a,double x) => v*Math.Cos(x)+Vector3D.Cross(a,v)*Math.Sin(x)+a*Vector3D.Dot(a,v)*(1-Math.Cos(x));
    static double Angle(Vector3D a,Vector3D b){double n=a.Length()*b.Length();return n>1e-12&&Finite(n)?Math.Acos(Math.Max(-1,Math.Min(1,Vector3D.Dot(a,b)/n)))*180/Math.PI:double.NaN;}
}

        return new ToolTopInfo { Marker = marker, Live = top, EntityId = top.EntityId, Subtype = top.BlockDefinition.SubtypeName, Local = pose * MatrixD.Invert(marker.WorldMatrix), Inventory = inventory };
    }
}
class ToolRegionScan
{
    Program P; IMyCubeBlock Marker; IMyCubeGrid Grid;
    long GridId; int Limit; string Axis, Fault;
    Queue<Vector3I> Pending = new Queue<Vector3I>();
    public HashSet<Vector3I> Cells = new HashSet<Vector3I>();
    public HashSet<Vector3I> Hidden = new HashSet<Vector3I>();
    public HashSet<long> Blocks = new HashSet<long>();
    public List<IMyShipMergeBlock> Merges = new List<IMyShipMergeBlock>();
    public static Vector3I[] Dirs = { Vector3I.UnitX, -Vector3I.UnitX, Vector3I.UnitY, -Vector3I.UnitY, Vector3I.UnitZ, -Vector3I.UnitZ };
    public ToolRegionScan(Program p, IMyCubeBlock marker, int cellLimit, string mergeAxis)
    {
        P = p; Marker = marker; Limit = Math.Min(4096, Math.Max(1, cellLimit)); Axis = mergeAxis;
        if (marker == null) { Fault = "Missing Head/Mount."; return; }
        Grid = marker.CubeGrid; GridId = Grid.EntityId;
        if (!Grid.CubeExists(marker.Position)) { Fault = "Head cell missing."; return; }
        Cells.Add(marker.Position); Pending.Enqueue(marker.Position);
    }
    public bool Step(out bool done, out string why)
    {
        done = false; why = Fault;
        if (Fault != null) return false;
        if (Marker.CubeGrid.EntityId != GridId) return ToolGeometry.Fail(out why, Fault = "Head grid changed; retry.");
        for (int count = 0; Pending.Count > 0 && count < 128; count++)
        {
            if (P.Runtime.MaxInstructionCount > 0 && P.Runtime.CurrentInstructionCount >= P.Runtime.MaxInstructionCount * .5) return true;
            Vector3I cell = Pending.Dequeue(); var slim = Grid.GetCubeBlock(cell);
            if (slim == null && !Grid.CubeExists(cell)) return ToolGeometry.Fail(out why, Fault = "Head cells changed; retry.");
            if (slim == null) Hidden.Add(cell);
            var fat = slim == null ? null : slim.FatBlock;
            if (fat != null && Blocks.Add(fat.EntityId))
            {
                var merge = fat as IMyShipMergeBlock;
                if (merge != null) { Vector3I d; if (!ToolGeometry.MergeDirection(merge, Axis, out d, out why)) { Fault = why; return false; } Merges.Add(merge); }
            }
            foreach (var d in Dirs)
            {
                Vector3I next = cell + d; if (Cells.Contains(next)) continue;
                var neighbor = Grid.GetCubeBlock(next); if (neighbor == null && !Grid.CubeExists(next)) continue;
                var a = fat as IMyShipMergeBlock; var b = neighbor == null ? null : neighbor.FatBlock as IMyShipMergeBlock;
                if (a != null && b != null && a.EntityId != b.EntityId)
                {
                    Vector3I ad, bd;
                    if (!ToolGeometry.MergeDirection(a, Axis, out ad, out why) || !ToolGeometry.MergeDirection(b, Axis, out bd, out why)) { Fault = why; return false; }
                    if (ad == d && bd == -d && ToolGeometry.OnFace(a, cell, ad) && ToolGeometry.OnFace(b, next, bd)) continue;
                }
                if (Cells.Count >= Limit) return ToolGeometry.Fail(out why, Fault = "Head cell limit (" + Limit + ").");
                Cells.Add(next); Pending.Enqueue(next);
            }
        }
        done = Pending.Count == 0; return true;
    }
}
static class ToolGeometry
{
    public static Program Owner;
    static bool Budget() => Owner == null || Owner.Runtime.MaxInstructionCount <= 0 || Owner.Runtime.CurrentInstructionCount < Owner.Runtime.MaxInstructionCount * .6;
    public static bool Fail(out string why, string message) { why = message; return false; }
    static string[] Axes = { "right", "left", "up", "down", "forward", "backward" };
    static Vector3D[] Normals = { Vector3D.Right, Vector3D.Left, Vector3D.Up, Vector3D.Down, Vector3D.Forward, Vector3D.Backward };
    static string[] Bases = { "LargeStator", "SmallStator", "LargeAdvancedStator", "SmallAdvancedStator", "SmallAdvancedStatorSmall" };
    static string[] Heads = { "LargeRotor", "SmallRotor", "LargeAdvancedRotor", "SmallAdvancedRotor", "SmallAdvancedRotorSmall" };
    public static bool Rotor(IMyCubeBlock b) => Array.IndexOf(b is IMyMotorRotor ? Heads : Bases, b.BlockDefinition.SubtypeName) >= 0;
    public static ToolTopInfo FindTip(IMyCubeBlock marker, HashSet<Vector3I> hidden, List<IMyTerminalBlock> inventory)
    {
        var grid = marker.CubeGrid;
        if (!Budget()) throw new Exception(GeometryBudgetMessage);
        foreach (var b in inventory)
        {
            if (!Budget()) throw new Exception(GeometryBudgetMessage);
            var m = b as IMyMechanicalConnectionBlock;
            var top = m == null ? null : m.Top;
            if (top != null && top.CubeGrid.EntityId == grid.EntityId)
                hidden.RemoveWhere(c => c.X >= top.Min.X && c.X <= top.Max.X && c.Y >= top.Min.Y && c.Y <= top.Max.Y && c.Z >= top.Min.Z && c.Z <= top.Max.Z);
        }
        if (hidden.Count == 0) return null;
        if (hidden.Count != 1 && hidden.Count != 18) throw new Exception("Need one loose rotor part on the end grid.");
        Vector3I min = new Vector3I(int.MaxValue), max = new Vector3I(int.MinValue);
        foreach (var c in hidden) { min = Vector3I.Min(min, c); max = Vector3I.Max(max, c); }
        var size = max - min + Vector3I.One;
        if (size.X * size.Y * size.Z != hidden.Count || hidden.Count == 18 && size.X + size.Y + size.Z != 8) throw new Exception("Ambiguous loose part shape.");
        Vector3I up = Vector3I.Zero; int faces = 0;
        foreach (var d in ToolRegionScan.Dirs)
        {
            if (hidden.Count == 18 && (d.X != 0 ? size.X : d.Y != 0 ? size.Y : size.Z) != 2) continue;
            bool joined = false;
            foreach (var c in hidden) if (!hidden.Contains(c + d) && grid.CubeExists(c + d)) joined = true;
            if (joined) { up = d; faces++; }
        }
        if (faces != 1) throw new Exception("Loose rotor mounting face is ambiguous.");
        var pivot = min + (size - Vector3I.One) / 2;
        if (hidden.Count == 18) pivot -= new Vector3I(Math.Min(0, up.X), Math.Min(0, up.Y), Math.Min(0, up.Z));
        Vector3D axis = Vector3D.TransformNormal((Vector3D)up, grid.WorldMatrix), forward = marker.WorldMatrix.Forward;
        forward -= axis * Vector3D.Dot(forward, axis);
        if (forward.LengthSquared() < .1) forward = marker.WorldMatrix.Up;
        if (!Orthonormalise(ref forward, ref axis)) throw new Exception("Invalid rotor mounting frame.");
        string type = hidden.Count == 18 ? "SmallAdvancedRotor" : grid.GridSizeEnum == MyCubeSize.Large ? "LargeRotor" : "SmallRotor";
        return new ToolTopInfo { Marker = marker, Subtype = type, Local = MatrixD.CreateWorld(grid.GridIntegerToWorld(pivot), forward, axis) * MatrixD.Invert(marker.WorldMatrix), Inventory = inventory };
    }
    static Vector3D[] Dummies = { new Vector3D(1.033530949712258e-7, .4208642244338989, 1.3662192088759184e-7), new Vector3D(2.17650750755638e-7, .036039892584085464, 3.7548051068370114e-7), new Vector3D(1.033530949712258e-7, .19979000091552734, 1.2796517978586053e-7), Vector3D.Zero, new Vector3D(2.553320221920785e-8, .047984808683395386, 5.927423671892029e-7) };
    class Face
    {
        public Vector3D Center, Normal, U, V; public double HalfU, HalfV;
    }


    public static bool MergeDirection(IMyShipMergeBlock b, string axis, out Vector3I d, out string why)
    {
        d = Vector3I.Zero; why = null; Vector3D n;
        string a = (axis ?? "Auto").ToLowerInvariant();
        if (a == "auto")
        {
            string s = b.BlockDefinition.SubtypeName;
            if (s == "LargeShipMergeBlock" || s == "SmallShipMergeBlock") a = "right";
            else if (s == "SmallShipSmallMergeBlock") a = "up";
            else return Fail(out why, "Unknown merge " + s + "; set mergeAxis.");
        }
        int index = Array.IndexOf(Axes, a);
        if (index < 0) return Fail(out why, "Invalid mergeAxis.");
        n = Vector3D.TransformNormal(Normals[index], b.WorldMatrix);
        n = Vector3D.TransformNormal(n, MatrixD.Transpose(b.CubeGrid.WorldMatrix));
        if (!Finite(n)) return Fail(out why, "Bad merge frame.");
        d = new Vector3I((int)Math.Round(n.X), (int)Math.Round(n.Y), (int)Math.Round(n.Z));
        if (Math.Abs(d.X) + Math.Abs(d.Y) + Math.Abs(d.Z) != 1) return Fail(out why, "Bad merge axis.");
        return true;
    }
    public static bool OnFace(IMyShipMergeBlock b, Vector3I c, Vector3I d) => d.X > 0 ? c.X == b.Max.X : d.X < 0 ? c.X == b.Min.X : d.Y > 0 ? c.Y == b.Max.Y : d.Y < 0 ? c.Y == b.Min.Y : d.Z > 0 ? c.Z == b.Max.Z : c.Z == b.Min.Z;
    static bool GetFace(IMyShipMergeBlock b, string axis, out Face f, out string why)
    {
        f = null; Vector3I d; if (!Budget()) return Fail(out why, GeometryBudgetMessage); if (!MergeDirection(b, axis, out d, out why)) return false;
        Vector3I size = b.Max - b.Min + Vector3I.One;
        int cells = d.X != 0 ? size.Y * size.Z : d.Y != 0 ? size.X * size.Z : size.X * size.Y;
        if (size.X < 1 || size.Y < 1 || size.Z < 1 || cells > 128 || cells < 1) return Fail(out why, "Merge face limit (128).");
        var g = b.CubeGrid; double s = g.GridSize;
        Vector3D local = (Vector3D)(b.Max - b.Min) * .5;
        local += new Vector3D(d.X * size.X, d.Y * size.Y, d.Z * size.Z) * .5;
        Vector3D u = d.X != 0 ? Vector3D.UnitY : Vector3D.UnitX;
        Vector3D v = Vector3D.Cross((Vector3D)d, u);
        f = new Face { Center = g.GridIntegerToWorld(b.Min) + Vector3D.TransformNormal(local * s, g.WorldMatrix), Normal = Vector3D.TransformNormal((Vector3D)d, g.WorldMatrix), U = Vector3D.TransformNormal(u, g.WorldMatrix), V = Vector3D.TransformNormal(v, g.WorldMatrix), HalfU = (d.X != 0 ? size.Y : size.X) * s * .5, HalfV = (d.Z != 0 ? size.Y : size.Z) * s * .5 };
        if (!Finite(f.Center) || !Finite(f.Normal) || !Finite(f.HalfU) || !Finite(f.HalfV)) return Fail(out why, "Bad merge face.");
        return true;
    }
    static Face TransformFace(Face a, MatrixD transform)
    {
        return new Face { Center = Vector3D.Transform(a.Center, transform), Normal = Vector3D.TransformNormal(a.Normal, transform), U = Vector3D.TransformNormal(a.U, transform), V = Vector3D.TransformNormal(a.V, transform), HalfU = a.HalfU, HalfV = a.HalfV };
    }
    static bool SameFace(Face a, Face b, double clearance)
    {
        if (Vector3D.Dot(a.Normal, b.Normal) > -.99999 || Vector3D.DistanceSquared(a.Center, b.Center + b.Normal * clearance) > .0004) return false;
        // Rectangles agree only when their cell axes and both extents agree;
        // quarter rolls may exchange the two axes.
        if (Math.Abs(Vector3D.Dot(a.U, b.U)) > .99999) return Math.Abs(a.HalfU - b.HalfU) <= .02 && Math.Abs(a.HalfV - b.HalfV) <= .02;
        return Math.Abs(Vector3D.Dot(a.U, b.V)) > .99999 && Math.Abs(a.HalfU - b.HalfV) <= .02 && Math.Abs(a.HalfV - b.HalfU) <= .02;
    }
    static bool AdjacentPartner(IMyShipMergeBlock a, IMyShipMergeBlock b, string axis)
    {
        Vector3I d; string why; if (!MergeDirection(a, axis, out d, out why)) return false;
        Vector3I min = a.Min, max = a.Max;
        if (d.X != 0) min.X = max.X = d.X > 0 ? max.X : min.X;
        else if (d.Y != 0) min.Y = max.Y = d.Y > 0 ? max.Y : min.Y;
        else min.Z = max.Z = d.Z > 0 ? max.Z : min.Z;
        for (int x = min.X; x <= max.X; x++) for (int y = min.Y; y <= max.Y; y++) for (int z = min.Z; z <= max.Z; z++)
        {
            if (!Budget()) return false;
            var slim = a.CubeGrid.GetCubeBlock(new Vector3I(x, y, z) + d);
            if (slim == null || slim.FatBlock == null || slim.FatBlock.EntityId != b.EntityId) return false;
        }
        return true;
    }
    public static bool PairLocked(ToolPortPair pair, string axis)
    {
        if (pair == null || pair.Head == null || pair.Stand == null) return false;
        var a = pair.Head; var b = pair.Stand;
        if (!a.IsConnected || !b.IsConnected || a.CubeGrid.EntityId != b.CubeGrid.EntityId) return false;
        Face af, bf; string why;
        return GetFace(a, axis, out af, out why) && GetFace(b, axis, out bf, out why) && SameFace(af, bf, 0) && AdjacentPartner(a, b, axis) && AdjacentPartner(b, a, axis);
    }
    public static bool TryLockedPartner(IMyShipMergeBlock a, string axis, out IMyShipMergeBlock partner)
    {
        partner = null; Vector3I d; string why;
        if (!Budget() || a == null || !a.IsConnected || !MergeDirection(a, axis, out d, out why)) return false;
        Vector3I c = a.Min;
        if (d.X > 0) c.X = a.Max.X; if (d.Y > 0) c.Y = a.Max.Y; if (d.Z > 0) c.Z = a.Max.Z;
        var slim = a.CubeGrid.GetCubeBlock(c + d);
        var other = slim == null ? null : slim.FatBlock as IMyShipMergeBlock;
        if (other == null || !PairLocked(new ToolPortPair { Head = a, Stand = other }, axis)) return false;
        partner = other; return true;
    }
    public static bool TryDockPoseAt(IMyTerminalBlock marker, List<IMyShipMergeBlock> own, List<IMyShipMergeBlock> rack, string axis, double clearance, long primaryStandID, out MatrixD pose, out List<ToolPortPair> pairs, out string why)
    {
        pose = MatrixD.Identity; pairs = null; why = null;
        if (marker == null || own == null || rack == null || own.Count == 0 || rack.Count == 0) return Fail(out why, "Missing merge ports.");
        if (own.Count > 16 || rack.Count > 32 || !Finite(clearance) || clearance < 0) return Fail(out why, "Bad clearance/port limit (16/32).");
        var a = new List<Face>(); var b = new List<Face>(); var ids = new HashSet<long>();
        foreach (var m in own) { Face f; if (m == null || !ids.Add(m.EntityId) || m.CubeGrid.EntityId != marker.CubeGrid.EntityId) return Fail(out why, "Bad Head merge set."); if (!GetFace(m, axis, out f, out why)) return false; a.Add(f); }
        ids.Clear(); foreach (var m in rack) { Face f; if (m == null || !ids.Add(m.EntityId)) return Fail(out why, "Bad stand merge set."); if (!GetFace(m, axis, out f, out why)) return false; b.Add(f); }
        double best = double.MaxValue;
        MatrixD source = MatrixD.CreateWorld(a[0].Center, a[0].U, a[0].Normal);
        for (int anchor = 0; anchor < rack.Count; anchor++)
        {
            if (!Budget()) { pairs = null; return Fail(out why, GeometryBudgetMessage); }
            if (primaryStandID > 0 && rack[anchor].EntityId != primaryStandID) continue;
            for (int roll = 0; roll < 4; roll++)
            {
            if (!Budget()) { pairs = null; return Fail(out why, GeometryBudgetMessage); }
            Vector3D tangent = roll == 0 ? b[anchor].U : roll == 1 ? b[anchor].V : roll == 2 ? -b[anchor].U : -b[anchor].V;
            MatrixD target = MatrixD.CreateWorld(b[anchor].Center + b[anchor].Normal * clearance, tangent, -b[anchor].Normal);
            MatrixD transform = MatrixD.Invert(source) * target;
            MatrixD trial = marker.WorldMatrix * transform;
            double cost = Vector3D.DistanceSquared(trial.Translation, marker.WorldMatrix.Translation) + 2 - Vector3D.Dot(trial.Up, marker.WorldMatrix.Up) - Vector3D.Dot(trial.Forward, marker.WorldMatrix.Forward);
            if (!Finite(cost) || cost >= best) continue;
            var found = new List<ToolPortPair>(); var used = new HashSet<long>(); bool valid = true;
            for (int i = 0; i < own.Count && valid; i++)
            {
                if (!Budget()) { pairs = null; return Fail(out why, GeometryBudgetMessage); }
                int match = -1; Face moved = TransformFace(a[i], transform);
                for (int j = 0; j < rack.Count; j++)
                {
                    if (!Budget()) { pairs = null; return Fail(out why, GeometryBudgetMessage); }
                    if (i == 0 && primaryStandID > 0 && j != anchor) continue;
                    if (used.Contains(rack[j].EntityId) || own[i].EntityId == rack[j].EntityId || !SameFace(moved, b[j], clearance)) continue;
                    var pair = new ToolPortPair { Head = own[i], Stand = rack[j] };
                    if ((rack[j].IsConnected || own[i].IsConnected) && !PairLocked(pair, axis)) continue;
                    if (match >= 0) { valid = false; break; } match = j;
                }
                if (match < 0) valid = false;
                else { used.Add(rack[match].EntityId); found.Add(new ToolPortPair { Head = own[i], Stand = rack[match] }); }
            }
            if (!valid) continue;
            best = cost; pose = trial; pairs = found;
            }
        }
        if (pairs == null) return Fail(out why, "No unique merge layout; check ports/mergeAxis.");
        return true;
    }
    public static bool TryTopPose(IMyMotorStator mount, ToolTopInfo top, out MatrixD pose, out string why, bool axial = false)
    {
        pose = MatrixD.Identity; MatrixD desiredBase;
        if (!TryAttachPose(mount, top, out desiredBase, out why)) return false;
        var delta = MatrixD.Invert(desiredBase) * mount.WorldMatrix;
        var desiredTop = top.WorldMatrix * delta;
        desiredTop.Translation = Vector3D.Transform(top.CubeGrid.GridIntegerToWorld(top.Position), delta);
        if (axial)
        {
            Vector3D a = top.WorldMatrix.Up, b = mount.WorldMatrix.Up;
            var cross = Vector3D.Cross(a, b);
            var rotation = cross.LengthSquared() > 1e-20 ? Unit(cross) * Math.Acos(Clamp(Vector3D.Dot(a, b), -1, 1)) : Vector3D.Dot(a, b) > 0 ? Vector3D.Zero : top.WorldMatrix.Forward * Math.PI;
            desiredTop = MatrixD.CreateWorld(desiredTop.Translation, Rotate(top.WorldMatrix.Forward, rotation), b);
        }
        pose = MatrixD.Invert(top.Local) * desiredTop;
        if (!Finite(pose.Translation) || !Finite(pose.Up) || !Finite(pose.Forward)) return Fail(out why, "Bad arm coupling frame.");
        return true;
    }
    public static bool TryAttachPose(IMyMotorStator mount, ToolTopInfo top, out MatrixD pose, out string why)
    {
        pose = MatrixD.Identity; why = null;
        if (mount == null || top == null) return Fail(out why, "Missing Mount/top.");
        string s = mount.BlockDefinition.SubtypeName, t = top.Subtype;
        int index = Array.IndexOf(Bases, s), headIndex = Array.IndexOf(Heads, t);
        if (index < 0) return Fail(out why, "Unsupported mount: " + s);
        if (headIndex < 0) return Fail(out why, "Incompatible factory head: " + t);
        Vector3D dummy = Dummies[index];
        double offset = index == 1 || index == 4 ? .045 : index == 3 ? .11 : 0;
        double lo = index == 3 ? -.02 : -.11, hi = .11;
        if ((index == 0 || index == 2) && top.CubeGrid.GridSizeEnum == MyCubeSize.Large) { lo = -.4; hi = .2; }
        double displacement = mount.Displacement;
        if (!Finite(displacement) || displacement < lo - .000001 || displacement > hi + .000001) return Fail(out why, "Attach would clamp displacement; use " + lo + " .. " + hi + ".");
        // Attach retains Angle and lock state. Preserve the phase for the rotor
        // instead of imposing a zero-angle head frame.
        double q = mount.Angle;
        if (!Finite(q)) return Fail(out why, "Bad mount angle.");
        double lower = mount.LowerLimitRad, upper = mount.UpperLimitRad;
        bool inLimits = false;
        for (int turn = -1; turn <= 1; turn++)
        {
            double candidate = q + turn * Math.PI * 2;
            if ((lower < -1e6 || candidate >= lower - .001) && (upper > 1e6 || candidate <= upper + .001)) inLimits = true;
        }
        if (!inLimits) return Fail(out why, "Angle outside mount limits.");
        Vector3D up = top.WorldMatrix.Up, forward = top.WorldMatrix.Forward;
        forward = forward * Math.Cos(q) + Vector3D.Cross(up, forward) * Math.Sin(q);
        pose = MatrixD.CreateWorld(Vector3D.Zero, forward, up);
        pose.Translation = top.CubeGrid.GridIntegerToWorld(top.Position) - Vector3D.TransformNormal(dummy + Vector3D.Up * (displacement - offset), pose);
        if (!Finite(pose.Translation) || !Finite(pose.Up) || !Finite(pose.Forward)) return Fail(out why, "Bad attach frame.");
        return true;
    }
}

    bool ScanStep(out bool done, out string why)
    {
        done = false; why = "";
        if (ScanIndex < Profiles.Count)
        {
            var h = Profiles[ScanIndex];
            IMyTerminalBlock current;
            if (!Exact(h.MarkerName, out current, out why) || current.EntityId != h.Marker.EntityId) { why = "Marker changed; Reload."; return false; }
            if (Scan == null) Scan = new ToolRegionScan(P, h.Marker, CellLimit, Axis);
            bool finished; if (!Scan.Step(out finished, out why)) { why = h.MarkerName + ": " + why; return false; }
            if (!finished) return true;
            if (!Scan.Blocks.Contains(h.Coupler.EntityId)) { why = "Tool Mount outside marker region."; return false; }
            foreach (var other in Profiles) if (other != h && Scan.Blocks.Contains(other.Marker.EntityId)) { why = "Tool regions overlap."; return false; }
            h.SafetyMerges.AddRange(Scan.Merges);
            if (ScanHead) h.Heads.AddRange(Scan.Merges);
            if (!MergeNames(h.HeadNames, h.Heads, out why) || !MergeNames(h.StandNames, h.Stands, out why)) return false;
            foreach (var merge in h.Heads) if (!Scan.Blocks.Contains(merge.EntityId)) { why = "Foreign head merge: " + merge.CustomName; return false; }
            if (ScanStand) Prefix(h.Prefix, h.Stands);
            foreach (var merge in h.Stands) if (h.Heads.Contains(merge)) { why = "Stand is on head: " + merge.CustomName; return false; }
            ScanIndex++; Scan = null; return true;
        }
        if (Tip == null)
        {
            if (ArmIndex < ArmOrder.Count)
            {
                var frame = GridFrame(ArmOrder[ArmIndex]);
                if (Scan == null) Scan = new ToolRegionScan(P, frame, CellLimit, Axis);
                bool finished; if (!Scan.Step(out finished, out why)) return false;
                if (!finished) return true;
                var candidate = ToolGeometry.FindTip(frame, Scan.Hidden, Inventory);
                if (candidate != null)
                { if (Found != null) { why = "Multiple loose arm rotor parts."; return false; } Found = candidate; }
                ArmIndex++; Scan = null; return true;
            }
            if (Found == null) { why = "No loose arm rotor part."; return false; }

        }

        foreach (var h in Profiles) foreach (var m in Parks) if (h.Heads.Contains(m)) { why = "Park is on head."; return false; }
        Scanned = true; done = true; return true;
    }

        }

        }
        if (Source == null) return BeginIncoming(out why);
        if (Vector3D.Dot(Source.Coupler.WorldMatrix.Up, Tip.WorldMatrix.Up) < -.95)
        { why = "Flipped mount; detach refused."; return false; }
        long anchor = 0;
        if (ParkOnly && ParkIndex > 0)
        {
            IMyTerminalBlock block;
            if (!Exact(ArmName + " - ParkMerge " + ParkIndex, out block, out why)) return false;
            var merge = block as IMyShipMergeBlock;
            if (merge == null || !Parks.Contains(merge)) { why = "Park port not configured."; return false; }
            anchor = merge.EntityId;
        }
        if (Connected(Source.SafetyMerges))
        {
            if (!FindSupport(Source, out why)) return false;
            if (anchor != 0 && Pairs[0].Stand.EntityId != anchor) { why = "Already merged to another park."; return false; }
            SetRetreat();
            P.StopMotion("Supported tool OFF"); Go("Lock"); return true;
        }
        var rack = ParkOnly ? Parks : Source.Stands;
        if (Source.Heads.Count == 0 || rack.Count == 0) { why = "Missing parking merges."; return false; }
        if (!ToolGeometry.TryDockPoseAt(Source.Marker, Source.Heads, rack, Axis, 0, anchor, out Dock, out Pairs, out why)) return false;
        Vector3I direction; if (!ToolGeometry.MergeDirection(Pairs[0].Stand, Axis, out direction, out why)) return false;
        Vector3D outward = Vector3D.TransformNormal((Vector3D)direction, Pairs[0].Stand.CubeGrid.WorldMatrix);
        foreach (var pair in Pairs)
        {
            if (pair.Stand.CubeGrid.EntityId == Reference.CubeGrid.EntityId) { why = "Stand is on arm."; return false; }
            Vector3I other; if (!ToolGeometry.MergeDirection(pair.Stand, Axis, out other, out why)) return false;
            if (Vector3D.Dot(outward, Vector3D.TransformNormal((Vector3D)other, pair.Stand.CubeGrid.WorldMatrix)) < .99999)
            { why = "Parking faces must be parallel."; return false; }
        }
        Approach = Dock; Approach.Translation += outward * Clearance;
        DockAnchor = Pairs[0].Stand;
        var inverse = MatrixD.Invert(DockAnchor.WorldMatrix); Dock *= inverse; Approach *= inverse;
        Supported = Source;
        Go(ApproachDockPhase); return true;
    }
    bool AllLocked()
    {
        if (Supported == null || Pairs.Count == 0 || Pairs.Count != Supported.Heads.Count || Supported.Coupler.CubeGrid.EntityId != Supported.Marker.CubeGrid.EntityId) return false;
        foreach (var pair in Pairs) if (pair.Head.CubeGrid.EntityId != Supported.Marker.CubeGrid.EntityId || pair.Stand.CubeGrid.EntityId == Reference.CubeGrid.EntityId || !ToolGeometry.PairLocked(pair, Axis)) return false;
        foreach (var m in Supported.SafetyMerges) if (m.IsConnected)
        { bool covered = false; foreach (var pair in Pairs) if (pair.Head.EntityId == m.EntityId) covered = true; if (!covered) return false; }
        return true;
    }
    bool FindSupport(Profile h, out string why)
    {
        why = ""; Pairs = new List<ToolPortPair>(); Supported = h;
        foreach (var head in h.Heads)
        {
            IMyShipMergeBlock stand;
            if (!head.IsConnected || !ToolGeometry.TryLockedPartner(head, Axis, out stand)) { why = "Unproven support: " + head.CustomName; return false; }
            foreach (var tool in Profiles) if (tool.SafetyMerges.Contains(stand)) { why = "Support is on another tool."; return false; }
            Pairs.Add(new ToolPortPair { Head = head, Stand = stand });
        }
        if (AllLocked()) return true;
        why = "Unselected/unlocked support."; return false;
    }
    bool BeginIncoming(out string why)
    {
        why = "";
        if (Destination == null) { Complete("Tool parked; bare mount OFF."); return true; }
        if (!FindSupport(Destination, out why))
        return false;
        return TargetTop(out why);
    }
    bool TargetTop(out string why)
    {
        why = "";
        if (!ToolGeometry.TryTopPose(Destination.Coupler, Tip, out Attach, out why, AxialPickup)) return false;
        TopApproach = Attach; TopApproach.Translation += Destination.Coupler.WorldMatrix.Up * Clearance;
        AttachAnchor = Destination.Marker;
        var inverse = MatrixD.Invert(AttachAnchor.WorldMatrix); Attach *= inverse; TopApproach *= inverse;
        Go(ApproachTopPhase); return true;
    }
    void SetRetreat()
    {
        RetreatAnchor = Source.Marker;
        Retreat = Reference.WorldMatrix; Retreat.Translation += Source.Coupler.WorldMatrix.Up * Clearance;
        Retreat *= MatrixD.Invert(RetreatAnchor.WorldMatrix);
    }
    bool MovePose(MatrixD pose, double dt, double obs, out bool settled, out string why)
    {
        settled = false; why = "";
        foreach (var h in Profiles) if (h.Coupler.PendingAttachment) { why = "Pending attachment; motion refused."; return false; }
        if (Source != null && Mounted(Source) && Connected(Source.SafetyMerges)) { why = "Mounted support; motion refused."; return false; }
        if (Phase == RetreatPhase || Phase == ApproachTopPhase || Phase == AlignTopPhase)
            foreach (var h in Profiles) if (h.Coupler.IsAttached || h.Coupler.Top != null) { why = "Unexpected attachment."; return false; }
        if ((Phase == ApproachDockPhase || Phase == "Dock") && !Mounted(Source)) { why = "Source attachment changed."; return false; }
        if (Focus != null) pose = Tip.Local * pose;
        return P.ToolPoseStep(pose, dt, obs, Move, Turn, Phase == AlignTopPhase ? Math.Min(PosTol, .003) : PosTol,
            Phase == AlignTopPhase ? Math.Min(AngleTol, .2) : AngleTol, out settled, out why);
    }
    public void Tick(double dt, double observationDt)
    {
        if (!Busy || !(dt > 0) || !(observationDt > 0)) return;
        try
        {
            if (dt > 1 || observationDt > 1) { Cancel("Tool observation gap."); return; }
            Elapsed += observationDt; string why;
            if (!Guard(out why)) { Cancel(why); return; }
            if (Elapsed > PhaseBudget) { Cancel("Tool phase timeout: " + Phase); return; }
            bool done, settled;
            if (Phase == ApproachDockPhase || Phase == "Dock" || Phase == RetreatPhase || Phase == ApproachTopPhase || Phase == AlignTopPhase)
            {
                if ((Phase == RetreatPhase || Phase == ApproachTopPhase || Phase == AlignTopPhase) && !AllLocked()) { Cancel("Support lost."); return; }
                MatrixD goal = Phase == ApproachDockPhase ? Approach : Phase == "Dock" ? Dock : Phase == RetreatPhase ? Retreat : Phase == ApproachTopPhase ? TopApproach : Attach;
                var anchor = Phase == ApproachDockPhase || Phase == "Dock" ? DockAnchor : Phase == RetreatPhase ? RetreatAnchor : AttachAnchor;
                if (anchor == null || anchor.Closed) { Cancel("Dock anchor lost."); return; }
                goal *= anchor.WorldMatrix;
                if (!MovePose(goal, dt, observationDt, out settled, out why)) { Cancel(why); return; }
                if (!Observe(settled)) return;
            }
            switch (Phase)
            {
                case "Scan":
                    if (!ScanStep(out done, out why)) { Cancel(why); return; }
                    if (!done) return;
                    foreach (var h in Profiles) if (h.Coupler.PendingAttachment)
                    {
                        if (h.Coupler.IsAttached || h.Coupler.Top != null || Last != "Scan" || PassiveScan) { Cancel("Explicit ToolScan required for pending Mount."); return; }
                        Clearing.Add(h.Coupler);
                    }
                    if (Clearing.Count > 0)
                    {
                        Journal("ScanPending"); foreach (var coupler in Clearing) coupler.Detach(); Go("ScanPending"); return;
                    }
                    if (!P.RebuildToolTopology(out why)) { Cancel(why); return; }
                    Go("ScanSettle"); return;
                case "ScanPending":
                    P.StopMotion("Clear pending OFF"); bool clear = true;
                    foreach (var coupler in Clearing)
                    {
                        if (coupler.IsAttached || coupler.Top != null) { Cancel("Pending cancellation changed."); return; }
                        if (coupler.PendingAttachment) clear = false;
                    }
                    if (!Observe(clear)) return;
                    if (!P.RebuildToolTopology(out why)) { Cancel(why); return; } Go("ScanSettle"); return;
                case "ScanSettle":
                    if (!P.SettleToolTopology(out why)) return;
                    if (Last == "Scan")
                    {
                        foreach (var h in Profiles) if (h.Relock && !h.Coupler.PendingAttachment)
                        {
                            MatrixD pose;
                            bool bare = !h.Coupler.IsAttached && h.Coupler.Top == null;
                            if (bare || Mounted(h) && ToolGeometry.TryTopPose(h.Coupler, Tip, out pose, out why, true) && Vector3D.DistanceSquared(Reference.GetPosition(), pose.Translation) < .000025 && Vector3D.Dot(Reference.WorldMatrix.Up, pose.Up) > .9999939)
                            { h.Coupler.RotorLock = true; h.Relock = false; }
                        }
                        Complete("Scanned; OFF.");
                    }
                    else if (!BeginChange(out why)) Cancel(why); return;
                case ApproachDockPhase:
                    Go("Dock"); return;
                case "Dock":
                    P.StopMotion("Supporting OFF"); Journal("EnableSupport");
                    foreach (var pair in Pairs) pair.Head.Enabled = true;
                    DockMoved = true;
                    Go("Lock"); return;
                case "Lock":
                    P.StopMotion("Support wait OFF");
                    if (!Mounted(Source)) { Cancel("Source detached prematurely."); return; }
                    if (!Observe(AllLocked())) return;
                    if (DockMoved) SetRetreat();
                    ParkedPairs = new List<ToolPortPair>(Pairs);
                    Source.Angle = Source.Coupler.Angle; Journal("Detach"); Source.Coupler.Detach(); Go("Detach"); return;
                case "Detach":
                    P.StopMotion("Detach wait OFF");
                    if (!AllLocked()) { Cancel("Support lost."); return; }
                    if (!Observe(!Source.Coupler.IsAttached && !Source.Coupler.PendingAttachment && Source.Coupler.Top == null && Tip.Base == null)) return;
                    Source.Angle = Source.Coupler.Angle;
                    if (!P.RebuildToolTopology(out why)) { Cancel(why); return; } Go("BareSettle"); return;
                case "BareSettle":
                    if (P.SettleToolTopology(out why)) Go(RetreatPhase); return;
                case RetreatPhase:
                    if (!BeginIncoming(out why)) Cancel(why); return;
                case ApproachTopPhase:
                    Go(AlignTopPhase); return;
                case AlignTopPhase:
                    foreach (var h in Profiles) if (h.Coupler.IsAttached || h.Coupler.PendingAttachment || h.Coupler.Top != null) { Cancel("Arm tip no longer bare."); return; }
                    P.StopMotion("Attaching OFF"); if (AxialPickup) Destination.Relock = Destination.Coupler.RotorLock;
                    Journal("Attach"); if (Destination.Relock) Destination.Coupler.RotorLock = false; Destination.Coupler.Attach(); Go("Attach"); return;
                case "Attach":
                    P.StopMotion("Attach wait OFF");
                    if (!AllLocked()) { Cancel("Support lost."); return; }
                    if (Destination.Coupler.IsAttached && !Mounted(Destination)) { Cancel("Wrong arm tip attached."); return; }
                    MatrixD expected;
                    if (!ToolGeometry.TryTopPose(Destination.Coupler, Tip, out expected, out why, AxialPickup)) { Cancel(why); return; }
                    bool aligned = Vector3D.DistanceSquared(Reference.GetPosition(), expected.Translation) < .000025 &&
                        Vector3D.Dot(Reference.WorldMatrix.Up, expected.Up) > .9999939 && Vector3D.Dot(Reference.WorldMatrix.Forward, expected.Forward) > .9999939;
                    if (!Observe(Mounted(Destination) && !Destination.Coupler.PendingAttachment && aligned)) return;
                    if (Destination.Relock) { Destination.Coupler.RotorLock = true; Destination.Relock = false; }
                    Journal("ReleaseSupport"); foreach (var pair in Pairs) pair.Head.Enabled = false; Go("Release"); return;
                case "Release":
                    P.StopMotion("Split wait OFF");
                    if (!Mounted(Destination)) { Cancel("Attachment lost."); return; }
                    bool split = !Connected(Destination.SafetyMerges);
                    foreach (var pair in Pairs) if (pair.Head.Closed || pair.Stand.Closed || pair.Stand.IsConnected || Destination.Marker.CubeGrid.EntityId == pair.Stand.CubeGrid.EntityId) split = false;
                    if (!Observe(split)) return;
                    if (!P.RebuildToolTopology(out why)) { Cancel(why); return; } Go("FullSettle"); return;
                case "FullSettle":
                    if (!Mounted(Destination) || Connected(Destination.SafetyMerges)) { Cancel("Attachment/support changed."); return; }
                    if (P.SettleToolTopology(out why)) Complete("Tool " + Destination.Name + " mounted.", true); return;
            }
        }
        catch (Exception e) { Cancel("Tools: " + e.Message); }
    }
    string Info()
    {
        var b = new System.Text.StringBuilder("Tools " + (EnabledTools ? EnabledKey : "disabled") + "; " + Phase + (Recovery ? "; recovery OFF" : "") + "\n" + Last);
        b.Append(Tip == null ? "\nNo arm rotor part found." : "\nArm rotor part detected.");
        for (int n = 0; n < Profiles.Count; n++) { var h = Profiles[n]; b.Append("\n").Append(n + 1).Append(" ").Append(h.Name).Append(Mounted(h) ? " mounted" : " detached").Append("; Mount ").Append(h.CouplerName).Append("; merges ").Append(h.Heads.Count).Append('/').Append(h.Stands.Count); }
        return b.ToString();
    }
}


// Swap motion uses the existing physical Jacobian, limits, budget guard and
// output commit. Only verified open-chain phases call this boundary.
string ToolGroupKey(AutoGroup g)
{
    string key = g.Rotary ? "R" : "P";
    foreach (var m in g.M) key += ":" + m.B.EntityId + (m.Sign > 0 ? "+" : "-");
    return key;
}

void RememberToolWeights(bool capture = true)
{
    if (!ToolWeightsLoaded)
    {
        var saved = AutoConfig.Saved(Storage);
        string rows = saved.Get(ToolWeightsSection, "Rows").ToString();
        foreach (string row in rows.Split('\n'))
        {
            if (row.Trim().Length == 0) continue;
            string[] cells = row.Split('|'); double m, r;
            if (cells.Length != 3 || !ReadDouble(cells[1], out m) ||
                !ReadDouble(cells[2], out r) ||
                !Finite(m) || !Finite(r) || m < 0 || m > 10 || r < 0 || r > 10)
                throw new Exception("Invalid saved tool actuator preferences.");
            ToolWeights[cells[0]] = new double[] { m, r };
        }
        ToolWeightsLoaded = true;
    }
    if (!capture) return;
    foreach (var g in Topology.Groups) ToolWeights[ToolGroupKey(g)] = new double[] { g.MoveW, g.TurnW };
    var keys = new List<string>(ToolWeights.Keys); keys.Sort(StringComparer.Ordinal);
    var text = new System.Text.StringBuilder();
    foreach (string key in keys)
    {
        if (text.Length > 0) text.Append('\n');
        text.Append(key).Append('|').Append(AutoSettings.Number(ToolWeights[key][0])).Append('|').Append(AutoSettings.Number(ToolWeights[key][1]));
    }
    var ini = AutoConfig.Saved(Storage); ini.Set(ToolWeightsSection, "Rows", text.ToString()); Storage = ini.ToString();
}

bool RebuildToolTopology(out string why)
{
    why = ""; StopMotion(""); ConfigValid = false;
    try
    {
        RememberToolWeights();
        Topology.SetEndpointOverride(Tools.Endpoint, Tools.IgnoredMount, Tools.EndFrame, Tools.Focus);
        if (!Topology.Discover(out why)) return false;
        foreach (var g in Topology.Groups)
        {
            double[] value;
            if (ToolWeights.TryGetValue(ToolGroupKey(g), out value)) { g.MoveW = value[0]; g.TurnW = value[1]; }
        }
        Allocate(Topology.Groups.Count); ConfigValid = true;
        SetupElapsed = ControlElapsed = 0;
        Runtime.UpdateFrequency = UpdateFrequency.Update1;
        return true;
    }
    catch (Exception e) { why = e.Message; return false; }
}

bool SettleToolTopology(out string why)
{
    if (!ConfigValid) { why = "Tool topology is not configured."; return false; }
    if (!Topology.StableSetup(out why)) return false;
    AutoConfig.UpdateGroups(this); LoadHome();
    return true;
}

bool ToolPoseStep(MatrixD world, double dt, double observationDt, double moveCap,
    double turnCap, double positionTolerance, double angleTolerance,
    out bool settled, out string why)
{
    settled = false; why = "";
    if (!Topology.Ready || !ConfigValid) { why = "Tool topology is not ready."; return false; }
    if (!Finite(dt) || dt <= 0 || !Finite(observationDt) || observationDt <= 0 ||
        !Finite(moveCap) || moveCap <= 0 || !Finite(turnCap) || turnCap <= 0)
    { why = "Invalid tool movement interval or cap."; return false; }
    MatrixD basis = Topology.Base.WorldMatrix;
    if (!SetPoseTarget(InFrame(world.Translation - Topology.Base.GetPosition(), basis),
        InFrame(world.Forward, basis), InFrame(world.Up, basis), 3))
    { why = "Invalid tool target frame."; return false; }
    ToolDriveMoveCap = moveCap; ToolDriveTurnCap = turnCap * Math.PI / 180;
    ToolDriving = true;
    try { UpdateControl(Math.Min(dt, .1), observationDt); }
    finally { ToolDriving = false; }
    if (!TargetValid || !Topology.Ready) { why = LastFault.Length > 0 ? LastFault : "Tool motion stopped."; return false; }
    settled = LastPositionError <= positionTolerance && LastOrientationErrorDeg <= angleTolerance &&
        FilteredV.Length() <= .005 && FilteredW.Length() * 180 / Math.PI <= .2;
    return true;
}
