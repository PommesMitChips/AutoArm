// AutoArm: auto-discovered mixed mechanical arm controller.
// Change only this value in code. All detected group participation settings are
// generated in this programmable block's Custom Data.
const string ArmName = "Arm 1";

// ===== IMPLEMENTATION =====
const string ControllerBuildVersion = "2.4";


// C# 6 programmable-block fragment. Paste inside Program.
class AutoMember
{
    public IMyMechanicalConnectionBlock B;
    public int Sign;
    public double Q0, Last, Home;
    public long U, V;
}

class AutoGroup
{
    public int Index;
    public bool Rotary;
    public string Sig, Label;
    public double MoveW = 1, TurnW = 1;
    public readonly List<AutoMember> M = new List<AutoMember>();
    public Vector3D Lin, Ang;
}

class AutoEdge
{
    public IMyMechanicalConnectionBlock B;
    public long A, Z;
    public int K;
    public bool BaseMark, HeadMark;
}

class AutoLink
{
    public AutoEdge E;
    public long U, V;
    public AutoLink(AutoEdge e, long u, long v) { E = e; U = u; V = v; }
}

class AutoTopology
{
    const int MaxMembers = 48, MaxGroups = 24, MaxWorkEdges = 128, MaxInventory = 4096;
    const double AxisCos = .999961923, RotaryLineM = .03;
    readonly Program P;
    public readonly string Arm;
    public readonly List<AutoGroup> Groups = new List<AutoGroup>();
    public readonly List<long> Rigid = new List<long>();
    readonly List<IMyMechanicalConnectionBlock> All = new List<IMyMechanicalConnectionBlock>();
    readonly List<IMyTerminalBlock> Inventory = new List<IMyTerminalBlock>();
    readonly List<string> SnapshotRows = new List<string>();
    readonly System.Text.StringBuilder SnapshotText = new System.Text.StringBuilder();
    readonly List<AutoEdge> Edges = new List<AutoEdge>();
    readonly HashSet<long> Used = new HashSet<long>();
    readonly HashSet<long> Corridor = new HashSet<long>();
    readonly HashSet<long> Watch = new HashSet<long>();
    public IMyMechanicalConnectionBlock Base;
    public IMyTerminalBlock Head;
    public string Fingerprint = "", Report = "";
    public bool Ready;
    long Hull, Start;
    long BaseId, HeadId, HeadGrid;
    string Snapshot = "";
    int StableCount;
    int ScanBases, ScanHeads;
    int FullScanIn = 30;
    bool Building;

    public AutoTopology(Program owner, string arm)
    {
        P = owner; Arm = arm == null ? "" : arm.Trim();
    }

    int Kind(IMyMechanicalConnectionBlock b)
    {
        if (b is IMyPistonBase) return 2;
        if (b is IMyMotorStator && !(b is IMyMotorSuspension)) return 1;
        return 0;
    }

    bool Attached(IMyMechanicalConnectionBlock b)
    {
        return b != null && !b.Closed && b.IsAttached && b.Top != null && b.TopGrid != null;
    }

    bool Mark(string name, string role)
    {
        return name != null && name.StartsWith(Arm + " - " + role, StringComparison.OrdinalIgnoreCase);
    }

    public bool Discover(out string why)
    {
        ZeroOwned(); Building = true;
        why = ""; Ready = false; StableCount = 0; Groups.Clear(); Rigid.Clear();
        Edges.Clear(); Used.Clear(); Corridor.Clear(); Watch.Clear(); Report = Fingerprint = "";
        if (Arm.Length == 0) return Bad("ArmName is empty.", out why);
        if (!Guard("pre-inventory reserve", out why)) return false;
        All.Clear(); Inventory.Clear(); P.GridTerminalSystem.GetBlocks(Inventory);
        if (Inventory.Count > MaxInventory) return Bad("Terminal inventory exceeds the " + MaxInventory + "-block discovery bound.", out why);
        if (!Guard("terminal inventory", out why)) return false;
        Base = null; Head = null; int bases = 0, heads = 0;
        for (int i = 0; i < Inventory.Count; i++)
        {
            if ((i & 63) == 0 && !Guard("terminal inventory filter", out why)) return false;
            var b = Inventory[i]; if (!b.IsSameConstructAs(P.Me)) continue;
            if (Mark(b.CustomName, "Head")) { Head = b; heads++; }
            var m = b as IMyMechanicalConnectionBlock; if (m == null) continue;
            All.Add(m); if (Mark(m.CustomName, "Base")) { Base = m; bases++; }
        }
        if (bases != 1) return Bad("Need exactly one mechanical block named '" + Arm + " - Base*'; found " + bases + ".", out why);
        if (!Attached(Base) || Kind(Base) == 0) return Bad("The Base marker must be an attached piston, rotor, or hinge.", out why);
        if (heads != 1) return Bad("Need exactly one terminal block named '" + Arm + " - Head*'; found " + heads + ".", out why);
        BaseId = Base.EntityId; HeadId = Head.EntityId; HeadGrid = Head.CubeGrid.EntityId;
        Hull = Base.CubeGrid.EntityId; Start = Base.TopGrid.EntityId;
        if (Hull == Start || Head.CubeGrid.EntityId == Hull) return Bad("Base top and Head must be on moving rigid segments beyond the base grid.", out why);
        for (int i = 0; i < All.Count; i++)
        {
            if ((i & 63) == 0 && !Guard("mechanical edge inventory", out why)) return false;
            if (!Attached(All[i])) continue;
            var e = new AutoEdge { B = All[i], A = All[i].CubeGrid.EntityId, Z = All[i].TopGrid.EntityId,
                K = Kind(All[i]), BaseMark = Mark(All[i].CustomName, "Base"), HeadMark = Mark(All[i].CustomName, "Head") };
            if (e.A == e.Z) return Bad("Self-loop mechanical connection: " + e.B.CustomName, out why);
            Edges.Add(e);
        }
        if (!ScopeEdges(out why)) return false;
        var root = Between(Hull, Start);
        if (!Has(root, Base.EntityId)) return Bad("Base marker attachment changed while scanning.", out why);
        if (!AddGroup(root, out why)) return false;
        Corridor.Add(Start); Corridor.Add(Hull);
        var dist = Distances(Head.CubeGrid.EntityId, out why); if (dist == null) return false;
        int d;
        if (!dist.TryGetValue(Start, out d)) return Bad("Head is not reachable from the Base top without returning through the base grid.", out why);
        long at = Start;
        while (at != Head.CubeGrid.EntityId)
        {
            if (!Guard("corridor recognition", out why)) return false;
            if (Groups.Count >= MaxGroups) return Bad("Active arm exceeds " + MaxGroups + " generalized groups.", out why);
            int here = dist[at];
            var nextEdges = new List<AutoLink>();
            var destinations = new List<long>();
            for (int i = 0; i < Edges.Count; i++)
            {
                var e = Edges[i]; if (Used.Contains(e.B.EntityId)) continue;
                long n = e.A == at ? e.Z : e.Z == at ? e.A : 0; int nd;
                if (n == 0 || !dist.TryGetValue(n, out nd) || nd != here - 1) continue;
                nextEdges.Add(new AutoLink(e, at, n)); if (!destinations.Contains(n)) destinations.Add(n);
            }
            if (destinations.Count == 0) return Bad("No unambiguous next stage from rigid grid " + at + ".", out why);
            if (destinations.Count == 1)
            {
                if (!AddGroup(nextEdges, out why)) return false;
                at = destinations[0]; Corridor.Add(at);
                continue;
            }
            var paths = new List<List<AutoLink>>();
            var candidate = new HashSet<long>();
            for (int i = 0; i < destinations.Count; i++)
            {
                long n = destinations[i]; var path = new List<AutoLink>(); int count = 0;
                for (int j = 0; j < nextEdges.Count; j++) if (nextEdges[j].V == n) { path.Add(nextEdges[j]); count++; }
                if (count != 1) return Bad("Unsupported nested/parallel bundle at split grid " + at + ".", out why);
                candidate.Add(path[0].E.B.EntityId);
                paths.Add(path);
            }
            if (OwnedCount() + candidate.Count > MaxMembers) return Bad("Split/rejoin candidate exceeds " + MaxMembers + " owned actuators.", out why);
            int guard = 0;
            while (true)
            {
                if (!Guard("split/rejoin recognition", out why)) return false;
                long meet = paths[0][paths[0].Count - 1].V; bool same = true;
                for (int i = 1; i < paths.Count; i++) if (paths[i][paths[i].Count - 1].V != meet) same = false;
                if (same) { at = meet; break; }
                if (++guard > MaxMembers) return Bad("Parallel paths do not reconverge within the supported bound.", out why);
                var occupied = new HashSet<long>();
                for (int i = 0; i < paths.Count; i++)
                {
                    if (!Guard("split/rejoin path expansion", out why)) return false;
                    var path = paths[i]; long u = path[path.Count - 1].V;
                    if (!occupied.Add(u)) return Bad("Parallel paths merge or cross before a common stage boundary.", out why);
                    AutoLink chosen = null; int count = 0; int ud = dist[u];
                    for (int j = 0; j < Edges.Count; j++)
                    {
                        if ((j & 31) == 0 && !Guard("split/rejoin edge search", out why)) return false;
                        var e = Edges[j]; if (Used.Contains(e.B.EntityId) || candidate.Contains(e.B.EntityId)) continue;
                        long v = e.A == u ? e.Z : e.Z == u ? e.A : 0; int vd;
                        if (v != 0 && dist.TryGetValue(v, out vd) && vd == ud - 1) { chosen = new AutoLink(e, u, v); count++; }
                    }
                    if (count != 1) return Bad("Unsupported branch/cross-link inside split path at grid " + u + ".", out why);
                    candidate.Add(chosen.E.B.EntityId);
                    if (OwnedCount() + candidate.Count > MaxMembers) return Bad("Split/rejoin candidate exceeds " + MaxMembers + " owned actuators.", out why);
                    path.Add(chosen);
                }
            }
            int stages = paths[0].Count;
            for (int i = 1; i < paths.Count; i++) if (paths[i].Count != stages)
                return Bad("Parallel arms have unequal stage counts; variable-ratio closure is unsupported.", out why);
            for (int s = 0; s < stages; s++)
            {
                var stage = new List<AutoLink>();
                for (int i = 0; i < paths.Count; i++) { stage.Add(paths[i][s]); Corridor.Add(paths[i][s].U); Corridor.Add(paths[i][s].V); }
                if (!AddGroup(stage, out why)) return false;
            }
        }
        if (!ClassifyUnused(out why)) return false;
        if (!BuildWatch(out why)) return false;
        if (!SnapshotFromInventory(out Snapshot, out why)) return Bad(why, out why);
        var fp = new System.Text.StringBuilder();
        for (int i = 0; i < Groups.Count; i++) { Groups[i].Index = i; if (i > 0) fp.Append('|'); fp.Append(Groups[i].Sig); }
        Fingerprint = fp.ToString();
        Rigid.AddRange(Corridor); Rigid.Sort();
        Report += Groups.Count + " groups / " + OwnedCount() + " actuators / " + Rigid.Count + " rigid segments detected.";
        for (int i = 0; i < Rigid.Count; i++) Report += "\nR" + (i + 1).ToString("00") + " = grid " + Rigid[i];
        for (int i = 0; i < Groups.Count; i++) if (Groups[i].M.Count > 1)
        { Report += " Parallel groups are live tolerance-checked 1:1 copies, not a proof for general closed linkages."; break; }
        Report += "\nKnown topology is checked every active pass; unknown additions/markers are fully scanned within 30 active passes.";
        FullScanIn = 30;
        Building = false; ZeroOwned();
        return true;
    }

    bool ScopeEdges(out string why)
    {
        why = "";
        var byGrid = new Dictionary<long, List<AutoEdge>>();
        for (int i = 0; i < Edges.Count; i++)
        {
            if ((i & 63) == 0 && !Guard("moving-component adjacency", out why)) return false;
            var e = Edges[i]; List<AutoEdge> a, z;
            if (!byGrid.TryGetValue(e.A, out a)) byGrid[e.A] = a = new List<AutoEdge>();
            if (!byGrid.TryGetValue(e.Z, out z)) byGrid[e.Z] = z = new List<AutoEdge>();
            a.Add(e); z.Add(e);
        }
        var keep = new HashSet<long>(); var nodes = new HashSet<long>(); var q = new Queue<long>();
        nodes.Add(Start); q.Enqueue(Start);
        while (q.Count > 0)
        {
            if (!Guard("moving-component scope", out why)) return false;
            long u = q.Dequeue(); List<AutoEdge> list;
            if (!byGrid.TryGetValue(u, out list)) continue;
            for (int i = 0; i < list.Count; i++)
            {
                if ((i & 31) == 0 && !Guard("moving-component expansion", out why)) return false;
                var e = list[i]; keep.Add(e.B.EntityId);
                if (keep.Count > MaxWorkEdges) return Bad("Moving mechanical component exceeds the " + MaxWorkEdges + "-edge discovery bound.", out why);
                long v = e.A == u ? e.Z : e.A;
                if (v != Hull && nodes.Add(v)) q.Enqueue(v);
            }
        }
        var scoped = new List<AutoEdge>();
        for (int i = 0; i < Edges.Count; i++)
        {
            if ((i & 63) == 0 && !Guard("moving-component filter", out why)) return false;
            if (keep.Contains(Edges[i].B.EntityId)) scoped.Add(Edges[i]);
        }
        Edges.Clear(); Edges.AddRange(scoped);
        return true;
    }

    bool Has(List<AutoLink> links, long id)
    {
        for (int i = 0; i < links.Count; i++) if (links[i].E.B.EntityId == id) return true;
        return false;
    }

    List<AutoLink> Between(long u, long v)
    {
        var r = new List<AutoLink>();
        for (int i = 0; i < Edges.Count; i++)
        {
            var e = Edges[i];
            if ((e.A == u && e.Z == v) || (e.A == v && e.Z == u)) r.Add(new AutoLink(e, u, v));
        }
        return r;
    }

    Dictionary<long, int> Distances(long target, out string why)
    {
        why = ""; var d = new Dictionary<long, int>(); var q = new Queue<long>(); d[target] = 0; q.Enqueue(target);
        while (q.Count > 0)
        {
            if (!Guard("Head-distance graph", out why)) return null;
            long u = q.Dequeue(); int nd = d[u] + 1;
            for (int i = 0; i < Edges.Count; i++)
            {
                var e = Edges[i]; long v = e.A == u ? e.Z : e.Z == u ? e.A : 0;
                if (v == 0 || v == Hull || d.ContainsKey(v)) continue;
                d[v] = nd; q.Enqueue(v);
            }
        }
        return d;
    }

    bool AddGroup(List<AutoLink> links, out string why)
    {
        why = ""; if (links.Count == 0) return Bad("Empty mechanical stage.", out why);
        if (Groups.Count >= MaxGroups) return Bad("Active Base-to-Head corridor exceeds " + MaxGroups + " generalized groups.", out why);
        if (OwnedCount() + links.Count > MaxMembers) return Bad("Active Base-to-Head corridor exceeds " + MaxMembers + " owned actuators.", out why);
        links.Sort((x, y) => x.E.B.EntityId.CompareTo(y.E.B.EntityId));
        int kind = Kind(links[0].E.B);
        if (kind == 0) return Bad("Unsupported connection on active route: " + links[0].E.B.CustomName, out why);
        var g = new AutoGroup { Rotary = kind == 1 };
        Vector3D baseL = Vector3D.Zero, baseA = Vector3D.Zero;
        for (int i = 0; i < links.Count; i++)
        {
            var l = links[i];
            if (Kind(l.E.B) != kind) return Bad("Parallel stage mixes piston and rotary actuators.", out why);
            var m = new AutoMember { B = l.E.B, U = l.U, V = l.V, Sign = 1 };
            Vector3D lin, ang;
            if (!Raw(m, Head.GetPosition(), out lin, out ang, out why)) return Bad(why, out why);
            if (i == 0) { baseL = lin; baseA = ang; }
            else
            {
                double dot = kind == 1 ? Vector3D.Dot(baseA, ang) : Vector3D.Dot(baseL, lin);
                m.Sign = dot < 0 ? -1 : 1; lin *= m.Sign; ang *= m.Sign;
                bool compatible = kind == 1
                    ? Vector3D.Dot(baseA, ang) >= AxisCos && (baseL - lin).Length() <= RotaryLineM
                    : Vector3D.Dot(baseL, lin) >= AxisCos;
                if (!compatible) return Bad("Parallel " + (kind == 1 ? "rotors/hinges lack one coaxial full twist" : "pistons are not collinear") +
                    " at stage containing '" + l.E.B.CustomName + "'. General linkage ratios are unsupported.", out why);
            }
            m.Q0 = m.Last = Position(m);
            if (!Finite(m.Q0)) return Bad("Non-finite actuator position: " + m.B.CustomName, out why);
            g.M.Add(m); Used.Add(m.B.EntityId);
        }
        var sig = new System.Text.StringBuilder(kind == 1 ? "R" : "P");
        var label = new System.Text.StringBuilder(kind == 1 ? "Rotary: " : "Piston: ");
        for (int i = 0; i < g.M.Count; i++)
        {
            var m = g.M[i]; sig.Append(':').Append(m.B.EntityId).Append(m.Sign > 0 ? '+' : '-').Append('@').Append(m.U).Append('>').Append(m.V);
            if (i > 0) label.Append(" | "); label.Append(m.B.CustomName);
        }
        g.Sig = sig.ToString(); g.Label = label.ToString(); Groups.Add(g);
        Report += "G" + Groups.Count + " " + g.Label + "\n";
        return true;
    }

    bool Raw(AutoMember m, Vector3D head, out Vector3D lin, out Vector3D ang, out string why)
    {
        lin = ang = Vector3D.Zero; why = ""; var b = m.B;
        if (!Attached(b)) { why = "Detached actuator: " + b.CustomName; return false; }
        long a = b.CubeGrid.EntityId, z = b.TopGrid.EntityId; int dir;
        if (a == m.U && z == m.V) dir = 1; else if (a == m.V && z == m.U) dir = -1;
        else { why = "Actuator endpoints changed: " + b.CustomName; return false; }
        Vector3D axis = b.WorldMatrix.Up; double n = axis.Length();
        if (!Finite(axis.X) || !Finite(axis.Y) || !Finite(axis.Z) || !Finite(head.X) || !Finite(head.Y) || !Finite(head.Z) ||
            !Finite(n) || n < .9) { why = "Invalid actuator axis or Head position: " + b.CustomName; return false; }
        axis /= n;
        if (b is IMyPistonBase) lin = axis * dir;
        else { ang = axis * -dir; lin = Vector3D.Cross(ang, head - b.GetPosition()); }
        return true;
    }

    public double Position(AutoMember m)
    {
        var p = m.B as IMyPistonBase; return p != null ? p.CurrentPosition : ((IMyMotorStator)m.B).Angle;
    }

    double Wrap(double a) { return a - Math.Floor((a + Math.PI) / (2 * Math.PI)) * (2 * Math.PI); }

    public bool CheckSync(AutoGroup g, out string why)
    {
        why = "";
        double lo = double.PositiveInfinity, hi = double.NegativeInfinity, spread = 0;
        for (int i = 0; i < g.M.Count; i++)
        {
            var m = g.M[i]; double q = Position(m);
            if (!Finite(q) || !Finite(m.Q0)) { why = "Non-finite actuator position in G" + (g.Index + 1) + ": " + m.B.CustomName; return false; }
            double d = q - m.Q0; if (g.Rotary) d = Wrap(d); d *= m.Sign;
            if (g.Rotary) for (int j = 0; j < i; j++)
            {
                var n = g.M[j]; double nd = Wrap(Position(n) - n.Q0) * n.Sign;
                spread = Math.Max(spread, Math.Abs(Wrap(d - nd)));
            }
            lo = Math.Min(lo, d); hi = Math.Max(hi, d);
        }
        if (g.M.Count < 2) return true;
        double tol = g.Rotary ? .25 * Math.PI / 180 : .02;
        if (!g.Rotary) spread = hi - lo;
        if (spread <= tol) return true;
        why = "Parallel actuator desynchronization in G" + (g.Index + 1) + ": " + spread.ToString("0.0000") + (g.Rotary ? " rad." : " m.");
        return false;
    }

    public bool Refresh(AutoGroup g, Vector3D headWorld, MatrixD reference, out double lo, out double hi, out string why)
    {
        lo = -1e6; hi = 1e6; why = ""; Vector3D sl = Vector3D.Zero, sa = Vector3D.Zero, firstL = Vector3D.Zero, firstA = Vector3D.Zero;
        if (!CheckSync(g, out why)) return Fail(why);
        for (int i = 0; i < g.M.Count; i++)
        {
            var m = g.M[i]; var b = m.B;
            if (!Attached(b) || !b.Enabled || !b.IsWorking) { why = "Unavailable owned actuator: " + b.CustomName; return Fail(why); }
            Vector3D l, a; if (!Raw(m, headWorld, out l, out a, out why)) return Fail(why);
            l *= m.Sign; a *= m.Sign;
            if (i == 0) { firstL = l; firstA = a; }
            else if (g.Rotary ? Vector3D.Dot(firstA, a) < AxisCos || (firstL - l).Length() > RotaryLineM : Vector3D.Dot(firstL, l) < AxisCos)
            { why = "Live parallel-stage screw mismatch in G" + (g.Index + 1) + "; closure is no longer 1:1 compatible."; return Fail(why); }
            sl += l; sa += a;
            double ml, mh; if (!Bounds(m, out ml, out mh, out why)) return Fail(why); lo = Math.Max(lo, ml); hi = Math.Min(hi, mh);
        }
        if (lo > hi) { why = "No common signed velocity remains in G" + (g.Index + 1) + "."; return Fail(why); }
        sl /= g.M.Count; sa /= g.M.Count;
        g.Lin = ToFrame(sl, reference); g.Ang = ToFrame(sa, reference);
        return true;
    }

    bool Bounds(AutoMember m, out double lo, out double hi, out string why)
    {
        lo = -1e6; hi = 1e6; why = ""; double q = Position(m), mn, mx;
        var p = m.B as IMyPistonBase;
        if (p != null) { lo = -p.MaxVelocity; hi = p.MaxVelocity; mn = p.MinLimit; mx = p.MaxLimit; }
        else
        {
            var r = (IMyMotorStator)m.B; mn = r.LowerLimitRad; mx = r.UpperLimitRad;
            if (r.RotorLock) lo = hi = 0;
        }
        if (!Finite(q) || !Finite(mn) || !Finite(mx) || !Finite(lo) || !Finite(hi) || mn > mx || lo > hi)
        { why = "Invalid position, limits, or maximum velocity: " + m.B.CustomName; return false; }
        if (p != null && p.MaxVelocity <= 0) { why = "Piston reports no positive maximum velocity: " + m.B.CustomName; return false; }
        double accel = p != null ? P.PistonAccelerationMps2 : P.JointAccelerationRpmPerSec * Math.PI / 30;
        double margin = p != null ? .001 : .1 * Math.PI / 180;
        if (!Finite(accel) || accel <= 0) { why = "Acceleration must be finite and positive for travel braking."; return false; }
        if (mn > -1e6) lo = Math.Max(lo, -Math.Sqrt(2 * accel * Math.Max(0, q - mn - margin)));
        if (mx < 1e6) hi = Math.Min(hi, Math.Sqrt(2 * accel * Math.Max(0, mx - q - margin)));
        if (m.Sign < 0) { double x = lo; lo = -hi; hi = -x; }
        return true;
    }

    Vector3D ToFrame(Vector3D v, MatrixD f)
    {
        return new Vector3D(Vector3D.Dot(v, f.Forward), Vector3D.Dot(v, f.Left), Vector3D.Dot(v, f.Up));
    }

    public void Set(AutoGroup g, double velocity)
    {
        for (int i = 0; i < g.M.Count; i++)
        {
            var m = g.M[i]; float v = (float)(m.Sign * velocity);
            var p = m.B as IMyPistonBase; if (p != null) p.Velocity = v; else ((IMyMotorStator)m.B).TargetVelocityRad = v;
        }
    }

    public void ZeroOwned()
    {
        for (int i = 0; i < Groups.Count; i++) for (int j = 0; j < Groups[i].M.Count; j++)
        {
            var b = Groups[i].M[j].B; if (b == null || b.Closed) continue;
            try { var p = b as IMyPistonBase; if (p != null) p.Velocity = 0; else { var r = b as IMyMotorStator; if (r != null) r.TargetVelocityRad = 0; } }
            catch (Exception) { }
        }
    }

    public bool StableSetup(out string why)
    {
        why = ""; if (!Validate(out why, true)) return false;
        bool moved = false;
        for (int i = 0; i < Groups.Count; i++) for (int j = 0; j < Groups[i].M.Count; j++)
        {
            var m = Groups[i].M[j]; double target, q = Position(m);
            if (!Finite(q)) { why = "Non-finite actuator position while settling: " + m.B.CustomName; StableCount = 0; return false; }
            var p = m.B as IMyPistonBase; target = p != null ? p.Velocity : ((IMyMotorStator)m.B).TargetVelocityRad;
            if (Math.Abs(target) > 1e-4) { why = "Waiting for owned actuator targets to reach zero."; StableCount = 0; return false; }
            double delta = q - m.Last; if (Groups[i].Rotary) delta = Wrap(delta);
            if (Math.Abs(delta) > 1e-4) moved = true; m.Last = q;
        }
        StableCount = moved ? 0 : StableCount + 1;
        if (StableCount < 2) { why = "Waiting for two stable stopped topology scans."; return false; }
        for (int i = 0; i < Groups.Count; i++) for (int j = 0; j < Groups[i].M.Count; j++) Groups[i].M[j].Q0 = Position(Groups[i].M[j]);
        Ready = true; return true;
    }

    public bool Validate(out string why, bool full = false)
    {
        why = "";
        if (!BudgetAvailable()) { why = "Instruction reserve was already low before cached topology validation; motion was refused."; return Fail(why); }
        if (Base == null || Head == null || Base.Closed || Head.Closed || Base.EntityId != BaseId || Head.EntityId != HeadId ||
            !Base.IsSameConstructAs(P.Me) || !Head.IsSameConstructAs(P.Me) || !Mark(Base.CustomName, "Base") || !Mark(Head.CustomName, "Head"))
        { why = "Base/Head marker disappeared or was renamed."; return Fail(why); }
        if (Head.CubeGrid == null || Head.CubeGrid.EntityId != HeadGrid) { why = "Head marker moved to another rigid grid; Rescan is required."; return Fail(why); }
        for (int i = 0; i < Edges.Count; i++)
        {
            var e = Edges[i]; var b = e.B;
            if (!Attached(b) || b.CubeGrid.EntityId != e.A || b.TopGrid.EntityId != e.Z || Kind(b) != e.K ||
                Mark(b.CustomName, "Base") != e.BaseMark || Mark(b.CustomName, "Head") != e.HeadMark)
            { why = "Known mechanical connection changed: " + (b == null ? "missing block" : b.CustomName) + ". Rescan is required."; return Fail(why); }
        }
        for (int i = 0; i < Groups.Count; i++) if (!CheckSync(Groups[i], out why)) return Fail(why);
        if (!full && --FullScanIn > 0) return true;
        string current;
        if (!CurrentSnapshot(out current, out why)) return Fail(why);
        if (ScanBases != 1 || ScanHeads != 1) { why = "Base/Head marker is no longer unique; Rescan is required."; return Fail(why); }
        if (current != Snapshot) { why = "Mechanical topology changed; Rescan is required."; return Fail(why); }
        FullScanIn = 30;
        return true;
    }

    bool ClassifyUnused(out string why)
    {
        why = ""; var seen = new HashSet<long>(); int ignored = 0;
        for (int i = 0; i < Edges.Count; i++)
        {
            var seed = Edges[i]; if (Used.Contains(seed.B.EntityId) || seen.Contains(seed.B.EntityId)) continue;
            var q = new Queue<AutoEdge>(); var boundary = new HashSet<long>(); q.Enqueue(seed); seen.Add(seed.B.EntityId); int count = 0;
            while (q.Count > 0)
            {
                if (!Guard("off-corridor classification", out why)) return false;
                var e = q.Dequeue(); count++;
                if ((e.A == Hull && Corridor.Contains(e.Z)) || (e.Z == Hull && Corridor.Contains(e.A)))
                    return Bad("A second mechanical anchor connects the base grid to the active corridor: " + e.B.CustomName, out why);
                if (Corridor.Contains(e.A)) boundary.Add(e.A); if (Corridor.Contains(e.Z)) boundary.Add(e.Z);
                for (int j = 0; j < Edges.Count; j++)
                {
                    var n = Edges[j]; if (Used.Contains(n.B.EntityId) || seen.Contains(n.B.EntityId)) continue;
                    if (n.A == e.A || n.A == e.Z || n.Z == e.A || n.Z == e.Z) { seen.Add(n.B.EntityId); q.Enqueue(n); }
                }
            }
            if (boundary.Count > 1) return Bad("Unsupported closure reconnects an unused mechanical component to multiple active rigid segments.", out why);
            ignored += count;
        }
        if (ignored > 0) Report += "Ignored " + ignored + " off-corridor actuators; they are never commanded.\n";
        return true;
    }

    bool BuildWatch(out string why)
    {
        why = "";
        var q = new Queue<long>(); Watch.Add(Start); q.Enqueue(Start);
        while (q.Count > 0)
        {
            if (!Guard("topology watch-set", out why)) return false;
            long u = q.Dequeue();
            for (int i = 0; i < Edges.Count; i++)
            {
                var e = Edges[i]; long v = e.A == u ? e.Z : e.Z == u ? e.A : 0;
                if (v == 0 || v == Hull || !Watch.Add(v)) continue; q.Enqueue(v);
            }
        }
        return true;
    }

    bool CurrentSnapshot(out string snapshot, out string why)
    {
        snapshot = ""; why = "";
        if (!BudgetAvailable()) { why = "Instruction reserve was already low before topology validation; motion was refused."; return false; }
        Inventory.Clear(); P.GridTerminalSystem.GetBlocks(Inventory);
        return SnapshotFromInventory(out snapshot, out why);
    }

    bool SnapshotFromInventory(out string snapshot, out string why)
    {
        snapshot = ""; why = "";
        if (Inventory.Count > MaxInventory) { why = "Terminal inventory exceeds the " + MaxInventory + "-block validation bound."; return false; }
        SnapshotRows.Clear(); ScanBases = ScanHeads = 0;
        for (int i = 0; i < Inventory.Count; i++)
        {
            if ((i & 31) == 0 && !BudgetAvailable()) { why = "Instruction reserve reached during topology validation; motion was refused."; return false; }
            var terminal = Inventory[i]; if (!terminal.IsSameConstructAs(P.Me)) continue;
            if (Mark(terminal.CustomName, "Head")) ScanHeads++;
            var b = terminal as IMyMechanicalConnectionBlock; if (b == null) continue;
            long a = b.CubeGrid.EntityId, z = Attached(b) ? b.TopGrid.EntityId : 0;
            if (Mark(b.CustomName, "Base")) ScanBases++;
            if (Watch.Contains(a) || Watch.Contains(z)) SnapshotRows.Add(b.EntityId + ":" + a + ":" + z + ":" + Kind(b));
        }
        if (!BudgetAvailable()) { why = "Instruction reserve reached after topology validation scan; motion was refused."; return false; }
        SnapshotRows.Sort(StringComparer.Ordinal); SnapshotText.Clear();
        for (int i = 0; i < SnapshotRows.Count; i++) { if (i > 0) SnapshotText.Append('|'); SnapshotText.Append(SnapshotRows[i]); }
        snapshot = SnapshotText.ToString(); return true;
    }

    int OwnedCount() { int n = 0; for (int i = 0; i < Groups.Count; i++) n += Groups[i].M.Count; return n; }

    bool Finite(double v) { return !double.IsNaN(v) && !double.IsInfinity(v); }

    bool BudgetAvailable()
    {
        return P.Runtime.MaxInstructionCount <= 0 || P.Runtime.CurrentInstructionCount < P.Runtime.MaxInstructionCount * .55;
    }

    bool Guard(string phase, out string why)
    {
        why = ""; if (BudgetAvailable()) return true;
        return Bad("Instruction reserve reached during " + phase + "; arm remains Off. Reduce attached topology or run Rescan again.", out why);
    }

    bool Fail(string why) { Ready = false; if (Building) { Groups.Clear(); Used.Clear(); } else ZeroOwned(); return false; }
    bool Bad(string text, out string why) { why = text; Fail(text); return false; }
}

        { why = "MovementMode must be HEAD, HRZ or VRT."; return false; }

    }
    public static bool ReadMode(string text, out PilotMode mode)
    {
        mode = PilotMode.HEAD;
        switch (text.Trim().ToUpperInvariant())
        {
            case "HEAD": return true;
            case "HRZ": mode = PilotMode.HRZ; return true;
            case "VRT": mode = PilotMode.VRT; return true;
            default: return false;
        }
    }
    static bool Finite(double v) { return !double.IsNaN(v) && !double.IsInfinity(v); }

public void Main(string argument, UpdateType source)
{
    DisplayWritten = false;
    double elapsed = Runtime.TimeSinceLastRun.TotalSeconds;
    bool timed = (source & (UpdateType.Update1 | UpdateType.Update10 | UpdateType.Update100 | UpdateType.Once)) != 0;
    if (Finite(elapsed) && elapsed > 0)
    {
        if (Enabled) ControlElapsed += elapsed;
        if (!Topology.Ready) SetupElapsed = Math.Min(0.5, SetupElapsed + elapsed);
        StatusElapsed = Math.Min(StatusRefreshSeconds, StatusElapsed + elapsed);
    }
    try
    {
        if (!string.IsNullOrWhiteSpace(argument)) RunCommands(argument);
        if (!Topology.Ready)
        {
            // A validation failure must require a new explicit On. Stable setup may
            // recover diagnostics while Off, but it must never resume old commands.
            if (Enabled || GoingHome)
            {
                StopMotion("Topology lost readiness; stopped. Run Check/Rescan/Reload, then On.");
                Say(Status);
                return;
            }
            if (!timed || !Finite(elapsed) || elapsed < 0 || SetupElapsed <= 0.000001) return;
            SetupElapsed = 0;
            MaintainSetup();
            if (!Topology.Ready) return;
        }
        if (PendingOn)
        {
            PendingOn = false;
            EnableControl();
        }
        if (!Enabled) return;
        if (ControlElapsed > 0.5) { Fault("Control update gap exceeded 0.5 seconds."); return; }
        // Commands can run in the same simulation frame as the previous call.
        // They change state immediately, but only timed updates advance motion.
        if (!timed) return;
        if (!Finite(elapsed) || elapsed < 0) { Fault("Invalid negative/non-finite control interval."); return; }
        double dt = ControlElapsed;
        if (dt <= 0.000001) return;
        ControlElapsed = 0;
        if (dt > 0.5) { Fault("Control update gap exceeded 0.5 seconds."); return; }
        if (GoingHome) UpdateHome(Math.Min(dt, 0.1));
        else UpdateControl(Math.Min(dt, 0.1), dt);
    }
    catch (Exception e)
    {
        Fault("Controller exception: " + e.Message);
    }
    finally
    {
        // SE clears the PB display on every invocation. Replay cached output;
        // routine numeric formatting happens at 4 Hz, not every control tick.
        Echo(DisplayText);
    }
}

void Say(string text)
{
    DisplayText = DisplayWritten ? DisplayText + "\n" + text : text;
    if (DisplayText.Length > 8000) DisplayText = DisplayText.Substring(DisplayText.Length - 8000);
    DisplayWritten = true;
}

void Discover(bool explicitRequest)
{
    StopMotion("");
    PivotProbe = null;
    SetupElapsed = 0;
    ConfigValid = false;
    string why;
    if (!Topology.Discover(out why))
    {
        Status = "Discovery failed: " + why;
        if (explicitRequest) Say(Status);
        return;
    }
    Topology.ZeroOwned();
    string updated;
    bool changed;
    if (!AutoConfig.LoadOrGenerate(this, Topology, Me.CustomData, out updated, out changed, out why))
    {
        Status = "Custom Data refused: " + why;
        if (explicitRequest) Say(Status);
        return;
    }
    ConfigValid = true;
    Allocate(Topology.Groups.Count);
    LoadHome();
    Status = (changed ? "Configuration generated/updated; " : "Configuration loaded; ") +
        Topology.Groups.Count + " groups found. Waiting for stable stopped scans.";
    Runtime.UpdateFrequency = UpdateFrequency.Update10;
    if (explicitRequest) Say(Status + "\n" + Topology.Report);
}

void MaintainSetup()
{
    if (!ConfigValid) return;
    string why;
    Topology.ZeroOwned();
    if (!Topology.StableSetup(out why))
    {
        Status = "Setup waiting: " + why;
        return;
    }
    Status = "Ready and OFF. Run On.";
    Say(Banner() + "\n" + Status);
}

void EnableControl()
{
    if (!ConfigValid)
    {
        PendingOn = false;
        Status = "On refused: Custom Data configuration is invalid. Fix it, then Reload.";
        Say(Status);
        return;
    }
    if (!Topology.Ready)
    {
        PendingOn = true;
        Status = "On pending until discovery/configuration has two stable stopped scans.";
        Runtime.UpdateFrequency = UpdateFrequency.Update10;
        Say(Status);
        return;
    }
    string why;
    if (!Topology.Validate(out why, true)) { Fault("On refused: " + why); return; }
    Vector3D p, f, u;
    if (!ReadPose(out p, out f, out u)) { Fault("On refused: Base/Head markers unavailable."); return; }
    TargetP = PreviousP = p;
    TargetF = PreviousF = f;
    TargetU = PreviousU = u;
    TargetValid = ObservationValid = true;
    AimingTarget = false;
    PivotProbe = null;
    FilteredV = FilteredW = PositionI = OrientationI = Vector3D.Zero;
    LastPilotLinear = LastPilotAngular = LastFeedbackLinear = LastFeedbackAngular = Vector3D.Zero;
    LastRequestedLinear = LastRequestedAngular = LastAchievedLinear = LastAchievedAngular = Vector3D.Zero;
    LastLinearAdvance = LastAngularAdvance = 1;
    LastPositionError = LastOrientationErrorDeg = 0;
    LastSolverPasses = LastActiveBounds = 0;
    LastSolverConverged = LastSolverBudgetLimited = false;
    LastFault = "";
    Enabled = true;

    StatusElapsed = StatusRefreshSeconds;

}

void StopMotion(string reason)
{
    if (Topology != null) Topology.ZeroOwned();
    Enabled = GoingHome = PendingOn = false;
    ControlElapsed = 0;
    DisplayPinned = false;
    TargetValid = ObservationValid = false;
    AimingTarget = false;
    PositionI = OrientationI = Vector3D.Zero;
    if (LastQdot != null) Array.Clear(LastQdot, 0, LastQdot.Length);
    if (BoundState != null) Array.Clear(BoundState, 0, BoundState.Length);
    Runtime.UpdateFrequency = UpdateFrequency.Update10;
    if (!string.IsNullOrWhiteSpace(reason))
    {
        Status = reason;
        LastFault = reason;
    }
}

void Fault(string reason)
{
    StopMotion(reason);
    Say("STOPPED: " + reason);
}

bool ReadPose(out Vector3D p, out Vector3D f, out Vector3D u)
{
    p = f = u = Vector3D.Zero;
    if (Topology == null || Topology.Base == null || Topology.Head == null || Topology.Base.Closed || Topology.Head.Closed) return false;
    MatrixD r = Topology.Base.WorldMatrix;
    p = InFrame(Topology.Head.GetPosition() - Topology.Base.GetPosition(), r);
    f = InFrame(Topology.Head.WorldMatrix.Forward, r);
    u = InFrame(Topology.Head.WorldMatrix.Up, r);
    if (!Finite(p.LengthSquared())) return false;
    return Orthonormalise(ref f, ref u);
}

void UpdateControl(double dt, double observationDt)
{
    string why;
    if (!Topology.Validate(out why)) { Fault("Topology changed: " + why); return; }
    Vector3D p, f, u;
    if (!ReadPose(out p, out f, out u)) { Fault("Base/Head marker became unavailable."); return; }
    if (!TargetValid) { Fault("No held pose. Run On."); return; }
    Vector3D measuredV = Vector3D.Zero, measuredW = Vector3D.Zero;
    if (ObservationValid)
    {
        measuredV = (p - PreviousP) / observationDt;
        measuredW = RotationError(PreviousF, PreviousU, f, u) / observationDt;
        double blend = observationDt / (VelocityFilterSeconds + observationDt);
        FilteredV += (measuredV - FilteredV) * blend;
        FilteredW += (measuredW - FilteredW) * blend;
    }
    PreviousP = p; PreviousF = f; PreviousU = u; ObservationValid = true;

    Vector3D pe = TargetP - p;
    Vector3D oe = RotationError(f, u, TargetF, TargetU);
    LastPositionError = pe.Length();
    LastOrientationErrorDeg = oe.Length() * 180 / Math.PI;
    if (!Finite(LastPositionError) || !Finite(LastOrientationErrorDeg)) { Fault("Non-finite pose feedback."); return; }

    ReadPilot(f, u, out LastPilotLinear, out LastPilotAngular);
    bool manualTurn = LastPilotAngular.LengthSquared() > 1e-12;
    if (manualTurn) AimingTarget = false;
    if (AimingTarget && LastOrientationErrorDeg <= AimToleranceDeg) AimingTarget = false;
    bool aiming = manualTurn || AimingTarget;
    double orientationBand = (aiming ? 0.5 * AimToleranceDeg : OrientationToleranceDeg) * Math.PI / 180;
    Vector3D rawPositionFeedback = Outside(pe, PositionDeadbandM) * PositionKp + PositionI - FilteredV * PositionKd;
    Vector3D rawOrientationFeedback = Outside(oe, orientationBand) * OrientationKp + OrientationI - FilteredW * OrientationKd;
    Vector3D boundedPositionFeedback = ClampLength(rawPositionFeedback, PositionCorrectionCap);
    LastFeedbackLinear = BlendManualCorrection(LastPilotLinear, boundedPositionFeedback,
        ManualCounterCorrectionFraction, out LastCounterCorrectionLimited);
    bool angularLimited;
    LastFeedbackAngular = BlendManualCorrection(LastPilotAngular, ClampLength(rawOrientationFeedback,
        Math.Min(TurnDeg, OrientationCorrectionCapDeg) * Math.PI / 180),
        ManualCounterCorrectionFraction, out angularLimited);
    LastCounterCorrectionLimited |= angularLimited;

    // Core invariant: tracking correction is added to live pilot demand. Pose error,
    // recovery state and target advancement never multiply or suppress pilot input.
    LastRequestedLinear = LastPilotLinear + LastFeedbackLinear;
    LastRequestedAngular = LastPilotAngular + LastFeedbackAngular;
    if (!Solve(p, dt, LastRequestedLinear, LastRequestedAngular, out LastAchievedLinear,
        out LastAchievedAngular, out why))
    {
        Fault("Velocity allocation failed: " + why);
        return;
    }
    if (!Commit()) return;

    Vector3D positionClipResidual = rawPositionFeedback - LastFeedbackLinear;
    Vector3D orientationClipResidual = rawOrientationFeedback - LastFeedbackAngular;
    // Only loss caused by command bounds inhibits bias learning. Damping and
    // unreachable task components exist even in the unconstrained reference.
    Vector3D positionAllocationResidual = LastUnconstrainedLinear - LastAchievedLinear;
    Vector3D orientationAllocationResidual = LastUnconstrainedAngular - LastAchievedAngular;
    UpdateIntegral(ref PositionI, pe, PositionDeadbandM, positionClipResidual, positionAllocationResidual,
        PositionKi, Math.Min(PositionIntegralCap, PositionCorrectionCap), PositionIntegralCapture, dt, LastSolverConverged && !LastSolverBudgetLimited);
    UpdateIntegral(ref OrientationI, oe, orientationBand, orientationClipResidual, orientationAllocationResidual, OrientationKi,
        Math.Min(OrientationIntegralCapDeg, Math.Min(TurnDeg, OrientationCorrectionCapDeg)) * Math.PI / 180, OrientationIntegralCaptureDeg * Math.PI / 180,
        dt, LastSolverConverged && !LastSolverBudgetLimited);

    LastLinearAdvance = PilotAdvance(LastPilotLinear, LastAchievedLinear - LastFeedbackLinear);
    LastAngularAdvance = PilotAdvance(LastPilotAngular, LastAchievedAngular - LastFeedbackAngular);
    // Allocation is a kinematic prediction, not measured delivery. Bound newly
    // requested target lead even when physical contact prevents predicted motion.
    LastLinearAdvance *= TargetLeadAdvance(pe, LastPilotLinear * (dt * LastLinearAdvance), MaximumPilotTargetLeadM);
    TargetP += LastPilotLinear * (dt * LastLinearAdvance);
    if (LastPilotAngular.LengthSquared() > 1e-20 && LastAngularAdvance > 0)
    {
        Vector3D step = LastPilotAngular * (dt * LastAngularAdvance);
        TargetF = Rotate(TargetF, step); TargetU = Rotate(TargetU, step);
        Orthonormalise(ref TargetF, ref TargetU);
    }
    RenderRoutineStatus();
}

void RenderRoutineStatus()
{
    if (DisplayWritten || DisplayPinned || StatusElapsed < StatusRefreshSeconds) return;
    StatusElapsed = 0;
    RoutineStatusBuilds++;
    Status = "ON | input+feedback | error " + LastPositionError.ToString("0.000") + " m / " +
        LastOrientationErrorDeg.ToString("0.00") + " deg";
    if (LastCounterCorrectionLimited) Status += " | manual priority";
    Say(Status + "\nPilot " + LastPilotLinear.Length().ToString("0.000") + " + feedback " +
        LastFeedbackLinear.Length().ToString("0.000") + " m/s; advance " +
        (100 * LastLinearAdvance).ToString("0") + "%\nSolver " + LastSolverPasses +
        " passes; " + LastActiveBounds + " bounds. Stop always zeros owned drives.");
}

bool Solve(Vector3D headReferencePosition, double dt, Vector3D linear, Vector3D angular,
    out Vector3D achievedLinear, out Vector3D achievedAngular, out string why)
{
    achievedLinear = achievedAngular = Vector3D.Zero;
    why = "";
    LastSolverPasses = LastActiveBounds = 0;
    LastSolverConverged = LastSolverBudgetLimited = false;
    int n = Topology.Groups.Count;
    double moveDemand = linear.Length(), turnDemand = angular.Length();
    double demand = moveDemand + turnDemand;
    MatrixD frame = Topology.Base.WorldMatrix;
    Vector3D headWorld = Topology.Head.GetPosition();
    for (int i = 0; i < n; i++)
    {
        AutoGroup g = Topology.Groups[i];
        double physicalLo, physicalHi;
        if (!Topology.Refresh(g, headWorld, frame, out physicalLo, out physicalHi, out why)) return false;
        double cap = g.Rotary ? JointRpm * Math.PI / 30 : PistonMps;
        double lo = Math.Max(physicalLo, -cap), hi = Math.Min(physicalHi, cap);
        double accel = g.Rotary ? JointAccelerationRpmPerSec * Math.PI / 30 : PistonAccelerationMps2;
        if (accel > 0)
        {
            // A changed limit/lock may make the old command infeasible. Centre the
            // acceleration window on the nearest currently feasible command.
            double centre = Clamp(LastQdot[i], lo, hi);
            lo = Math.Max(lo, centre - accel * dt);
            hi = Math.Min(hi, centre + accel * dt);
        }
        if (lo > hi + 1e-10) { why = "No common command interval for " + g.Label; return false; }
        double preference = demand > 1e-9 ? (moveDemand * g.MoveW + turnDemand * g.TurnW) / demand :
            0.5 * (g.MoveW + g.TurnW);
        double s = Scale[i] = preference > 0 ? cap * Math.Sqrt(preference) : 0;
        Lo[i] = s > 1e-12 ? lo / s : 0;
        Hi[i] = s > 1e-12 ? hi / s : 0;
        X[i] = s > 1e-12 ? Clamp(LastQdot[i] / s, Lo[i], Hi[i]) : 0;
        int old = BoundState[i];
        BoundState[i] = Hi[i] - Lo[i] <= 1e-12 ? 2 :
            old == -1 && Math.Abs(X[i] - Lo[i]) <= 1e-9 ? -1 :
            old == 1 && Math.Abs(X[i] - Hi[i]) <= 1e-9 ? 1 : 0;
        J[0, i] = g.Lin.X; J[1, i] = g.Lin.Y; J[2, i] = g.Lin.Z;
        J[3, i] = g.Ang.X; J[4, i] = g.Ang.Y; J[5, i] = g.Ang.Z;
        for (int r = 0; r < 6; r++)
        {
            double weight = r < 3 ? PositionTaskWeight : OrientationTaskWeight;
            B[r, i] = weight * J[r, i] * s;
        }
    }
    for (int r = 0; r < 6; r++)
    {
        double weight = r < 3 ? PositionTaskWeight : OrientationTaskWeight;
        Task[r] = weight * (r == 0 ? linear.X : r == 1 ? linear.Y : r == 2 ? linear.Z :
            r == 3 ? angular.X : r == 4 ? angular.Y : angular.Z);
    }
    double ridge = SolverDamping * SolverDamping;
    int maxPasses = Math.Min(SolverPassLimit, Math.Max(8, 2 * n + 4));
    for (int pass = 0; pass < maxPasses; pass++)
    {
        if (Runtime.CurrentInstructionCount > Runtime.MaxInstructionCount * 0.68)
        {
            LastSolverBudgetLimited = true;
            break;
        }
        LastSolverPasses = pass + 1;
        for (int r = 0; r < 6; r++)
        {
            double value = Task[r];
            for (int i = 0; i < n; i++) if (BoundState[i] != 0) value -= B[r, i] * X[i];
            Rhs[r] = value;
            for (int c = 0; c <= r; c++)
            {
                double total = r == c ? ridge : 0;
                for (int i = 0; i < n; i++) if (BoundState[i] == 0) total += B[r, i] * B[c, i];
                Gram[r, c] = total;
            }
        }
        if (!Cholesky(Gram, Rhs, Chol, Temp, Solution)) { why = "singular/non-finite damped task matrix"; return false; }
        double alpha = 1;
        int hit = -1, side = 0;
        for (int i = 0; i < n; i++)
        {
            Trial[i] = X[i];
            if (BoundState[i] != 0) continue;
            double value = 0;
            for (int r = 0; r < 6; r++) value += B[r, i] * Solution[r];
            Trial[i] = value;
            double delta = value - X[i], a;
            if (delta < -1e-14 && value < Lo[i] - 1e-12)
            {
                a = Math.Max(0, (Lo[i] - X[i]) / delta);
                if (a < alpha) { alpha = a; hit = i; side = -1; }
            }
            else if (delta > 1e-14 && value > Hi[i] + 1e-12)
            {
                a = Math.Max(0, (Hi[i] - X[i]) / delta);
                if (a < alpha) { alpha = a; hit = i; side = 1; }
            }
        }
        for (int i = 0; i < n; i++) if (BoundState[i] == 0)
            X[i] = Clamp(X[i] + alpha * (Trial[i] - X[i]), Lo[i], Hi[i]);
        if (hit >= 0)
        {
            X[hit] = side < 0 ? Lo[hit] : Hi[hit];
            BoundState[hit] = side;
            continue;
        }
        for (int r = 0; r < 6; r++)
        {
            double value = -Task[r];
            for (int i = 0; i < n; i++) value += B[r, i] * X[i];
            Residual[r] = value;
        }
        int release = -1;
        double worst = 1e-9;
        for (int i = 0; i < n; i++)
        {
            if (BoundState[i] == 0 || BoundState[i] == 2) continue;
            double gradient = ridge * X[i];
            for (int r = 0; r < 6; r++) gradient += B[r, i] * Residual[r];
            double violation = BoundState[i] < 0 ? -gradient : gradient;
            if (violation > worst) { worst = violation; release = i; }
        }
        if (release >= 0) { BoundState[release] = 0; continue; }
        LastSolverConverged = true;
        break;
    }
    if (LastSolverPasses == 0) { why = "no instruction budget for a solver pass"; return false; }
    for (int i = 0; i < n; i++)
    {
        double velocity = X[i] * Scale[i];
        if (!Finite(velocity)) { why = "non-finite group velocity"; return false; }
        LastQdot[i] = velocity;
        if (BoundState[i] != 0) LastActiveBounds++;
        achievedLinear += Topology.Groups[i].Lin * velocity;
        achievedAngular += Topology.Groups[i].Ang * velocity;
    }
    LastUnconstrainedLinear = achievedLinear; LastUnconstrainedAngular = achievedAngular;
    if (LastActiveBounds > 0 && LastSolverConverged && !LastSolverBudgetLimited)
    {
        if (Runtime.CurrentInstructionCount > Runtime.MaxInstructionCount * 0.68) LastSolverBudgetLimited = true;
        else if (!UnconstrainedTask(n, ridge, out LastUnconstrainedLinear, out LastUnconstrainedAngular))
        { why = "non-finite unconstrained task reference"; return false; }
    }
    return Finite(achievedLinear.LengthSquared()) && Finite(achievedAngular.LengthSquared());
}

bool UnconstrainedTask(int n, double ridge, out Vector3D linear, out Vector3D angular)
{
    linear = angular = Vector3D.Zero;
    for (int r = 0; r < 6; r++)
    {
        Rhs[r] = Task[r];
        for (int c = 0; c <= r; c++)
        {
            double total = r == c ? ridge : 0;
            for (int i = 0; i < n; i++) total += B[r, i] * B[c, i];
            Gram[r, c] = total;
        }
    }
    if (!Cholesky(Gram, Rhs, Chol, Temp, Solution)) return false;
    for (int i = 0; i < n; i++)
    {
        double rate = 0;
        for (int r = 0; r < 6; r++) rate += B[r, i] * Solution[r];
        rate *= Scale[i];
        linear += Topology.Groups[i].Lin * rate; angular += Topology.Groups[i].Ang * rate;
    }
    return Finite(linear.LengthSquared()) && Finite(angular.LengthSquared());
}

bool Commit()
{
    if (Runtime.CurrentInstructionCount > Runtime.MaxInstructionCount * 0.88)
    {
        Fault("Instruction budget exhausted before output commit.");
        return false;
    }
    for (int i = 0; i < Topology.Groups.Count; i++) Topology.Set(Topology.Groups[i], LastQdot[i]);
    // Every member of a proven parallel group receives its signed group command.
    return true;
}

void UpdateIntegral(ref Vector3D integral, Vector3D error, double deadband,
    Vector3D clipResidual, Vector3D allocationResidual, double ki,
    double cap, double capture, double dt, bool permit)
{
    Vector3D outside = Outside(error, deadband);
    if (ki <= 0 || !permit || error.Length() > capture)
    {
        if (IntegralUnwindSeconds <= 0) integral = Vector3D.Zero;
        else integral /= 1 + dt / IntegralUnwindSeconds;
        return;
    }
    // A held pose still needs the learned load bias. Reaching the deadband is
    // success, not a reason to leak gravity compensation away.
    integral = ClampLength(integral, cap);
    if (outside.LengthSquared() < 1e-20) return;
    Vector3D delta = outside * (ki * dt);
    // Treat output/limiter clipping and actuator allocation as independent
    // constraints: their residuals may oppose and numerically cancel.
    double scale = Math.Min(IntegralGrowthScale(delta, integral, clipResidual),
        IntegralGrowthScale(delta, integral, allocationResidual));
    integral = ClampLength(integral + delta * scale, cap);
}

static double IntegralGrowthScale(Vector3D delta, Vector3D integral, Vector3D residual)
{
    double growth = Vector3D.Dot(delta, residual);
    if (growth <= 1e-12) return 1;
    // Scale the original increment jointly for both constraints. Sequential
    // vector projections can rotate it back into a previously blocked direction.
    return Clamp(-Vector3D.Dot(integral, residual) / growth, 0, 1);
}

static Vector3D BlendManualCorrection(Vector3D pilotVelocity, Vector3D correction,
    double maximumOppositionFraction, out bool limited)
{
    limited = false;
    double speed = pilotVelocity.Length();
    if (speed <= 1e-12) return correction;
    Vector3D direction = pilotVelocity / speed;
    double component = Vector3D.Dot(correction, direction);
    double minimum = -speed * maximumOppositionFraction;
    if (component >= minimum) return correction;
    limited = true;
    return correction + direction * (minimum - component);
}

double PilotAdvance(Vector3D pilot, Vector3D achievedMinusFeedback)
{
    double d = pilot.LengthSquared();
    if (d < 1e-12) return 1;
    return Clamp(Vector3D.Dot(achievedMinusFeedback, pilot) / d, 0, 1);
}

static double TargetLeadAdvance(Vector3D error, Vector3D step, double limit)
{
    double a = step.LengthSquared();
    if (a < 1e-20) return 1;
    double e = error.LengthSquared(), radius = Math.Max(limit * limit, e);
    if ((error + step).LengthSquared() <= radius) return 1;
    double b = Vector3D.Dot(error, step);
    double root = Math.Sqrt(Math.Max(0, b * b + a * (radius - e)));
    // The alternate root avoids subtracting near-equal positive values.
    return Clamp(b >= 0 ? (root + b > 0 ? (radius - e) / (root + b) : 0) : (root - b) / a, 0, 1);
}

static Vector3D Outside(Vector3D error, double deadband)
{
    double n = error.Length();
    return n > deadband && n > 1e-12 ? error * ((n - deadband) / n) : Vector3D.Zero;
}

void ReadPilot(Vector3D headForward, Vector3D headUp, out Vector3D linear, out Vector3D angular)
{
    linear = angular = Vector3D.Zero;
    Vector3D input; Vector2 mouse; double roll;
    IMyShipController c = PilotInputs(out input, out mouse, out roll);
    if (c == null) return;
    Vector3D forward, left, up;
    if (MovementMode == PilotMode.HEAD)
    {
        forward = headForward; up = headUp;
        if (!Orthonormalise(ref forward, ref up)) throw new Exception("Invalid Head control frame.");
        left = Unit(Vector3D.Cross(up, forward));
    }
    else ControllerAxes(c, out forward, out left, out up);
    Vector3D moveForward = forward, moveUp = up;
    if (MovementMode == PilotMode.VRT) { moveForward = up; moveUp = -forward; }
    Vector3D local = new Vector3D(-input.Z, -input.X, input.Y);
    if (local.LengthSquared() > 1) local = Unit(local);
    linear = (moveForward * local.X + left * local.Y + moveUp * local.Z) * MoveMps;
    // Preserve the established Q/E convention: positive raw roll rotates +Head Forward.
    angular = headForward * roll;
    if (ReadMouse) angular += (left * mouse.X - up * mouse.Y) * MouseSensitivity;
    angular = ClampLength(angular, 1) * (TurnDeg * Math.PI / 180);
}

IMyShipController PilotInputs(out Vector3D move, out Vector2 mouse, out double roll)
{
    move = Vector3D.Zero; mouse = Vector2.Zero; roll = 0;
    var c = FindInputController(); if (c == null) return null;
    if (ReadKeyboard) move = c.MoveIndicator;
    if (ReadMouse) mouse = c.RotationIndicator;
    if (ReadRoll) roll = c.RollIndicator;
    if (!Finite(move.LengthSquared()) || !Finite(mouse.X) || !Finite(mouse.Y) || !Finite(roll))
        throw new Exception("Non-finite pilot input.");
    roll = Clamp(roll, -1, 1); return c;
}

bool HasPilotInput()
{
    Vector3D move; Vector2 mouse; double roll;
    if (PilotInputs(out move, out mouse, out roll) == null) return false;
    // Detect the operator's enabled raw demand, even at Speed=0 or when
    // simultaneous roll/mouse vectors cancel in the selected view frame.
    return move.LengthSquared() > 1e-8 || mouse.LengthSquared() > 1e-8 || Math.Abs(roll) > 1e-4;
}

bool InputScope(IMyShipController c)
{
    return c != null && !c.Closed && c.CubeGrid != null && c.IsSameConstructAs(Me);
}

IMyShipController FindInputController(bool force = false)
{
    if (!force && InputScope(InputController) && InputController.IsUnderControl)
        return InputController;
    if (!force && ++InputScan < 10) return null;
    InputScan = 0;
    InputScratch.Clear();
    GridTerminalSystem.GetBlocksOfType<IMyShipController>(InputScratch, b => InputScope(b) && b.IsUnderControl);
    InputController = null;
    foreach (var b in InputScratch) if (InputController == null || b.EntityId < InputController.EntityId) InputController = b;
    return InputController;
}

void ControllerAxes(IMyShipController c, out Vector3D forward, out Vector3D left, out Vector3D up)
{
    if (!InputScope(c) || !c.IsUnderControl) throw new Exception("HRZ/VRT direction control requires an active cockpit on this construct.");
    MatrixD frame = Topology.Base.WorldMatrix;
    forward = InFrame(c.WorldMatrix.Forward, frame); up = InFrame(c.WorldMatrix.Up, frame);
    if (!Orthonormalise(ref forward, ref up)) throw new Exception("Invalid active cockpit orientation axes.");
    left = Unit(Vector3D.Cross(up, forward));
}

void HoldCurrent()
{

}

void Face(string direction)
{
    if (!Enabled) throw new Exception("Run On before Face.");
    Vector3D forward = new Vector3D(1, 0, 0), left = new Vector3D(0, 1, 0), up = new Vector3D(0, 0, 1);
    if (MovementMode != PilotMode.HEAD) ControllerAxes(FindInputController(true), out forward, out left, out up);
    Vector3D targetForward, targetUp;
    switch (direction.ToLowerInvariant())
    {
        case "forward": targetForward = forward; targetUp = up; break;
        case "back": targetForward = -forward; targetUp = up; break;
        case "left": targetForward = left; targetUp = up; break;
        case "right": targetForward = -left; targetUp = up; break;
        case "down": targetForward = -up; targetUp = forward; break;
        case "up": targetForward = up; targetUp = -forward; break;
        default: throw new Exception("Face expects Forward, Back, Left, Right, Up or Down.");
    }
    if (GoingHome) HoldCurrent();
    if (!Enabled) return;
    TargetF = targetForward; TargetU = targetUp;
    AimingTarget = true;
    Orthonormalise(ref TargetF, ref TargetU);
}

void Step(string axis, double degrees)
{
    if (!Enabled) throw new Exception("Run On before a rotation step.");
    string kind = axis.ToLowerInvariant();
    if (kind != "yaw" && kind != "pitch" && kind != "roll") throw new Exception("Step expects Yaw, Pitch or Roll.");
    Vector3D viewForward = Vector3D.Zero, viewLeft = Vector3D.Zero, viewUp = Vector3D.Zero;
    if (MovementMode != PilotMode.HEAD && kind != "roll")
        ControllerAxes(FindInputController(true), out viewForward, out viewLeft, out viewUp);
    if (GoingHome) HoldCurrent();
    if (!Enabled) return;
    Vector3D a;
    switch (kind)
    {
        case "yaw": a = MovementMode == PilotMode.HEAD ? new Vector3D(0, 0, 1) : viewUp; break;
        case "pitch": a = MovementMode == PilotMode.HEAD ? -Unit(Vector3D.Cross(TargetU, TargetF)) : -viewLeft; break;
        default: a = TargetF; break;
    }
    Vector3D step = Unit(a) * (degrees * Math.PI / 180);
    TargetF = Rotate(TargetF, step); TargetU = Rotate(TargetU, step);
    AimingTarget = true;
    Orthonormalise(ref TargetF, ref TargetU);
}

void RunCommands(string argument)
{
    string[] commands = argument.Replace("\r", "").Split(new char[] { ';', '\n' }, StringSplitOptions.RemoveEmptyEntries);
    foreach (string raw in commands) RunCommand(raw.Trim());
}

void RunCommand(string command)
{
    if (command.StartsWith("MiningArm ", StringComparison.OrdinalIgnoreCase)) command = command.Substring(10).Trim();
    string[] w = command.Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
    if (w.Length == 0) return;
    string c = w[0].ToLowerInvariant();
    if (c.Length == 1 && c[0] >= '1' && c[0] <= '9') { NumericAction(c[0] - '0'); return; }
    switch (c)
    {
        case "version": Say(Banner()); break;
        case "on": EnableControl(); break;
        case "off": case "stop": StopMotion(c == "stop" ? "Stop command." : "Off command."); Say(Status); break;
        case "toggle": if (Enabled) { StopMotion("Toggle OFF."); Say(Status); } else EnableControl(); break;
        case "onoff": case "-oo":
            if (w.Length < 2) throw new Exception("OnOff expects On, Off or Toggle.");
            if (w[1].Equals("on", StringComparison.OrdinalIgnoreCase)) EnableControl();
            else if (w[1].Equals("off", StringComparison.OrdinalIgnoreCase)) { StopMotion("Off command."); Say(Status); }
            else if (w[1].Equals("toggle", StringComparison.OrdinalIgnoreCase)) { if (Enabled) { StopMotion("Toggle OFF."); Say(Status); } else EnableControl(); }
            else throw new Exception("OnOff expects On, Off or Toggle.");
            break;
        case "hold": HoldCurrent(); break;
        case "check": Check(); break;
        case "rescan": Discover(true); break;
        case "reload": Reload(); break;
        case "info": case "headinfo": case "help": PrintInfo(); break;
        case "pivotprobe":
            DisplayPinned = true;
            if (PivotProbe == null) PivotProbe = new PivotProbeTool(this, Topology);
            PivotProbe.Run(w);
            break;
        case "status": DisplayPinned = false; StatusElapsed = StatusRefreshSeconds; Say("Live status display resumed."); break;
        case "settings": PrintInfo(); break;
        case "joints": PrintJoints(); break;
        case "bindings": PrintBindings(); break;
        case "sethome": case "-sh": CaptureHome(); break;
        case "gohome": case "-gh": BeginHome(); break;
        case "homeinfo": PrintHome(); break;
        case "speed": case "-s": MoveMps = Number(w, 1, 0, 5, "Speed"); break;
        case "readkeyboard": case "-rk": ReadKeyboard = InputToggle(w, ReadKeyboard); break;
        case "readroll": ReadRoll = InputToggle(w, ReadRoll); break;
        case "readmouse": case "-rm": ReadMouse = InputToggle(w, ReadMouse); break;
        case "mode": case "movementmode": case "controlmode":
            if (w.Length != 2) throw new Exception("MovementMode expects HEAD, HRZ or VRT.");
            SetMovementMode(w[1]); break;
        case "orientationtolerance":
            double tolerance = Number(w, 1, 0, MaximumOrientationToleranceDeg, "OrientationTolerance");
            StopMotion("Stopped: OrientationTolerance changed; run On to recapture.");
            OrientationToleranceDeg = tolerance;
            Say(Status);
            break;
        case "turnspeed": TurnDeg = Number(w, 1, 0.01, 30, "TurnSpeed"); break;
        case "stepsize": case "stepangle": StepDeg = Number(w, 1, 0.01, 180, "StepSize"); break;
        case "jointspeed": JointRpm = Number(w, 1, 0.01, 5, "JointSpeed"); break;
        case "pistonspeed": PistonMps = Number(w, 1, 0.01, 2, "PistonSpeed"); break;
        case "homespeed": HomeJointRpm = Number(w, 1, 0.01, 5, "Home joint RPM"); HomePistonMps = Number(w, 2, 0.01, 2, "Home piston m/s"); break;
        case "face": if (w.Length < 2) throw new Exception("Face expects a direction."); Face(w[1]); break;
        case "forward": case "back": case "left": case "right": case "up": case "down": Face(c); break;
        case "step": if (w.Length < 2) throw new Exception("Step expects an axis."); Step(w[1], w.Length > 2 ? Parse(w[2]) : StepDeg); break;
        case "yaw": Step("yaw", Number(w, 1, -180, 180, "Yaw")); break;
        case "pitch": Step("pitch", Number(w, 1, -180, 180, "Pitch")); break;
        case "roll": Step("roll", Number(w, 1, -180, 180, "Roll")); break;
        case "yawleft": Step("yaw", StepDeg); break;
        case "yawright": Step("yaw", -StepDeg); break;
        case "pitchup": Step("pitch", StepDeg); break;
        case "pitchdown": Step("pitch", -StepDeg); break;
        case "rollleft": Step("roll", -StepDeg); break;
        case "rollright": Step("roll", StepDeg); break;
        case "dumptrace": PrintInfo(); Say("Compact v2 keeps live diagnostics in Info; Custom Data remains configuration-safe."); break;
        case "traceclear": Say("No persistent trace buffer is retained in compact v2."); break;
        case "move": case "moveto": case "relmove": case "runfrompanel":
            throw new Exception(c + " is not supported by AutoArm's discovered pose control.");
        default: throw new Exception("Unknown command '" + w[0] + "'. Use Info.");
    }
}

static bool InputToggle(string[] words, bool current)
{
    if (words.Length != 2) throw new Exception(words[0] + " expects On, Off or Toggle.");
    string value = words[1].ToLowerInvariant();
    if (value == "on") return true;
    if (value == "off") return false;
    if (value == "toggle") return !current;
    throw new Exception(words[0] + " expects On, Off or Toggle.");
}

void NumericAction(int n)
{
    string action = AutoSettings.DefaultBindings[n - 1].ToLowerInvariant();
    switch (action)
    {
        case "none": return;
        case "forward": case "back": case "left": case "right": case "up": case "down": Face(action); return;
        default: RunCommand(action); return;
    }
}

void Reload()
{
    StopMotion("Reload command.");
    PivotProbe = null;
    SetupElapsed = 0;
    string updated, why;
    bool changed;
    if (Topology.Groups.Count > 0 && !Topology.Validate(out why, true))
    {
        ConfigValid = false;
        Say("Reload refused: " + why + " Run Rescan after repairing topology.");
        return;
    }
    if (!AutoConfig.LoadOrGenerate(this, Topology, Me.CustomData, out updated, out changed, out why))
    {
        ConfigValid = false;
        Status = "Reload refused without changing Custom Data: " + why;
        Say(Status);
        return;
    }
    ConfigValid = true;
    Allocate(Topology.Groups.Count);
    LoadHome();
    Status = "Configuration reloaded; controller remains OFF.";
    Say(Status);
}

void CaptureHome()
{
    StopMotion("SetHome command.");
    if (!ConfigValid) { Say("SetHome refused: fix Custom Data, then Reload."); return; }
    string why = "topology is not ready";
    if (!Topology.Ready || !Topology.Validate(out why, true)) { Say("SetHome refused: " + why); return; }

    HomeFingerprint = Topology.Fingerprint;
    WriteHome();

}

void LoadHome()
{
    Home.Clear(); HomeFingerprint = "";
    if (string.IsNullOrWhiteSpace(Storage)) return;
    var ini = new MyIni();
    MyIniParseResult result;
    if (!ini.TryParse(Storage, out result)) return;
    string section = "AutoArm Home";
    if (!ini.ContainsSection(section))
    {
        var sections = new List<string>(); ini.GetSections(sections);
        foreach (string candidate in sections)
            if (candidate.EndsWith(" Home", StringComparison.Ordinal) && ini.Get(candidate, "Fingerprint").ToString() == Topology.Fingerprint)
            {
                if (section != "AutoArm Home") return;
                section = candidate;
            }
    }
    string fingerprint = ini.Get(section, "Fingerprint").ToString("");
    if (fingerprint != Topology.Fingerprint) return;
    foreach (AutoGroup g in Topology.Groups) foreach (AutoMember m in g.M)
    {
        string key = "E" + m.B.EntityId;
        double value = ini.Get(section, key).ToDouble(double.NaN);
        if (!Finite(value)) { Home.Clear(); return; }
        Home[m.B.EntityId] = value;
    }
    HomeFingerprint = fingerprint;
    if (section != "AutoArm Home")
    {
        ini.DeleteSection(section); Storage = ini.ToString(); WriteHome();
    }
}

void WriteHome()
{
    var ini = AutoConfig.Saved(Storage);
    ini.DeleteSection("AutoArm Home");
    ini.Set("AutoArm Home", "Fingerprint", HomeFingerprint);
    foreach (var pair in Home) ini.Set("AutoArm Home", "E" + pair.Key, pair.Value.ToString("R", System.Globalization.CultureInfo.InvariantCulture));
    Storage = ini.ToString();
}

void BeginHome()
{
    StopMotion("");
    if (!ConfigValid) { Say("GoHome refused: fix Custom Data, then Reload."); return; }
    if (!Topology.Ready || HomeFingerprint != Topology.Fingerprint)
    { Say("GoHome refused: use SetHome for the current topology."); return; }
    string why;
    if (!Topology.Validate(out why, true)) { Say("GoHome refused: " + why); return; }
    foreach (AutoGroup g in Topology.Groups) foreach (AutoMember m in g.M)
    {
        double resolved;
        if (!ResolveHome(m, Home[m.B.EntityId], out resolved))
        { Say("GoHome refused: saved target is outside current limits at " + m.B.CustomName + "."); return; }
        m.Home = resolved;
    }
    Enabled = GoingHome = true;
    PivotProbe = null;
    ControlElapsed = 0;
    Runtime.UpdateFrequency = UpdateFrequency.Update1;
    Status = "GoHome in progress; live pilot input cancels.";
    Say(Status);
}

void UpdateHome(double dt)
{
    if (HasPilotInput()) { Fault("GoHome cancelled by pilot input."); return; }
    string why;
    if (!Topology.Validate(out why)) { Fault("GoHome topology fault: " + why); return; }
    bool complete = true;
    for (int i = 0; i < Topology.Groups.Count; i++)
    {
        AutoGroup g = Topology.Groups[i];
        double error = 0, branch = 0;
        bool haveBranch = false;
        double agreement = g.Rotary ? 0.5 * Math.PI / 180 : 0.005;
        // A limited member fixes the legal native-angle branch. Unlimited peers
        // may then select the equivalent circular error nearest that branch.
        if (g.Rotary) foreach (AutoMember m in g.M)
        {
            IMyMotorStator r = (IMyMotorStator)m.B;
            if (r.LowerLimitRad < -1000000 && r.UpperLimitRad > 1000000) continue;
            double e = m.Sign * MemberHomeError(m, true);
            if (!haveBranch) { branch = e; haveBranch = true; }
            else if (Math.Abs(e - branch) > agreement) { Fault("GoHome limited-member routes disagree at " + g.Label); return; }
        }
        foreach (AutoMember m in g.M)
        {
            double e = m.Sign * MemberHomeError(m, g.Rotary);
            bool unlimited = false;
            if (g.Rotary)
            {
                IMyMotorStator r = (IMyMotorStator)m.B;
                unlimited = r.LowerLimitRad < -1000000 && r.UpperLimitRad > 1000000;
            }
            if (!haveBranch) { branch = e; haveBranch = true; }
            else if (unlimited) e = branch + Wrap(e - branch);
            if (Math.Abs(e - branch) > agreement) { Fault("GoHome member errors disagree at " + g.Label); return; }
            error += e;
        }
        error /= g.M.Count;
        double tolerance = g.Rotary ? 0.25 * Math.PI / 180 : 0.005;
        if (Math.Abs(error) > tolerance) complete = false;
        double lo, hi;
        if (!Topology.Refresh(g, Topology.Head.GetPosition(), Topology.Base.WorldMatrix, out lo, out hi, out why))
        { Fault("GoHome: " + why); return; }
        double cap = g.Rotary ? HomeJointRpm * Math.PI / 30 : HomePistonMps;
        double lower = Math.Max(lo, -cap), upper = Math.Min(hi, cap);
        if (lower > upper) { Fault("GoHome has no common velocity interval at " + g.Label); return; }
        double velocity = Math.Abs(error) <= tolerance ? 0 : Clamp(error * 0.5, lower, upper);
        double accel = g.Rotary ? JointAccelerationRpmPerSec * Math.PI / 30 : PistonAccelerationMps2;
        double centre = Clamp(LastQdot[i], lower, upper);
        velocity = Clamp(velocity, Math.Max(lower, centre - accel * dt), Math.Min(upper, centre + accel * dt));
        LastQdot[i] = velocity;
    }
    if (complete)
    {
        StopMotion("GoHome complete.");
        Say(Status);
    }
    else Commit();
}

bool ResolveHome(AutoMember m, double requested, out double resolved)
{
    resolved = requested;
    IMyPistonBase p = m.B as IMyPistonBase;
    if (p != null)
    {
        if (!Finite(resolved) || resolved < p.MinLimit - 0.005 || resolved > p.MaxLimit + 0.005) return false;
        resolved = Clamp(resolved, p.MinLimit, p.MaxLimit);
        return true;
    }
    IMyMotorStator r = (IMyMotorStator)m.B;
    if (!Finite(resolved)) return false;
    double lo = r.LowerLimitRad, hi = r.UpperLimitRad, turn = 2 * Math.PI;
    double k = Math.Round((r.Angle - requested) / turn);
    double kMin = lo < -1000000 ? double.NegativeInfinity : Math.Ceiling((lo - requested - 0.25 * Math.PI / 180) / turn);
    double kMax = hi > 1000000 ? double.PositiveInfinity : Math.Floor((hi - requested + 0.25 * Math.PI / 180) / turn);
    if (kMin > kMax) return false;
    resolved = requested + Clamp(k, kMin, kMax) * turn;
    if (!Finite(resolved) || resolved < lo - 0.25 * Math.PI / 180 || resolved > hi + 0.25 * Math.PI / 180) return false;
    resolved = Clamp(resolved, lo, hi);
    return true;
}

double MemberHomeError(AutoMember m, bool rotary)
{
    double error = m.Home - MemberPosition(m);
    if (!rotary) return error;
    IMyMotorStator r = (IMyMotorStator)m.B;
    return r.LowerLimitRad < -1000000 && r.UpperLimitRad > 1000000 ? Wrap(error) : error;
}

double MemberPosition(AutoMember m)
{
    IMyMotorStator rotor = m.B as IMyMotorStator;
    if (rotor != null) return rotor.Angle;
    IMyPistonBase piston = m.B as IMyPistonBase;
    if (piston != null) return piston.CurrentPosition;
    throw new Exception("Unsupported owned actuator type.");
}

double Number(string[] words, int index, double min, double max, string label)
{
    if (words.Length <= index) throw new Exception(label + " needs a value.");
    double value = Parse(words[index]);
    if (value < min || value > max) throw new Exception(label + " must be " + min + ".." + max + ".");
    Say(label + " = " + value.ToString("0.###") + ".");
    return value;
}

double Parse(string text)
{
    double value;
    if (!double.TryParse(text, System.Globalization.NumberStyles.Float,
        System.Globalization.CultureInfo.InvariantCulture, out value) || !Finite(value))
        throw new Exception("Expected a finite number with decimal point.");
    return value;
}

static bool Cholesky(double[,] a, double[] b, double[,] l, double[] temp, double[] answer)
{
    for (int i = 0; i < 6; i++) for (int j = 0; j <= i; j++)
    {
        double value = a[i, j];
        for (int k = 0; k < j; k++) value -= l[i, k] * l[j, k];
        if (i == j)
        {
            if (!Finite(value) || value <= 1e-15) return false;
            l[i, j] = Math.Sqrt(value);
        }
        else l[i, j] = value / l[j, j];
    }
    for (int i = 0; i < 6; i++)
    {
        double value = b[i];
        for (int j = 0; j < i; j++) value -= l[i, j] * temp[j];
        temp[i] = value / l[i, i];
    }
    for (int i = 5; i >= 0; i--)
    {
        double value = temp[i];
        for (int j = i + 1; j < 6; j++) value -= l[j, i] * answer[j];
        answer[i] = value / l[i, i];
        if (!Finite(answer[i])) return false;
    }
    return true;
}

static Vector3D RotationError(Vector3D f, Vector3D u, Vector3D tf, Vector3D tu)
{
    if (!Orthonormalise(ref f, ref u) || !Orthonormalise(ref tf, ref tu)) return new Vector3D(double.NaN);
    Vector3D l = Vector3D.Cross(u, f), tl = Vector3D.Cross(tu, tf);
    double d00 = tf.X * f.X + tl.X * l.X + tu.X * u.X,
        d01 = tf.X * f.Y + tl.X * l.Y + tu.X * u.Y,
        d02 = tf.X * f.Z + tl.X * l.Z + tu.X * u.Z;
    double d10 = tf.Y * f.X + tl.Y * l.X + tu.Y * u.X,
        d11 = tf.Y * f.Y + tl.Y * l.Y + tu.Y * u.Y,
        d12 = tf.Y * f.Z + tl.Y * l.Z + tu.Y * u.Z;
    double d20 = tf.Z * f.X + tl.Z * l.X + tu.Z * u.X,
        d21 = tf.Z * f.Y + tl.Z * l.Y + tu.Z * u.Y,
        d22 = tf.Z * f.Z + tl.Z * l.Z + tu.Z * u.Z;
    Vector3D skew = new Vector3D(d21 - d12, d02 - d20, d10 - d01) * 0.5;
    double angle = Math.Acos(Clamp((d00 + d11 + d22 - 1) * 0.5, -1, 1));
    if (angle < 1e-7) return skew;
    if (Math.PI - angle > 1e-5) return skew * (angle / Math.Sin(angle));
    // Near pi, the skew is ill-conditioned. Extract an eigenaxis from the full
    // relative rotation, selecting the largest diagonal for numeric stability.
    double x, y, z;
    if (d00 >= d11 && d00 >= d22)
    {
        x = Math.Sqrt(Math.Max(0, (d00 + 1) * 0.5));
        y = (d01 + d10) / (4 * Math.Max(x, 1e-10));
        z = (d02 + d20) / (4 * Math.Max(x, 1e-10));
    }
    else if (d11 >= d22)
    {
        y = Math.Sqrt(Math.Max(0, (d11 + 1) * 0.5));
        x = (d01 + d10) / (4 * Math.Max(y, 1e-10));
        z = (d12 + d21) / (4 * Math.Max(y, 1e-10));
    }
    else
    {
        z = Math.Sqrt(Math.Max(0, (d22 + 1) * 0.5));
        x = (d02 + d20) / (4 * Math.Max(z, 1e-10));
        y = (d12 + d21) / (4 * Math.Max(z, 1e-10));
    }
    Vector3D axis = Unit(new Vector3D(x, y, z));
    if (skew.LengthSquared() > 1e-16 && Vector3D.Dot(axis, skew) < 0) axis = -axis;
    return Unit(axis) * angle;
}

static Vector3D Rotate(Vector3D value, Vector3D rotation)
{
    double angle = rotation.Length();
    if (angle < 1e-12) return value;
    Vector3D axis = rotation / angle;
    return value * Math.Cos(angle) + Vector3D.Cross(axis, value) * Math.Sin(angle) +
        axis * Vector3D.Dot(axis, value) * (1 - Math.Cos(angle));
}

static bool Orthonormalise(ref Vector3D f, ref Vector3D u)
{
    double fn = f.Length();
    if (!Finite(fn) || fn < 1e-10) return false;
    f /= fn;
    u -= f * Vector3D.Dot(u, f);
    double un = u.Length();
    if (!Finite(un) || un < 1e-10) return false;
    u /= un;
    return true;
}

static Vector3D Unit(Vector3D value)
{
    double n = value.Length();
    if (!Finite(n) || n < 1e-12) throw new Exception("Invalid zero/non-finite vector.");
    return value / n;
}

static Vector3D ClampLength(Vector3D value, double cap)
{
    double n = value.Length();
    if (!Finite(n)) return new Vector3D(double.NaN);
    return n > cap && n > 1e-12 ? value * (cap / n) : value;
}

static Vector3D InFrame(Vector3D value, MatrixD frame)
{
    return new Vector3D(Vector3D.Dot(value, frame.Forward), Vector3D.Dot(value, frame.Left), Vector3D.Dot(value, frame.Up));
}

static double Clamp(double value, double lo, double hi)
{
    return Math.Max(lo, Math.Min(hi, value));
}

static double Wrap(double angle)
{
    return angle - Math.Floor((angle + Math.PI) / (2 * Math.PI)) * (2 * Math.PI);
}

static bool Finite(double value)
{
    return !double.IsNaN(value) && !double.IsInfinity(value);
}


// Passive RAM-only pivot probe. The player supplies all motion. Configuration
// and Storage are untouched; existing topology-fault handling can stop drives.
class PivotProbeTool
{
    const double MinA=Math.PI/180,MaxA=15*Math.PI/180,OtherA=.1*Math.PI/180;
    const double PairA=.1*Math.PI/180,OtherP=.001,MinTravel=.01,AxisCos=.999993908;
    readonly Program P; readonly AutoTopology T; readonly List<Result> R=new List<Result>();
    Shot A; int G=-1; string Fingerprint="";
    class Shot { public Vector3D P,F,U; public double[][] Q; public Vector3D[] Axis,Base,Top; }
    public class Result
    {
        public int Group; public double Delta,Travel,Orientation;
        public readonly double[] Error=new double[3],Spread=new double[3]; public string Verdict;
    }
    public IList<Result> Trials { get { return R.AsReadOnly(); } }
    public PivotProbeTool(Program owner,AutoTopology topology){P=owner;T=topology;}

    public void Run(string[] w)
    {
        int n=w.Length>0&&w[0].Equals("pivotprobe",StringComparison.OrdinalIgnoreCase)?1:0;
        if(w.Length<=n){Help();return;} string c=w[n].ToLowerInvariant();
        if(c=="start"&&w.Length==n+2){int g;if(!int.TryParse(w[n+1],out g)){Say("Group must be an integer.");return;}Start(g-1);return;}
        if(c=="end"&&w.Length==n+1){End();return;} if(c=="report"&&w.Length==n+1){Report();return;}
        if(c=="reset"&&w.Length==n+1){Clear();R.Clear();Say("PivotProbe RAM cleared; no blocks changed.");return;} Help();
    }
    void Start(int g)
    {
        string why;if(!Ready(out why)){Say("PivotProbe refused: "+why);return;}
        if(A!=null){Say("A Start snapshot exists; run End or Reset.");return;}
        if(g<0||g>=T.Groups.Count){Say("Group must be 1.."+T.Groups.Count+".");return;}
        if(!T.Groups[g].Rotary){Say("Only rotary/hinge groups have a pivot.");return;}
        Shot s;if(!Capture(g,out s,out why)){Say("PivotProbe refused: "+why);return;}
        A=s;G=g;Fingerprint=T.Fingerprint;Say("Captured G"+(g+1)+". Manually jog ONLY this group 1..15 deg, zero all targets, settle, then run PivotProbe End. Parallel members must follow discovered signs.");
    }
    void End()
    {
        if(A==null){Say("Run PivotProbe Start first.");return;}string why;
        if(!Ready(out why)){Say("End refused: "+why);return;}
        if(Fingerprint!=T.Fingerprint){Clear();Say("Inconclusive: topology changed.");return;}
        Shot b;if(!Capture(G,out b,out why)){Clear();Say("Inconclusive: "+why);return;}
        Result r;if(!Analyse(A,b,G,out r,out why)){Clear();Say("Inconclusive: "+why);return;}
        if(R.Count==12)R.RemoveAt(0);R.Add(r);Clear();Say(Format(r));
    }
    bool Ready(out string why)
    {
        why="";if(P.Enabled||P.GoingHome){why="controller must already be Off; probe will not stop it.";return false;}
        if(!T.Ready){why="topology is not ready.";return false;}if(!T.Validate(out why,true))return false;
        foreach(AutoGroup g in T.Groups)foreach(AutoMember m in g.M){var p=m.B as IMyPistonBase;double v=p!=null?p.Velocity:((IMyMotorStator)m.B).TargetVelocityRad;if(!Finite(v)||Math.Abs(v)>1e-4){why="owned target not zero: "+m.B.CustomName;return false;}}
        return true;
    }
    bool Capture(int selected,out Shot s,out string why)
    {
        s=new Shot();why="";MatrixD f=T.Base.WorldMatrix;Vector3D o=T.Base.GetPosition();
        s.P=Point(T.Head.GetPosition(),o,f);s.F=Vec(T.Head.WorldMatrix.Forward,f);s.U=Vec(T.Head.WorldMatrix.Up,f);
        if(!Good(s.P)||!Unit(ref s.F)||!Unit(ref s.U)){why="invalid Base/Head pose.";return false;}
        s.Q=new double[T.Groups.Count][];
        for(int i=0;i<T.Groups.Count;i++){s.Q[i]=new double[T.Groups[i].M.Count];for(int j=0;j<s.Q[i].Length;j++){s.Q[i][j]=T.Position(T.Groups[i].M[j]);if(!Finite(s.Q[i][j])){why="non-finite joint position.";return false;}}}
        AutoGroup g=T.Groups[selected];int count=g.M.Count;s.Axis=new Vector3D[count];s.Base=new Vector3D[count];s.Top=new Vector3D[count];
        for(int i=0;i<count;i++)
        {
            AutoMember m=g.M[i];IMyMotorStator r=m.B as IMyMotorStator;
            if(r==null||r.RotorLock||m.B.Top==null){why="selected member unavailable/locked.";return false;}
            double q=s.Q[selected][i],margin=.25*Math.PI/180;
            if(!Finite(r.LowerLimitRad)||!Finite(r.UpperLimitRad)||r.LowerLimitRad>r.UpperLimitRad){why="selected limits are invalid.";return false;}
            if((r.LowerLimitRad>-1e6&&q<=r.LowerLimitRad+margin)||(r.UpperLimitRad<1e6&&q>=r.UpperLimitRad-margin)){why="selected member near limit: "+m.B.CustomName;return false;}
            int dir=m.B.CubeGrid.EntityId==m.U&&m.B.TopGrid.EntityId==m.V?1:m.B.CubeGrid.EntityId==m.V&&m.B.TopGrid.EntityId==m.U?-1:0;
            if(dir==0){why="selected endpoints changed.";return false;}
            s.Axis[i]=Vec(m.B.WorldMatrix.Up*(-dir*m.Sign),f);s.Base[i]=Point(m.B.GetPosition(),o,f);s.Top[i]=Point(m.B.Top.GetPosition(),o,f);
            if(!Unit(ref s.Axis[i])||!Good(s.Base[i])||!Good(s.Top[i])){why="invalid axis/pivot candidate.";return false;}
        }return true;
    }
    bool Analyse(Shot a,Shot b,int group,out Result result,out string why)
    {
        result=null;why="";
        for(int i=0;i<T.Groups.Count;i++)if(i!=group)for(int j=0;j<T.Groups[i].M.Count;j++){double d=b.Q[i][j]-a.Q[i][j];if(T.Groups[i].Rotary)d=Wrap(d);if(Math.Abs(d)>(T.Groups[i].Rotary?OtherA:OtherP)){why="another group moved: G"+(i+1);return false;}}
        AutoGroup g=T.Groups[group];double first=0,total=0,pair=0;
        for(int i=0;i<g.M.Count;i++)
        {
            double d=Wrap(b.Q[group][i]-a.Q[group][i])*g.M[i].Sign;if(i==0)first=d;else{d=first+Wrap(d-first);pair=Math.Max(pair,Math.Abs(d-first));}total+=d;
            if(Vector3D.Dot(a.Axis[0],a.Axis[i])<AxisCos||Vector3D.Dot(a.Axis[i],b.Axis[i])<AxisCos){why="axis/member consistency exceeds 0.2 deg.";return false;}
        }
        double delta=total/g.M.Count;if(pair>PairA){why="parallel delta spread exceeds 0.1 deg.";return false;}
        if(Math.Abs(delta)<MinA||Math.Abs(delta)>MaxA){why="motion must be 1..15 deg; measured "+Deg(delta).ToString("0.000");return false;}
        double travel=(b.P-a.P).Length();if(travel<MinTravel){why="Head travel below 0.01 m.";return false;}
        double ori=Math.Max(Angle(Rot(a.F,a.Axis[0],delta),b.F),Angle(Rot(a.U,a.Axis[0],delta),b.U));
        if(!Finite(ori)||ori>1){why="orientation residual "+ori.ToString("0.000")+" deg suggests slip/other motion.";return false;}
        result=new Result{Group=group,Delta=delta,Travel=travel,Orientation=ori};
        for(int k=0;k<3;k++)
        {
            Vector3D mean=Vector3D.Zero;var predicted=new Vector3D[g.M.Count];
            for(int i=0;i<g.M.Count;i++){Vector3D c=Candidate(a,i,k);predicted[i]=c+Rot(a.P-c,a.Axis[i],delta);mean+=predicted[i];}mean/=g.M.Count;
            for(int i=0;i<predicted.Length;i++)result.Spread[k]=Math.Max(result.Spread[k],(predicted[i]-mean).Length());
            result.Error[k]=(b.P-mean).Length();if(!Finite(result.Error[k])||!Finite(result.Spread[k])){why="non-finite residual.";return false;}
        }
        int best=0;for(int k=1;k<3;k++)if(result.Error[k]<result.Error[best])best=k;double next=double.PositiveInfinity;
        for(int k=0;k<3;k++)if(k!=best)next=Math.Min(next,result.Error[k]);
        result.Verdict=result.Spread[best]>.01?"inconclusive: counterpart predictions disagree":next-result.Error[best]<.001?"axis-equivalent/inconclusive":Name(best)+" smallest (not a calibration)";return true;
    }
    void Report()
    {
        if(R.Count==0){Say("No accepted trials. Capture positive and negative manual motions.");return;}
        var s=new System.Text.StringBuilder("PIVOT PROBE RAM REPORT - no pivot changed\n");for(int i=0;i<R.Count;i++)s.Append('#').Append(i+1).Append(' ').Append(Format(R[i])).Append('\n');
        s.Append("Collect both directions; residuals are evidence, not proof. Flex/load/slip can dominate.");Say(s.ToString());
    }
    string Format(Result r){return "G"+(r.Group+1)+" d="+Deg(r.Delta).ToString("0.000")+"deg travel="+r.Travel.ToString("0.0000")+"m ori="+r.Orientation.ToString("0.000")+"deg\n Base e/spread="+Row(r,0)+" Top="+Row(r,1)+" Mid="+Row(r,2)+" RESULT="+r.Verdict;}
    string Row(Result r,int k){return r.Error[k].ToString("0.0000")+"/"+r.Spread[k].ToString("0.0000")+"m";}
    Vector3D Candidate(Shot s,int i,int k){return k==0?s.Base[i]:k==1?s.Top[i]:(s.Base[i]+s.Top[i])*.5;}
    string Name(int k){return k==0?"Base center":k==1?"Top center":"Midpoint";}
    void Clear(){A=null;G=-1;Fingerprint="";}void Help(){Say("PivotProbe Start <group> | End | Report | Reset");}void Say(string s){P.Say(s);}
    static double Wrap(double a){return a-Math.Floor((a+Math.PI)/(2*Math.PI))*(2*Math.PI);}static double Deg(double a){return a*180/Math.PI;}
    static bool Finite(double v){return !double.IsNaN(v)&&!double.IsInfinity(v);}static bool Good(Vector3D v){return Finite(v.X)&&Finite(v.Y)&&Finite(v.Z);}
    static bool Unit(ref Vector3D v){double n=v.Length();if(!Good(v)||!Finite(n)||n<1e-10)return false;v/=n;return true;}
    static Vector3D Point(Vector3D p,Vector3D o,MatrixD f){return Vec(p-o,f);}static Vector3D Vec(Vector3D v,MatrixD f){return new Vector3D(Vector3D.Dot(v,f.Forward),Vector3D.Dot(v,f.Left),Vector3D.Dot(v,f.Up));}
    static Vector3D Rot(Vector3D v,Vector3D a,double x){return v*Math.Cos(x)+Vector3D.Cross(a,v)*Math.Sin(x)+a*Vector3D.Dot(a,v)*(1-Math.Cos(x));}
    static double Angle(Vector3D a,Vector3D b){double n=a.Length()*b.Length();return n>1e-12&&Finite(n)?Math.Acos(Math.Max(-1,Math.Min(1,Vector3D.Dot(a,b)/n)))*180/Math.PI:double.NaN;}
}
