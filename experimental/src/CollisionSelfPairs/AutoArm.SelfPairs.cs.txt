// Experimental conservative self-pair reachability filter. No actuator writes.
sealed class SelfJoint {
    public IMyMechanicalConnectionBlock Block;
    public long From, To, Top;
    public double Lower, Upper;
    public Vector3D LocalPivot, LocalAxis;
}
sealed class SelfBall { public Vector3D Centre; public double Radius; }
sealed class SelfPair {
    public long A, B, Anchor;
    public SelfBall First, Second;
    public double Gap;
}
sealed class SelfTable {
    public string Stamp;
    public long Revision;
    public readonly List<SelfJoint> Joints = new List<SelfJoint>();
    public readonly Dictionary<long, List<SelfJoint>> Paths = new Dictionary<long, List<SelfJoint>>();
    public readonly Dictionary<long, SelfBall> Geometry = new Dictionary<long, SelfBall>();
    public readonly Dictionary<long, Vector3I[]> Bounds = new Dictionary<long, Vector3I[]>();
    public readonly Dictionary<long, double> Sizes = new Dictionary<long, double>();
    public readonly List<SelfPair> Pairs = new List<SelfPair>();
    public readonly Dictionary<string, SelfPair> Proven = new Dictionary<string, SelfPair>();
    public int Cursor;
    public bool Supported;
}
readonly Dictionary<string, SelfTable> SelfTables = new Dictionary<string, SelfTable>();
SelfTable ActiveSelfTable;
string SelfPairMode = "Shadow";
long SelfGeometryRevision;
int SelfProven, SelfCandidates, SelfSkipped, SelfWouldSkip, SelfInvalidations;
double SelfSlack = .05;
void ConfigureSelfPairs(MyIni ini) {
    string mode = ini.Get("Collision", "SelfPairs").ToString("Shadow");
    if (mode != "Off" && mode != "Shadow" && mode != "On") throw new Exception("SelfPairs must be Off, Shadow or On.");
    SelfPairMode = mode; SelfTables.Clear(); ActiveSelfTable = null; SelfInvalidations = 0;
}
string SelfPairInfo() {
    return "\nSelfPairs " + SelfPairMode + ": " + SelfProven + "/" + SelfCandidates + " proven; " +
        SelfWouldSkip + " eligible / " + SelfSkipped + " skipped this query; rebuilds " + SelfInvalidations;
}
static string SelfKey(long a, long b) { return a < b ? a + "/" + b : b + "/" + a; }
bool SelfWork() { return Runtime.MaxInstructionCount <= 0 || Runtime.CurrentInstructionCount < Runtime.MaxInstructionCount * .20; }
static bool SelfVector(Vector3D v) { return Finite(v.X) && Finite(v.Y) && Finite(v.Z); }
static bool SelfFrame(MatrixD m) {
    return SelfVector(m.Translation) && SelfVector(m.Right) && SelfVector(m.Up) && SelfVector(m.Backward) &&
        Math.Abs(m.Right.LengthSquared() - 1) < 1e-5 && Math.Abs(m.Up.LengthSquared() - 1) < 1e-5 &&
        Math.Abs(m.Backward.LengthSquared() - 1) < 1e-5 && Math.Abs(Vector3D.Dot(m.Right, m.Up)) < 1e-5 &&
        Math.Abs(Vector3D.Dot(m.Up, m.Backward)) < 1e-5 && Math.Abs(Vector3D.Dot(m.Backward, m.Right)) < 1e-5;
}
bool SelfLimits(IMyMechanicalConnectionBlock block, out double lower, out double upper) {
    var rotor = block as IMyMotorStator; var piston = block as IMyPistonBase;
    lower = rotor != null ? rotor.LowerLimitRad : piston != null ? piston.MinLimit : double.NaN;
    upper = rotor != null ? rotor.UpperLimitRad : piston != null ? piston.MaxLimit : double.NaN;
    return Finite(lower) && Finite(upper) && lower <= upper && (rotor != null || lower >= 0 && upper <= 1000);
}
SelfTable CaptureSelfTable(Arm arm) {
    var table = new SelfTable { Stamp = arm.Fingerprint, Revision = SelfGeometryRevision };
    var incoming = new Dictionary<long, SelfJoint>();
    foreach (var group in arm.Groups) {
        // Branched/parallel and reversed connections retain all live checks in
        // this prototype. A serial proof must not invent independent closures.
        if (group.Members.Count != 1) return table;
        var member = group.Members[0]; var block = member.Block; double lower, upper;
        if (!Attached(block) || block.CubeGrid.EntityId != member.From || block.TopGrid.EntityId != member.To ||
            incoming.ContainsKey(member.To) || !SelfLimits(block, out lower, out upper) || !SelfFrame(block.CubeGrid.WorldMatrix)) return table;
        var inverse = MatrixD.Invert(block.CubeGrid.WorldMatrix);
        var joint = new SelfJoint { Block = block, From = member.From, To = member.To, Top = block.Top.EntityId,
            Lower = lower, Upper = upper, LocalPivot = Vector3D.Transform(block.GetPosition(), inverse),
            LocalAxis = Vector3D.TransformNormal(block.WorldMatrix.Up, inverse) };
        if (!SelfVector(joint.LocalPivot) || !SelfVector(joint.LocalAxis) || Math.Abs(joint.LocalAxis.LengthSquared() - 1) > 1e-5) return table;
        incoming.Add(member.To, joint); table.Joints.Add(joint);
    }
    foreach (long id in arm.Bodies.Keys) {
        GridShape grid; if (!Scene.TryGetValue(id, out grid) || !grid.Ready || grid.Shapes.Count == 0 || !SelfFrame(grid.Grid.WorldMatrix)) return table;
        var path = new List<SelfJoint>(); var seen = new HashSet<long>(); long cursor = id;
        while (cursor != arm.Hull) {
            SelfJoint joint; if (!seen.Add(cursor) || !incoming.TryGetValue(cursor, out joint)) return table;
            path.Insert(0, joint); cursor = joint.From;
        }
        // Occupied-grid bounds enclose every scanned shape, including armour,
        // unnamed tops and offset equipment. Empty allocation is conservative.
        double size = grid.Grid.GridSize;
        var centre = ((Vector3D)grid.Min + (Vector3D)grid.Max) * (size * .5);
        double radius = (((Vector3D)grid.Max - (Vector3D)grid.Min + Vector3D.One) * (size * .5)).Length();
        if (!Finite(size) || size <= 0 || !SelfVector(centre) || !Finite(radius)) return table;
        table.Paths.Add(id, path); table.Geometry.Add(id, new SelfBall { Centre = centre, Radius = radius });
        table.Bounds.Add(id, new[] { grid.Min, grid.Max }); table.Sizes.Add(id, size);
    }
    var ids = new List<long>(table.Paths.Keys); ids.Sort();
    for (int i = 0; i < ids.Count; i++) for (int j = i + 1; j < ids.Count; j++) table.Pairs.Add(new SelfPair { A = ids[i], B = ids[j] });
    table.Supported = true; return table;
}
bool SelfTableValid(SelfTable table, Arm arm) {
    if (table.Stamp != arm.Fingerprint || table.Revision != SelfGeometryRevision) return false;
    if (!table.Supported) return true;
    foreach (var joint in table.Joints) {
        double lower, upper; var block = joint.Block;
        if (!Attached(block) || block.CubeGrid.EntityId != joint.From || block.TopGrid.EntityId != joint.To || block.Top.EntityId != joint.Top ||
            !SelfLimits(block, out lower, out upper) || lower != joint.Lower || upper != joint.Upper || !SelfFrame(block.CubeGrid.WorldMatrix)) return false;
        var inverse = MatrixD.Invert(block.CubeGrid.WorldMatrix);
        if (Vector3D.DistanceSquared(Vector3D.Transform(block.GetPosition(), inverse), joint.LocalPivot) > 1e-10 ||
            Vector3D.DistanceSquared(Vector3D.TransformNormal(block.WorldMatrix.Up, inverse), joint.LocalAxis) > 1e-10) return false;
    }
    foreach (var item in table.Bounds) {
        GridShape grid;
        if (!Scene.TryGetValue(item.Key, out grid) || !grid.Ready || grid.Min != item.Value[0] || grid.Max != item.Value[1] ||
            grid.Grid.GridSize != table.Sizes[item.Key] || !SelfFrame(grid.Grid.WorldMatrix)) return false;
    }
    return true;
}
SelfBall ReachBall(SelfTable table, long body, List<SelfJoint> path, int shared, MatrixD inverse) {
    var geometry = table.Geometry[body];
    var centre = Vector3D.Transform(Vector3D.Transform(geometry.Centre, Scene[body].Grid.WorldMatrix), inverse);
    double motion = 0;
    for (int i = shared; i < path.Count; i++) {
        var joint = path[i];
        var block = joint.Block; var rotor = block as IMyMotorStator;
        if (rotor != null) {
            double q = rotor.Angle;
            if (!Finite(q)) return null;
            // Full rotation if wrapping or unlimited limits make the reduced
            // interval ambiguous. Never infer a lock from commanded zero speed.
            double angle = Math.PI;
            if (joint.Lower > -10000 && joint.Upper < 10000 && q >= joint.Lower && q <= joint.Upper)
                angle = Math.Min(Math.PI, Math.Max(q - joint.Lower, joint.Upper - q));
            var pivot = Vector3D.Transform(block.GetPosition(), inverse);
            var axis = Vector3D.TransformNormal(block.WorldMatrix.Up, inverse); axis.Normalize();
            double radial = Vector3D.Cross(centre - pivot, axis).Length();
            motion += 2 * Math.Sin(angle * .5) * radial;
        } else {
            var piston = block as IMyPistonBase; double q = piston.CurrentPosition;
            if (!Finite(q) || q < joint.Lower || q > joint.Upper) return null;
            motion += Math.Max(q - joint.Lower, joint.Upper - q);
        }
        if (!Finite(motion)) return null;
    }
    // Product-of-rigid-transforms bound: rotations preserve the radius of the
    // downstream error ball. Sum each joint's displacement of the reference
    // centre; do not assume axes stay collinear or sample poses as a proof.
    return new SelfBall { Centre = centre, Radius = geometry.Radius + motion + SelfSlack + 1e-6 * (1 + centre.Length() + motion) };
}
bool ProveSelfPair(SelfTable table, Arm arm, SelfPair pair) {
    var first = table.Paths[pair.A]; var second = table.Paths[pair.B]; int shared = 0;
    while (shared < first.Count && shared < second.Count && first[shared] == second[shared]) shared++;
    pair.Anchor = shared == 0 ? arm.Hull : first[shared - 1].To;
    GridShape anchor; if (!Scene.TryGetValue(pair.Anchor, out anchor) || !SelfFrame(anchor.Grid.WorldMatrix)) return false;
    var inverse = MatrixD.Invert(anchor.Grid.WorldMatrix);
    pair.First = ReachBall(table, pair.A, first, shared, inverse); pair.Second = ReachBall(table, pair.B, second, shared, inverse);
    if (pair.First == null || pair.Second == null) return false;
    pair.Gap = Vector3D.Distance(pair.First.Centre, pair.Second.Centre) - pair.First.Radius - pair.Second.Radius;
    return Finite(pair.Gap) && pair.Gap > Math.Max(Influence, Clearance) + SelfSlack;
}
void PrepareSelfPairs(Arm arm) {
    ActiveSelfTable = null; SelfCandidates = SelfProven = SelfWouldSkip = SelfSkipped = 0;
    if (SelfPairMode == "Off" || !SelfWork()) return;
    foreach (long id in arm.Bodies.Keys) { GridShape grid; if (!Scene.TryGetValue(id, out grid) || !grid.Ready) return; }
    string key = arm.PB.EntityId + "/" + arm.Name; SelfTable table;
    if (!SelfTables.TryGetValue(key, out table) || !SelfTableValid(table, arm)) {
        SelfInvalidations++; table = CaptureSelfTable(arm); SelfTables[key] = table;
    }
    if (!table.Supported) return;
    // Bound preprocessing work per request. Unprocessed/uncertain pairs simply
    // continue down the original live collision path.
    int work = 0;
    while (table.Cursor < table.Pairs.Count && work++ < 2 && SelfWork()) {
        var pair = table.Pairs[table.Cursor++];
        if (ProveSelfPair(table, arm, pair)) table.Proven[SelfKey(pair.A, pair.B)] = pair;
    }
    ActiveSelfTable = table; SelfCandidates = table.Pairs.Count; SelfProven = table.Proven.Count;
}
bool SkipSelfPair(Arm arm, GridShape a, GridShape b, double range) {
    var table = ActiveSelfTable; SelfPair proof;
    if (table == null || !arm.Bodies.ContainsKey(a.Grid.EntityId) || !arm.Bodies.ContainsKey(b.Grid.EntityId) ||
        !table.Proven.TryGetValue(SelfKey(a.Grid.EntityId, b.Grid.EntityId), out proof) || !Finite(range) || proof.Gap <= range + SelfSlack) return false;
    var anchor = Scene[proof.Anchor].Grid.WorldMatrix; if (!SelfFrame(anchor)) return false;
    var inverse = MatrixD.Invert(anchor);
    // Check actual geometry remains inside the certified enclosure. Flex,
    // out-of-limit travel or an imperfect pivot model cannot silently extend it.
    foreach (long id in new[] { proof.A, proof.B }) {
        var geometry = table.Geometry[id]; var ball = id == proof.A ? proof.First : proof.Second;
        var centre = Vector3D.Transform(Vector3D.Transform(geometry.Centre, Scene[id].Grid.WorldMatrix), inverse);
        if (!SelfVector(centre) || Vector3D.Distance(centre, ball.Centre) + geometry.Radius > ball.Radius) {
            table.Proven.Remove(SelfKey(proof.A, proof.B)); SelfProven = table.Proven.Count; return false;
        }
    }
    SelfWouldSkip++;
    if (SelfPairMode != "On") return false;
    SelfSkipped++; return true;
}
