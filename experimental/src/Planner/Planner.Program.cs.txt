public void Main(string argument,UpdateType source)
{
    double dt=Runtime.TimeSinceLastRun.TotalSeconds;if(Finite(dt)&&dt>0)Clock+=dt;
    try
    {
        Receive();
        if(!string.IsNullOrWhiteSpace(argument)&&argument!=Tag)Command(argument.Trim());
        if(Phase!="Idle"&&Phase!="Complete"&&Phase!="Preview")
        {
            if(Peer==null||Peer.Closed||!Peer.IsFunctional||!Peer.IsSameConstructAs(Me))throw new Exception("Arm PB lost.");
            if(Remote!=0&&Clock-ReplyAt>.4)throw new Exception("Arm reply lost.");
            if(Clock-LastPoll>=.1){LastPoll=Clock;Send(Remote==0?"HELLO":Phase=="Model"&&ModelRequest==0?"MODEL":"PING");}
            if(Phase!="Model"&&World!=null&&!World.Still(Submitted?new double[0]:null))throw new Exception("Scene/start moved: "+World.Problem+"; cancel and replan.");
            if(World!=null)World.Audit();
            if(!Submitted&&Clock-Started>45)throw new Exception("Planning time budget in "+Phase+"; "+(World==null?"":World.Problem)+"; no motion sent.");
            if(Phase=="Scan"){World.Scan();if(World.Scanned){World.Prepare();BeginIK(new double[World.Lower.Length],Goal,false);Phase="GoalIK";}}
            else if(Phase=="GoalIK"||Phase=="Project"||Phase=="StraightIK")IKStep();
            else if(Phase=="StartCheck"||Phase=="GoalCheck")
            {
                World.CheckStep();if(World.Checked){if(!World.Clear){if(Phase=="StartCheck")throw new Exception("Start blocked: "+World.Problem);RetryGoal();}
                else if(Phase=="StartCheck"){World.Begin(TargetQ);Phase="GoalCheck";}
                else{A.Clear();B.Clear();A.Add(new Node(new double[TargetQ.Length],-1));B.Add(new Node(TargetQ,-1));Swapped=false;Straight=new List<double[]>{A[0].Q};StraightIndex=1;StraightCount=Math.Max(1,Math.Min(64,(int)Math.Ceiling(Vector3D.Distance(Start.Translation,Goal.Translation)/.5)));HeadRoute=true;BeginStraight();}}
            }
            else if(Phase=="Edge")EdgeStep();
            else if(Phase=="Search")SearchStep();
            else if(Phase=="Validate")ValidateStep();
            else if(Phase=="Moving")MoveStep();
        }
    }
    catch(Exception e){Abort(e.Message);}
    Echo("AutoArm "+ControllerBuildVersion+" Planner | "+ArmName+"\n"+Phase+" | "+Status+"\nNodes "+Nodes+" / "+NodeLimit+". Cancel requests a stop.");
}
void Command(string text)
{
    string[] w=text.Split(new char[]{' ','\t'},StringSplitOptions.RemoveEmptyEntries);string verb=w[0].ToLowerInvariant();
    if(verb=="cancel"||verb=="stop"){Abort("Canceled.");return;}
    if(verb=="info"){Status=World==null?Status:Status+"; route "+(Route==null?0:Route.Count)+" configurations";return;}
    if(Phase!="Idle"&&Phase!="Complete"&&Phase!="Preview")throw new Exception("Planner busy; Cancel first.");
    if(verb=="select"&&w.Length>=2){ArmName=string.Join(" ",w.Skip(1));var ini=new MyIni();ini.TryParse(Me.CustomData);ini.Set("Planner","Arm",ArmName);Me.CustomData=ini.ToString();Peer=null;Remote=Seen=Received=Generation=0;Configure();return;}
    if((verb!="moveto"&&verb!="previewto")||w.Length!=4)throw new Exception("Commands: MoveTo x y z, PreviewTo x y z, Select arm, Cancel, Info.");
    Requested=new Vector3D(Parse(w[1]),Parse(w[2]),Parse(w[3]));Execute=verb=="moveto";Configure();World=null;Route=null;Validated=null;Remote=Received=Seen=Generation=0;ModelRequest=0;PathId=0;Submitted=false;Nodes=Attempts=EdgeChecks=0;Started=Clock;Phase="Model";Status="Requesting idle arm model; no motion.";LastPoll=Clock;Send("HELLO");
}

void Receive()
{
    int count=0;while(IGC.UnicastListener.HasPendingMessage&&count++<8)
    {
        var message=IGC.UnicastListener.AcceptMessage();var text=message.Data as string;var ini=new MyIni();
        if(Peer==null||message.Source!=Peer.EntityId||message.Tag!=Tag||text==null||text.Length>16000||!ini.TryParse(text)||ini.Get(S,"Version").ToInt32()!=5||ini.Get(S,"Service").ToInt32()!=1||ini.Get(S,"Arm").ToString()!=ArmName||ini.Get(S,"Remote").ToInt64()!=Epoch||ini.Get(S,"Operation").ToString()!="STATE")continue;
        long epoch=ini.Get(S,"Epoch").ToInt64(),sequence=ini.Get(S,"Sequence").ToInt64(),ack=ini.Get(S,"Ack").ToInt64();if(epoch<=0||sequence<=0||ack<HelloRequest||ack<=0||ack>Sequence||ini.Get(S,"Role").ToString()!="Plan"||epoch<Remote||epoch==Remote&&sequence<=Received)continue;
        if(Remote!=0&&epoch!=Remote)throw new Exception("Arm restarted; replan.");Remote=epoch;Received=Seen=sequence;ReplyAt=Clock;Generation=ini.Get(S,"Generation").ToInt64();State=ini;
        string error=ini.Get(S,"Error").ToString();if(error.Length>0&&Phase!="Idle")throw new Exception(error);
        if(Phase!="Model"&&Phase!="Idle"&&Phase!="Complete"&&Phase!="Preview"&&!Submitted&&ini.Get(S,"State").ToString()=="Canceled")throw new Exception("Arm canceled planning hold.");
        if(Phase=="Model"&&ini.Get(S,"Ack").ToInt64()==ModelRequest&&ini.ContainsKey(S,"PlanFingerprint"))
        {
            if(!ini.Get(S,"Ready").ToBoolean()||!ini.Get(S,"Enabled").ToBoolean()||ini.Get(S,"Busy").ToBoolean()||!ini.Get(S,"SafetyRequired").ToBoolean()||!ini.Get(S,"SafetyReady").ToBoolean()||ini.Get(S,"SafetyHold").ToBoolean())throw new Exception("Planning requires an idle ON arm with fresh Safety.");
            Stamp=ini.Get(S,"PlanFingerprint").ToString();World=new PlanningWorld(this,ini,Clearance);Start=Frame(ini,"Pose");Goal=Start;
            var anchor=GridTerminalSystem.GetBlockWithId(World.BaseId);if(anchor==null)throw new Exception("Arm base lost.");AnchorPose=anchor.WorldMatrix;
            Goal.Translation=anchor.GetPosition()+anchor.WorldMatrix.Forward*Requested.X+anchor.WorldMatrix.Left*Requested.Y+anchor.WorldMatrix.Up*Requested.Z;
            Phase="Scan";Status="Scanning all construct cells, including empty cells.";
        }
    }
}
static Vector3D Rotation(MatrixD a,MatrixD b)
{
    var value=(Vector3D.Cross(a.Forward,b.Forward)+Vector3D.Cross(a.Up,b.Up)+Vector3D.Cross(a.Right,b.Right))*.5;
    double dot=Clamp((Vector3D.Dot(a.Forward,b.Forward)+Vector3D.Dot(a.Up,b.Up)+Vector3D.Dot(a.Right,b.Right)-1)*.5,-1,1),angle=Math.Acos(dot);
    if(angle<1e-8)return Vector3D.Zero;double length=value.Length();return length>1e-8?value*(angle/length):a.Up*angle;
}
void BeginIK(double[] seed,MatrixD goal,bool orientation)
{SolveQ=(double[])seed.Clone();SolveGoal=goal;OrientationOnly=orientation;Iteration=0;}
void RetryGoal()
{
    if(++Attempts>12)throw new Exception("No permitted goal configuration within budget: "+World.Problem);
    var seed=new double[World.Lower.Length];for(int i=0;i<seed.Length;i++)seed[i]=World.Lower[i]+Random.NextDouble()*(World.Upper[i]-World.Lower[i]);BeginIK(seed,Goal,false);Phase="GoalIK";
}

{
    AxisRoute=new List<double[]>();AxisVerified=new List<double[]>();var q=(double[])A[0].Q.Clone();AxisRoute.Add(q);AxisVerified.Add(q);
    for(int k=0;k<q.Length;k++){q=(double[])q.Clone();int i=AxisOrder==0?k:q.Length-1-k;q[i]=B[0].Q[i];AxisRoute.Add(q);}

void IKStep()
{
    if(!Work())return;var pose=World.FK(SolveQ);var linear=OrientationOnly?Vector3D.Zero:SolveGoal.Translation-pose.Translation;var angular=Rotation(pose,SolveGoal);
    if(linear.Length()<.004&&angular.Length()<.002)
    {
        if(Phase=="StraightIK"){BeginEdge(Straight[Straight.Count-1],SolveQ,4);return;}
        if(OrientationOnly){BeginEdge((Swapped?B:A)[NearIndex].Q,SolveQ,1);return;}
        TargetQ=(double[])SolveQ.Clone();World.Begin(new double[TargetQ.Length]);Phase="StartCheck";return;
    }
    if(++Iteration>80)
    {
        if(Phase=="StraightIK"){BeginDetour();return;}if(OrientationOnly){Phase="Search";return;}RetryGoal();return;
    }
    int n=SolveQ.Length;var columns=new double[6,n];var gram=new double[6,6];var rhs=new double[]{linear.X,linear.Y,linear.Z,angular.X,angular.Y,angular.Z};
    for(int i=0;i<n;i++)
    {
        Vector3D l,a;World.Column(i,pose.Translation,out l,out a);double scale=World.Model.Groups[i].Rotary?AngleStep:PistonStep;
        for(int r=0;r<6;r++)columns[r,i]=scale*(r<3?(OrientationOnly?0:r==0?l.X:r==1?l.Y:l.Z):r==3?a.X:r==4?a.Y:a.Z);
    }
    for(int r=0;r<6;r++)for(int c=0;c<6;c++){double value=r==c?.00001:0;for(int i=0;i<n;i++)value+=columns[r,i]*columns[c,i];gram[r,c]=value;}
    var solution=Linear(gram,rhs);var delta=new double[n];double maximum=1;
    for(int i=0;i<n;i++){for(int r=0;r<6;r++)delta[i]+=columns[r,i]*solution[r];maximum=Math.Max(maximum,Math.Abs(delta[i]));}
    for(int i=0;i<n;i++)SolveQ[i]=Clamp(SolveQ[i]+(World.Model.Groups[i].Rotary?AngleStep:PistonStep)*delta[i]/maximum,World.Lower[i],World.Upper[i]);
}
static double[] Linear(double[,] a,double[] b)
{
    int n=b.Length;var l=new double[n,n];var x=new double[n];
    for(int i=0;i<n;i++)for(int j=0;j<=i;j++){double s=a[i,j];for(int k=0;k<j;k++)s-=l[i,k]*l[j,k];l[i,j]=i==j?Math.Sqrt(Math.Max(s,1e-14)):s/l[j,j];}
    for(int i=0;i<n;i++){double s=b[i];for(int k=0;k<i;k++)s-=l[i,k]*x[k];x[i]=s/l[i,i];}
    for(int i=n-1;i>=0;i--){double s=x[i];for(int k=i+1;k<n;k++)s-=l[k,i]*x[k];x[i]=s/l[i,i];}return x;
}
double Distance(double[] x,double[] y){double s=0;for(int i=0;i<x.Length;i++){double d=(x[i]-y[i])/(World.Model.Groups[i].Rotary?AngleStep:PistonStep);s+=d*d;}return Math.Sqrt(s);}
void SearchStep()
{
    if(!Work())return;if(++Nodes>NodeLimit)throw new Exception("No route found within node budget; no motion sent.");
    var tree=Swapped?B:A;var other=Swapped?A:B;var sample=new double[World.Lower.Length];
    if(Nodes%4==0)Array.Copy(other[0].Q,sample,sample.Length);else for(int i=0;i<sample.Length;i++)sample[i]=World.Lower[i]+Random.NextDouble()*(World.Upper[i]-World.Lower[i]);
    NearIndex=Nearest(tree,sample);double distance=Distance(tree[NearIndex].Q,sample);double fraction=Math.Min(1,4/Math.Max(distance,1e-12));
    for(int i=0;i<sample.Length;i++)sample[i]=tree[NearIndex].Q[i]+fraction*(sample[i]-tree[NearIndex].Q[i]);
    var pose=World.FK(sample);pose.Forward=Goal.Forward;pose.Up=Goal.Up;pose.Right=Goal.Right;BeginIK(sample,pose,true);Phase="Project";
}
int Nearest(List<Node> tree,double[] q){int found=0;double best=double.MaxValue;for(int i=0;i<tree.Count;i++){double d=Distance(tree[i].Q,q);if(d<best){best=d;found=i;}}return found;}
void BeginEdge(double[] from,double[] to,int purpose)
{
    EdgeFrom=(double[])from.Clone();EdgeTo=(double[])to.Clone();EdgePurpose=purpose;Segments.Clear();Segments.Push(new double[][]{EdgeFrom,EdgeTo});EdgeRoute.Clear();EdgeRoute.Add(EdgeFrom);EdgeRunning=false;EdgeResult=true;Phase="Edge";
}
double[][] Segment;
int EdgeChecks;
void EdgeStep()
{
    if(!Work())return;
    if(!EdgeRunning)
    {
        if(Segments.Count==0){FinishEdge();return;}if(++EdgeChecks>8192)throw new Exception("Swept configuration budget; no route certified.");
        Segment=Segments.Pop();var mid=new double[EdgeFrom.Length];var half=new double[mid.Length];for(int i=0;i<mid.Length;i++){mid[i]=(Segment[0][i]+Segment[1][i])*.5;half[i]=(Segment[1][i]-Segment[0][i])*.5;}
        World.Begin(mid,half);EdgeRunning=true;
    }
    World.CheckStep();if(!World.Checked)return;EdgeRunning=false;
    var pose=World.CheckingHead;double rotation=Rotation(pose,Goal).Length();for(int i=0;i<EdgeFrom.Length;i++)if(World.Model.Groups[i].Rotary)rotation+=Math.Abs(Segment[1][i]-Segment[0][i])*.5;
    bool orientation=rotation<.035;
    if(EdgePurpose==4){var direction=Goal.Translation-Start.Translation;double t=direction.LengthSquared()>1e-12?Clamp(Vector3D.Dot(pose.Translation-Start.Translation,direction)/direction.LengthSquared(),0,1):0;orientation&=Vector3D.Distance(pose.Translation,Start.Translation+direction*t)+World.HeadPadding<=.05;}
    if(World.Clear&&orientation){EdgeRoute.Add(Segment[1]);return;}
    double d=Distance(Segment[0],Segment[1]);if(d<.01||Segments.Count>64){EdgeResult=false;Segments.Clear();FinishEdge();return;}
    var middle=new double[EdgeFrom.Length];for(int i=0;i<middle.Length;i++)middle[i]=(Segment[0][i]+Segment[1][i])*.5;
    Segments.Push(new double[][]{middle,Segment[1]});Segments.Push(new double[][]{Segment[0],middle});
}
void FinishEdge()
{
    if(EdgePurpose==5){if(!EdgeResult){if(++AxisOrder<2)BeginAxis();else Phase="Search";return;}AxisVerified.AddRange(EdgeRoute.Skip(1));if(++AxisIndex==AxisRoute.Count){Route=AxisVerified;ReadyRoute();}else BeginEdge(AxisRoute[AxisIndex-1],AxisRoute[AxisIndex],5);return;}
    if(EdgePurpose==4){if(!EdgeResult){BeginDetour();return;}Straight.AddRange(EdgeRoute.Skip(1));if(++StraightIndex>StraightCount){Route=Straight;ReadyRoute();}else BeginStraight();return;}
    if(EdgePurpose==0)
    {
        if(EdgeResult){Route=new List<double[]>(EdgeRoute.Select(q=>(double[])q.Clone()));ReadyRoute();return;}AxisOrder=0;BeginAxis();return;
    }
    var tree=Swapped?B:A;var other=Swapped?A:B;
    if(EdgePurpose==1)
    {
        if(!EdgeResult){Swapped=!Swapped;Phase="Search";return;}tree.Add(new Node(EdgeTo,NearIndex));JoinIndex=Nearest(other,EdgeTo);BeginEdge(EdgeTo,other[JoinIndex].Q,2);return;
    }
    if(EdgePurpose==2)
    {
        if(!EdgeResult){Swapped=!Swapped;Phase="Search";return;}
        var left=Trace(tree,tree.Count-1);var right=Trace(other,JoinIndex);right.Reverse();left.AddRange(right);
        if(Swapped)left.Reverse();Route=left;RouteCursor=1;var first=Route[0];EdgeRoute.Clear();EdgeRoute.Add(first);Phase="Validate";return;
    }
    if(!EdgeResult)throw new Exception("Route edge validation failed.");
    Validated.AddRange(EdgeRoute.Skip(1));RouteCursor++;Phase="Validate";
}
List<double[]> Trace(List<Node> tree,int index){var result=new List<double[]>();while(index>=0){result.Add(tree[index].Q);index=tree[index].Parent;}result.Reverse();return result;}
List<double[]> Validated;
void ValidateStep()
{
    if(Validated==null){Validated=new List<double[]>();Validated.Add(Route[0]);}
    if(RouteCursor==Route.Count){Route=Validated;Validated=null;ReadyRoute();return;}
    BeginEdge(Route[RouteCursor-1],Route[RouteCursor],3);
}
void ReadyRoute()
{
    // Always retain the subdivision points whose interval boxes were checked.
    if(Route.Count<2)throw new Exception("Empty planning route.");
    Status="Checked "+Route.Count+" joint configurations and connecting swept boxes; "+(HeadRoute?"head corridor retained.":"head detour used.");
    if(!Execute){Send("RELEASEMODEL");Phase="Preview";return;}RouteCursor=0;SendChunk();
}
void SendChunk()
{
    int end=Math.Min(Route.Count-1,RouteCursor+7);var poses=new List<string>();var joints=new List<string>();var inverse=MatrixD.Invert(AnchorPose);
    for(int k=RouteCursor;k<=end;k++)
    {
        var pose=World.FK(Route[k])*inverse;var p=pose.Translation;var f=pose.Forward;var u=pose.Up;
        poses.Add(string.Join(" ",new double[]{p.X,p.Y,p.Z,f.X,f.Y,f.Z,u.X,u.Y,u.Z}.Select(Number)));joints.Add(World.NativeRow(Route[k]));
    }
    var ini=new MyIni();ini.Set(S,"Waypoints",string.Join("|",poses));ini.Set(S,"Joints",string.Join("|",joints));ini.Set(S,"Anchor",World.BaseId);ini.Set(S,"PlanFingerprint",Stamp);SentEnd=end;Send("PATH",ini);Phase="Moving";Status="Following checked joint route "+end+"/"+(Route.Count-1)+"; live Safety retained.";
}
void MoveStep()
{
    if(State==null)return;string state=State.Get(S,"State").ToString();long path=State.Get(S,"PathId").ToInt64();
    if(path!=PathId){if(State.Get(S,"Ack").ToInt64()>=PathId)throw new Exception("Joint path refused/replaced.");return;}
    if(state=="Canceled"||!State.Get(S,"Enabled").ToBoolean())throw new Exception("Arm canceled joint route.");
    if(state!="Complete")return;Submitted=false;RouteCursor=SentEnd;
    if(RouteCursor==Route.Count-1){var anchor=GridTerminalSystem.GetBlockWithId(World.BaseId);if(anchor==null)throw new Exception("Goal anchor lost.");var goal=Goal*MatrixD.Invert(AnchorPose)*anchor.WorldMatrix;var pose=Frame(State,"Pose");if(Vector3D.Distance(pose.Translation,goal.Translation)>.005||Rotation(pose,goal).Length()>.2*Math.PI/180)throw new Exception("Measured goal residual exceeds 5 mm / .2 deg.");Send("RELEASEMODEL");Phase="Complete";Status="Head goal reached; arm ON hold.";return;}
    SendChunk();
}

    Submitted=false;Phase="Idle";Status="Stopped: "+reason+" No automatic return.";World=null;Validated=null;
}

void RefreshScene(bool force)
{
     Blocks.Clear(); GridTerminalSystem.GetBlocks(Blocks); Joints.Clear();
    var alive = new HashSet<long>();
    foreach (var block in Blocks)
    {
        if (block.Closed || !block.IsSameConstructAs(Me)) continue;
        AddGrid(block.CubeGrid, alive, force);
        var joint = block as IMyMechanicalConnectionBlock;
        if (joint != null) { Joints.Add(joint); if (joint.TopGrid != null) AddGrid(joint.TopGrid, alive, force); }
    }
    Parts.Clear(); Bases.Clear(); Models.Clear();
    foreach (var joint in Joints) {
        Parts[joint.EntityId] = joint; if (joint.Top != null) Parts[joint.Top.EntityId] = joint.Top;
        List<IMyMechanicalConnectionBlock> list;
        if (!Bases.TryGetValue(joint.CubeGrid.EntityId, out list)) { list = new List<IMyMechanicalConnectionBlock>(); Bases.Add(joint.CubeGrid.EntityId, list); }
        list.Add(joint);
    }
    var removed = new List<long>(); foreach (var pair in Scene) if (!alive.Contains(pair.Key)) removed.Add(pair.Key);
    foreach (long id in removed) Scene.Remove(id);
    foreach (var grid in Scene.Values) if (!grid.Ready)
    {
        grid.Tags.Clear();
        foreach (var block in Blocks) if (block.CubeGrid.EntityId == grid.Grid.EntityId) TagCells(grid, block, block.EntityId);
        foreach (var joint in Joints) if (joint.Top != null && joint.Top.CubeGrid.EntityId == grid.Grid.EntityId) TagCells(grid, joint.Top, joint.Top.EntityId);
    }
}
void AddGrid(IMyCubeGrid native, HashSet<long> alive, bool force)
{
    if (native == null) return; alive.Add(native.EntityId); GridShape shape;
    if (!Scene.TryGetValue(native.EntityId, out shape)) { shape = new GridShape { Grid = native, Previous = native.WorldMatrix, Observed = Clock }; Scene.Add(native.EntityId, shape); force = true; }
    if (force || shape.Min != native.Min || shape.Max != native.Max)
    { shape.Min = native.Min; shape.Max = native.Max; shape.Cursor = shape.Min; shape.Shapes.Clear(); shape.Tiles.Clear(); shape.Ready = false; }
}
void TagCells(GridShape grid, IMyCubeBlock block, long tag)
{
    var min = block.Min; var max = block.Max;
    double count = (double)(max.X - min.X + 1) * (max.Y - min.Y + 1) * (max.Z - min.Z + 1);
    if (count > 512) return;
    for (int z = min.Z; z <= max.Z; z++) for (int y = min.Y; y <= max.Y; y++) for (int x = min.X; x <= max.X; x++) grid.Tags[new Vector3I(x, y, z)] = tag;
}
void ScanStep()
{
    foreach (var grid in Scene.Values)
    {
        if (grid.Ready) continue;
        double cells = ((double)grid.Max.X - grid.Min.X + 1) * ((double)grid.Max.Y - grid.Min.Y + 1) * ((double)grid.Max.Z - grid.Min.Z + 1);
        if (cells > MaxCells) { Status = "Grid " + grid.Grid.EntityId + " exceeds MaxCells; increase limit or simplify geometry."; return; }
        int work = 0;
        while (grid.Cursor.Z <= grid.Max.Z && work++ < 512 && Budget())
        {
            var cell = grid.Cursor;
            if (grid.Grid.CubeExists(cell))
            {
                long tag; grid.Tags.TryGetValue(cell, out tag);
                int last = cell.X;
                while (last < grid.Max.X && work++ < 512 && Budget())
                {
                    var next = new Vector3I(last + 1, cell.Y, cell.Z); long other; grid.Tags.TryGetValue(next, out other);
                    if (other != tag || !grid.Grid.CubeExists(next)) break; last++;
                }
                double size = grid.Grid.GridSize;
                var shape = new Shape { Centre = new Vector3D((cell.X + (double)last) * .5, cell.Y, cell.Z) * size,
                    Half = new Vector3D((last - cell.X + 1) * .5, .5, .5) * size, Tag = tag };
                grid.Shapes.Add(shape);
                for (int x = Tile(cell.X); x <= Tile(last); x++)
                {
                    var key = new Vector3I(x, Tile(cell.Y), Tile(cell.Z)); List<Shape> list;
                    if (!grid.Tiles.TryGetValue(key, out list)) { list = new List<Shape>(); grid.Tiles.Add(key, list); } list.Add(shape);
                }
                if (grid.Shapes.Count > MaxShapes) { Status = "Grid exceeds MaxShapes; increase limit or simplify geometry."; return; }
                grid.Cursor.X = last;
            }
            grid.Cursor.X++;
            if (grid.Cursor.X > grid.Max.X) { grid.Cursor.X = grid.Min.X; grid.Cursor.Y++; }
            if (grid.Cursor.Y > grid.Max.Y) { grid.Cursor.Y = grid.Min.Y; grid.Cursor.Z++; }
        }
        grid.Ready = grid.Cursor.Z > grid.Max.Z;
        if (!Budget()) return;
    }
}
void ParseArm(Arm arm, MyIni ini)
{
    int limit = 8;
    if (ini.ContainsKey(Section, "ConstraintLimit") && (!ini.Get(Section, "ConstraintLimit").TryGetInt32(out limit) || limit != 8 && limit != 16)) throw new Exception("Invalid collision constraint capacity.");
    arm.ConstraintLimit = limit;
    arm.Groups.Clear(); arm.Bodies.Clear(); arm.Contacts.Clear();
    arm.Hull = ini.Get(Section, "Hull").ToInt64(); arm.Tool = ini.Get(Section, "ToolGrid").ToInt64();
    arm.TipGrid = ini.Get(Section, "TipGrid").ToInt64(); arm.Socket = ini.Get(Section, "Socket").ToBoolean();
    var cell = Numbers(ini.Get(Section, "TipCell").ToString(), 3); arm.TipCell = new Vector3I((int)cell[0], (int)cell[1], (int)cell[2]);
    var up = Numbers(ini.Get(Section, "TipUp").ToString(), 3); arm.TipUp = new Vector3D(up[0], up[1], up[2]);
    foreach (string row in ini.Get(Section, "Groups").ToString().Split('|'))
    {
        string[] parts = row.Split(':'); if (parts.Length < 2 || parts[0] != "R" && parts[0] != "P" || arm.Groups.Count >= 24) throw new Exception("Invalid collision topology.");
        var group = new Group { Rotary = parts[0] == "R" };
        for (int i = 1; i < parts.Length; i++)
        {
            string part = parts[i]; int at = part.IndexOf('@'), to = part.IndexOf('>'); long id, from, dest;
            if (at < 2 || to <= at || !long.TryParse(part.Substring(0, at - 1), out id) || !long.TryParse(part.Substring(at + 1, to - at - 1), out from) || !long.TryParse(part.Substring(to + 1), out dest) || part[at - 1] != '+' && part[at - 1] != '-') throw new Exception("Invalid collision member.");
            var block = GridTerminalSystem.GetBlockWithId(id) as IMyMechanicalConnectionBlock;
            if (block == null || block.Closed || !block.IsSameConstructAs(Me) || !block.IsAttached || block.TopGrid == null ||
                (group.Rotary ? !(block is IMyMotorStator) || block is IMyMotorSuspension : !(block is IMyPistonBase)) ||
                !(block.CubeGrid.EntityId == from && block.TopGrid.EntityId == dest || block.CubeGrid.EntityId == dest && block.TopGrid.EntityId == from)) throw new Exception("Collision topology changed.");
            group.Members.Add(new Joint { Block = block, From = from, To = dest, Sign = part[at - 1] == '+' ? 1 : -1 });
            if (Attached(block)) { arm.OwnParts.Add(block.EntityId); arm.OwnParts.Add(block.Top.EntityId); }
            if (from != arm.Hull) arm.Bodies[from] = 0; if (dest != arm.Hull) arm.Bodies[dest] = 0;
        }
        arm.Groups.Add(group);
    }
    if (arm.Groups.Count == 0 || arm.Tool == arm.Hull) throw new Exception("Collision endpoint/hull invalid.");
    if (!arm.Groups[0].Members.Any(m => m.Block.CustomName.StartsWith(arm.Name + " - Base", StringComparison.OrdinalIgnoreCase) && m.Block.CubeGrid.EntityId == arm.Hull)) throw new Exception("Collision base identity mismatch.");
    int n = arm.Groups.Count; arm.Caps = Numbers(ini.Get(Section, "Caps").ToString(), n); arm.Rates = Numbers(ini.Get(Section, "Rates").ToString(), n);
    for (int i = 0; i < n; i++)
    {
        if (arm.Caps[i] <= 0 || arm.Caps[i] > 2 || Math.Abs(arm.Rates[i]) > arm.Caps[i] + 1e-6) throw new Exception("Invalid collision joint rates.");
        var downstream = new HashSet<long>(); foreach (var member in arm.Groups[i].Members) downstream.Add(member.To);
        for (int j = i + 1; j < n; j++) foreach (var member in arm.Groups[j].Members) if (downstream.Contains(member.From)) downstream.Add(member.To);
        foreach (long grid in downstream) if (arm.Bodies.ContainsKey(grid)) arm.Bodies[grid] |= 1 << i;
    }
    if (!arm.Bodies.ContainsKey(arm.Tool)) arm.Bodies[arm.Tool] = (1 << n) - 1;
    foreach (string row in ini.Get(Section, "Contacts").ToString().Split('|'))
    {
        if (row.Length == 0) continue; string[] cells = row.Split(','); long a, b;
        if (cells.Length < 2 || cells.Length > 3 || !long.TryParse(cells[0], out a) || !long.TryParse(cells[1], out b) || arm.Contacts.Count >= 17) throw new Exception("Invalid collision contact.");
        arm.Contacts.Add(new ContactPair { A = a, B = b, Axis = cells.Length == 3 ? cells[2] : "Auto" });
    }
}
static double[] Numbers(string text, int n)
{
    string[] parts = text.Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries); if (parts.Length != n) throw new Exception("Collision vector size.");
    var values = new double[n]; for (int i = 0; i < n; i++) if (!double.TryParse(parts[i], System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out values[i]) || !Finite(values[i])) throw new Exception("Nonfinite collision vector."); return values;
}
static bool Finite(double value) { return !double.IsNaN(value) && !double.IsInfinity(value); }
static string Number(double value) { return value.ToString("R", System.Globalization.CultureInfo.InvariantCulture); }
static string Row(double[] values) { var text = new System.Text.StringBuilder(); foreach (double value in values) { if (text.Length > 0) text.Append(' '); text.Append(Number(value)); } return text.ToString(); }



static int Tile(int cell) { return (int)Math.Floor(cell / 8.0); }
List<Shape> BodyShapes(Arm arm, GridShape grid)
{
    if (!arm.Socket || arm.TipGrid != grid.Grid.EntityId) return grid.Shapes;
    var shapes = new List<Shape>(); double half = grid.Grid.GridSize * .5; var tip = (Vector3D)arm.TipCell * grid.Grid.GridSize;
    foreach (var box in grid.Shapes)
    {
        if (!Contains(box, tip, half) || box.Half.X <= half + 1e-6) { shapes.Add(box); continue; }
        double left = tip.X - half - (box.Centre.X - box.Half.X), right = box.Centre.X + box.Half.X - tip.X - half;
        if (left > 1e-6) shapes.Add(new Shape { Centre = new Vector3D(tip.X - half - left * .5, tip.Y, tip.Z), Half = new Vector3D(left * .5, half, half), Tag = box.Tag });
        shapes.Add(new Shape { Centre = tip, Half = new Vector3D(half), Tag = box.Tag });
        if (right > 1e-6) shapes.Add(new Shape { Centre = new Vector3D(tip.X + half + right * .5, tip.Y, tip.Z), Half = new Vector3D(right * .5, half, half), Tag = box.Tag });
    }
    return shapes;
}
List<Shape> Candidates(GridShape body, Shape shape, GridShape obstacle, double range)
{
    var world = World(body); var inverse = MatrixD.Transpose(World(obstacle).GetOrientation());
    Vector3D centre = Vector3D.TransformNormal(Vector3D.Transform(shape.Centre, world) - World(obstacle).Translation, inverse);
    var axes = Axes(world); Vector3D half = Vector3D.Zero;
    foreach (var axis in Axes(World(obstacle)))
    {
        double extent = Radius(shape, axes, axis) + range;
        int i = axis == World(obstacle).Right ? 0 : axis == World(obstacle).Up ? 1 : 2;
        if (i == 0) half.X = extent; else if (i == 1) half.Y = extent; else half.Z = extent;
    }
    // Outward slack for native float orientations and coordinate projection.
    half += new Vector3D(1e-6 * (1 + half.Length() + centre.Length()));
    double size = obstacle.Grid.GridSize;
    var min = new Vector3I(Tile((int)Math.Floor((centre.X - half.X) / size)), Tile((int)Math.Floor((centre.Y - half.Y) / size)), Tile((int)Math.Floor((centre.Z - half.Z) / size)));
    var max = new Vector3I(Tile((int)Math.Ceiling((centre.X + half.X) / size)), Tile((int)Math.Ceiling((centre.Y + half.Y) / size)), Tile((int)Math.Ceiling((centre.Z + half.Z) / size)));
    var found = new HashSet<Shape>();
    Vector3D low = centre - half, high = centre + half;
    for (int z = min.Z; z <= max.Z; z++) for (int y = min.Y; y <= max.Y; y++) for (int x = min.X; x <= max.X; x++)
    {
        if (!Budget()) throw new Exception("Collision spatial query budget."); List<Shape> list;
        TilesVisited++;
        if (obstacle.Tiles.TryGetValue(new Vector3I(x, y, z), out list)) foreach (var candidate in list)
        {
            if ((++CandidatesSeen & 31) == 0 && !Budget()) throw new Exception("Collision candidate query budget.");
            // Tiles are 20 m wide on large grids. Reject their distant contents
            // against the conservative projected search box before SAT.
            if (candidate.Centre.X + candidate.Half.X < low.X || candidate.Centre.X - candidate.Half.X > high.X ||
                candidate.Centre.Y + candidate.Half.Y < low.Y || candidate.Centre.Y - candidate.Half.Y > high.Y ||
                candidate.Centre.Z + candidate.Half.Z < low.Z || candidate.Centre.Z - candidate.Half.Z > high.Z) continue;
            found.Add(candidate);
        }
    }
    return new List<Shape>(found);
}

bool NearGrids(GridShape a, GridShape b, double range)
{
    var ac = new Shape { Centre = ((Vector3D)a.Min + (Vector3D)a.Max) * (a.Grid.GridSize * .5), Half = ((Vector3D)a.Max - (Vector3D)a.Min + Vector3D.One) * (a.Grid.GridSize * .5) };
    var bc = new Shape { Centre = ((Vector3D)b.Min + (Vector3D)b.Max) * (b.Grid.GridSize * .5), Half = ((Vector3D)b.Max - (Vector3D)b.Min + Vector3D.One) * (b.Grid.GridSize * .5) };
    Vector3D normal; return Separation(a, ac, b, bc, out normal) < range;
}
static Vector3D[] Axes(MatrixD frame) { return new Vector3D[] { frame.Right, frame.Up, frame.Backward }; }
static double Radius(Shape box, Vector3D[] axes, Vector3D normal)
{ return Math.Abs(Vector3D.Dot(normal, axes[0])) * box.Half.X + Math.Abs(Vector3D.Dot(normal, axes[1])) * box.Half.Y + Math.Abs(Vector3D.Dot(normal, axes[2])) * box.Half.Z; }
double Separation(GridShape a, Shape x, GridShape b, Shape y, out Vector3D normal)
{
    var aa = Axes(World(a)); var bb = Axes(World(b));
    Vector3D delta = Vector3D.Transform(x.Centre, World(a)) - Vector3D.Transform(y.Centre, World(b));
    double distance = -double.MaxValue; normal = Vector3D.Up;
    for (int i = 0; i < 15; i++)
    {
        Vector3D axis = i < 3 ? aa[i] : i < 6 ? bb[i - 3] : Vector3D.Cross(aa[(i - 6) / 3], bb[(i - 6) % 3]);
        double length = axis.Length(); if (length < 1e-7) continue; axis /= length;
        double dot = Vector3D.Dot(delta, axis), gap = Math.Abs(dot) - Radius(x, aa, axis) - Radius(y, bb, axis);
        if (gap > distance) { distance = gap; normal = dot < 0 ? -axis : axis; }
    }
    return distance;
}





bool Contact(Arm arm, GridShape a, Shape x, GridShape b, Shape y)
{
    IMyCubeBlock part;
    if (Parts.TryGetValue(x.Tag, out part)) { var joint = part as IMyMechanicalConnectionBlock; if (joint != null && Attached(joint) && y.Tag == joint.Top.EntityId) return true; }
    if (Parts.TryGetValue(y.Tag, out part)) { var joint = part as IMyMechanicalConnectionBlock; if (joint != null && Attached(joint) && x.Tag == joint.Top.EntityId) return true; }
    if (MountedCollars(x.Tag, y.Tag) || MountedCollars(y.Tag, x.Tag)) return true;
    foreach (var pair in arm.Contacts)
    {
        if (pair.A != 0)
        {
            if (!(x.Tag == pair.A && y.Tag == pair.B || y.Tag == pair.A && x.Tag == pair.B)) continue;
            var first = GridTerminalSystem.GetBlockWithId(pair.A) as IMyShipMergeBlock; var second = GridTerminalSystem.GetBlockWithId(pair.B) as IMyShipMergeBlock;
            if (first != null && second != null && Vector3D.Dot(MergeNormal(first, pair.Axis), MergeNormal(second, pair.Axis)) < -.99) return true;
        }
        else if (arm.Socket)
        {
            GridShape tip = a.Grid.EntityId == arm.TipGrid ? a : b.Grid.EntityId == arm.TipGrid ? b : null;
            Shape top = tip == a ? x : y, mount = tip == a ? y : x;
            var rotor = GridTerminalSystem.GetBlockWithId(pair.B) as IMyMotorStator;
            if (tip != null && top.Half.X <= tip.Grid.GridSize * .5 + 1e-6 && mount.Tag == pair.B && rotor != null && Vector3D.Dot(rotor.WorldMatrix.Up, arm.TipUp) > .998 &&
                Contains(top, (Vector3D)arm.TipCell * tip.Grid.GridSize, tip.Grid.GridSize * .5)) return true;
        }
    }
    return false;
}
// A vanilla shaft collar used as the mounting face of the next stator is
// part of that stator's immediate joint interface. Its padded box must not
// oppose the next head's rotation. This exception covers those two exact
// heads only; actuator bodies, armour and all other cells keep their checks.
bool MountedCollars(long fixedId, long movingId)
{
    IMyCubeBlock fixedPart, movingPart; Vector3D min, max;
    if (!Parts.TryGetValue(fixedId, out fixedPart) || !Parts.TryGetValue(movingId, out movingPart) ||
        !(fixedPart is IMyMotorRotor || fixedPart is IMyPistonTop) || !(movingPart is IMyMotorRotor) ||
        !ModelBounds(fixedPart, out min, out max) || !ModelBounds(movingPart, out min, out max)) return false;
    bool owned = false;
    foreach (var previous in Joints) if (Attached(previous) && previous.Top.EntityId == fixedId) { owned = true; break; }
    if (!owned) return false;
    List<IMyMechanicalConnectionBlock> joints;
    if (!Bases.TryGetValue(fixedPart.CubeGrid.EntityId, out joints)) return false;
    foreach (var joint in joints) {
        var stator = joint as IMyMotorStator;
        if (stator == null || !Attached(joint) || joint.Top.EntityId != movingId || !ModelBounds(stator, out min, out max)) continue;
        string kind = stator.BlockDefinition.SubtypeName, fixedKind = fixedPart.BlockDefinition.SubtypeName;
        Vector3D face = kind == "LargeHinge" ? stator.WorldMatrix.Right : stator.WorldMatrix.Down;
        Vector3D mount = fixedKind == "LargeHingeHead" ? fixedPart.WorldMatrix.Left : fixedPart.WorldMatrix.Up;
        double size = fixedPart.CubeGrid.GridSize;
        if (Vector3D.Dot(face, mount) < -.999 &&
            Vector3D.Distance(stator.GetPosition() + face * size, fixedPart.GetPosition()) < size * .01 &&
            Vector3D.Distance(stator.GetPosition(), movingPart.GetPosition()) < size * .5 + .05) return true;
    }
    return false;
}
// Only joint-base volumes against their own mounted assembly get model bounds.
// Armour, other arms and unrelated ship contacts keep their original cell boxes.
bool InterfacePair(GridShape a, Shape x, GridShape b, Shape y)
{
    return InterfaceSide(a, x, b, y) || InterfaceSide(b, y, a, x);
}
bool InterfaceSide(GridShape parent, Shape baseBox, GridShape child, Shape box)
{
    IMyCubeBlock part; List<IMyMechanicalConnectionBlock> joints;
    if (!Parts.TryGetValue(baseBox.Tag, out part) || part.CubeGrid.EntityId != parent.Grid.EntityId || !Bases.TryGetValue(parent.Grid.EntityId, out joints)) return false;
    foreach (var joint in joints)
    {
        if (parent.Grid.EntityId != joint.CubeGrid.EntityId || !Attached(joint)) continue;
        bool valid = baseBox.Tag == joint.EntityId;
        if (!valid) foreach (var previous in Joints) if (Attached(previous) && previous.Top.CubeGrid.EntityId == parent.Grid.EntityId && previous.Top.EntityId == baseBox.Tag) { valid = true; break; }
        if (!valid) continue;
        if (child.Grid.EntityId == joint.TopGrid.EntityId) return true;
        List<IMyMechanicalConnectionBlock> nexts;
        if (!Bases.TryGetValue(joint.TopGrid.EntityId, out nexts)) continue;
        foreach (var next in nexts) if (Attached(next) && next.CubeGrid.EntityId == joint.TopGrid.EntityId &&
            child.Grid.EntityId == next.TopGrid.EntityId && box.Tag == next.Top.EntityId) return true;
    }
    return false;
}
static bool Attached(IMyMechanicalConnectionBlock joint)
{ return !joint.Closed && joint.IsAttached && !joint.PendingAttachment && joint.Top != null && !joint.Top.Closed && joint.TopGrid != null && joint.Top.CubeGrid.EntityId == joint.TopGrid.EntityId; }
bool ClipModel(GridShape grid, Shape box, out Shape result, out bool known)
{
    result = box; known = false; IMyCubeBlock block; Shape model; Vector3D min, max;
    if (!Parts.TryGetValue(box.Tag, out block) || block.Closed || block.CubeGrid.EntityId != grid.Grid.EntityId) return true;
    if (!Models.TryGetValue(box.Tag, out model)) {
        if (ModelBounds(block, out min, out max)) {
            var basis = block.WorldMatrix * MatrixD.Invert(World(grid)); var half = (max - min) * .5;
            model = new Shape { Centre = Vector3D.Transform((max + min) * .5, basis), Half = Abs(basis.Right) * half.X + Abs(basis.Up) * half.Y + Abs(basis.Backward) * half.Z, Tag = box.Tag };
        }
        Models[box.Tag] = model;
    }
    if (model == null) return true; known = true;
    min = Vector3D.Max(box.Centre - box.Half, model.Centre - model.Half); max = Vector3D.Min(box.Centre + box.Half, model.Centre + model.Half);
    if (min.X >= max.X || min.Y >= max.Y || min.Z >= max.Z) return false;
    result = new Shape { Centre = (min + max) * .5, Half = (max - min) * .5, Tag = box.Tag }; return true;
}
static Vector3D Abs(Vector3D v) { return new Vector3D(Math.Abs(v.X), Math.Abs(v.Y), Math.Abs(v.Z)); }
static bool ModelBounds(IMyCubeBlock block, out Vector3D min, out Vector3D max)
{
    min = max = Vector3D.Zero;
    if (Math.Abs(block.CubeGrid.GridSize - 2.5) > 1e-6) return false;
    string name = block.BlockDefinition.SubtypeName;
    // Bounds from the installed vanilla MWM assets, rounded outward + 25 mm.
    if (block is IMyMotorStator && !(block is IMyMotorSuspension)) {
        if (name == "LargeAdvancedStator" || name == "LargeStator") { min = new Vector3D(-1.247, -1.265, -1.247); max = new Vector3D(1.247, .55, 1.247); }
        else if (name == "LargeHinge") { min = new Vector3D(-.925, -1.103, -.925); max = new Vector3D(1.251, 1.103, .925); }
        else return false;
    } else if (block is IMyPistonBase) {
        if (name == "LargePistonBase") { min = new Vector3D(-1.247, -2.527, -1.247); max = new Vector3D(1.247, 1.804, 1.247); }
        else if (name == "LargePistonBaseReskin") { min = new Vector3D(-1.255, -2.502, -1.259); max = new Vector3D(1.255, 1.417, 1.259); }
        else return false;
    } else if (block is IMyMotorRotor) {
        if (name == "LargeHingeHead") { min = new Vector3D(-1.251, -.782, -1.131); max = new Vector3D(1.131, .782, 1.131); }
        else if (name == "LargeAdvancedRotor" || name == "LargeRotor") { min = new Vector3D(-1.126, -.612, -1.053); max = new Vector3D(1.126, 1.251, 1.053); }
        else return false;
    } else if (block is IMyPistonTop) {
        if (name == "LargePistonTop") { min = new Vector3D(-1.249, .757, -1.249); max = new Vector3D(1.249, 1.251, 1.249); }
        else if (name == "LargePistonTopReskin") { min = new Vector3D(-1.257, .886, -1.251); max = new Vector3D(1.257, 1.252, 1.251); }
        else return false;
    }
    else return false;
    min -= new Vector3D(.025); max += new Vector3D(.025);
    // Fail closed if a replacement/changed model exceeds the baked envelope.
    var native = block.WorldAABB;
    if (!Finite(native.Min.X) || !Finite(native.Min.Y) || !Finite(native.Min.Z) || !Finite(native.Max.X) || !Finite(native.Max.Y) || !Finite(native.Max.Z) ||
        native.Max.X <= native.Min.X || native.Max.Y <= native.Min.Y || native.Max.Z <= native.Min.Z) return false;
    var world = block.WorldMatrix; var centre = Vector3D.Transform((min + max) * .5, world); var half = (max - min) * .5;
    var extent = Abs(world.Right) * half.X + Abs(world.Up) * half.Y + Abs(world.Backward) * half.Z + new Vector3D(.001);
    return native.Min.X >= centre.X - extent.X && native.Min.Y >= centre.Y - extent.Y && native.Min.Z >= centre.Z - extent.Z &&
        native.Max.X <= centre.X + extent.X && native.Max.Y <= centre.Y + extent.Y && native.Max.Z <= centre.Z + extent.Z;
}
static Vector3D MergeNormal(IMyShipMergeBlock block, string axis)
{
    string name = axis.ToLowerInvariant();
    if (name == "auto")
    {
        string subtype = block.BlockDefinition.SubtypeName;
        name = subtype == "SmallShipSmallMergeBlock" ? "up" : subtype == "LargeShipMergeBlock" || subtype == "SmallShipMergeBlock" ? "right" : "";
    }
    switch (name)
    {
        case "right": return block.WorldMatrix.Right; case "left": return block.WorldMatrix.Left;
        case "up": return block.WorldMatrix.Up; case "down": return block.WorldMatrix.Down;
        case "forward": return block.WorldMatrix.Forward; case "backward": return block.WorldMatrix.Backward;
        default: return Vector3D.Zero;
    }
}

            // for the guide's .0005 native-unit settling tolerance.
            if(rotor!=null){lo=rotor.RotorLock?q:Math.Max(q-Math.PI,rotor.LowerLimitRad> -1e6?rotor.LowerLimitRad+.1*Math.PI/180+.0005:rotor.LowerLimitRad);hi=rotor.RotorLock?q:Math.Min(q+Math.PI,rotor.UpperLimitRad<1e6?rotor.UpperLimitRad-.1*Math.PI/180-.0005:rotor.UpperLimitRad);}
            else{var piston=(IMyPistonBase)m.Block;lo=piston.MinLimit==piston.MaxLimit?q:piston.MinLimit+.0015;hi=piston.MinLimit==piston.MaxLimit?q:piston.MaxLimit-.0015;}

        }

        MemberStart.Add(values);NativeStart[i]=values[0];var first=group.Members[0];JointLocal.Add(first.Block.WorldMatrix*MatrixD.Invert(Captured[first.Block.CubeGrid.EntityId]));

public void Audit()
{
    if(AuditGrids==null)return;
    for(int work=0;work<64&&Budget();work++)
    {
        var grid=AuditGrids[AuditIndex];bool expected=false;List<Shape> boxes;
        if(grid.Tiles.TryGetValue(new Vector3I(Tile(AuditCell.X),Tile(AuditCell.Y),Tile(AuditCell.Z)),out boxes))
            foreach(var box in boxes)if(Contains(box,(Vector3D)AuditCell*grid.Grid.GridSize,0)){expected=true;break;}
        if(grid.Grid.CubeExists(AuditCell)!=expected)throw new Exception("Interior geometry changed; Rescan Collision and plan again.");
        AuditCell.X++;if(AuditCell.X>grid.Max.X){AuditCell.X=grid.Min.X;AuditCell.Y++;}if(AuditCell.Y>grid.Max.Y){AuditCell.Y=grid.Min.Y;AuditCell.Z++;}
        if(AuditCell.Z>grid.Max.Z){AuditIndex=(AuditIndex+1)%AuditGrids.Count;AuditCell=AuditGrids[AuditIndex].Min;}
    }
}
public bool Still(double[] current=null)
{
    var root=Scene[Model.Hull].Grid.WorldMatrix;var oldRoot=Captured[Model.Hull];
    Dictionary<long,MatrixD> expected=null;
    if(current==null){
        var q=new double[Model.Groups.Count];for(int i=0;i<q.Length;i++){var m=Model.Groups[i].Members[0];double d=Position(m.Block)-MemberStart[i][0];if(Model.Groups[i].Rotary)d=Math.Atan2(Math.Sin(d),Math.Cos(d));q[i]=d*m.Sign;}
        var saved=new Dictionary<long,MatrixD>(Poses);try{FK(q);expected=new Dictionary<long,MatrixD>(Poses);}finally{Poses.Clear();foreach(var pair in saved)Poses.Add(pair.Key,pair.Value);}
    }
    foreach(var grid in Scene.Values)
    {
        if(grid.Grid.Closed||grid.Grid.Min!=grid.Min||grid.Grid.Max!=grid.Max){Status="Grid bounds/identity changed: "+grid.Grid.EntityId;return false;}
        if(current!=null&&Model.Bodies.ContainsKey(grid.Grid.EntityId))continue;
        var old=(expected!=null&&Model.Bodies.ContainsKey(grid.Grid.EntityId)?expected[grid.Grid.EntityId]:Captured[grid.Grid.EntityId])*MatrixD.Invert(oldRoot);var now=grid.Grid.WorldMatrix*MatrixD.Invert(root);
        if(Vector3D.Distance(old.Translation,now.Translation)>.003||Vector3D.Dot(old.Forward,now.Forward)<.999999||Vector3D.Dot(old.Up,now.Up)<.999999){Status="Grid moved: "+grid.Grid.EntityId+" by "+Number(Vector3D.Distance(old.Translation,now.Translation))+" m";return false;}
    }
    for(int i=0;i<Model.Groups.Count;i++)for(int j=0;j<Model.Groups[i].Members.Count;j++)
    {var m=Model.Groups[i].Members[j];if(!Attached(m.Block)){Status="Joint detached: "+m.Block.EntityId;return false;}double error=Position(m.Block)-MemberStart[i][j];if(m.Block is IMyMotorStator)error=Math.Atan2(Math.Sin(error),Math.Cos(error));if(current==null&&Math.Abs(error)>.0005){Status="Joint moved: "+m.Block.EntityId;return false;}}
    return true;
}
public MatrixD FK(double[] q)
{
    Poses.Clear();foreach(var pair in Captured)Poses[pair.Key]=pair.Value;
    for(int i=0;i<Model.Groups.Count;i++)
    {
        var group=Model.Groups[i];var member=group.Members[0];var block=JointLocal[i]*Poses[member.Block.CubeGrid.EntityId];
        double sign=(member.Block.CubeGrid.EntityId==member.From?1:-1)*member.Sign;
        var rotate=MatrixD.CreateFromAxisAngle(-block.Up*sign,q[i]);
        foreach(var pair in Model.Bodies)if((pair.Value&(1<<i))!=0)
        {
            var pose=Poses[pair.Key];if(group.Rotary){var point=block.Translation+Vector3D.TransformNormal(pose.Translation-block.Translation,rotate);pose=pose.GetOrientation()*rotate;pose.Translation=point;}
            else pose.Translation+=block.Up*(sign*q[i]);Poses[pair.Key]=pose;
        }
    }
    return HeadLocal*Poses[Model.Tool];
}
public void Column(int i,Vector3D head,out Vector3D linear,out Vector3D angular)
{
    if(Upper[i]-Lower[i]<1e-10){linear=angular=Vector3D.Zero;return;}
    var group=Model.Groups[i];var m=group.Members[0];var block=JointLocal[i]*Poses[m.Block.CubeGrid.EntityId];double sign=(m.Block.CubeGrid.EntityId==m.From?1:-1)*m.Sign;
    angular=group.Rotary?-block.Up*sign:Vector3D.Zero;linear=group.Rotary?Vector3D.Cross(angular,head-block.Translation):block.Up*sign;
}

public void Begin(double[] q,double[] halfDelta=null)
{
    if(halfDelta!=null)halfDelta=halfDelta.Select(value=>Math.Abs(value)+.0005).ToArray();
    Sweep=halfDelta;PadBody=PadObstacle=-1;CheckingHead=FK(q);Padding.Clear();Rods.Clear();
    foreach(var joint in Joints)if(joint is IMyPistonBase&&Attached(joint))
    {
        MatrixD local;if(!PartLocal.TryGetValue(joint.Top.EntityId,out local)||!Poses.ContainsKey(joint.TopGrid.EntityId))continue;
        var parent=Poses[joint.CubeGrid.EntityId];var inverse=MatrixD.Invert(parent);
        var start=Vector3D.Transform((PartLocal[joint.EntityId]*parent).Translation,inverse);
        var end=Vector3D.Transform((local*Poses[joint.TopGrid.EntityId]).Translation,inverse);
        var axis=Vector3D.TransformNormal(joint.WorldMatrix.Up,MatrixD.Transpose(joint.CubeGrid.WorldMatrix.GetOrientation()));
        var rod=new Shape{Centre=(start+end)*.5,Half=Abs(end-start)*.5+(Vector3D.One-Abs(axis))*(joint.CubeGrid.GridSize*.5),Tag=joint.EntityId,Rod=true};
        List<Shape> list;if(!Rods.TryGetValue(joint.CubeGrid.EntityId,out list)){list=new List<Shape>();Rods.Add(joint.CubeGrid.EntityId,list);}list.Add(rod);
    }
    foreach(var pair in Model.Bodies)
    {
        var grid=Scene[pair.Key];var centre=Vector3D.Transform(((Vector3D)grid.Min+(Vector3D)grid.Max)*(grid.Grid.GridSize*.5),World(grid));
        double radius=(((Vector3D)grid.Max-(Vector3D)grid.Min+Vector3D.One)*(grid.Grid.GridSize*.5)).Length(),pad=0;
        List<Shape> rods;if(Rods.TryGetValue(pair.Key,out rods))foreach(var rod in rods)
        {radius=Math.Max(radius,Vector3D.Distance(Vector3D.Transform(rod.Centre,World(grid)),centre)+rod.Half.Length());int group;if(halfDelta!=null&&PistonGroups.TryGetValue(rod.Tag,out group))pad=Math.Max(pad,Math.Abs(halfDelta[group]));}
        if(halfDelta!=null)for(int i=Model.Groups.Count-1;i>=0;i--)if((pair.Value&(1<<i))!=0)
        {
            var m=Model.Groups[i].Members[0];var pivot=(JointLocal[i]*Poses[m.Block.CubeGrid.EntityId]).Translation;
            pad+=Model.Groups[i].Rotary?Math.Abs(halfDelta[i])*(Vector3D.Distance(centre,pivot)+radius+pad):Math.Abs(halfDelta[i]);
        }
        Padding[pair.Key]=pad;
    }
    Bodies=new List<long>(Model.Bodies.Keys.OrderBy(id=>id));Obstacles=new List<long>(Scene.Keys.OrderBy(id=>id));
    BodyCursor=ObstacleCursor=BoxCursor=CandidateCursor=0;BodyBoxes=CandidateBoxes=null;Checked=false;Clear=true;

public void CheckStep()
{
    while(!Checked&&Budget())
    {
        if(BodyCursor==Bodies.Count){Checked=true;return;}
        var body=Scene[Bodies[BodyCursor]];
        if(ObstacleCursor==Obstacles.Count){BodyCursor++;ObstacleCursor=BoxCursor=0;BodyBoxes=CandidateBoxes=null;continue;}
        var obstacle=Scene[Obstacles[ObstacleCursor]];
        if(PadBody!=body.Grid.EntityId||PadObstacle!=obstacle.Grid.EntityId){PadBody=body.Grid.EntityId;PadObstacle=obstacle.Grid.EntityId;int first,second;Model.Bodies.TryGetValue(PadBody,out first);Model.Bodies.TryGetValue(PadObstacle,out second);int shared=first&second;PairBodyPad=RelativePadding(body,shared);PairObstaclePad=RelativePadding(obstacle,shared);}
        double pad=PairBodyPad,otherPad=PairObstaclePad;
        if(obstacle.Grid.EntityId!=Model.Hull&&Model.Bodies.ContainsKey(obstacle.Grid.EntityId)&&obstacle.Grid.EntityId<body.Grid.EntityId||!Rods.ContainsKey(body.Grid.EntityId)&&!Rods.ContainsKey(obstacle.Grid.EntityId)&&!NearGrids(body,obstacle,Clearance+pad+otherPad))
        {ObstacleCursor++;BoxCursor=0;BodyBoxes=CandidateBoxes=null;continue;}
        if(BodyBoxes==null){BodyBoxes=body.Grid.EntityId==Model.Hull?new List<Shape>():new List<Shape>(BodyShapes(Model,body));List<Shape> rods;if(Rods.TryGetValue(body.Grid.EntityId,out rods))BodyBoxes.AddRange(rods.Where(r=>Model.OwnParts.Contains(r.Tag)));}
        if(BoxCursor==BodyBoxes.Count){ObstacleCursor++;BoxCursor=0;BodyBoxes=CandidateBoxes=null;continue;}
        var a=BodyBoxes[BoxCursor];
        if(obstacle==body&&!a.Rod){BoxCursor++;CandidateBoxes=null;continue;}
        if(CandidateBoxes==null){CandidateBoxes=obstacle==body?new List<Shape>(obstacle.Shapes):Candidates(body,a,obstacle,Clearance+pad+otherPad);List<Shape> rods;if(Rods.TryGetValue(obstacle.Grid.EntityId,out rods))CandidateBoxes.AddRange(rods);CandidateCursor=0;}
        if(CandidateCursor==CandidateBoxes.Count){BoxCursor++;CandidateBoxes=null;continue;}
        var b=CandidateBoxes[CandidateCursor++];if(a.Tag==b.Tag&&a.Rod||Contact(Model,body,a,obstacle,b))continue;
        Shape x=a,y=b;bool knownX=false,knownY=false;bool assembly=InterfacePair(body,a,obstacle,b);
        if(assembly||Model.OwnParts.Contains(a.Tag)&&Model.OwnParts.Contains(b.Tag))
        {if(!a.Rod&&!ClipModel(body,a,out x,out knownX)||!b.Rod&&!ClipModel(obstacle,b,out y,out knownY))continue;}
        x=new Shape{Centre=x.Centre,Half=x.Half+new Vector3D(pad),Tag=x.Tag};y=new Shape{Centre=y.Centre,Half=y.Half+new Vector3D(otherPad),Tag=y.Tag};
        double clearance=assembly?knownX&&knownY?-.075:knownX||knownY?-.04:Clearance:Clearance;Vector3D axis;
        double gap=Separation(body,x,obstacle,y,out axis);
        if(gap<clearance){Status="Blocked grids "+body.Grid.EntityId+"/"+obstacle.Grid.EntityId+" parts "+x.Tag+"/"+y.Tag+" gap "+Number(gap);Checked=true;Clear=false;return;}
    }
}
double RelativePadding(GridShape grid,int excluded)
{
    if(Sweep==null)return 0;int mask;if(!Model.Bodies.TryGetValue(grid.Grid.EntityId,out mask))return 0;
    if(excluded==0)return Padding[grid.Grid.EntityId];
    var centre=Vector3D.Transform(((Vector3D)grid.Min+(Vector3D)grid.Max)*(grid.Grid.GridSize*.5),World(grid));
    double radius=(((Vector3D)grid.Max-(Vector3D)grid.Min+Vector3D.One)*(grid.Grid.GridSize*.5)).Length(),pad=0;List<Shape> rods;
    if(Rods.TryGetValue(grid.Grid.EntityId,out rods))foreach(var rod in rods){radius=Math.Max(radius,Vector3D.Distance(Vector3D.Transform(rod.Centre,World(grid)),centre)+rod.Half.Length());int group;if(PistonGroups.TryGetValue(rod.Tag,out group))pad=Math.Max(pad,Math.Abs(Sweep[group]));}
    for(int i=Model.Groups.Count-1;i>=0;i--)if((mask&(1<<i))!=0&&(excluded&(1<<i))==0){var m=Model.Groups[i].Members[0];var pivot=(JointLocal[i]*Poses[m.Block.CubeGrid.EntityId]).Translation;pad+=Model.Groups[i].Rotary?Math.Abs(Sweep[i])*(Vector3D.Distance(centre,pivot)+radius+pad):Math.Abs(Sweep[i]);}return pad;
}
