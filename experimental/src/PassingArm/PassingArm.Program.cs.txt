public Program()
{
    Runtime.UpdateFrequency = UpdateFrequency.Update10;
    IGC.UnicastListener.SetMessageCallback(WireTag);
    try
    {
        if (!Saved.TryParse(Storage)) { Saved.Clear(); Saved.EndContent = Storage; }
        if (Saved.Get("AutoArm Host", "PB").ToInt64() == Me.EntityId) { InputClock = Saved.Get("AutoArm Host", "InputClock").ToInt64(); InputOwner = Saved.Get("AutoArm Host", "InputOwner").ToInt64(); }
        { Selection = IGC.RegisterBroadcastListener(SelectionTag); if (Selection != null) Selection.SetMessageCallback(WireTag); }
        {StopSaved();
}
        LoadHost();
        string prior = Saved.Get("AutoArm Host", "Selected").ToString();
        if (Saved.Get("AutoArm Host", "PB").ToInt64() == Me.EntityId && Arms.ContainsKey(prior)) Selected = prior;
        SelectArm(Selected.Length == 0 ? Order[0] : Selected, InputOwner == 0);
        Save();
    }
    catch (Exception e) { StopAll(e.Message); }
    Display();
}
public void Save()
{
    if (Order.Count == 0) return;
    bool changed = Saved.Get("AutoArm Host", "PB").ToInt64() != Me.EntityId || Saved.Get("AutoArm Host", "Selected").ToString() != Selected || Saved.Get("AutoArm Host", "InputClock").ToInt64() != InputClock || Saved.Get("AutoArm Host", "InputOwner").ToInt64() != InputOwner;
    if (changed) { Saved.Set("AutoArm Host", "Format", 1); Saved.Set("AutoArm Host", "PB", Me.EntityId); Saved.Set("AutoArm Host", "Selected", Selected); Saved.Set("AutoArm Host", "InputClock", InputClock); Saved.Set("AutoArm Host", "InputOwner", InputOwner); }
    foreach (var pair in Arms) if (pair.Value.Storage != pair.Value.SavedData) { Saved.Set("State " + pair.Key, "Data", pair.Value.Storage); pair.Value.SavedData = pair.Value.Storage; changed = true; }
    if (!changed) return;
    string text = Saved.ToString(); if (text != Storage) Storage = text;
}
void StopSaved()
{
    var sections = new List<string>(); Saved.GetSections(sections);
    foreach (string section in sections)
    {
        if (!section.StartsWith("State ", StringComparison.Ordinal)) continue;
        var state = new MyIni(); if (!state.TryParse(Saved.Get(section, "Data").ToString())) continue;
        ArmCore.SavedDrives.Stop(state, section.Substring(6), Me, GridTerminalSystem);
    }
}
void LoadHost()
{
    var ini = new MyIni();
    if (string.IsNullOrWhiteSpace(Me.CustomData))
    {
        ini.Set(GlobalSection, "Format", HostFormat); ini.Set(GlobalSection, "SelectedArm", DefaultArm);
        var bases = new List<IMyMechanicalConnectionBlock>(); GridTerminalSystem.GetBlocksOfType(bases); var detected = new List<string>();
        foreach (var block in bases)
        {
            int split = block.CustomName.IndexOf(" - Base", StringComparison.Ordinal);
            if (split > 0 && !block.Closed && block.IsSameConstructAs(Me) && !detected.Contains(block.CustomName.Substring(0, split))) detected.Add(block.CustomName.Substring(0, split));
        }
        if (detected.Count == 0) detected.Add(DefaultArm); detected.Sort(StringComparer.Ordinal);
        foreach (string name in detected) ini.Set(name, "ToolSwapPB", "");
        Me.CustomData = ini.ToString();
    }
    if (!ini.TryParse(Me.CustomData) || ini.Get(GlobalSection, "Format").ToInt32() != HostFormat)
        throw new Exception("Custom Data requires [global] Format=" + HostFormat + ". Delete Custom Data.");
    var names = new List<string>(); ini.GetSections(names); names.Remove(GlobalSection);
    if (names.Count < 1 || names.Count > 8) throw new Exception("Configure 1..8 named arm sections.");
    foreach (string name in names) if (name.Trim() != name || name.Length == 0 || name.IndexOfAny(new char[] { ',', '|', '(', ')', ';', '\r', '\n' }) >= 0 || name.Equals(GlobalSection, StringComparison.OrdinalIgnoreCase)) throw new Exception("Invalid arm section name.");
    {
        var peers = new List<IMyProgrammableBlock>(); GridTerminalSystem.GetBlocksOfType(peers);
        foreach (var peer in peers)
        {
            if (peer == Me || peer.Closed || !peer.IsSameConstructAs(Me)) continue;
            var data = new MyIni(); if (!data.TryParse(peer.CustomData) || data.Get(GlobalSection, "Format").ToInt32() != 7) continue;
            foreach (string name in names) if (data.ContainsSection(name)) throw new Exception("Arm already assigned to another PB: " + name);
        }
    }
    var removed = new List<string>(); foreach (string name in Order) if (!names.Contains(name)) removed.Add(name);
    foreach (string name in removed) { Arms[name].Stop("Arm removed from configuration."); Saved.Set("State " + name, "Data", Arms[name].Storage); Arms.Remove(name); }
    Configuration = ini; Order.Clear(); Order.AddRange(names);
    AutoLinks();
    foreach (string name in names) if (!Arms.ContainsKey(name))
    {
        string state = Saved.Get("AutoArm Host", "PB").ToInt64() == Me.EntityId ? Saved.Get("State " + name, "Data").ToString() : "";
        Arms.Add(name, new ArmCore(this, name, ConfigFor(name), state));
    }
    if (!Arms.ContainsKey(Selected))
    { string configured = ini.Get(GlobalSection, "SelectedArm").ToString(); Selected = Arms.ContainsKey(configured) ? configured : names[0]; }
    {foreach (var core in Arms.Values) core.Select(core.Name == Selected && InputOwner == Me.EntityId);
}
    HostError = "";
}
string Effective(string arm, string key)
{ return Configuration.Get(Configuration.ContainsKey(arm, key) ? arm : GlobalSection, key).ToString().Replace("{arm}", arm); }
void AutoLinks()
{
    var peers = new List<IMyProgrammableBlock>(); GridTerminalSystem.GetBlocksOfType(peers);
    foreach (string name in Order)
    {
        string key = "ToolSwapPB";
        if (Effective(name, key).Length > 0) continue;
        IMyProgrammableBlock found = null;
        foreach (var peer in peers)
        {
            if (peer == Me || peer.Closed || !peer.IsFunctional || !peer.IsSameConstructAs(Me)) continue;
            var data = new MyIni();
            if (!data.TryParse(peer.CustomData) || data.Get(GlobalSection, "Format").ToInt32() != (2) || !data.ContainsSection(name)) continue;
            if ((!data.Get(data.ContainsKey(name, "Tools.Enabled") ? name : GlobalSection, "Tools.Enabled").ToBoolean(true))) continue;
            if (found != null) throw new Exception("Multiple module PBs for " + name + ". Set " + key + ".");
            found = peer;
        }
        if (found != null) Configuration.Set(name, key, "@" + found.EntityId);
    }
}
MyIni ConfigFor(string name)
{
    var ini = new MyIni();
    var keys = new List<MyIniKey>(); Configuration.GetKeys(GlobalSection, keys);
    var own = new List<MyIniKey>(); Configuration.GetKeys(name, own); keys.AddRange(own);
    var seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
    foreach (var item in keys)
    {
        string key = item.Name; if (!seen.Add(key) || key.Equals("Format", StringComparison.OrdinalIgnoreCase) || key.Equals("SelectedArm", StringComparison.OrdinalIgnoreCase) || key.Equals("Structure", StringComparison.OrdinalIgnoreCase) || key.Equals("Actuators", StringComparison.OrdinalIgnoreCase)) continue;
        string section = "Config", field = key; int dot = key.IndexOf('.');
        if (dot > 0) { section = key.Substring(0, dot); field = key.Substring(dot + 1); }
        else if (key.Equals("ToolSwapPB", StringComparison.OrdinalIgnoreCase) || key.Equals("Peers", StringComparison.OrdinalIgnoreCase)) section = "Modules";
        ini.Set(section, field, Effective(name, key));
    }
    return ini;
}
bool Layout(ArmCore core, out string why)
{
    why = ""; string name = core.Name, table = Effective(name, "Actuators"), expected = Effective(name, "Structure");
    bool shared = !Configuration.ContainsKey(name, "Actuators") && Configuration.ContainsKey(GlobalSection, "Actuators") && table.Trim().Length > 0;
    if (shared && Configuration.ContainsKey(name, "Structure")) { why = "A local Structure requires local Actuators; clear it to inherit the global template."; return false; }
    if (shared && expected.Length == 0)
    {
        ArmCore prototype;
        if (!Arms.TryGetValue(Order[0], out prototype) || !prototype.HasLayout) { why = "Waiting for global structure prototype: " + Order[0]; return false; }
        expected = prototype.Shape; Configuration.Set(GlobalSection, "Structure", expected);
    }
    string shape = core.Shape;
    if (expected.Length > 0 && expected != shape) { why = name + " does not match configured joint kinds/parallel layout."; return false; }
    if (!core.ApplyWeights(table, out why)) return false;
    foreach (var joint in core.Joints())
    {
        ArmCore owner;
        if (Owners.TryGetValue(joint.EntityId, out owner) && owner != core && Arms.ContainsKey(owner.Name))
        { owner.Stop("Actuator also claimed by " + name); why = "Actuator shared with " + owner.Name + ": " + joint.CustomName; return false; }
    }
    var release = new List<long>(); foreach (var pair in Owners) if (pair.Value == core) release.Add(pair.Key);
    foreach (long id in release) Owners.Remove(id);
    foreach (var joint in core.Joints()) Owners[joint.EntityId] = core;
    if (!shared && table.Trim().Length == 0) Configuration.Set(name, "Actuators", core.WeightTable);
    PublishConfig(); return true;
}
void PublishConfig()
{
    // Generated values only fill missing keys. Preserve edits made while running.
    var live = new MyIni(); if (!live.TryParse(Me.CustomData) || live.Get(GlobalSection, "Format").ToInt32() != HostFormat) return;
    var keys = new List<MyIniKey>(); Configuration.GetKeys(keys);
    foreach (var key in keys) if (!live.ContainsKey(key.Section, key.Name)) live.Set(key.Section, key.Name, Configuration.Get(key).ToString());
    string text = live.ToString(); if (text != Me.CustomData) Me.CustomData = text;
}
void SelectArm(string name, bool claim = true)
{
    if (!Arms.ContainsKey(name)) throw new Exception("Unknown arm: " + name);
    Selected = name;
    { if (claim) { InputClock++; InputOwner = Me.EntityId; } foreach (var core in Arms.Values) core.Select(core.Name == name && InputOwner == Me.EntityId); if (InputOwner == Me.EntityId) AnnounceInput(); }
    Save();
}
void Command(string raw)
{
    string text = raw.Trim(); if (text.Length == 0) return;
    if (text.StartsWith("Select ", StringComparison.OrdinalIgnoreCase)) { SelectArm(text.Substring(7).Trim()); return; }
    if (text.Equals("StopAll", StringComparison.OrdinalIgnoreCase)) { StopAll("StopAll command."); return; }
    int end = text.IndexOfAny(new char[] { ' ', '(', ',' }); string first = (end < 0 ? text : text.Substring(0, end)).ToLowerInvariant();
    if (first == "on" || first == "reload" || first == "rescan")
    { try { LoadHost(); if ((InputOwner == 0)) SelectArm(Selected); } catch (Exception e) { StopAll(e.Message); throw; } }
    string name = Selected, command = text; int open = text.IndexOf('(');
    if (open > 0)
    {
        if (!text.EndsWith(")", StringComparison.Ordinal)) throw new Exception("Command needs closing parenthesis.");
        string verb = text.Substring(0, open).Trim(); var values = new List<string>();
        foreach (string value in text.Substring(open + 1, text.Length - open - 2).Split(',')) if (value.Trim().Length > 0) values.Add(value.Trim());
        if (values.Count > 0 && Arms.ContainsKey(values[values.Count - 1])) { name = values[values.Count - 1]; values.RemoveAt(values.Count - 1); }
        else if (values.Count > Arity(verb)) throw new Exception("Unknown arm argument.");
        command = verb + (values.Count == 0 ? "" : " " + string.Join(" ", values));
    }
    else
    {
        int comma = text.LastIndexOf(',');
        if (comma >= 0) { name = text.Substring(comma + 1).Trim(); command = text.Substring(0, comma).Trim(); }
    }
    ArmCore target; if (!Arms.TryGetValue(name, out target)) throw new Exception("Unknown arm: " + name);
    string action = command.Split(' ')[0].ToLowerInvariant();
    if (action == "on" || action == "reload" || action == "rescan")
    { target.Config = ConfigFor(name); target.ConfigChanged = false; }
    target.Update(command, UpdateType.Terminal, 0);
    CaptureDefaults(target);
}
static int Arity(string verb)
{
    switch (verb.ToLowerInvariant())
    {
        case "on": case "off": case "stop": case "hold": case "check": case "info": case "joints": case "reload": case "rescan": case "sethome": case "gohome": case "toolscan": case "toolinfo": case "swapcancel": case "version": case "status": case "bindings": case "homeinfo": case "forward": case "back": case "left": case "right": case "up": case "down": case "yawleft": case "yawright": case "pitchup": case "pitchdown": case "rollleft": case "rollright": return 0;
        case "moveto": return 3;
        case "orientto": return 6;
        case "movetoposori": return 9;
        case "homespeed": case "step": return 2;
        default: return 1;
    }
}
void AnnounceInput()
{ if (Selection != null) IGC.SendBroadcastMessage(SelectionTag, InputClock + "|" + Selected, TransmissionDistance.CurrentConstruct); }
void ReceiveSelection(double elapsed)
{
    if (Selection == null) return;
    int read = 0;
    while (Selection.HasPendingMessage && read++ < 8)
    {
        var message = Selection.AcceptMessage(); string text = message.Data as string; long clock;
        if (message.Tag != SelectionTag || message.Source == Me.EntityId || text == null || text.Length > 512) continue;
        string[] cells = text.Split('|');
        if (cells.Length != 2 || !long.TryParse(cells[0], out clock) || clock <= 0 || clock < InputClock || clock == InputClock && message.Source <= InputOwner) continue;
        var peer = GridTerminalSystem.GetBlockWithId(message.Source) as IMyProgrammableBlock; var config = new MyIni();
        if (peer == null || peer.Closed || !peer.IsFunctional || !peer.IsSameConstructAs(Me) || !config.TryParse(peer.CustomData) || config.Get(GlobalSection, "Format").ToInt32() != 7 || !config.ContainsSection(cells[1])) continue;
        InputClock = clock; InputOwner = message.Source;
        foreach (var core in Arms.Values) core.Select(false);
    }
    if (elapsed > 0 && !double.IsInfinity(elapsed) && !double.IsNaN(elapsed)) InputAge += elapsed;
    if (InputAge >= .1 && InputOwner == Me.EntityId) { InputAge = 0; AnnounceInput(); }
}
void CaptureDefaults(ArmCore core)
{
    if (!core.ConfigChanged) return;
    core.ConfigChanged = false;
    var ini = core.Config;
    var keys = new List<MyIniKey>(); ini.GetKeys(keys);
    foreach (var key in keys)
    {
        if (key.Section == "AutoArm" || key.Name == "Actuators") continue;
        string field = key.Section == "Config" || key.Section == "Modules" ? key.Name : key.Section + "." + key.Name;
        if (!Configuration.ContainsKey(core.Name, field) && !Configuration.ContainsKey(GlobalSection, field))
            Configuration.Set(core.SharedSetting(field) ? GlobalSection : core.Name, field, ini.Get(key).ToString());
    }
    PublishConfig();
}
public void Main(string argument, UpdateType source)
{
    double elapsed = Runtime.TimeSinceLastRun.TotalSeconds;
    try { {ReceiveSelection(elapsed); } } catch (Exception e) { StopAll(e.Message); }
    try
    {
        if (!string.IsNullOrWhiteSpace(argument) && !((source & UpdateType.IGC) != 0 && argument == WireTag))
            foreach (string raw in argument.Split(new char[] { ';', '\n' }, StringSplitOptions.RemoveEmptyEntries)) Command(raw);
    }
    catch (Exception e) { HostError = e.Message; }
    try
    {
        int read = 0;
        while (IGC.UnicastListener.HasPendingMessage && read++ < 16)
        {
            if (Runtime.MaxInstructionCount > 0 && Runtime.CurrentInstructionCount >= Runtime.MaxInstructionCount * .4) break;
            var message = IGC.UnicastListener.AcceptMessage(); var data = message.Data as string; var ini = new MyIni(); ArmCore core;
            if (message.Tag != WireTag || data == null || data.Length > 16000 || !ini.TryParse(data) || !Arms.TryGetValue(ini.Get("Link", "Arm").ToString(), out core)) continue;
            if (core.Inbox.Count < 32) core.Inbox.Enqueue(message);
        }
        int count = Order.Count;
        for (int i = 0; i < count; i++)
        {
            if (Runtime.MaxInstructionCount > 0 && Runtime.CurrentInstructionCount >= Runtime.MaxInstructionCount * .7) { StopAll("Shared PB instruction budget exhausted."); break; }
            var core = Arms[Order[(Next + i) % count]]; core.Update("", source, elapsed); CaptureDefaults(core);
        }
        if (count > 0) Next = (Next + 1) % count;
        Save();
    }
    catch (Exception e) { StopAll(e.Message); }
    finally
    {
        UpdateFrequency frequency = UpdateFrequency.Update10;
        foreach (var core in Arms.Values) if ((core.Frequency & UpdateFrequency.Update1) != 0) frequency = UpdateFrequency.Update1;
        Runtime.UpdateFrequency = frequency; Display();
    }
}
void StopAll(string why)
{ HostError = why; foreach (var core in Arms.Values) core.Stop(why); }
void Display()
{
    var lines = new List<string>(); lines.Add("AutoArm " + ControllerBuildVersion + " " + ("Arm") + " | Selected: " + Selected);
    if ((InputOwner != Me.EntityId)) lines.Add("Cockpit input inactive here. Run Select " + Selected + ".");
    if (HostError.Length > 0) lines.Add(HostError);
    foreach (string name in Order) lines.Add(name + ": " + Arms[name].Summary);
    ArmCore current; if (Arms.TryGetValue(Selected, out current)) lines.Add(current.Report);
    Echo(string.Join("\n", lines));
}

    public bool Discover(out string why)
    {
        ZeroOwned(); Building = true;
        why = ""; Ready = false; StableCount = 0; Groups.Clear(); Rigid.Clear();
        Edges.Clear(); Unowned.Clear(); Used.Clear(); Corridor.Clear(); Watch.Clear(); Report = Fingerprint = "";
        if (Arm.Length == 0) return Bad("ArmName is empty.", out why);
        if (!Guard("pre-inventory reserve", out why)) return false;
        All.Clear(); Inventory.Clear(); P.GridTerminalSystem.GetBlocks(Inventory);
        if (Inventory.Count > MaxInventory) return Bad("Terminal inventory exceeds the " + MaxInventory + "-block discovery bound.", out why);
        if (!Guard("terminal inventory", out why)) return false;
        Base = null; Head = null; int bases = 0, heads = 0;
        for (int i = 0; i < Inventory.Count; i++)
        {
            if ((i & 63) == 0 && !Guard("terminal inventory filter", out why)) return false;
            var b = Inventory[i]; if (!b.IsSameConstructAs(P.Me)) continue;
            if (HeadMark(b)) { Head = b; heads++; }

            All.Add(m); if (Mark(m.CustomName, "Base")) { Base = m; bases++; }
        }
        if (bases != 1) return Bad("Need exactly one mechanical block named '" + Arm + " - Base*'; found " + bases + ".", out why);

            if (!Attached(All[i])) continue;
            var e = new AutoEdge { B = All[i], A = All[i].CubeGrid.EntityId, Z = All[i].TopGrid.EntityId,
                K = Kind(All[i]), BaseMark = Mark(All[i].CustomName, "Base"), HeadMark = HeadMark(All[i]) };
            if (e.A == e.Z) return Bad("Self-loop mechanical connection: " + e.B.CustomName, out why);
            Edges.Add(e);
        }
        if (!ScopeEdges(out why)) return false;
        var root = Between(Hull, Start);
        if (!Has(root, Base.EntityId)) return Bad("Base marker attachment changed while scanning.", out why);
        if (!AddGroup(root, out why)) return false;
        Corridor.Add(Start); Corridor.Add(Hull);
        var dist = Distances(EndGrid, out why); if (dist == null) return false;
        int d;
        if (!dist.TryGetValue(Start, out d)) return Bad("Head route returns through hull or is unreachable.", out why);
        long at = Start;
        while (at != EndGrid)
        {
            if (!Guard("corridor recognition", out why)) return false;
            if (Groups.Count >= MaxGroups) return Bad("Active arm exceeds " + MaxGroups + " generalized groups.", out why);
            int here = dist[at];
            var nextEdges = new List<AutoLink>();
            var destinations = new List<long>();
            for (int i = 0; i < Edges.Count; i++)
            {
                var e = Edges[i]; if (Used.Contains(e.B.EntityId)) continue;
                long n = e.A == at ? e.Z : e.Z == at ? e.A : 0; int nd;
                if (n == 0 || !dist.TryGetValue(n, out nd) || nd != here - 1) continue;
                nextEdges.Add(new AutoLink(e, at, n)); if (!destinations.Contains(n)) destinations.Add(n);
            }
            if (destinations.Count == 0) return Bad("No unambiguous next stage from rigid grid " + at + ".", out why);
            if (destinations.Count == 1)
            {
                if (!AddGroup(nextEdges, out why)) return false;
                at = destinations[0]; Corridor.Add(at);
                continue;
            }
            var paths = new List<List<AutoLink>>();
            var candidate = new HashSet<long>();
            for (int i = 0; i < destinations.Count; i++)
            {
                long n = destinations[i]; var path = new List<AutoLink>(); int count = 0;
                for (int j = 0; j < nextEdges.Count; j++) if (nextEdges[j].V == n) { path.Add(nextEdges[j]); count++; }
                if (count != 1) return Bad("Unsupported nested/parallel bundle at split grid " + at + ".", out why);
                candidate.Add(path[0].E.B.EntityId);
                paths.Add(path);
            }
            if (OwnedCount() + candidate.Count > MaxMembers) return Bad("Split/rejoin candidate exceeds " + MaxMembers + " owned actuators.", out why);
            int guard = 0;
            while (true)
            {
                if (!Guard("split/rejoin recognition", out why)) return false;
                long meet = paths[0][paths[0].Count - 1].V; bool same = true;
                for (int i = 1; i < paths.Count; i++) if (paths[i][paths[i].Count - 1].V != meet) same = false;
                if (same) { at = meet; break; }
                if (++guard > MaxMembers) return Bad("Parallel paths do not reconverge within the supported bound.", out why);
                var occupied = new HashSet<long>();
                for (int i = 0; i < paths.Count; i++)
                {
                    if (!Guard("split/rejoin path expansion", out why)) return false;
                    var path = paths[i]; long u = path[path.Count - 1].V;
                    if (!occupied.Add(u)) return Bad("Parallel paths merge or cross before a common stage boundary.", out why);
                    AutoLink chosen = null; int count = 0; int ud = dist[u];
                    for (int j = 0; j < Edges.Count; j++)
                    {
                        if ((j & 31) == 0 && !Guard("split/rejoin edge search", out why)) return false;
                        var e = Edges[j]; if (Used.Contains(e.B.EntityId) || candidate.Contains(e.B.EntityId)) continue;
                        long v = e.A == u ? e.Z : e.Z == u ? e.A : 0; int vd;
                        if (v != 0 && dist.TryGetValue(v, out vd) && vd == ud - 1) { chosen = new AutoLink(e, u, v); count++; }
                    }
                    if (count != 1) return Bad("Unsupported branch/cross-link inside split path at grid " + u + ".", out why);
                    candidate.Add(chosen.E.B.EntityId);
                    if (OwnedCount() + candidate.Count > MaxMembers) return Bad("Split/rejoin candidate exceeds " + MaxMembers + " owned actuators.", out why);
                    path.Add(chosen);
                }
            }
            int stages = paths[0].Count;
            for (int i = 1; i < paths.Count; i++) if (paths[i].Count != stages)
                return Bad("Parallel arms have unequal stage counts; variable-ratio closure is unsupported.", out why);
            for (int s = 0; s < stages; s++)
            {
                var stage = new List<AutoLink>();
                for (int i = 0; i < paths.Count; i++) { stage.Add(paths[i][s]); Corridor.Add(paths[i][s].U); Corridor.Add(paths[i][s].V); }
                if (!AddGroup(stage, out why)) return false;
            }
        }
        if (!ClassifyUnused(out why)) return false;
        foreach (var edge in Edges) if (!Used.Contains(edge.B.EntityId)) Unowned.Add(edge);
        if (!BuildWatch(out why)) return false;
        if (!SnapshotFromInventory(out Snapshot, out why)) return Bad(why, out why);
        var fp = new System.Text.StringBuilder();
        for (int i = 0; i < Groups.Count; i++) { Groups[i].Index = i; if (i > 0) fp.Append('|'); fp.Append(Groups[i].Sig); }
        Fingerprint = fp.ToString();
        Rigid.AddRange(Corridor); Rigid.Sort();
        Report += Groups.Count + " groups / " + OwnedCount() + " actuators / " + Rigid.Count + " rigid segments detected.";
        for (int i = 0; i < Rigid.Count; i++) Report += "\nR" + (i + 1).ToString("00") + " = grid " + Rigid[i];
        for (int i = 0; i < Groups.Count; i++) if (Groups[i].M.Count > 1)
        { Report += " Parallel 1:1 copies: rechecked live."; break; }
        Report += "\nCached checks each pass; full scan every 30.";
        FullScanIn = 30;
        Building = false; ZeroOwned();
        return true;
    }

    bool ScopeEdges(out string why)
    {
        why = "";
        var byGrid = new Dictionary<long, List<AutoEdge>>();
        for (int i = 0; i < Edges.Count; i++)
        {
            if ((i & 63) == 0 && !Guard("moving-component adjacency", out why)) return false;
            var e = Edges[i]; List<AutoEdge> a, z;
            if (!byGrid.TryGetValue(e.A, out a)) byGrid[e.A] = a = new List<AutoEdge>();
            if (!byGrid.TryGetValue(e.Z, out z)) byGrid[e.Z] = z = new List<AutoEdge>();
            a.Add(e); z.Add(e);
        }
        var keep = new HashSet<long>(); var nodes = new HashSet<long>(); var q = new Queue<long>();
        nodes.Add(Start); q.Enqueue(Start);
        while (q.Count > 0)
        {
            if (!Guard("moving-component scope", out why)) return false;
            long u = q.Dequeue(); List<AutoEdge> list;
            if (!byGrid.TryGetValue(u, out list)) continue;
            for (int i = 0; i < list.Count; i++)
            {
                if ((i & 31) == 0 && !Guard("moving-component expansion", out why)) return false;
                var e = list[i]; keep.Add(e.B.EntityId);
                if (keep.Count > MaxWorkEdges) return Bad("Moving mechanical component exceeds the " + MaxWorkEdges + "-edge discovery bound.", out why);
                long v = e.A == u ? e.Z : e.A;
                if (v != Hull && nodes.Add(v)) q.Enqueue(v);
            }
        }
        var scoped = new List<AutoEdge>();
        for (int i = 0; i < Edges.Count; i++)
        {
            if ((i & 63) == 0 && !Guard("moving-component filter", out why)) return false;
            if (keep.Contains(Edges[i].B.EntityId)) scoped.Add(Edges[i]);
        }
        Edges.Clear(); Edges.AddRange(scoped);
        return true;
    }

    bool Has(List<AutoLink> links, long id)
    {
        for (int i = 0; i < links.Count; i++) if (links[i].E.B.EntityId == id) return true;
        return false;
    }

    List<AutoLink> Between(long u, long v)
    {
        var r = new List<AutoLink>();
        for (int i = 0; i < Edges.Count; i++)
        {
            var e = Edges[i];
            if ((e.A == u && e.Z == v) || (e.A == v && e.Z == u)) r.Add(new AutoLink(e, u, v));
        }
        return r;
    }

    Dictionary<long, int> Distances(long target, out string why)
    {
        why = ""; var d = new Dictionary<long, int>(); var q = new Queue<long>(); d[target] = 0; q.Enqueue(target);
        while (q.Count > 0)
        {
            if (!Guard("Head-distance graph", out why)) return null;
            long u = q.Dequeue(); int nd = d[u] + 1;
            for (int i = 0; i < Edges.Count; i++)
            {
                var e = Edges[i]; long v = e.A == u ? e.Z : e.Z == u ? e.A : 0;
                if (v == 0 || v == Hull || d.ContainsKey(v)) continue;
                d[v] = nd; q.Enqueue(v);
            }
        }
        return d;
    }

    bool AddGroup(List<AutoLink> links, out string why)
    {
        why = ""; if (links.Count == 0) return Bad("Empty mechanical stage.", out why);
        if (Groups.Count >= MaxGroups) return Bad("Active Base-to-Head corridor exceeds " + MaxGroups + " generalized groups.", out why);
        if (OwnedCount() + links.Count > MaxMembers) return Bad("Active Base-to-Head corridor exceeds " + MaxMembers + " owned actuators.", out why);
        links.Sort((x, y) => x.E.B.EntityId.CompareTo(y.E.B.EntityId));
        int kind = Kind(links[0].E.B);
        if (kind == 0) return Bad("Unsupported connection on active route: " + links[0].E.B.CustomName, out why);
        var g = new AutoGroup { Rotary = kind == 1 };
        Vector3D baseL = Vector3D.Zero, baseA = Vector3D.Zero;
        for (int i = 0; i < links.Count; i++)
        {
            var l = links[i];
            if (Kind(l.E.B) != kind) return Bad("Parallel stage mixes piston and rotary actuators.", out why);
            var m = new AutoMember { B = l.E.B, Edge = l.E, U = l.U, V = l.V, Sign = 1 };
            Vector3D lin, ang;
            if (!Raw(m, EndPosition, out lin, out ang, out why)) return Bad(why, out why);
            if (i == 0) { baseL = lin; baseA = ang; }
            else
            {
                double dot = kind == 1 ? Vector3D.Dot(baseA, ang) : Vector3D.Dot(baseL, lin);
                m.Sign = dot < 0 ? -1 : 1; lin *= m.Sign; ang *= m.Sign;
                bool compatible = kind == 1
                    ? Vector3D.Dot(baseA, ang) >= AxisCos && (baseL - lin).Length() <= RotaryLineM
                    : Vector3D.Dot(baseL, lin) >= AxisCos;
                if (!compatible) return Bad("Parallel " + (kind == 1 ? "rotors/hinges lack one coaxial full twist" : "pistons are not collinear") +
                    " at stage containing '" + l.E.B.CustomName + "'. General linkage ratios are unsupported.", out why);
            }
            m.Q0 = m.Last = Position(m);
            if (!Finite(m.Q0)) return Bad("Non-finite actuator position: " + m.B.CustomName, out why);
            g.M.Add(m); Used.Add(m.B.EntityId);
        }
        var sig = new System.Text.StringBuilder(kind == 1 ? "R" : "P");
        var label = new System.Text.StringBuilder(kind == 1 ? "Rotary: " : "Piston: ");
        for (int i = 0; i < g.M.Count; i++)
        {
            var m = g.M[i]; sig.Append(':').Append(m.B.EntityId).Append(m.Sign > 0 ? '+' : '-').Append('@').Append(m.U).Append('>').Append(m.V);
            if (i > 0) label.Append(" | "); label.Append(m.B.CustomName);
        }
        g.Sig = sig.ToString(); g.Label = label.ToString(); Groups.Add(g);
        Report += "G" + Groups.Count + " " + g.Label + "\n";
        return true;
    }

    bool Raw(AutoMember m, Vector3D head, out Vector3D lin, out Vector3D ang, out string why, bool liveChecked = false)
    {
        lin = ang = Vector3D.Zero; why = ""; var b = m.B;
        if (!liveChecked && !Attached(b)) { why = "Detached actuator: " + b.CustomName; return false; }
        long a = b.CubeGrid.EntityId, z = b.TopGrid.EntityId; int dir;
        if (m.Edge != null && (a != m.Edge.A || z != m.Edge.Z))
        { why = "Actuator endpoints changed: " + b.CustomName; return false; }
        if (a == m.U && z == m.V) dir = 1; else if (a == m.V && z == m.U) dir = -1;
        else { why = "Actuator endpoints changed: " + b.CustomName; return false; }
        Vector3D axis = b.WorldMatrix.Up; double n = axis.Length();
        if (!Finite(axis.X) || !Finite(axis.Y) || !Finite(axis.Z) || !Finite(head.X) || !Finite(head.Y) || !Finite(head.Z) ||
            !Finite(n) || n < .9) { why = "Invalid actuator axis or Head position: " + b.CustomName; return false; }
        axis /= n;
        if (b is IMyPistonBase) lin = axis * dir;
        else { ang = axis * -dir; lin = Vector3D.Cross(ang, head - b.GetPosition()); }
        return true;
    }

    public double Position(AutoMember m)
    {
        var p = m.B as IMyPistonBase; return p != null ? p.CurrentPosition : ((IMyMotorStator)m.B).Angle;
    }



    public bool CheckSync(AutoGroup g, out string why)
    {
        why = "";
        double lo = double.PositiveInfinity, hi = double.NegativeInfinity, spread = 0;
        for (int i = 0; i < g.M.Count; i++)
        {
            var m = g.M[i]; double q = Position(m);
            if (!Finite(q) || !Finite(m.Q0)) { why = "Non-finite actuator position in G" + (g.Index + 1) + ": " + m.B.CustomName; return false; }
            double d = q - m.Q0; if (g.Rotary) d = Wrap(d); d *= m.Sign;
            if (g.Rotary) for (int j = 0; j < i; j++)
            {
                var n = g.M[j]; double nd = Wrap(Position(n) - n.Q0) * n.Sign;
                spread = Math.Max(spread, Math.Abs(Wrap(d - nd)));
            }
            lo = Math.Min(lo, d); hi = Math.Max(hi, d);
        }
        if (g.M.Count < 2) return true;
        double tol = g.Rotary ? .25 * Math.PI / 180 : .02;
        if (!g.Rotary) spread = hi - lo;
        if (spread <= tol) return true;
        why = "Parallel actuator desynchronization in G" + (g.Index + 1) + ": " + spread.ToString("0.0000") + (g.Rotary ? " rad." : " m.");
        return false;
    }

    public bool Refresh(AutoGroup g, Vector3D headWorld, MatrixD reference, out double lo, out double hi, out string why)
    {
        lo = -1e6; hi = 1e6; why = ""; Vector3D sl = Vector3D.Zero, sa = Vector3D.Zero, firstL = Vector3D.Zero, firstA = Vector3D.Zero;
        if (g.M.Count > 1 && !CheckSync(g, out why)) return Fail(why);
        for (int i = 0; i < g.M.Count; i++)
        {
            var m = g.M[i]; var b = m.B;
            if (!Attached(b) || !b.Enabled || !b.IsWorking) { why = "Unavailable owned actuator: " + b.CustomName; return Fail(why); }
            if (!Finite(m.Q0) || Mark(b.CustomName, "Base") != m.Edge.BaseMark || HeadMark(b) != m.Edge.HeadMark)
            { why = "Owned actuator identity/marker changed: " + b.CustomName; return Fail(why); }
            Vector3D l, a; if (!Raw(m, headWorld, out l, out a, out why, true)) return Fail(why);
            l *= m.Sign; a *= m.Sign;
            if (i == 0) { firstL = l; firstA = a; }
            else if (g.Rotary ? Vector3D.Dot(firstA, a) < AxisCos || (firstL - l).Length() > RotaryLineM : Vector3D.Dot(firstL, l) < AxisCos)
            { why = "Parallel screw mismatch G" + (g.Index + 1) + "; not 1:1."; return Fail(why); }
            sl += l; sa += a;
            double ml, mh; if (!Bounds(m, out ml, out mh, out why)) return Fail(why); lo = Math.Max(lo, ml); hi = Math.Min(hi, mh);
        }
        if (lo > hi) { why = "No common signed velocity remains in G" + (g.Index + 1) + "."; return Fail(why); }
        sl /= g.M.Count; sa /= g.M.Count;
        g.Lin = InFrame(sl, reference); g.Ang = InFrame(sa, reference);
        return true;
    }

    bool Bounds(AutoMember m, out double lo, out double hi, out string why)
    {
        lo = -1e6; hi = 1e6; why = ""; double q = Position(m), mn, mx;
        var p = m.B as IMyPistonBase;
        if (p != null) { lo = -p.MaxVelocity; hi = p.MaxVelocity; mn = p.MinLimit; mx = p.MaxLimit; }
        else
        {
            var r = (IMyMotorStator)m.B; mn = r.LowerLimitRad; mx = r.UpperLimitRad;
            if (r.RotorLock) lo = hi = 0;
        }
        if (!Finite(q) || !Finite(mn) || !Finite(mx) || !Finite(lo) || !Finite(hi) || mn > mx || lo > hi)
        { why = "Invalid position, limits, or maximum velocity: " + m.B.CustomName; return false; }
        if (p != null && p.MaxVelocity <= 0) { why = "Piston reports no positive maximum velocity: " + m.B.CustomName; return false; }
        double accel = p != null ? P.PistonAccelerationMps2 : P.JointAccelerationRpmPerSec * Math.PI / 30;
        double margin = p != null ? .001 : .1 * Math.PI / 180;
        if (!Finite(accel) || accel <= 0) { why = "Acceleration must be finite and positive for travel braking."; return false; }
        if (mn > -1e6) lo = Math.Max(lo, -Math.Sqrt(2 * accel * Math.Max(0, q - mn - margin)));
        if (mx < 1e6) hi = Math.Min(hi, Math.Sqrt(2 * accel * Math.Max(0, mx - q - margin)));
        if (m.Sign < 0) { double x = lo; lo = -hi; hi = -x; }
        return true;
    }



    public void Set(AutoGroup g, double velocity)
    {
        for (int i = 0; i < g.M.Count; i++)
        {
            var m = g.M[i]; float v = (float)(m.Sign * velocity);
            var p = m.B as IMyPistonBase; if (p != null) p.Velocity = v; else ((IMyMotorStator)m.B).TargetVelocityRad = v;
        }
    }

    public void ZeroOwned()
    {
        for (int i = 0; i < Groups.Count; i++) for (int j = 0; j < Groups[i].M.Count; j++)
        {
            var b = Groups[i].M[j].B; if (b == null || b.Closed) continue;
            try { var p = b as IMyPistonBase; if (p != null) p.Velocity = 0; else { var r = b as IMyMotorStator; if (r != null) r.TargetVelocityRad = 0; } }
            catch (Exception) { }
        }
    }

    public bool StableSetup(out string why)
    {
        why = ""; if (!Validate(out why, true)) return false;
        bool moved = false;
        for (int i = 0; i < Groups.Count; i++) for (int j = 0; j < Groups[i].M.Count; j++)
        {
            var m = Groups[i].M[j]; double target, q = Position(m);
            if (!Finite(q)) { why = "Non-finite actuator position while settling: " + m.B.CustomName; StableCount = 0; return false; }
            var p = m.B as IMyPistonBase; target = p != null ? p.Velocity : ((IMyMotorStator)m.B).TargetVelocityRad;
            if (Math.Abs(target) > 1e-4) { why = "Waiting for owned actuator targets to reach zero."; StableCount = 0; return false; }
            double delta = q - m.Last; if (Groups[i].Rotary) delta = Wrap(delta);
            if (Math.Abs(delta) > 1e-4) moved = true; m.Last = q;
        }
        StableCount = moved ? 0 : StableCount + 1;
        if (StableCount < 2) { why = "Waiting for two stable stopped topology scans."; return false; }
        for (int i = 0; i < Groups.Count; i++) for (int j = 0; j < Groups[i].M.Count; j++) Groups[i].M[j].Q0 = Position(Groups[i].M[j]);
        Ready = true; return true;
    }

    public bool Validate(out string why, bool full = false, bool refreshBeforeCommit = false)
    {
        why = "";
        if (!BudgetAvailable()) { why = "Budget low; motion refused."; return Fail(why); }
        if (Base == null || Head == null || Base.Closed || Head.Closed || Base.EntityId != BaseId || Head.EntityId != HeadId ||
            !Base.IsSameConstructAs(P.Me) || !Head.IsSameConstructAs(P.Me) || !Mark(Base.CustomName, "Base") || !HeadMark(Head))
        { why = "Base/Head marker disappeared or was renamed."; return Fail(why); }
        if (Head.CubeGrid == null || Head.CubeGrid.EntityId != HeadGrid) { why = "Head marker moved to another rigid grid; Rescan is required."; return Fail(why); }
        var marker = Head as IMyMechanicalConnectionBlock;
        if (End == null || End.Closed || End.EntityId != EndId || End.CubeGrid == null || End.CubeGrid.EntityId != EndGrid ||
            Focus != null && (Focus.Closed || Focus.CubeGrid.EntityId != EndGrid || Focus.Marker.CubeGrid.EntityId != EndGrid) ||
            Focus == null && End != Head && marker != null && (!Attached(marker) || marker.Top.EntityId != EndId))
        { why = "Endpoint changed; Rescan is required."; return Fail(why); }
        var checkedEdges = !full && refreshBeforeCommit ? Unowned : Edges;
        for (int i = 0; i < checkedEdges.Count; i++)
        {
            var e = checkedEdges[i]; var b = e.B;
            // Active control refreshes every owned group before any output. Its
            // Raw/Bounds/sync checks cover these edges once, rather than twice.
            if (!Attached(b) || b.CubeGrid.EntityId != e.A || b.TopGrid.EntityId != e.Z || Kind(b) != e.K ||
                Mark(b.CustomName, "Base") != e.BaseMark || HeadMark(b) != e.HeadMark)
            { why = "Known mechanical connection changed: " + (b == null ? "missing block" : b.CustomName) + ". Rescan is required."; return Fail(why); }
        }
        if (full || !refreshBeforeCommit)
            for (int i = 0; i < Groups.Count; i++) if (!CheckSync(Groups[i], out why)) return Fail(why);
        if (!full && --FullScanIn > 0) return true;
        string current;
        if (!CurrentSnapshot(out current, out why)) return Fail(why);
        if (ScanBases != 1 || ScanHeads != 1) { why = "Base/Head marker is no longer unique; Rescan is required."; return Fail(why); }
        if (current != Snapshot) { why = "Mechanical topology changed; Rescan is required."; return Fail(why); }
        FullScanIn = 30;
        return true;
    }

    bool ClassifyUnused(out string why)
    {
        why = ""; var seen = new HashSet<long>(); int ignored = 0;
        for (int i = 0; i < Edges.Count; i++)
        {
            var seed = Edges[i]; if (Used.Contains(seed.B.EntityId) || seen.Contains(seed.B.EntityId)) continue;
            var q = new Queue<AutoEdge>(); var boundary = new HashSet<long>(); q.Enqueue(seed); seen.Add(seed.B.EntityId); int count = 0;
            while (q.Count > 0)
            {
                if (!Guard("off-corridor classification", out why)) return false;
                var e = q.Dequeue(); count++;
                if ((e.A == Hull && Corridor.Contains(e.Z)) || (e.Z == Hull && Corridor.Contains(e.A)))
                    return Bad("A second mechanical anchor connects the base grid to the active corridor: " + e.B.CustomName, out why);
                if (Corridor.Contains(e.A)) boundary.Add(e.A); if (Corridor.Contains(e.Z)) boundary.Add(e.Z);
                for (int j = 0; j < Edges.Count; j++)
                {
                    var n = Edges[j]; if (Used.Contains(n.B.EntityId) || seen.Contains(n.B.EntityId)) continue;
                    if (n.A == e.A || n.A == e.Z || n.Z == e.A || n.Z == e.Z) { seen.Add(n.B.EntityId); q.Enqueue(n); }
                }
            }
            if (boundary.Count > 1) return Bad("Unsupported closure: cross-linked segments.", out why);
            ignored += count;
        }
        if (ignored > 0) Report += "Ignored " + ignored + " off-corridor actuators; they are never commanded.\n";
        return true;
    }

    bool BuildWatch(out string why)
    {
        why = "";
        var q = new Queue<long>(); Watch.Add(Start); q.Enqueue(Start);
        while (q.Count > 0)
        {
            if (!Guard("topology watch-set", out why)) return false;
            long u = q.Dequeue();
            for (int i = 0; i < Edges.Count; i++)
            {
                var e = Edges[i]; long v = e.A == u ? e.Z : e.Z == u ? e.A : 0;
                if (v == 0 || v == Hull || !Watch.Add(v)) continue; q.Enqueue(v);
            }
        }
        return true;
    }

    bool CurrentSnapshot(out string snapshot, out string why)
    {
        snapshot = ""; why = "";
        if (!BudgetAvailable()) { why = "Budget low; scan refused."; return false; }
        Inventory.Clear(); P.GridTerminalSystem.GetBlocks(Inventory);
        return SnapshotFromInventory(out snapshot, out why);
    }

    bool SnapshotFromInventory(out string snapshot, out string why)
    {
        snapshot = ""; why = "";
        if (Inventory.Count > MaxInventory) { why = "Terminal inventory exceeds the " + MaxInventory + "-block validation bound."; return false; }
        SnapshotRows.Clear(); ScanBases = ScanHeads = 0;
        for (int i = 0; i < Inventory.Count; i++)
        {
            if ((i & 31) == 0 && !BudgetAvailable()) { why = "Scan budget exhausted."; return false; }
            var terminal = Inventory[i]; if (!terminal.IsSameConstructAs(P.Me)) continue;
            if (HeadMark(terminal)) ScanHeads++;
            var b = terminal as IMyMechanicalConnectionBlock; if (b == null) continue;
            if (b.EntityId == ExcludedMechanical) continue;
            long a = b.CubeGrid.EntityId, z = Attached(b) ? b.TopGrid.EntityId : 0;
            if (Mark(b.CustomName, "Base")) ScanBases++;
            if (Watch.Contains(a) || Watch.Contains(z)) SnapshotRows.Add(b.EntityId + ":" + a + ":" + z + ":" + Kind(b));
        }
        if (!BudgetAvailable()) { why = "Post-scan budget exhausted."; return false; }
        SnapshotRows.Sort(StringComparer.Ordinal); SnapshotText.Clear();
        for (int i = 0; i < SnapshotRows.Count; i++) { if (i > 0) SnapshotText.Append('|'); SnapshotText.Append(SnapshotRows[i]); }
        snapshot = SnapshotText.ToString(); return true;
    }

    int OwnedCount() { int n = 0; for (int i = 0; i < Groups.Count; i++) n += Groups[i].M.Count; return n; }


    bool BudgetAvailable()
    {
        return P.Runtime.MaxInstructionCount <= 0 || P.Runtime.CurrentInstructionCount < P.Runtime.MaxInstructionCount * .55;
    }

    bool Guard(string phase, out string why)
    {
        why = ""; if (BudgetAvailable()) return true;
        return Bad("Budget: " + phase + "; Off. Reduce topology/Rescan.", out why);
    }

    bool Fail(string why) { Ready = false; if (Building) { Groups.Clear(); Used.Clear(); } else ZeroOwned(); return false; }
    bool Bad(string text, out string why) { why = text; Fail(text); return false; }
}

    public static bool ReadWeights(string table, List<AutoGroup> groups, out List<double[]> values, out string why)
    {
        values = new List<double[]>(); why = "";

        if (string.IsNullOrWhiteSpace(table)) return true;

                !long.TryParse(part.Substring(to + 1), out b) || id == 0 || a == 0 || b == 0 || a == b || !ids.Add(id)) return false;
        }
        return true;
    }

public void Main(string argument, UpdateType source)
{
    DisplayWritten = false;
    double elapsed = Elapsed;
    bool timed = (source & (UpdateType.Update1 | UpdateType.Update10 | UpdateType.Update100 | UpdateType.Once)) != 0;
    if (Finite(elapsed) && elapsed > 0)
    {
        if (Enabled || (Tools != null && Tools.Busy)) ControlElapsed += elapsed;
        if (!Topology.Ready) SetupElapsed = Math.Min(0.5, SetupElapsed + elapsed);
        StatusElapsed = Math.Min(StatusRefreshSeconds, StatusElapsed + elapsed);
    }
    try
    {
        if (Tools != null) Tools.Receive(elapsed);
        if (Services != null) Services.Tick(elapsed);
        if (!string.IsNullOrWhiteSpace(argument) && !((source & UpdateType.IGC) != 0 && argument == LinkTag)) RunCommands(argument);
        if (Tools != null && Tools.Busy)
        {
            if (!ConfigValid) { Tools.Cancel("Invalid configuration; Reload required."); return; }
            if (Tools.Automatic && HasPilotInput()) { Tools.Cancel("Pilot canceled swap."); return; }
            if (ControlElapsed > 0.5) { Fault("Tool operation update gap exceeded 0.5 seconds."); return; }
            if (!timed) return;
            if (!Finite(elapsed) || elapsed < 0) { Fault("Invalid tool operation interval."); return; }
            double toolDt = ControlElapsed;
            if (toolDt <= 0.000001) return;
            ControlElapsed = 0;
            Tools.Tick(Math.Min(toolDt, .1), toolDt);
            return;
        }
        if (!Topology.Ready)
        {
            // A validation failure must require a new explicit On. Stable setup may
            // recover diagnostics while Off, but it must never resume old commands.
            if (Enabled || GoingHome)
            {
                StopMotion("Topology lost readiness; stopped. Run Check/Rescan/Reload, then On.");
                Say(Status);
                return;
            }
            if (!timed || !Finite(elapsed) || elapsed < 0 || SetupElapsed <= 0.000001) return;
            SetupElapsed = 0;
            MaintainSetup();
            if (!Topology.Ready) return;
        }
        if (PendingOn)
        {
            PendingOn = false;
            EnableControl();
        }
        if (!Enabled) return;
        if (ControlElapsed > 0.5) { Fault("Control update gap exceeded 0.5 seconds."); return; }
        // Commands can run in the same simulation frame as the previous call.
        // They change state immediately, but only timed updates advance motion.
        if (!timed) return;
        if (!Finite(elapsed) || elapsed < 0) { Fault("Invalid negative/non-finite control interval."); return; }
        double dt = ControlElapsed;
        if (dt <= 0.000001) return;
        ControlElapsed = 0;
        if (dt > 0.5) { Fault("Control update gap exceeded 0.5 seconds."); return; }
        if (LocalPath != null)
        {
            string why; if (HasPilotInput()) { StopMotion("Pilot canceled path."); Say(Status); return; }
            if (!Tools.AllowManual(out why) || !StepPath(LocalPath, Math.Min(dt, .1), dt, out why)) { Fault(why); return; }
            if (LocalPath.Done) { LocalPath = null; EnableControl(); Say("Path complete; ON hold."); }
        }
        else if (GoingHome) UpdateHome(Math.Min(dt, 0.1));
        else UpdateControl(Math.Min(dt, 0.1), dt);
    }
    catch (Exception e)
    {
        Fault("Controller exception: " + e.Message);
    }
    finally
    {
        // SE clears the PB display on every invocation. Replay cached output;
        // routine numeric formatting happens at 4 Hz, not every control tick.
        Echo(DisplayText);
    }
}

{
    DisplayText = DisplayWritten ? DisplayText + "\n" + text : text;
    if (DisplayText.Length > 8000) DisplayText = DisplayText.Substring(DisplayText.Length - 8000);
    DisplayWritten = true;
}

void Discover(bool explicitRequest)
{
    StopMotion("");
    PivotProbe = null;
    SetupElapsed = 0;
    ConfigValid = false;
    string why;
    AutoConfig.EnsureDefaults(this);
    MyIni toolConfig; var options = new AutoSettings();
    if (!AutoConfig.ReadCurrent(this, CustomData, out toolConfig, out why) || !options.Read(toolConfig, out why)) throw new Exception(why);
    if (Tools != null && !Tools.Load(out why)) throw new Exception(why);
    Services.Load(toolConfig);
    if (Tools != null && Tools.Configured)
    {
        string preserved; bool differs;
        if (!AutoConfig.LoadOrGenerate(this, Topology, CustomData, out preserved, out differs, out why, true)) throw new Exception(why);
        foreach (var group in Topology.Groups)
        { double[] values; if (ToolWeights.TryGetValue(ToolGroupKey(group), out values)) { group.MoveW = values[0]; group.TurnW = values[1]; } }
        ApplySettings(options); ConfigValid = true; Frequency = UpdateFrequency.Update1;
        Status = "Waiting for ToolSwap setup."; if (explicitRequest) Say(Status); return;
    }
    Topology.SetEndpointOverride(Tools == null ? null : Tools.Endpoint, Tools == null ? 0 : Tools.IgnoredMount, Tools == null ? null : Tools.EndFrame, Tools == null ? null : Tools.Focus);
    if (!Topology.Discover(out why))
    {
        Status = "Discovery failed: " + why;
        if (explicitRequest) Say(Status);
        return;
    }
    Topology.ZeroOwned();
    string updated;
    bool changed;
    if (!AutoConfig.LoadOrGenerate(this, Topology, CustomData, out updated, out changed, out why))
    {
        Status = "Custom Data refused: " + why;
        if (explicitRequest) Say(Status);
        return;
    }
    ConfigValid = true;
    Allocate(Topology.Groups.Count);
    LoadHome();
    Status = (changed ? "Configuration generated/updated; " : "Configuration loaded; ") +
        Topology.Groups.Count + " groups found. Waiting for stable stopped scans.";
    Frequency = Tools != null && Tools.Busy ? UpdateFrequency.Update1 : UpdateFrequency.Update10;
    if (explicitRequest) Say(Status + "\n" + Topology.Report);
}

bool ReadPose(out Vector3D p, out Vector3D f, out Vector3D u)
{
    p = f = u = Vector3D.Zero;
    if (Topology == null || Topology.Base == null || Topology.End == null || Topology.Base.Closed || Topology.End.Closed) return false;
    MatrixD r = Topology.Base.WorldMatrix;
    p = InFrame(Topology.EndPosition - Topology.Base.GetPosition(), r);
    var pose = Topology.EndPose;

    u = InFrame(pose.Up, r);

}

void UpdateControl(double dt, double observationDt)
{
    string why;
    if (!ToolDriving && Tools != null && !Tools.AllowManual(out why)) { Fault(why); return; }
    if (!Topology.Validate(out why, false, true)) { Fault("Topology changed: " + why); return; }
    Vector3D p, f, u;
    if (!ReadPose(out p, out f, out u)) { Fault("Base/Head marker became unavailable."); return; }
    if (!TargetValid) { Fault("No held pose. Run On."); return; }
    Vector3D measuredV = Vector3D.Zero, measuredW = Vector3D.Zero;
    if (ObservationValid)
    {
        measuredV = (p - PreviousP) / observationDt;
        measuredW = RotationError(PreviousF, PreviousU, f, u) / observationDt;
        double blend = observationDt / (0.15 + observationDt);
        FilteredV += (measuredV - FilteredV) * blend;
        FilteredW += (measuredW - FilteredW) * blend;
    }
    PreviousP = p; PreviousF = f; PreviousU = u; ObservationValid = true;
    var safety = Services == null ? null : Services.Safety;
    if (safety != null && (!safety.Ready || safety.Blocked)) { safety.Hold(); return; }

    Vector3D pe = TargetP - p;
    Vector3D oe = RotationError(f, u, TargetF, TargetU);
    LastPositionError = pe.Length();
    LastOrientationErrorDeg = oe.Length() * 180 / Math.PI;
    if (!Finite(LastPositionError) || !Finite(LastOrientationErrorDeg)) { Fault("Non-finite pose feedback."); return; }

    if (ToolDriving) LastPilotLinear = LastPilotAngular = Vector3D.Zero;
    else ReadPilot(f, u, out LastPilotLinear, out LastPilotAngular);
    bool manualTurn = LastPilotAngular.LengthSquared() > 1e-12;
    if (manualTurn) AimingTarget = false;
    if (AimingTarget && LastOrientationErrorDeg <= AimToleranceDeg) AimingTarget = false;
    bool aiming = ToolDriving || manualTurn || AimingTarget;
    double orientationBand = ToolDriving ? ToolOrientationBand : (aiming ? 0.5 * AimToleranceDeg : OrientationToleranceDeg) * Math.PI / 180;
    Vector3D positionDemand = Outside(pe, ToolDriving ? ToolPositionBand : PositionDeadbandM) * PositionKp;
    Vector3D orientationDemand = Outside(oe, orientationBand) * OrientationKp;
    if (ToolDriving)
    {
        positionDemand = ClampLength(positionDemand, ToolDriveMoveCap);
        orientationDemand = ClampLength(orientationDemand, ToolDriveTurnCap);
    }
    // Measured-motion damping stays inside additive, bounded correction.
    // Preserve the idle deadbands rather than chase measurement jitter.
    double pd = ToolDriving || LastPilotLinear.LengthSquared() > 1e-12 || LastPositionError > PositionDeadbandM ? PositionDamping : 0;
    double od = ToolDriving || manualTurn || oe.Length() > orientationBand ? OrientationDamping : 0;
    Vector3D rawPositionFeedback = positionDemand + PositionI - FilteredV * pd;
    Vector3D rawOrientationFeedback = orientationDemand + OrientationI - FilteredW * od;
    Vector3D boundedPositionFeedback = ClampLength(rawPositionFeedback, ToolDriving ? ToolDriveMoveCap : PositionCorrectionCap);
    LastFeedbackLinear = BlendManualCorrection(LastPilotLinear, boundedPositionFeedback,
        ManualCounterCorrectionFraction, out LastCounterCorrectionLimited);
    bool angularLimited;
    LastFeedbackAngular = BlendManualCorrection(LastPilotAngular, ClampLength(rawOrientationFeedback,
        ToolDriving ? ToolDriveTurnCap : Math.Min(TurnDeg, OrientationCorrectionCapDeg) * Math.PI / 180),
        ManualCounterCorrectionFraction, out angularLimited);
    LastCounterCorrectionLimited |= angularLimited;

    // Core invariant: tracking correction is added to live pilot demand. Pose error,
    // recovery state and target advancement never suppress pilot input. Explicit
    // collision supervision can restrict accepted motion after demand formation.
    LastRequestedLinear = LastPilotLinear + LastFeedbackLinear;
    LastRequestedAngular = LastPilotAngular + LastFeedbackAngular;
    double motionScale = safety == null ? 1 : safety.Scale;
    LastRequestedLinear *= motionScale; LastRequestedAngular *= motionScale;
    if (!Solve(p, dt, LastRequestedLinear, LastRequestedAngular, out LastAchievedLinear,
        out LastAchievedAngular, out why))
    {
        Fault("Velocity allocation failed: " + why);
        return;
    }
    if (!Commit()) return;

    Vector3D positionClipResidual = rawPositionFeedback - LastFeedbackLinear * motionScale;
    Vector3D orientationClipResidual = rawOrientationFeedback - LastFeedbackAngular * motionScale;
    // Only loss caused by command bounds inhibits bias learning. Damping and
    // unreachable task components exist even in the unconstrained reference.
    Vector3D positionAllocationResidual = LastUnconstrainedLinear - LastAchievedLinear;
    Vector3D orientationAllocationResidual = LastUnconstrainedAngular - LastAchievedAngular;
    UpdateIntegral(ref PositionI, ToolDriving ? positionDemand - FilteredV : pe, ToolDriving ? 0 : PositionDeadbandM, positionClipResidual, positionAllocationResidual,
        ToolDriving ? PositionKi / PositionKp : PositionKi, Math.Min(PositionIntegralCap, ToolDriving ? ToolDriveMoveCap : PositionCorrectionCap), ToolDriving ? double.MaxValue : 0.10, dt, LastSolverConverged && !LastSolverBudgetLimited);
    UpdateIntegral(ref OrientationI, ToolDriving ? orientationDemand - FilteredW : oe, ToolDriving ? 0 : orientationBand, orientationClipResidual, orientationAllocationResidual, ToolDriving ? OrientationKi / OrientationKp : OrientationKi,
        Math.Min(OrientationIntegralCapDeg * Math.PI / 180, ToolDriving ? ToolDriveTurnCap : Math.Min(TurnDeg, OrientationCorrectionCapDeg) * Math.PI / 180), ToolDriving ? double.MaxValue : 10.0 * Math.PI / 180,
        dt, LastSolverConverged && !LastSolverBudgetLimited);

    LastLinearAdvance = PilotAdvance(LastPilotLinear, LastAchievedLinear - LastFeedbackLinear * motionScale);
    LastAngularAdvance = PilotAdvance(LastPilotAngular, LastAchievedAngular - LastFeedbackAngular * motionScale);
    if (safety != null) { LastLinearAdvance = Math.Min(LastLinearAdvance, motionScale * safety.Fraction); LastAngularAdvance = Math.Min(LastAngularAdvance, motionScale * safety.Fraction); }
    // Allocation is a kinematic prediction, not measured delivery. Bound newly
    // requested target lead even when physical contact prevents predicted motion.
    LastLinearAdvance *= TargetLeadAdvance(pe, LastPilotLinear * (dt * LastLinearAdvance), MaximumPilotTargetLeadM);
    TargetP += LastPilotLinear * (dt * LastLinearAdvance);
    if (LastPilotAngular.LengthSquared() > 1e-20 && LastAngularAdvance > 0)
    {
        Vector3D step = LastPilotAngular * (dt * LastAngularAdvance);
        TargetF = Rotate(TargetF, step); TargetU = Rotate(TargetU, step);
        Orthonormalise(ref TargetF, ref TargetU);
    }
    RenderRoutineStatus();
}

{
    if (DisplayWritten || DisplayPinned || StatusElapsed < StatusRefreshSeconds) return;
    StatusElapsed = 0;
    RoutineStatusBuilds++;
    Status = (ToolDriving ? "PATH motion | remaining " : "ON | input+feedback | error ") + (ToolDriving ? PathRemainingDistance : LastPositionError).ToString(ThreeDecimals) + " m / " +
        LastOrientationErrorDeg.ToString("0.00") + " deg";
    if (LastCounterCorrectionLimited) Status += " | manual priority";
    if (ToolDriving)
    { Say(Status + "\nSpeed command " + LastRequestedLinear.Length().ToString(ThreeDecimals) + " m/s. Stop cancels the path."); return; }
    Say(Status + "\nPilot " + LastPilotLinear.Length().ToString(ThreeDecimals) + " + feedback " +
        LastFeedbackLinear.Length().ToString(ThreeDecimals) + " m/s; advance " +
        (100 * LastLinearAdvance).ToString("0") + "%\nSolver " + LastSolverPasses +
        " passes; " + LastActiveBounds + " bounds. Stop zeros drives.");
}

bool Solve(Vector3D headReferencePosition, double dt, Vector3D linear, Vector3D angular,
    out Vector3D achievedLinear, out Vector3D achievedAngular, out string why)
{
    achievedLinear = achievedAngular = Vector3D.Zero;
    why = "";
    LastSolverPasses = LastActiveBounds = 0;
    LastSolverConverged = LastSolverBudgetLimited = false;
    int n = Topology.Groups.Count;
    double moveDemand = linear.Length(), turnDemand = angular.Length();
    double demand = moveDemand + turnDemand;
    MatrixD frame = Topology.Base.WorldMatrix;
    Vector3D headWorld = Topology.EndPosition;
    for (int i = 0; i < n; i++)
    {
        AutoGroup g = Topology.Groups[i];
        double physicalLo, physicalHi;
        if (!Topology.Refresh(g, headWorld, frame, out physicalLo, out physicalHi, out why)) return false;
        double cap = g.Rotary ? JointRpm * Math.PI / 30 : PistonMps;
        double lo = Math.Max(physicalLo, -cap), hi = Math.Min(physicalHi, cap);
        double accel = g.Rotary ? JointAccelerationRpmPerSec * Math.PI / 30 : PistonAccelerationMps2;
        if (accel > 0)
        {
            // A changed limit/lock may make the old command infeasible. Centre the
            // acceleration window on the nearest currently feasible command.
            double centre = Clamp(LastQdot[i], lo, hi);
            lo = Math.Max(lo, centre - accel * dt);
            hi = Math.Min(hi, centre + accel * dt);
        }
        if (lo > hi + 1e-10) { why = "No common command interval for " + g.Label; return false; }
        double preference = demand > 1e-9 ? (moveDemand * g.MoveW + turnDemand * g.TurnW) / demand :
            0.5 * (g.MoveW + g.TurnW);
        double s = Scale[i] = preference > 0 ? cap * Math.Sqrt(preference) : 0;
        Bias[i] = s > 1e-12 && Services != null && Services.Safety != null ? Services.Safety.Preferred(i) / s : 0;
        Lo[i] = s > 1e-12 ? lo / s : 0;
        Hi[i] = s > 1e-12 ? hi / s : 0;
        X[i] = s > 1e-12 ? Clamp(LastQdot[i] / s, Lo[i], Hi[i]) : 0;
        int old = BoundState[i];
        BoundState[i] = Hi[i] - Lo[i] <= 1e-12 ? 2 :
            old == -1 && Math.Abs(X[i] - Lo[i]) <= 1e-9 ? -1 :
            old == 1 && Math.Abs(X[i] - Hi[i]) <= 1e-9 ? 1 : 0;
        for (int r = 0; r < 6; r++)
        {
            double weight = r < 3 ? PositionTaskWeight : OrientationTaskWeight;
            J[r, i] = Component(r < 3 ? g.Lin : g.Ang, r % 3);
            B[r, i] = weight * J[r, i] * s;
        }
    }
    for (int r = 0; r < 6; r++)
    {
        double weight = r < 3 ? PositionTaskWeight : OrientationTaskWeight;
        Task[r] = weight * Component(r < 3 ? linear : angular, r % 3);
    }
    double ridge = SolverDamping * SolverDamping;
    int maxPasses = Math.Min(64, Math.Max(8, 2 * n + 4));
    for (int pass = 0; pass < maxPasses; pass++)
    {
        if (Runtime.CurrentInstructionCount > Runtime.MaxInstructionCount * 0.68)
        {
            LastSolverBudgetLimited = true;
            break;
        }
        LastSolverPasses = pass + 1;
        BuildTaskMatrix(n, ridge, true);
        if (!Cholesky(Gram, Rhs, Chol, Temp, Solution)) { why = "singular/non-finite damped task matrix"; return false; }
        double alpha = 1;
        int hit = -1, side = 0;
        for (int i = 0; i < n; i++)
        {
            Trial[i] = X[i];
            if (BoundState[i] != 0) continue;
            double value = Bias[i];
            for (int r = 0; r < 6; r++) value += B[r, i] * Solution[r];
            Trial[i] = value;
            double delta = value - X[i];
            int bound = delta < -1e-14 && value < Lo[i] - 1e-12 ? -1 :
                delta > 1e-14 && value > Hi[i] + 1e-12 ? 1 : 0;
            if (bound != 0)
            {
                double a = Math.Max(0, ((bound < 0 ? Lo[i] : Hi[i]) - X[i]) / delta);
                if (a < alpha) { alpha = a; hit = i; side = bound; }
            }
        }
        for (int i = 0; i < n; i++) if (BoundState[i] == 0)
            X[i] = Clamp(X[i] + alpha * (Trial[i] - X[i]), Lo[i], Hi[i]);
        if (hit >= 0)
        {
            X[hit] = side < 0 ? Lo[hit] : Hi[hit];
            BoundState[hit] = side;
            continue;
        }
        for (int r = 0; r < 6; r++)
        {
            double value = -Task[r];
            for (int i = 0; i < n; i++) value += B[r, i] * X[i];
            Residual[r] = value;
        }
        int release = -1;
        double worst = 1e-9;
        for (int i = 0; i < n; i++)
        {
            if (BoundState[i] == 0 || BoundState[i] == 2) continue;
            double gradient = ridge * (X[i] - Bias[i]);
            for (int r = 0; r < 6; r++) gradient += B[r, i] * Residual[r];
            double violation = BoundState[i] < 0 ? -gradient : gradient;
            if (violation > worst) { worst = violation; release = i; }
        }
        if (release >= 0) { BoundState[release] = 0; continue; }
        LastSolverConverged = true;
        break;
    }
    if (LastSolverPasses == 0) { why = "no instruction budget for a solver pass"; return false; }
    for (int i = 0; i < n; i++)
    {
        double velocity = X[i] * Scale[i];
        if (!Finite(velocity)) { why = "non-finite group velocity"; return false; }
        LastQdot[i] = velocity;
        if (BoundState[i] != 0) LastActiveBounds++;
    }
    var protection = Services == null ? null : Services.Safety;
    if (protection != null) protection.Constrain(LastQdot);
    for (int i = 0; i < n; i++) { achievedLinear += Topology.Groups[i].Lin * LastQdot[i]; achievedAngular += Topology.Groups[i].Ang * LastQdot[i]; }
    LastUnconstrainedLinear = achievedLinear; LastUnconstrainedAngular = achievedAngular;
    if ((LastActiveBounds > 0 || protection != null && protection.Limited) && LastSolverConverged && !LastSolverBudgetLimited)
    {
        if (Runtime.CurrentInstructionCount > Runtime.MaxInstructionCount * 0.68) LastSolverBudgetLimited = true;
        else if (!UnconstrainedTask(n, ridge, out LastUnconstrainedLinear, out LastUnconstrainedAngular))
        { why = "non-finite unconstrained task reference"; return false; }
    }
    return Finite(achievedLinear.LengthSquared()) && Finite(achievedAngular.LengthSquared());
}

static double Component(Vector3D v, int axis)
{
    return axis == 0 ? v.X : axis == 1 ? v.Y : v.Z;
}

void BuildTaskMatrix(int n, double ridge, bool bounded)
{
    for (int r = 0; r < 6; r++)
    {
        Rhs[r] = Task[r];
        if (bounded) for (int i = 0; i < n; i++) if (BoundState[i] != 0) Rhs[r] -= B[r, i] * X[i];
        for (int i = 0; i < n; i++) if (!bounded || BoundState[i] == 0) Rhs[r] -= B[r, i] * Bias[i];
        for (int c = 0; c <= r; c++)
        {
            double total = r == c ? ridge : 0;
            for (int i = 0; i < n; i++) if (!bounded || BoundState[i] == 0) total += B[r, i] * B[c, i];
            Gram[r, c] = total;
        }
    }
}

bool UnconstrainedTask(int n, double ridge, out Vector3D linear, out Vector3D angular)
{
    linear = angular = Vector3D.Zero;
    BuildTaskMatrix(n, ridge, false);
    if (!Cholesky(Gram, Rhs, Chol, Temp, Solution)) return false;
    for (int i = 0; i < n; i++)
    {
        double rate = Bias[i];
        for (int r = 0; r < 6; r++) rate += B[r, i] * Solution[r];
        rate *= Scale[i];
        linear += Topology.Groups[i].Lin * rate; angular += Topology.Groups[i].Ang * rate;
    }
    return Finite(linear.LengthSquared()) && Finite(angular.LengthSquared());
}

bool Commit()
{
    if (GoingHome && Services != null && Services.Safety != null)
    { if (!Services.Safety.Ready || Services.Safety.Blocked) { Services.Safety.Hold(); return false; } Services.Safety.Limit(LastQdot); }
    if (Runtime.CurrentInstructionCount > Runtime.MaxInstructionCount * 0.88)
    {
        Fault("Instruction budget exhausted before output commit.");
        return false;
    }
    for (int i = 0; i < Topology.Groups.Count; i++) Topology.Set(Topology.Groups[i], LastQdot[i]);
    // Every member of a proven parallel group receives its signed group command.
    return true;
}

void UpdateIntegral(ref Vector3D integral, Vector3D error, double deadband,
    Vector3D clipResidual, Vector3D allocationResidual, double ki,
    double cap, double capture, double dt, bool permit)
{
    Vector3D outside = Outside(error, deadband);
    if (ki <= 0 || !permit || error.Length() > capture)
    {
        integral = (integral / (1 + dt / IntegralUnwindSeconds));
        return;
    }
    // A held pose still needs the learned load bias. Reaching the deadband is
    // success, not a reason to leak gravity compensation away.
    integral = ClampLength(integral, cap);
    if (outside.LengthSquared() < 1e-20) return;
    Vector3D delta = outside * (ki * dt);
    // Treat output/limiter clipping and actuator allocation as independent
    // constraints: their residuals may oppose and numerically cancel.
    double scale = Math.Min(IntegralGrowthScale(delta, integral, clipResidual),
        IntegralGrowthScale(delta, integral, allocationResidual));
    integral = ClampLength(integral + delta * scale, cap);
}

static double IntegralGrowthScale(Vector3D delta, Vector3D integral, Vector3D residual)
{
    double growth = Vector3D.Dot(delta, residual);
    if (growth <= 1e-12) return 1;
    // Scale the original increment jointly for both constraints. Sequential
    // vector projections can rotate it back into a previously blocked direction.
    return Clamp(-Vector3D.Dot(integral, residual) / growth, 0, 1);
}

static Vector3D BlendManualCorrection(Vector3D pilotVelocity, Vector3D correction,
    double maximumOppositionFraction, out bool limited)
{
    limited = false;
    double speed = pilotVelocity.Length();
    if (speed <= 1e-12) return correction;
    Vector3D direction = pilotVelocity / speed;
    double component = Vector3D.Dot(correction, direction);
    double minimum = -speed * maximumOppositionFraction;
    if (component >= minimum) return correction;
    limited = true;
    return correction + direction * (minimum - component);
}

double PilotAdvance(Vector3D pilot, Vector3D achievedMinusFeedback)
{
    double d = pilot.LengthSquared();
    if (d < 1e-12) return 1;
    return Clamp(Vector3D.Dot(achievedMinusFeedback, pilot) / d, 0, 1);
}

static double TargetLeadAdvance(Vector3D error, Vector3D step, double limit)
{
    double a = step.LengthSquared();
    if (a < 1e-20) return 1;
    double e = error.LengthSquared(), radius = Math.Max(limit * limit, e);
    if ((error + step).LengthSquared() <= radius) return 1;
    double b = Vector3D.Dot(error, step);
    double root = Math.Sqrt(Math.Max(0, b * b + a * (radius - e)));
    // The alternate root avoids subtracting near-equal positive values.
    return Clamp(b >= 0 ? (root + b > 0 ? (radius - e) / (root + b) : 0) : (root - b) / a, 0, 1);
}

static Vector3D Outside(Vector3D error, double deadband)
{
    double n = error.Length();
    return n > deadband && n > 1e-12 ? error * ((n - deadband) / n) : Vector3D.Zero;
}

void ReadPilot(Vector3D headForward, Vector3D headUp, out Vector3D linear, out Vector3D angular)
{
    linear = angular = Vector3D.Zero;
    Vector3D input; Vector2 mouse; double roll;
    IMyShipController c = PilotInputs(out input, out mouse, out roll);
    if (c == null) return;
    ManualAxes(ref headForward, ref headUp);
    Vector3D forward, left, up;
    if (MovementMode == PilotMode.HEAD)
    {
        forward = headForward; up = headUp;
        if (!Orthonormalise(ref forward, ref up)) throw new Exception("Invalid Head control frame.");
        left = Unit(Vector3D.Cross(up, forward));
    }
    else ControllerAxes(c, out forward, out left, out up);
    Vector3D moveForward = forward, moveUp = up;
    if (MovementMode == PilotMode.VRT) { moveForward = up; moveUp = -forward; }
    Vector3D local = new Vector3D(-input.Z, -input.X, input.Y);
    if (local.LengthSquared() > 1) local = Unit(local);
    linear = (moveForward * local.X + left * local.Y + moveUp * local.Z) * MoveMps;
    // Preserve the established Q/E convention: positive raw roll rotates +Head Forward.
    angular = headForward * roll;
    if (ReadMouse) angular += (left * mouse.X - up * mouse.Y) * MouseSensitivity;
    angular = ClampLength(angular, 1) * (TurnDeg * Math.PI / 180);
}

IMyShipController PilotInputs(out Vector3D move, out Vector2 mouse, out double roll)
{
    move = Vector3D.Zero; mouse = Vector2.Zero; roll = 0;
    if (!PilotSelected) return null;
    var c = FindInputController(); if (c == null) return null;
    if (ReadKeyboard) move = c.MoveIndicator;
    if (ReadMouse) mouse = c.RotationIndicator;
    if (ReadRoll) roll = c.RollIndicator;
    if (!Finite(move.LengthSquared()) || !Finite(mouse.X) || !Finite(mouse.Y) || !Finite(roll))
        throw new Exception("Non-finite pilot input.");
    roll = Clamp(roll, -1, 1); return c;
}

bool HasPilotInput()
{
    Vector3D move; Vector2 mouse; double roll;
    if (PilotInputs(out move, out mouse, out roll) == null) return false;
    // Detect the operator's enabled raw demand, even at Speed=0 or when
    // simultaneous roll/mouse vectors cancel in the selected view frame.
    return move.LengthSquared() > 1e-8 || mouse.LengthSquared() > 1e-8 || Math.Abs(roll) > 1e-4;
}

bool InputScope(IMyShipController c)
{
    return c != null && !c.Closed && c.CubeGrid != null && c.IsSameConstructAs(Me);
}

IMyShipController FindInputController(bool force = false)
{
    if (!force && InputScope(InputController) && InputController.IsUnderControl)
        return InputController;
    if (!force && ++InputScan < 10) return null;
    InputScan = 0;
    InputScratch.Clear();
    GridTerminalSystem.GetBlocksOfType<IMyShipController>(InputScratch, b => InputScope(b) && b.IsUnderControl);
    InputController = null;
    foreach (var b in InputScratch) if (InputController == null || b.EntityId < InputController.EntityId) InputController = b;
    return InputController;
}

void ControllerAxes(IMyShipController c, out Vector3D forward, out Vector3D left, out Vector3D up)
{
    if (!InputScope(c) || !c.IsUnderControl) throw new Exception("HRZ/VRT requires an active cockpit.");
    MatrixD frame = Topology.Base.WorldMatrix;
    forward = InFrame(c.WorldMatrix.Forward, frame); up = InFrame(c.WorldMatrix.Up, frame);
    if (!Orthonormalise(ref forward, ref up)) throw new Exception("Invalid active cockpit orientation axes.");
    left = Unit(Vector3D.Cross(up, forward));
}

    MovementMode = mode; InputController = null; InputScan = 10;

void HoldCurrent()
{
    LocalPath = null;
    if (!Enabled) { EnableControl(); return; }
    Vector3D p, f, u;
    if (!ReadPose(out p, out f, out u)) { Fault("Hold failed: markers unavailable."); return; }
    CapturePose(p, f, u);
    GoingHome = false;
    ControlElapsed = 0;
    StatusElapsed = StatusRefreshSeconds;
    DisplayPinned = false;
    Topology.ZeroOwned();
    if (LastQdot != null) Array.Clear(LastQdot, 0, LastQdot.Length);
    if (BoundState != null) Array.Clear(BoundState, 0, BoundState.Length);
    Status = "ON: holding recaptured Head pose";
    Say("Hold: pose recaptured; ON.");
}

void CapturePose(Vector3D p, Vector3D f, Vector3D u)
{
    SetPoseTarget(p, f, u, 3);
    PreviousP = p; PreviousF = f; PreviousU = u;
    ObservationValid = true; AimingTarget = false;
    FilteredV = FilteredW = PositionI = OrientationI = Vector3D.Zero;
}

bool SetPoseTarget(Vector3D p, Vector3D f, Vector3D u, int parts)
{
    if ((parts & 1) != 0 && !Finite(p.LengthSquared()) || (parts & 2) != 0 && !Orthonormalise(ref f, ref u)) return false;
    if ((parts & 1) != 0) TargetP = p;
    if ((parts & 2) != 0) { TargetF = f; TargetU = u; }
    TargetValid = true;
    return true;
}

void PoseCommand(string[] w, int parts)
{
    if (!Enabled) throw new Exception("Run On before pose commands.");
    string why;
    if (SwapBatchLock || Tools != null && !Tools.AllowManual(out why)) throw new Exception("Pose command refused: tool lock.");
    int n = parts == 3 ? 9 : parts == 1 ? 3 : 6;
    if (w.Length != n + 1) throw new Exception("Pose command has wrong arity.");
    var v = new double[n];
    for (int i = 0; i < n; i++) v[i] = Parse(w[i + 1]);
    LocalPath = null;
    Vector3D p = parts == 2 ? TargetP : new Vector3D(v[0], v[1], v[2]);
    int j = parts == 3 ? 3 : 0;
    Vector3D f = parts == 1 ? TargetF : new Vector3D(v[j], v[j + 1], v[j + 2]);
    Vector3D u = parts == 1 ? TargetU : new Vector3D(v[j + 3], v[j + 4], v[j + 5]);
    if ((parts & 1) != 0 && !Finite(p.LengthSquared()) || (parts & 2) != 0 && !Orthonormalise(ref f, ref u)) throw new Exception("Invalid pose target.");
    if (GoingHome) HoldCurrent();
    if (!Enabled) return;
    if ((parts & 2) != 0) NativeAim(ref f, ref u);
    SetPoseTarget(p, f, u, parts);
    if ((parts & 2) != 0) AimingTarget = true;
}

void Face(string direction)
{
    if (LocalPath != null) HoldCurrent();
    if (!Enabled) throw new Exception("Run On before Face.");
    Vector3D forward = new Vector3D(1, 0, 0), left = new Vector3D(0, 1, 0), up = new Vector3D(0, 0, 1);
    if (MovementMode != PilotMode.HEAD) ControllerAxes(FindInputController(true), out forward, out left, out up);
    Vector3D targetForward, targetUp = up;
    switch (direction.ToLowerInvariant())
    {
        case "forward": targetForward = forward; break;
        case "back": targetForward = -forward; break;
        case "left": targetForward = left; break;
        case "right": targetForward = -left; break;
        case "down": targetForward = -up; targetUp = forward; break;
        case "up": targetForward = up; targetUp = -forward; break;
        default: throw new Exception("Face expects Forward, Back, Left, Right, Up or Down.");
    }
    if (GoingHome) HoldCurrent();
    if (!Enabled) return;
    NativeAim(ref targetForward, ref targetUp);
    SetPoseTarget(TargetP, targetForward, targetUp, 2);
    AimingTarget = true;
}

void Step(string axis, double degrees)
{
    if (LocalPath != null) HoldCurrent();
    if (!Enabled) throw new Exception("Run On before a rotation step.");
    string kind = axis.ToLowerInvariant();
    if (kind != "yaw" && kind != "pitch" && kind != "roll") throw new Exception("Step expects Yaw, Pitch or Roll.");
    Vector3D viewForward = Vector3D.Zero, viewLeft = Vector3D.Zero, viewUp = Vector3D.Zero;
    if (MovementMode != PilotMode.HEAD && kind != "roll")
        ControllerAxes(FindInputController(true), out viewForward, out viewLeft, out viewUp);
    if (GoingHome) HoldCurrent();
    if (!Enabled) return;
    Vector3D a;
    switch (kind)
    {
        case "yaw": a = MovementMode == PilotMode.HEAD ? new Vector3D(0, 0, 1) : viewUp; break;
        case "pitch": a = MovementMode == PilotMode.HEAD ? -Unit(Vector3D.Cross(TargetU, TargetF)) : -viewLeft; break;
        default: a = BareControl ? -TargetU : TargetF; break;
    }
    Vector3D step = Unit(a) * (degrees * Math.PI / 180);
    SetPoseTarget(TargetP, Rotate(TargetF, step), Rotate(TargetU, step), 2);
    AimingTarget = true;
}

void RunCommands(string argument)
{
    SwapBatchLock = Tools != null && Tools.Automatic;
    string[] commands = argument.Replace("\r", "").Split(new char[] { ';', '\n' }, StringSplitOptions.RemoveEmptyEntries);
    try { foreach (string raw in commands) RunCommand(raw.Trim()); }
    finally { SwapBatchLock = false; }
}

void RunCommand(string command)
{
    if (command.StartsWith("MiningArm ", StringComparison.OrdinalIgnoreCase)) command = command.Substring(10).Trim();
    string[] w = command.Split(new char[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
    if (w.Length == 0) return;
    string c = w[0].ToLowerInvariant();
    if (Tools != null && Tools.Handle(command)) { SwapBatchLock = true; return; }
    if (Tools != null && Tools.Busy && c != "on" && c != "toggle" && c != "onoff" && c != "-oo" && c != "off" && c != "stop" && c != "info" && c != "status" && c != "version")
        throw new Exception("Tool busy; SwapCancel/Stop first.");
    if (c.Length == 1 && c[0] >= '1' && c[0] <= '9') { NumericAction(c[0] - '0'); return; }
    switch (c)
    {
        case "version": Say(Banner()); break;
        case "on": StartControl(); break;
        case "off": case "stop": if (Tools != null) { Tools.InhibitRecovery(); if (Tools.Busy) Tools.Cancel(c); } StopMotion(c == "stop" ? "Stop command." : "Off command."); Say(Status); break;
        case "toggle": if (Enabled) { if (Tools != null) Tools.InhibitRecovery(); StopMotion("Toggle OFF."); Say(Status); } else StartControl(); break;
        case "onoff": case "-oo":
            c = w.Length < 2 ? "" : w[1].ToLowerInvariant();
            if (c != "on" && c != "off" && c != "toggle") throw new Exception("OnOff expects On, Off or Toggle.");
            RunCommand(w[1]);
            break;
        case "hold": HoldCurrent(); break;
        case "check": Check(); break;
        case "rescan": Discover(true); break;
        case "reload": Reload(); break;
        case "info": case "headinfo": case "help": PrintInfo(); break;
        case "pivotprobe":
            DisplayPinned = true;
            if (PivotProbe == null) PivotProbe = new PivotProbeTool(this, Topology);
            PivotProbe.Run(w);
            break;
        case "status": DisplayPinned = false; StatusElapsed = StatusRefreshSeconds; Say("Live status display resumed."); break;
        case "settings": PrintInfo(); break;
        case "joints": PrintJoints(); break;
        case "bindings": PrintBindings(); break;
        case "sethome": case "-sh": CaptureHome(); break;
        case "gohome": case "-gh": BeginHome(); break;
        case "homeinfo": PrintHome(); break;
        case "speed": case "-s": MoveMps = Number(w, 1, 0, 5, "Speed"); break;
        case "readkeyboard": case "-rk": ReadKeyboard = InputToggle(w, ReadKeyboard); break;
        case "readroll": ReadRoll = InputToggle(w, ReadRoll); break;
        case "readmouse": case "-rm": ReadMouse = InputToggle(w, ReadMouse); break;
        case "mode": case "movementmode": case "controlmode":
            if (w.Length != 2) throw new Exception("MovementMode expects HEAD, HRZ or VRT.");
            SetMovementMode(w[1]); break;
        case "orientationtolerance":
            double tolerance = Number(w, 1, 0, 30.0, OrientationToleranceKey);
            StopMotion("Stopped: OrientationTolerance changed; run On to recapture.");
            OrientationToleranceDeg = tolerance;
            Say(Status);
            break;
        case "turnspeed": TurnDeg = Number(w, 1, 0.01, 30, "TurnSpeed"); break;
        case "stepsize": case "stepangle": StepDeg = Number(w, 1, 0.01, 180, "StepSize"); break;
        case "jointspeed": JointRpm = Number(w, 1, 0.01, 5, "JointSpeed"); break;
        case "pistonspeed": PistonMps = Number(w, 1, 0.01, 2, "PistonSpeed"); break;
        case "homespeed": HomeJointRpm = Number(w, 1, 0.01, 5, "Home joint RPM"); HomePistonMps = Number(w, 2, 0.01, 2, "Home piston m/s"); break;
        case "face": if (w.Length < 2) throw new Exception("Face expects a direction."); Face(w[1]); break;
        case "forward": case "back": case "left": case "right": case "up": case "down": Face(c); break;
        case "step": if (w.Length < 2) throw new Exception("Step expects an axis."); Step(w[1], w.Length > 2 ? Parse(w[2]) : StepDeg); break;
        case "yaw": Step("yaw", Number(w, 1, -180, 180, "Yaw")); break;
        case "pitch": Step("pitch", Number(w, 1, -180, 180, "Pitch")); break;
        case "roll": Step("roll", Number(w, 1, -180, 180, "Roll")); break;
        case "yawleft": case "yawright": Step("yaw", c == "yawleft" ? StepDeg : -StepDeg); break;
        case "pitchup": case "pitchdown": Step("pitch", c == "pitchup" ? StepDeg : -StepDeg); break;
        case "rollleft": case "rollright": Step("roll", c == "rollright" ? StepDeg : -StepDeg); break;
        case "dumptrace": PrintInfo(); Say("See Info; Custom Data unchanged."); break;
        case "traceclear": Say("No trace buffer."); break;
        case "moveto": PoseCommand(w, 1); break;
        case "orientto": PoseCommand(w, 2); break;
        case "movetoposori": PoseCommand(w, 3); break;
        case "path": PathCommand(command); break;
        case "move": case "relmove": case "runfrompanel":
            throw new Exception(c + " is not supported by AutoArm's discovered pose control.");
        default: throw new Exception("Unknown command '" + w[0] + "'. Use Info.");
    }
}

static bool InputToggle(string[] words, bool current)
{
    string value = words.Length == 2 ? words[1].ToLowerInvariant() : "";
    if (value == "on") return true;
    if (value == "off") return false;
    if (value == "toggle") return !current;
    throw new Exception(words[0] + " expects On, Off or Toggle.");
}

void Reload()
{
    StopMotion("Reload command.");
    PivotProbe = null;
    SetupElapsed = 0;
    string updated, why;
    bool changed;
    ConfigValid = false;
    MyIni configIni;
    if (!AutoConfig.ReadCurrent(this, CustomData, out configIni, out why)) throw new Exception(why);
    Services.Load(configIni);
    if (Topology.Groups.Count == 0 || (Tools != null && Tools.Configured) || configIni.Get(ToolsSection, EnabledKey).ToBoolean(false)) { Discover(true); return; }
    if (Topology.Groups.Count > 0 && !Topology.Validate(out why, true))
    {
        ConfigValid = false;
        Say("Reload refused: " + why + " Run Rescan after repairing topology.");
        return;
    }
    if (!AutoConfig.LoadOrGenerate(this, Topology, CustomData, out updated, out changed, out why))
    {
        ConfigValid = false;
        Status = "Reload refused without changing Custom Data: " + why;
        Say(Status);
        return;
    }
    ConfigValid = true;
    Allocate(Topology.Groups.Count);
    LoadHome();
    Status = "Configuration reloaded; controller remains OFF.";
    Say(Status);
}

void LoadHome()
{
    Home.Clear(); HomeFingerprint = "";
    if (string.IsNullOrWhiteSpace(Storage)) return;
    var ini = new MyIni();
    MyIniParseResult result;
    if (!ini.TryParse(Storage, out result)) return;
    string section = HomeSection;
    if (!ini.ContainsSection(section))
    {
        var sections = new List<string>(); ini.GetSections(sections);
        foreach (string candidate in sections)
            if (candidate.EndsWith(" Home", StringComparison.Ordinal) && ini.Get(candidate, "Fingerprint").ToString() == Topology.Fingerprint)
            {
                if (section != HomeSection) return;
                section = candidate;
            }
    }
    string fingerprint = ini.Get(section, "Fingerprint").ToString("");
    if (fingerprint != Topology.Fingerprint) return;
    foreach (AutoGroup g in Topology.Groups) foreach (AutoMember m in g.M)
    {
        string key = "E" + m.B.EntityId;
        double value = ini.Get(section, key).ToDouble(double.NaN);
        if (!Finite(value)) { Home.Clear(); return; }
        Home[m.B.EntityId] = value;
    }
    HomeFingerprint = fingerprint;
    if (section != HomeSection)
    {
        ini.DeleteSection(section); Storage = ini.ToString(); WriteHome();
    }
}

void WriteHome()
{
    var ini = AutoConfig.Saved(Storage);
    ini.DeleteSection(HomeSection);
    ini.Set(HomeSection, "Fingerprint", HomeFingerprint);
    foreach (var pair in Home) ini.Set(HomeSection, "E" + pair.Key, pair.Value.ToString("R", System.Globalization.CultureInfo.InvariantCulture));
    Storage = ini.ToString();
}

void BeginHome()
{
    string permission;
    if (SwapBatchLock || (Tools != null && !Tools.AllowManual(out permission))) { Say("GoHome refused: tool lock."); return; }
    StopMotion("");

    { Say("GoHome refused: use SetHome for the current topology."); return; }

    Status = "GoHome in progress; live pilot input cancels.";
    Say(Status);
}

void UpdateHome(double dt)
{
    if (HasPilotInput()) { Fault("GoHome cancelled by pilot input."); return; }
    string why;
    if (Tools != null && !Tools.AllowManual(out why)) { Fault(why); return; }
    if (!Topology.Validate(out why, false, true)) { Fault("GoHome topology fault: " + why); return; }
    bool complete = true;
    for (int i = 0; i < Topology.Groups.Count; i++)
    {
        AutoGroup g = Topology.Groups[i];
        double error = 0, branch = 0;
        bool haveBranch = false;
        double agreement = g.Rotary ? 0.5 * Math.PI / 180 : 0.005;
        // A limited member fixes the legal native-angle branch. Unlimited peers
        // may then select the equivalent circular error nearest that branch.
        if (g.Rotary) foreach (AutoMember m in g.M)
        {
            if (Unlimited((IMyMotorStator)m.B)) continue;
            double e = m.Sign * MemberHomeError(m, true);
            if (!haveBranch) { branch = e; haveBranch = true; }
            else if (Math.Abs(e - branch) > agreement) { Fault("GoHome limited-member routes disagree at " + g.Label); return; }
        }
        foreach (AutoMember m in g.M)
        {
            double e = m.Sign * MemberHomeError(m, g.Rotary);
            bool unlimited = g.Rotary && Unlimited((IMyMotorStator)m.B);
            if (!haveBranch) { branch = e; haveBranch = true; }
            else if (unlimited) e = branch + Wrap(e - branch);
            if (Math.Abs(e - branch) > agreement) { Fault("GoHome member errors disagree at " + g.Label); return; }
            error += e;
        }
        error /= g.M.Count;
        double tolerance = g.Rotary ? 0.25 * Math.PI / 180 : 0.005;
        if (Math.Abs(error) > tolerance) complete = false;
        double lo, hi;
        if (!Topology.Refresh(g, Topology.EndPosition, Topology.Base.WorldMatrix, out lo, out hi, out why))
        { Fault("GoHome: " + why); return; }
        double cap = g.Rotary ? HomeJointRpm * Math.PI / 30 : HomePistonMps;
        double lower = Math.Max(lo, -cap), upper = Math.Min(hi, cap);
        if (lower > upper) { Fault("GoHome has no common velocity interval at " + g.Label); return; }
        double velocity = Math.Abs(error) <= tolerance ? 0 : Clamp(error * 0.5, lower, upper);
        double accel = g.Rotary ? JointAccelerationRpmPerSec * Math.PI / 30 : PistonAccelerationMps2;
        double centre = Clamp(LastQdot[i], lower, upper);
        velocity = Clamp(velocity, Math.Max(lower, centre - accel * dt), Math.Min(upper, centre + accel * dt));
        LastQdot[i] = velocity;
    }
    if (complete)
    {
        StopMotion("GoHome complete.");
        Say(Status);
    }
    else Commit();
}

bool ResolveHome(AutoMember m, double requested, out double resolved)
{
    resolved = requested;
    var p = m.B as IMyPistonBase;
    if (p != null)
    {
        if (!Finite(resolved) || resolved < p.MinLimit - 0.005 || resolved > p.MaxLimit + 0.005) return false;
        resolved = Clamp(resolved, p.MinLimit, p.MaxLimit);
        return true;
    }
    var r = (IMyMotorStator)m.B;
    if (!Finite(resolved)) return false;
    double lo = r.LowerLimitRad, hi = r.UpperLimitRad, turn = 2 * Math.PI;
    double k = Math.Round((r.Angle - requested) / turn);
    double kMin = lo < -1000000 ? double.NegativeInfinity : Math.Ceiling((lo - requested - 0.25 * Math.PI / 180) / turn);
    double kMax = hi > 1000000 ? double.PositiveInfinity : Math.Floor((hi - requested + 0.25 * Math.PI / 180) / turn);
    if (kMin > kMax) return false;
    resolved = requested + Clamp(k, kMin, kMax) * turn;
    if (!Finite(resolved) || resolved < lo - 0.25 * Math.PI / 180 || resolved > hi + 0.25 * Math.PI / 180) return false;
    resolved = Clamp(resolved, lo, hi);
    return true;
}

}

static bool Cholesky(double[,] a, double[] b, double[,] l, double[] temp, double[] answer)
{
    for (int i = 0; i < 6; i++) for (int j = 0; j <= i; j++)
    {
        double value = a[i, j];
        for (int k = 0; k < j; k++) value -= l[i, k] * l[j, k];
        if (i == j)
        {
            if (!Finite(value) || value <= 1e-15) return false;
            l[i, j] = Math.Sqrt(value);
        }
        else l[i, j] = value / l[j, j];
    }
    for (int i = 0; i < 6; i++)
    {
        double value = b[i];
        for (int j = 0; j < i; j++) value -= l[i, j] * temp[j];
        temp[i] = value / l[i, i];
    }
    for (int i = 5; i >= 0; i--)
    {
        double value = temp[i];
        for (int j = i + 1; j < 6; j++) value -= l[j, i] * answer[j];
        answer[i] = value / l[i, i];
        if (!Finite(answer[i])) return false;
    }
    return true;
}

static Vector3D RotationError(Vector3D f, Vector3D u, Vector3D tf, Vector3D tu)
{
    if (!Orthonormalise(ref f, ref u) || !Orthonormalise(ref tf, ref tu)) return new Vector3D(double.NaN);
    Vector3D l = Vector3D.Cross(u, f), tl = Vector3D.Cross(tu, tf);
    double d00 = tf.X * f.X + tl.X * l.X + tu.X * u.X,
        d01 = tf.X * f.Y + tl.X * l.Y + tu.X * u.Y,
        d02 = tf.X * f.Z + tl.X * l.Z + tu.X * u.Z;
    double d10 = tf.Y * f.X + tl.Y * l.X + tu.Y * u.X,
        d11 = tf.Y * f.Y + tl.Y * l.Y + tu.Y * u.Y,
        d12 = tf.Y * f.Z + tl.Y * l.Z + tu.Y * u.Z;
    double d20 = tf.Z * f.X + tl.Z * l.X + tu.Z * u.X,
        d21 = tf.Z * f.Y + tl.Z * l.Y + tu.Z * u.Y,
        d22 = tf.Z * f.Z + tl.Z * l.Z + tu.Z * u.Z;
    var skew = new Vector3D(d21 - d12, d02 - d20, d10 - d01) * 0.5;
    double angle = Math.Acos(Clamp((d00 + d11 + d22 - 1) * 0.5, -1, 1));
    if (angle < 1e-7) return skew;
    if (Math.PI - angle > 1e-5) return skew * (angle / Math.Sin(angle));
    // Near pi, the skew is ill-conditioned. Extract an eigenaxis from the full
    // relative rotation, selecting the largest diagonal for numeric stability.
    double x, y, z;
    if (d00 >= d11 && d00 >= d22)
    {
        x = Math.Sqrt(Math.Max(0, (d00 + 1) * 0.5));
        y = (d01 + d10) / (4 * Math.Max(x, 1e-10));
        z = (d02 + d20) / (4 * Math.Max(x, 1e-10));
    }
    else if (d11 >= d22)
    {
        y = Math.Sqrt(Math.Max(0, (d11 + 1) * 0.5));
        x = (d01 + d10) / (4 * Math.Max(y, 1e-10));
        z = (d12 + d21) / (4 * Math.Max(y, 1e-10));
    }
    else
    {
        z = Math.Sqrt(Math.Max(0, (d22 + 1) * 0.5));
        x = (d02 + d20) / (4 * Math.Max(z, 1e-10));
        y = (d12 + d21) / (4 * Math.Max(z, 1e-10));
    }
    Vector3D axis = Unit(new Vector3D(x, y, z));
    if (skew.LengthSquared() > 1e-16 && Vector3D.Dot(axis, skew) < 0) axis = -axis;
    return Unit(axis) * angle;
}

static Vector3D Rotate(Vector3D value, Vector3D rotation)
{
    double angle = rotation.Length();
    if (angle < 1e-12) return value;
    Vector3D axis = rotation / angle;
    return value * Math.Cos(angle) + Vector3D.Cross(axis, value) * Math.Sin(angle) +
        axis * Vector3D.Dot(axis, value) * (1 - Math.Cos(angle));
}

static bool Orthonormalise(ref Vector3D f, ref Vector3D u)
{
    double fn = f.Length();
    if (!Finite(fn) || fn < 1e-10) return false;
    f /= fn;
    u -= f * Vector3D.Dot(u, f);
    double un = u.Length();
    if (!Finite(un) || un < 1e-10) return false;
    u /= un;
    return true;
}

static Vector3D Unit(Vector3D value)
{
    double n = value.Length();
    if (!Finite(n) || n < 1e-12) throw new Exception("Invalid zero/non-finite vector.");
    return value / n;
}

static Vector3D ClampLength(Vector3D value, double cap)
{
    double n = value.Length();
    if (!Finite(n)) return new Vector3D(double.NaN);
    return n > cap && n > 1e-12 ? value * (cap / n) : value;
}

static Vector3D InFrame(Vector3D value, MatrixD frame)

{

    return new Vector3D(Vector3D.Dot(value, frame.Forward), Vector3D.Dot(value, frame.Left), Vector3D.Dot(value, frame.Up));

}

static double Clamp(double value, double lo, double hi)

{

    return Math.Max(lo, Math.Min(hi, value));

}

static double Wrap(double angle)

{

    return angle - Math.Floor((angle + Math.PI) / (2 * Math.PI)) * (2 * Math.PI);

}

static bool Finite(double value)

{

    return !double.IsNaN(value) && !double.IsInfinity(value);

}

static bool Finite(Vector3D value)

{

    return Finite(value.X) && Finite(value.Y) && Finite(value.Z);

}

static bool Unlimited(IMyMotorStator r)
{
    return r.LowerLimitRad < -1000000 && r.UpperLimitRad > 1000000;
}


static bool ReadDouble(string text, out double value)

{

    return double.TryParse(text, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out value) && Finite(value);

}

// and Storage are untouched; existing topology-fault handling can stop drives.
class PivotProbeTool
{

        if(!Finite(ori)||ori>1){why="orientation residual "+ori.ToString(ThreeDecimals)+" deg: slip/other motion.";return false;}
        result=new Result{Group=group,Delta=delta,Travel=travel,Orientation=ori};
        for(int k=0;k<3;k++)
        {
            Vector3D mean=Vector3D.Zero;var predicted=new Vector3D[g.M.Count];
            for(int i=0;i<g.M.Count;i++){Vector3D c=Candidate(a,i,k);predicted[i]=c+Rot(a.P-c,a.Axis[i],delta);mean+=predicted[i];}mean/=g.M.Count;
            for(int i=0;i<predicted.Length;i++)result.Spread[k]=Math.Max(result.Spread[k],(predicted[i]-mean).Length());
            result.Error[k]=(b.P-mean).Length();if(!Finite(result.Error[k])||!Finite(result.Spread[k])){why="non-finite residual.";return false;}
        }
        int best=0;for(int k=1;k<3;k++)if(result.Error[k]<result.Error[best])best=k;double next=double.PositiveInfinity;
        for(int k=0;k<3;k++)if(k!=best)next=Math.Min(next,result.Error[k]);
        result.Verdict=result.Spread[best]>.01?"inconclusive: pair mismatch":next-result.Error[best]<.001?"axis-equivalent/inconclusive":Name(best)+" smallest (not a calibration)";return true;
    }
    void Report()
    {
        if(R.Count==0){Say("No accepted trials. Capture positive and negative manual motions.");return;}
        var s=new System.Text.StringBuilder("PIVOT PROBE RAM REPORT - no pivot changed\n");for(int i=0;i<R.Count;i++)s.Append('#').Append(i+1).Append(' ').Append(Format(R[i])).Append('\n');
        s.Append("Collect both directions; residuals are evidence, not proof. Flex/load/slip can dominate.");Say(s.ToString());
    }
    string Format(Result r){return "G"+(r.Group+1)+" d="+Deg(r.Delta).ToString(ThreeDecimals)+"deg travel="+r.Travel.ToString("0.0000")+"m ori="+r.Orientation.ToString(ThreeDecimals)+"deg\n Base e/spread="+Row(r,0)+" Top="+Row(r,1)+" Mid="+Row(r,2)+" RESULT="+r.Verdict;}
    string Row(Result r,int k){return r.Error[k].ToString("0.0000")+"/"+r.Spread[k].ToString("0.0000")+"m";}
    Vector3D Candidate(Shot s,int i,int k){return k==0?s.Base[i]:k==1?s.Top[i]:(s.Base[i]+s.Top[i])*.5;}
    string Name(int k){return k==0?"Base center":k==1?"Top center":"Midpoint";}
    void Clear(){A=null;G=-1;Fingerprint="";}void Help(){Say("PivotProbe Start <group> | End | Report | Reset");}void Say(string s){P.Say(s);}
    static double Deg(double a)

    return a * 180 / Math.PI;

}

    static bool Unit(ref Vector3D v){double n=v.Length();if(!Finite(v)||!Finite(n)||n<1e-10)return false;v/=n;return true;}

    static Vector3D Rot(Vector3D v, Vector3D a, double x)

    static double Angle(Vector3D a,Vector3D b){double n=a.Length()*b.Length();return n>1e-12&&Finite(n)?Math.Acos(Math.Max(-1,Math.Min(1,Vector3D.Dot(a,b)/n)))*180/Math.PI:double.NaN;}

string ToolGroupKey(AutoGroup g)
{
    string key = g.Rotary ? "R" : "P";
    foreach (var m in g.M) key += ":" + m.B.EntityId + (m.Sign > 0 ? "+" : "-");

void RememberToolWeights(bool capture = true)
{
    if (!ToolWeightsLoaded)
    {
        var saved = AutoConfig.Saved(Storage);
        var loaded = new Dictionary<string, double[]>();
        string rows = saved.Get(ToolWeightsSection, "Rows").ToString();
        foreach (string row in rows.Split('\n'))
        {
            if (row.Trim().Length == 0) continue;
            string[] cells = row.Split('|'); double m, r;
            if (cells.Length != 3 || !ReadDouble(cells[1], out m) ||
                !ReadDouble(cells[2], out r) ||
                !Finite(m) || !Finite(r) || m < 0 || m > 10 || r < 0 || r > 10)
                throw new Exception("Invalid saved tool actuator preferences.");
            ToolWeightIds(cells[0]);
            if (loaded.ContainsKey(cells[0])) throw new Exception("Duplicate saved tool actuator preferences.");
            loaded.Add(cells[0], new double[] { m, r });
        }
        foreach (var pair in loaded) ToolWeights[pair.Key] = pair.Value;
        ToolWeightsLoaded = true;
    }
    if (!capture) return;
    foreach (var g in Topology.Groups) ToolWeights[ToolGroupKey(g)] = new double[] { g.MoveW, g.TurnW };
    WriteToolWeights();
}

static List<long> ToolWeightIds(string key)
{
    string[] parts = key.Split(':'); var ids = new List<long>();
    if (parts.Length < 2 || parts.Length > 49 || parts[0] != "R" && parts[0] != "P") throw new Exception("Invalid saved tool actuator identity.");
    for (int i = 1; i < parts.Length; i++)
    {
        string part = parts[i]; long id;
        if (part.Length < 2 || part[part.Length - 1] != '+' && part[part.Length - 1] != '-' ||
            !long.TryParse(part.Substring(0, part.Length - 1), out id) || id == 0 || ids.Contains(id)) throw new Exception("Invalid saved tool actuator identity.");
        ids.Add(id);
    }
    return ids;
}

// Prune only after successful stopped discovery. A different grid ID or an
// unlocked/parked actuator is not deletion. Failed discovery never prunes.
void PruneToolWeights()
{
    RememberToolWeights(false);
    var dead = new List<string>();
    foreach (var pair in ToolWeights)
    {
        if (Runtime.MaxInstructionCount > 0 && Runtime.CurrentInstructionCount >= Runtime.MaxInstructionCount * .5) return;
        foreach (long id in ToolWeightIds(pair.Key))
        {
            var block = GridTerminalSystem.GetBlockWithId(id);
            if (block == null || block.Closed || (pair.Key[0] == 'P' ? !(block is IMyPistonBase) : !(block is IMyMotorStator) || block is IMyMotorSuspension))
            { dead.Add(pair.Key); break; }
        }
    }
    if (dead.Count == 0) return;
    foreach (string key in dead) ToolWeights.Remove(key);
    WriteToolWeights();
}

void WriteToolWeights()
{
    var keys = new List<string>(ToolWeights.Keys); keys.Sort(StringComparer.Ordinal);
    var text = new System.Text.StringBuilder();
    foreach (string key in keys)
    {
        if (text.Length > 0) text.Append('\n');
        text.Append(key).Append('|').Append(AutoSettings.Number(ToolWeights[key][0])).Append('|').Append(AutoSettings.Number(ToolWeights[key][1]));
    }
    var ini = AutoConfig.Saved(Storage);
    if (ini.Get(ToolWeightsSection, "Rows").ToString() == text.ToString()) return;
    ini.Set(ToolWeightsSection, "Rows", text.ToString()); Storage = ini.ToString();
}

bool RebuildToolTopology(out string why)
{
    why = ""; StopMotion(""); ConfigValid = false;
    try
    {
        RememberToolWeights();
        Topology.SetEndpointOverride(Tools.Endpoint, Tools.IgnoredMount, Tools.EndFrame, Tools.Focus);
        if (!Topology.Discover(out why)) return false;
        foreach (var g in Topology.Groups)
        {
            double[] value;
            if (ToolWeights.TryGetValue(ToolGroupKey(g), out value)) { g.MoveW = value[0]; g.TurnW = value[1]; }
        }
        Allocate(Topology.Groups.Count); ConfigValid = true;
        SetupElapsed = ControlElapsed = 0;
        Frequency = UpdateFrequency.Update1;
        return true;
    }
    catch (Exception e) { why = e.Message; return false; }
}

bool SettleToolTopology(out string why)
{
    if (!ConfigValid) { why = "Tool topology is not configured."; return false; }
    if (!Topology.StableSetup(out why)) return false;
    if (!CheckArmLayout(out why)) { Topology.Ready = false; if (why.StartsWith("Waiting", StringComparison.Ordinal)) return false; ConfigValid = false; Fault(why); return false; }
    PruneToolWeights();
    AutoConfig.UpdateGroups(this); LoadHome();
    return true;
}

bool ToolPoseStep(MatrixD world, double dt, double observationDt, double moveCap,
    double turnCap, double positionTolerance, double angleTolerance,
    out bool settled, out string why, double settleScale = 0)
{
    settled = false; why = "";
    if (!Topology.Ready || !ConfigValid) { why = "Tool topology is not ready."; return false; }
    if (!Finite(dt) || dt <= 0 || !Finite(observationDt) || observationDt <= 0 ||
        !Finite(moveCap) || moveCap <= 0 || !Finite(turnCap) || turnCap <= 0 ||
        !Finite(positionTolerance) || positionTolerance <= 0 || !Finite(angleTolerance) || angleTolerance <= 0)
    { why = "Invalid tool interval, cap or tolerance."; return false; }
    MatrixD basis = Topology.Base.WorldMatrix;
    if (!SetPoseTarget(InFrame(world.Translation - Topology.Base.GetPosition(), basis),
        InFrame(world.Forward, basis), InFrame(world.Up, basis), 3))
    { why = "Invalid tool target frame."; return false; }
    ToolDriveMoveCap = moveCap; ToolDriveTurnCap = turnCap * Math.PI / 180;
    ToolPositionBand = positionTolerance * .5; ToolOrientationBand = angleTolerance * Math.PI / 360;
    ToolDriving = true;
    try { UpdateControl(Math.Min(dt, .1), observationDt); }
    finally { ToolDriving = false; }
    if (!TargetValid || !Topology.Ready) { why = LastFault.Length > 0 ? LastFault : "Tool motion stopped."; return false; }
    settled = LastPositionError <= positionTolerance && LastOrientationErrorDeg <= angleTolerance &&
        FilteredV.Length() <= Math.Max(.005, positionTolerance * settleScale) && FilteredW.Length() * 180 / Math.PI <= Math.Max(.2, angleTolerance * settleScale);
    return true;
}

static MatrixD LinkFrame(MyIni ini, string key)
{
    var p = LinkVector(ini, key + "P"); var f = LinkVector(ini, key + "F"); var u = LinkVector(ini, key + "U");

}

}

ArmPath ReadPath(string text, long anchor, long id, double move, double turn, double positionTolerance, double angleTolerance)
{
    var path = new ArmPath { Id = id, Move = move, Turn = turn, PositionTolerance = positionTolerance, AngleTolerance = angleTolerance };
    if (text.Length > 12000 || !Finite(move) || move <= 0 || move > 5 || !Finite(turn) || turn <= 0 || turn > 30) throw new Exception("Invalid path caps/size.");
    if (anchor != 0)
    { path.Anchor = GridTerminalSystem.GetBlockWithId(anchor); if (path.Anchor == null || path.Anchor.Closed || !path.Anchor.IsSameConstructAs(Me)) throw new Exception("Path anchor unavailable."); }
    foreach (string row in text.Split('|'))
    {
        string[] cells = row.Split(new char[] { ' ', '\t', '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries);
        if (cells.Length != 9 || path.Waypoints.Count >= 32) throw new Exception("Path needs 1..32 rows of position/forward/up (9 numbers).");
        var v = new double[9]; for (int i = 0; i < 9; i++) if (!ReadDouble(cells[i], out v[i])) throw new Exception("Non-finite path waypoint.");
        var forward = new Vector3D(v[3], v[4], v[5]); var up = new Vector3D(v[6], v[7], v[8]);
        if (!Orthonormalise(ref forward, ref up)) throw new Exception("Invalid path orientation.");
        path.Waypoints.Add(MatrixD.CreateWorld(new Vector3D(v[0], v[1], v[2]), forward, up));
    }
    return path;
}

{

}
static double[] PathSpeeds(MyIni ini, string key, int count, double fallback, double inherited, double maximum)
{
    string text = ini.Get(LinkSection, key).ToString(); if (text.Length == 0) return null;
    string[] rows = text.Split('|'); if (rows.Length != count) throw new Exception("Path " + key + " count mismatch.");
    var values = new double[count];
    for (int i = 0; i < count; i++)
    {
        double value; if (!ReadDouble(rows[i], out value) || value < 0 || value > maximum) throw new Exception("Invalid path " + key);
        values[i] = value == 0 ? inherited : value;
        if (!(values[i] > 0)) throw new Exception("Arm default path speed is disabled.");
    }

}
bool StepPath(ArmPath path, double dt, double observed, out string why)
{
    why = ""; if (path.Done) return true;
    if (path.Anchor != null && (path.Anchor.Closed || !path.Anchor.IsSameConstructAs(Me))) { why = "Path anchor lost."; return false; }
    if (path.Budget == 0)
    {
        var target = path.Current; double distance = Vector3D.Distance(target.Translation, Topology.EndPosition);
        double speed = Math.Min(path.CurrentMove, path.Move);
        double angle = RotationError(Topology.EndPose.Forward, Topology.EndPose.Up, target.Forward, target.Up).Length() * 180 / Math.PI;
        double turn = Math.Min(path.CurrentTurn, path.Turn);
        path.Budget = 30 + (speed > 0 ? distance / speed : 0) + (turn > 0 ? angle / turn : 0);
    }
    path.Elapsed += observed;
    if (path.Elapsed > path.Budget) { why = "Path waypoint timeout: " + (path.Completed + 1); return false; }
    MatrixD goal = path.Current, demand = goal;
    if (Services != null && Services.Safety != null) Services.Safety.Navigate(path, ref demand);
    bool entry = path.Docking && path.Completed + 1 < path.Waypoints.Count;
    double pt = entry ? Math.Max(.001, Math.Min(.02, Vector3D.Distance(path.Waypoints[path.Completed].Translation, path.Waypoints[path.Completed + 1].Translation) * .25)) : path.PositionTolerance;
    double at = entry ? Math.Max(1, path.AngleTolerance) : path.AngleTolerance;
    PathRemainingDistance = Vector3D.Distance(goal.Translation, Topology.EndPosition);
    if (path.Linear)
    {
        MatrixD basis = path.Anchor == null ? MatrixD.Identity : path.Anchor.WorldMatrix;
        var position = Vector3D.Transform(Topology.EndPosition, MatrixD.Invert(basis));
        if (!path.LineStarted) { path.LineStart = position; path.LineStarted = true; }
        var delta = path.Waypoints[path.Completed].Translation - path.LineStart; double length = delta.Length();
        var reference = path.LineStart;
        // Align before insertion/withdrawal, then hold one orientation and steer
        // to a lookahead point on the captured Cartesian line, never a shortcut.
        double angle = RotationError(Topology.EndPose.Forward, Topology.EndPose.Up, goal.Forward, goal.Up).Length() * 180 / Math.PI;
        double gate = path.Docking ? Math.Max(at, Math.Min(1, PathRemainingDistance * 10)) : at;
        if (length > 1e-10)
        {
            var direction = delta / length; double progress = Vector3D.Dot(position - path.LineStart, direction);
            // Losing alignment pauses at current progress, never back at the
            // insertion start. Retain perpendicular correction onto the line.
            reference += direction * Clamp(progress + (path.CurrentMove / PositionKp + pt * .5) * Clamp(2 - angle / gate, 0, 1), 0, length);
        }
        demand.Translation = Vector3D.Transform(reference, basis);
    }
    bool settled;
    if (!ToolPoseStep(demand, dt, observed, path.CurrentMove, path.CurrentTurn, pt, at, out settled, out why, path.Docking ? 2 : 0)) return false;
    if (entry) settled = LastPositionError <= pt && LastOrientationErrorDeg <= at && (path.Capture || FilteredV.Length() <= Math.Max(.005, Math.Min(path.Move, pt * 2.5)) && FilteredW.Length() * 180 / Math.PI <= Math.Max(.2, path.Turn));
    settled &= PathRemainingDistance <= pt;
    path.Stable = settled ? path.Stable + 1 : 0;
    if (path.Stable >= 2) { if (!entry) { Topology.ZeroOwned(); Tools.ZeroMount(); } path.Completed++; path.Elapsed = path.Budget = 0; path.Stable = 0; path.LineStarted = false; Say("Path " + path.Id + ": " + path.Completed + "/" + path.Waypoints.Count); }
    return true;
}
void PathCommand(string command)
{
    string why;
    if (!Enabled || !Topology.Ready || !Tools.AllowManual(out why)) throw new Exception("Run On before Path; equipment must be ready.");
    var path = ReadPath(command.Substring(command.IndexOf(' ') + 1), 0, 0, MoveMps, TurnDeg, .005, .2);
    // Validate the entire list before changing any motion state.
    Vector3D p, f, u; if (!ReadPose(out p, out f, out u)) throw new Exception("Path focus unavailable.");
    StopMotion(""); LocalPath = path; CapturePose(p, f, u); Enabled = true; Frequency = UpdateFrequency.Update1;
    Say("Path accepted: " + path.Waypoints.Count + " waypoints.");
}

    public void Receive(double elapsed)
    {
        if (Finite(elapsed) && elapsed > 0) Link.Age += elapsed;
        int count = 0;
        while (P.Inbox.Count > 0 && count++ < 16)
        {
            if (P.Runtime.MaxInstructionCount > 0 && P.Runtime.CurrentInstructionCount >= P.Runtime.MaxInstructionCount * .5) break;
            var message = P.Inbox.Dequeue(); MyIni ini;
            if (P.Services != null && P.Services.Receive(message)) continue;
            if (!Link.Read(message, out ini)) continue;
            long epoch = ini.Get(LinkSection, "Epoch").ToInt64(), seq = ini.Get(LinkSection, "Sequence").ToInt64();
            string op = ini.Get(LinkSection, "Operation").ToString();
            if (op == "HELLO")
            {
                if (ini.Get(LinkSection, "Remote").ToInt64() != 0 && ini.Get(LinkSection, "Remote").ToInt64() != Link.Epoch) continue;
                bool acquire = ini.Get(LinkSection, "Acquire").ToBoolean();
                if (epoch < Link.RemoteEpoch) continue;
                if (Link.RemoteEpoch != epoch) { P.StopMotion(""); Link.RemoteEpoch = epoch; Link.Received = 0; Lease = Driving = Rebuilding = Permitted = Transition = ResumePending = Resumed = false; }
                if (seq <= Link.Received) continue;
                Link.Received = seq;
                if (acquire && P.ConfigValid && ini.Get(LinkSection, "Remote").ToInt64() == Link.Epoch && ini.Get(LinkSection, "Generation").ToInt64() == Link.Generation)
                { P.StopMotion(""); Lease = true; Driving = Rebuilding = Permitted = Transition = ResumePending = Resumed = false; Changing = ini.Get(LinkSection, "Changing").ToBoolean(); Error = ""; }
                Link.Age = 0; Request = seq; Reply("STATE"); continue;
            }
            if (epoch != Link.RemoteEpoch || seq <= Link.Received || ini.Get(LinkSection, "Remote").ToInt64() != Link.Epoch || ini.Get(LinkSection, "Generation").ToInt64() != Link.Generation) continue;
            Link.Received = seq; Link.Age = 0; Request = seq;
            try
            {
                if (op == "INFO") { P.DisplayPinned = true; P.Say(ini.Get(LinkSection, "Text").ToString()); }
                else if (op == "PING")
                {
                    Permitted = ini.Get(LinkSection, "Permit").ToBoolean();
                    if (P.Enabled && !Permitted) { P.Fault("ToolSwap equipment not ready."); }
                }
                else if (op == "RESUME" && !Lease && Resumed) { Reply("STATE"); continue; }
                else if (!Lease) throw new Exception("No movement lease.");
                else if (op == "TRANSITION")
                { Driving = false; Path = null; Transition = true; P.StopMotion(""); P.Topology.Ready = false; }
                else if (op == "ABORT")
                { Driving = Rebuilding = Lease = Permitted = Resumed = false; if (ini.Get(LinkSection, "ExplicitStop").ToBoolean()) InhibitRecovery(); Path = null; Error = ini.Get(LinkSection, "Reason").ToString(); P.StopMotion(Error); P.Say("ToolSwap stopped: " + Error); }
                else if (op == "STOP" || op == "PAUSE") { Driving = false; Path = null; if (ini.Get(LinkSection, "ExplicitStop").ToBoolean()) InhibitRecovery(); P.StopMotion(""); }
                else if (op == "BUILD")
                {
                    if (!P.ConfigValid) throw new Exception("Arm configuration invalid; run On on arm PB.");
                    Driving = false; P.StopMotion(""); ApplyEndpoint(ini);
                    string why; if (!P.RebuildToolTopology(out why)) throw new Exception(why);
                    Rebuilding = true; Settled = false;
                }
                else if (op == "PATH")
                {
                    if (!P.Topology.Ready || Rebuilding || Transition) throw new Exception("Arm topology is not ready.");
                    Move = Number(ini, "Move", .001, 5); Turn = Number(ini, "Turn", .01, 30);
                    PosTol = Number(ini, "PosTol", .001, .05); AngleTol = Number(ini, "AngleTol", .01, 2);
                    Path = P.ReadPath(ini.Get(LinkSection, "Waypoints").ToString(), ini.Get(LinkSection, "Anchor").ToInt64(), seq, Move, Turn, PosTol, AngleTol);
                    P.ConfigurePathStages(Path, ini);
                    Path.Capture = ini.Get(LinkSection, "Capture").ToBoolean();
                    Contacts = ini.Get(LinkSection, "Contacts").ToString();
                    if (Contacts.Length > 2048) throw new Exception("Docking contact limit.");
                    Driving = true; Settled = false; MoveRequest = seq;
                }
                else if (op == "RESUME" || op == "RELEASE")
                {
                    Driving = false; P.StopMotion(""); ApplyEquipment(ini); Permitted = ini.Get(LinkSection, "Permit").ToBoolean();
                    if (ini.Get(LinkSection, "ExplicitStop").ToBoolean()) InhibitRecovery();
                    Lease = false; ResumePending = Resumed = op == "RESUME"; Error = "";
                    ResumeElapsed = 0;
                    RecoverBare |= ResumePending;
                }
                else throw new Exception("Unknown module operation.");
                Reply("STATE");
            }
            catch (Exception e) { Error = e.Message; Driving = Rebuilding = false; P.StopMotion(Error); Reply("STATE"); }
        }
        if (Configured && Link.Age > .5 && (P.Enabled || Lease || ResumePending))
        { Driving = Rebuilding = false; P.Fault("ToolSwap communication timeout."); }
        if (Detached())
        { RecoverBare = false; P.StopMotion("Tool detached; rediscovering bare arm."); P.StartControl(); }
    }
    static double Number(MyIni ini, string key, double lo, double hi)
    { double v; if (!ini.Get(LinkSection, key).TryGetDouble(out v) || !Finite(v) || v < lo || v > hi) throw new Exception("Bad module " + key); return v; }
    void ApplyEndpoint(MyIni ini)
    {
        End = P.GridTerminalSystem.GetBlockWithId(ini.Get(LinkSection, "Endpoint").ToInt64());
        var owner = P.GridTerminalSystem.GetBlockWithId(ini.Get(LinkSection, "AnchorOwner").ToInt64());
        Frame = LinkAnchor(owner, ini.Get(LinkSection, "Anchor").ToInt64());
        if (End == null || End.Closed || !End.IsSameConstructAs(P.Me) || !owner.IsSameConstructAs(P.Me)) throw new Exception("Foreign/lost module endpoint.");
        Ignore = ini.Get(LinkSection, "Ignore").ToInt64(); Tip = null;
        if (ini.Get(LinkSection, "Focus").ToBoolean())
        {
            Tip = new ArmTipInfo { Marker = Frame, Local = LinkFrame(ini, "Local") };
            if (!Tip.CubeGrid.CubeExists(Tip.Position)) throw new Exception("Module focus is outside the arm.");
        }
        ApplyEquipment(ini);
    }
    void ApplyEquipment(MyIni ini)
    {
        var couplers = new List<IMyMotorStator>(); var markers = new List<IMyTerminalBlock>();
        var tops = new List<long>(); var merges = new List<List<IMyShipMergeBlock>>();
        string[] rows = ini.Get(LinkSection, "Equipment").ToString().Split('\n');
        if (rows.Length > 16) throw new Exception("Module equipment limit.");
        foreach (var row in rows)
        {
            string[] cells = row.Split('|'); long rotorId, markerId, topId;
            if (cells.Length != 4 || !long.TryParse(cells[0], out rotorId) || !long.TryParse(cells[1], out markerId) || !long.TryParse(cells[2], out topId)) throw new Exception("Bad module equipment row.");
            var rotor = P.GridTerminalSystem.GetBlockWithId(rotorId) as IMyMotorStator; var marker = P.GridTerminalSystem.GetBlockWithId(markerId);
            if (rotor == null || marker == null || !rotor.IsSameConstructAs(P.Me) || !marker.IsSameConstructAs(P.Me) || couplers.Contains(rotor)) throw new Exception("Lost/duplicate module equipment.");
            var ports = new List<IMyShipMergeBlock>();
            foreach (var cell in cells[3].Split(',')) if (cell.Length > 0)
            {
                long id; if (!long.TryParse(cell, out id) || ports.Count >= 16) throw new Exception("Bad module merge list.");
                var port = P.GridTerminalSystem.GetBlockWithId(id) as IMyShipMergeBlock;
                if (port == null || !port.IsSameConstructAs(P.Me)) throw new Exception("Lost module merge."); ports.Add(port);
            }
            couplers.Add(rotor); markers.Add(marker); tops.Add(topId); merges.Add(ports);
        }
        Couplers.Clear(); Couplers.AddRange(couplers); Markers.Clear(); Markers.AddRange(markers); Tops.Clear(); Tops.AddRange(tops); Merges.Clear(); Merges.AddRange(merges);
    }
    public void Tick(double dt, double observationDt)
    {
        if (Queued.Length > 0) { string command = Queued; Queued = ""; Lease = true; SendCommand(command); }
        if (Link.Age > .5) return;
        if (Rebuilding)
        { string why; if (P.SettleToolTopology(out why)) { Rebuilding = Transition = false; Reply("STATE"); } }
        if (ResumePending)
        {
            ResumeElapsed += observationDt; string why = "Resume waiting for topology/budget.";
            if (P.Topology.Ready && !Rebuilding && !Transition && (P.Runtime.MaxInstructionCount <= 0 || P.Runtime.CurrentInstructionCount < P.Runtime.MaxInstructionCount * .3))
            {
                ResumePending = false;
                if (!AllowManual(out why)) ResumePending = true;
                else
                {
                    P.EnableControl(); Error = P.Enabled ? "" : "Resume refused: " + P.Status;
                    Reply("STATE");
                }
            }
            if (ResumePending && ResumeElapsed > 2)
            { ResumePending = false; Error = why; P.StopMotion(Error); P.Say(Error); Reply("STATE"); }
        }
        if (Driving)
        {
            if (Path.Capture) for (int i = 0; i < Tops.Count; i++) if (Tops[i] != 0) foreach (var m in Merges[i]) if (m.IsConnected)
            { Driving = false; Path = null; Transition = true; P.StopMotion(""); P.Topology.Ready = false; Reply("STATE"); return; }
            string why; int previous = Path.Completed;
            if (!P.StepPath(Path, dt, observationDt, out why))
            { Error = why; Driving = false; P.StopMotion(why); }
            else if (Path.Done) { Settled = true; Driving = false; P.StopMotion(""); }
            ReportElapsed += observationDt;
            if (!Driving || previous != Path.Completed || ReportElapsed >= .1) { ReportElapsed = 0; Reply("STATE"); }
        }
    }
    void Reply(string operation)
    {
        if (Resumed && !ResumePending && !P.Enabled && Error.Length == 0)
            Error = P.LastFault.Length > 0 ? P.LastFault : "Arm stopped after tool handoff.";
        var ini = Link.Packet(operation); ini.Set(LinkSection, "Ack", Request); ini.Set(LinkSection, "MoveAck", MoveRequest);
        ini.Set(LinkSection, "Lease", Lease); ini.Set(LinkSection, "Ready", P.ConfigValid && P.Topology.Ready && !Rebuilding && !Transition);
        ini.Set(LinkSection, "Enabled", P.Enabled);
        ini.Set(LinkSection, "Stopped", !Driving && !P.Enabled); ini.Set(LinkSection, "Settled", Settled);
        ini.Set(LinkSection, "PathId", Path == null ? 0 : Path.Id); ini.Set(LinkSection, "Completed", Path == null ? 0 : Path.Completed); ini.Set(LinkSection, "Count", Path == null ? 0 : Path.Waypoints.Count);
        ini.Set(LinkSection, "Error", Error); ini.Set(LinkSection, "Pilot", P.HasPilotInput());
        ini.Set(LinkSection, "PilotSelected", P.PilotSelected);
        ini.Set(LinkSection, "Keyboard", P.ReadKeyboard); ini.Set(LinkSection, "Mouse", P.ReadMouse); ini.Set(LinkSection, "Roll", P.ReadRoll);
        ini.Set(LinkSection, "DefaultMove", P.MoveMps); ini.Set(LinkSection, "DefaultTurn", P.TurnDeg);
        ini.Set(LinkSection, "Head", P.Topology.Head == null ? 0 : P.Topology.Head.EntityId);
        var rows = new List<string>(); foreach (var group in P.Topology.Groups)
        { var ids = new List<string>(); foreach (var member in group.M) ids.Add(member.B.EntityId.ToString()); rows.Add(string.Join(",", ids)); }
        ini.Set(LinkSection, "Groups", string.Join("\n", rows)); Link.Send(ini);
    }
}


// Optional direct peers. The arm remains the only joint writer; planners submit
// complete paths, observers read state, safety peers may stop every controller.
sealed class ArmServices
{
    sealed class Client
    {
        public ModuleLink Link;
        public string Role, State = "Idle", Error = "";
        public ArmPath Path;
        public long Seen;
        public double ReplyAge;
    }
    readonly ArmCore P;
    readonly Dictionary<long, Client> Peers = new Dictionary<long, Client>();
    Client Owner;
    string Config = "";
    long Revision = 1;
    public SafetyOverlay Safety;
    public ArmServices(ArmCore p) { P = p; }
    public void Load(MyIni ini)
    {
        var peers = new Dictionary<long, Client>();
        string text = ini.Get("Modules", "Peers").ToString();
        foreach (string row in text.Split('\n'))
        {
            if (row.Trim().Length == 0) continue;
            string[] cells = row.Split('|');
            if (cells.Length != 2 || peers.Count >= 16) throw new Exception("Modules.Peers needs PB name | Observe, Plan, Stop or Safety (max 16).");
            string name = cells[0].Trim(), role = cells[1].Trim();
            if (role != "Observe" && role != "Plan" && role != "Stop" && role != "Safety" || name == ini.Get("Modules", "ToolSwapPB").ToString().Trim()) throw new Exception("Invalid/duplicate module role.");
            var link = new ModuleLink(P); link.Resolve(name);
            if (peers.ContainsKey(link.Peer.EntityId)) throw new Exception("Duplicate module PB.");
            peers.Add(link.Peer.EntityId, new Client { Link = link, Role = role });
        }
        SafetyOverlay safety = null;
        foreach (var pair in peers) if (pair.Value.Role == "Safety") { if (safety != null) throw new Exception("One Safety service per arm."); safety = new SafetyOverlay(P, pair.Value.Link); }
        Revoke("Configuration reloaded."); Peers.Clear(); Safety = safety;
        foreach (var pair in peers) Peers.Add(pair.Key, pair.Value);
        Config = text;
    }
    public void WriteConfig(MyIni ini) { if (Config.Length > 0) ini.Set("Modules", "Peers", Config); }
    public void Revoke(string reason)
    {
        Revision++;
        if (Safety != null) Safety.Reset();
        if (Owner != null) { Owner.State = "Canceled"; Owner.Error = reason; Owner = null; }
    }
    public void Tick(double elapsed)
    {
        if (Safety != null) Safety.Tick(elapsed, Revision);
        if (Owner == null) return;
        var client = Owner;
        if (Finite(elapsed) && elapsed > 0) client.Link.Age += elapsed;
        if (Finite(elapsed) && elapsed > 0) client.ReplyAge += elapsed;
        var peer = client.Link.Peer;
        if (peer.Closed || !peer.IsFunctional || !peer.IsSameConstructAs(P.Me) || client.Link.Age > .5 || client.ReplyAge > .5)
        { P.Fault("Planner communication timeout/unavailable."); return; }
        if (P.LocalPath == null)
        {
            client.State = client.Path.Done && P.Enabled ? "Complete" : "Canceled";
            Owner = null; Revision++;
        }
    }
    public bool Receive(MyIGCMessage message)
    {
        Client client;
        if (!Peers.TryGetValue(message.Source, out client)) return false;
        MyIni ini; var link = client.Link;
        if (!link.Read(message, out ini) || ini.Get(LinkSection, "Service").ToInt32() != 1) return true;
        long epoch = ini.Get(LinkSection, "Epoch").ToInt64(), seq = ini.Get(LinkSection, "Sequence").ToInt64();
        string op = ini.Get(LinkSection, "Operation").ToString();
        long remote = ini.Get(LinkSection, "Remote").ToInt64();
        if (op == "HELLO")
        {
            if (epoch < link.RemoteEpoch || remote != 0 && remote != link.Epoch) return true;
            if (epoch != link.RemoteEpoch)
            {
                if (client.Role == "Safety") Safety.Reset();
                if (Owner == client) P.StopMotion("Planner restarted.");
                link.RemoteEpoch = epoch; link.Received = 0;
                client.Seen = 0;
            }
        }
        else if (epoch != link.RemoteEpoch || remote != link.Epoch ||
            op != "STATUS" && op != "PING" && op != "STOP" && ini.Get(LinkSection, "Generation").ToInt64() != Revision) return true;
        if (seq <= link.Received) return true;
        long seen = ini.Get(LinkSection, "Seen").ToInt64();
        if (op != "HELLO" && (seen < 0 || seen > link.Sequence)) return true;
        if (op != "HELLO" && seen > client.Seen) { client.Seen = seen; client.ReplyAge = 0; }
        link.Received = seq; link.Age = 0; client.Error = "";
        try
        {
            if (op == "HELLO" || op == "STATUS" || op == "PING") { }
            else if (op == "STOP")
            {
                if (client.Role != "Stop" && client.Role != "Safety" && Owner != client) throw new Exception("Module has no stop authority.");
                P.Tools.InhibitRecovery();
                if (P.Tools.Busy) P.Tools.Cancel("Module emergency stop.");
                P.StopMotion("Module stop: " + link.Peer.CustomName); P.Say(P.Status);
            }
            else if (op == "GUIDE")
            { if (client.Role != "Safety" || seen == 0) throw new Exception("No collision authority/acknowledgement."); Safety.Read(ini); }
            else if (op == "PATH")
            {
                string why;
                if (client.Role != "Plan") throw new Exception("Module is read only.");
                if (seen == 0) throw new Exception("Acknowledge the arm STATE before PATH.");
                if (Owner != null || P.Tools.Busy || P.LocalPath != null || P.GoingHome) throw new Exception("Arm motion already owned.");
                if (!P.Enabled || !P.ConfigValid || !P.Topology.Ready || !P.Tools.AllowManual(out why)) throw new Exception("Arm not ready; run On.");
                var path = P.ReadPath(ini.Get(LinkSection, "Waypoints").ToString(), ini.Get(LinkSection, "Anchor").ToInt64(), seq, P.MoveMps, P.TurnDeg, .005, .2);
                P.ConfigurePathStages(path, ini);
                Vector3D p, f, u;
                if (!P.ReadPose(out p, out f, out u)) throw new Exception("Arm focus unavailable.");
                // No state changes until every waypoint and policy has passed.
                P.StopMotion(""); P.LocalPath = path; P.CapturePose(p, f, u); P.Enabled = true;
                P.Frequency = UpdateFrequency.Update1;
                client.Path = path; client.State = "Moving"; Owner = client;
                client.ReplyAge = 0;
                P.Say("Module path: " + link.Peer.CustomName);
            }
            else throw new Exception("Unsupported service operation.");
        }
        catch (Exception e) { client.Error = e.Message; }
        if (op != "GUIDE") Reply(client, seq); return true;
    }
    void Reply(Client client, long ack)
    {
        var link = client.Link; link.Generation = Revision;
        var ini = link.Packet("STATE"); ini.Set(LinkSection, "Service", 1); ini.Set(LinkSection, "Ack", ack);
        ini.Set(LinkSection, "Role", client.Role); ini.Set(LinkSection, "Owner", Owner == null ? 0 : Owner.Link.Peer.EntityId);
        ini.Set(LinkSection, "Ready", P.ConfigValid && P.Topology.Ready); ini.Set(LinkSection, "Enabled", P.Enabled);
        ini.Set(LinkSection, "Busy", P.Tools.Busy || P.LocalPath != null || P.GoingHome);
        ini.Set(LinkSection, "State", client.State); ini.Set(LinkSection, "Error", client.Error);
        ini.Set(LinkSection, "PathId", client.Path == null ? 0 : client.Path.Id);
        ini.Set(LinkSection, "Completed", client.Path == null ? 0 : client.Path.Completed);
        ini.Set(LinkSection, "Count", client.Path == null ? 0 : client.Path.Waypoints.Count);
        ini.Set(LinkSection, "DefaultMove", P.MoveMps); ini.Set(LinkSection, "DefaultTurn", P.TurnDeg);
        ini.Set(LinkSection, "SafetyRequired", Safety != null);
        if (Safety != null) Safety.WriteStatus(ini);
        ini.Set(LinkSection, "PositionError", P.LastPositionError); ini.Set(LinkSection, "AngleError", P.LastOrientationErrorDeg);
        ini.Set(LinkSection, "SolverPasses", P.LastSolverPasses); ini.Set(LinkSection, "SolverBudget", P.LastSolverBudgetLimited);
        if (P.Topology.Ready) WriteFrame(ini, "Pose", P.Topology.EndPose);
        // Failure to reply to an idle observer must not fault a healthy arm.
        try { link.Send(ini); } catch { if (Owner == client) P.Fault("Planner reply failed."); }
    }
}


// Optional safety overlay. Geometry stays in the shared Collision PB.
sealed class SafetyOverlay
{
    readonly ArmCore P;
    public readonly ModuleLink Link;
    public double Age = 1e6, SendAge, Scale;
    public bool Limited;
    public bool Blocked = true;
    public double Fraction = 1;
    string Stamp = "", Reason = "Waiting for collision service.";
    double[] Bias = new double[0];
    readonly List<double[]> Rows = new List<double[]>();
    readonly Dictionary<long, double> Sent = new Dictionary<long, double>();
    double Clock;
    int ConstraintLimit = 8;
    public void Reset() { Age = 1e6; Sent.Clear(); SendAge = 1; }
    public SafetyOverlay(ArmCore p, ModuleLink link) { P = p; Link = link; }
    double Cap(int i) { return P.Topology.Groups[i].Rotary ? (P.GoingHome ? P.HomeJointRpm : P.JointRpm) * Math.PI / 30 : P.GoingHome ? P.HomePistonMps : P.PistonMps; }
    string Identity() { return P.Topology.Fingerprint + "/" + P.Topology.Head.CubeGrid.EntityId + "/" + P.Tools.Contacts; }
    public void Tick(double elapsed, long revision)
    {
        if (Finite(elapsed) && elapsed > 0) { Clock += elapsed; Age += elapsed; SendAge += elapsed; }
        if (!P.Topology.Ready || P.Topology.Groups.Count == 0) { Age = 1e6; Stamp = ""; return; }
        string stamp = Identity();
        if (stamp != Stamp) { Stamp = stamp; Age = 1e6; Sent.Clear(); SendAge = 1; }
        if (SendAge < .1) return;
        SendAge = 0;
        var ini = Link.Packet("CHECK"); ini.Set(LinkSection, "Service", 1); ini.Set(LinkSection, "Generation", revision);
        ini.Set(LinkSection, "ConstraintLimit", 16);
        ini.Set(LinkSection, "Fingerprint", Stamp); ini.Set(LinkSection, "Groups", P.Topology.Fingerprint);
        ini.Set(LinkSection, "Hull", P.Topology.Base.CubeGrid.EntityId); ini.Set(LinkSection, "ToolGrid", P.Topology.Head.CubeGrid.EntityId);
        ini.Set(LinkSection, "Contacts", P.Tools.Contacts);
        var focus = P.Tools.Focus;
        var cell = focus == null ? P.Topology.End.Position : focus.Position;
        ini.Set(LinkSection, "Socket", focus != null);
        ini.Set(LinkSection, "TipGrid", P.Topology.End.CubeGrid.EntityId);
        ini.Set(LinkSection, "TipCell", cell.X + " " + cell.Y + " " + cell.Z);
        WriteVector(ini, "TipUp", P.Topology.EndPose.Up);
        var caps = new System.Text.StringBuilder(); var rates = new System.Text.StringBuilder();
        for (int i = 0; i < P.Topology.Groups.Count; i++)
        {
            if (i > 0) { caps.Append(' '); rates.Append(' '); }
            caps.Append(AutoSettings.Number(Cap(i)));
            rates.Append(AutoSettings.Number(P.LastQdot == null ? 0 : P.LastQdot[i]));
        }
        WriteNavigation(ini); ini.Set(LinkSection, "Caps", caps.ToString()); ini.Set(LinkSection, "Rates", rates.ToString());
        if (Sent.Count >= 16) Sent.Clear(); Sent[Link.Sequence] = Clock;
        try { Link.Send(ini); } catch { Age = 1e6; Reason = "Collision service unavailable."; }
    }
    public void Read(MyIni ini)
    {
        long ack = ini.Get(LinkSection, "Ack").ToInt64(); double time;
        if (!Sent.TryGetValue(ack, out time) || Clock - time > .25 || ini.Get(LinkSection, "Fingerprint").ToString() != Stamp) return;
        int n = P.Topology.Groups.Count; double scale; bool blocked;
        if (!ini.Get(LinkSection, "Hold").TryGetBoolean(out blocked)) throw new Exception("Missing collision permission.");
        if (!ini.Get(LinkSection, "Scale").TryGetDouble(out scale) || !Finite(scale) || scale < 0 || scale > 1) throw new Exception("Invalid collision speed scale.");
        var bias = Numbers(ini.Get(LinkSection, "Bias").ToString(), n);
        for (int i = 0; i < n; i++)
            if (Math.Abs(bias[i]) > Cap(i) * .25 + 1e-9) throw new Exception("Collision preference exceeds cap.");
        var rows = new List<double[]>();
        int limit = 8;
        if (ini.ContainsKey(LinkSection, "ConstraintLimit") && (!ini.Get(LinkSection, "ConstraintLimit").TryGetInt32(out limit) || limit != 8 && limit != 16)) throw new Exception("Invalid collision constraint capacity.");
        foreach (string text in ini.Get(LinkSection, "Rows").ToString().Split('\n'))
        {
            if (text.Length == 0) continue;
            if (rows.Count >= limit) throw new Exception("Too many collision constraints.");
            var row = Numbers(text, n + 1);
            foreach (double value in row) if (Math.Abs(value) > 1e6) throw new Exception("Collision constraint out of range.");
            rows.Add(row);
        }
        // Commit only a complete, fresh, matching reply. Replay cannot extend expiry.
        ReadNavigation(ini); Sent.Remove(ack); Bias = bias; Rows.Clear(); Rows.AddRange(rows); Scale = scale; Blocked = blocked;
        ConstraintLimit = limit;
        Reason = ini.Get(LinkSection, "Reason").ToString(); Age = Clock - time;
    }
    static double[] Numbers(string text, int count)
    {
        string[] parts = text.Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
        if (parts.Length != count) throw new Exception("Collision row size mismatch.");
        var values = new double[count];
        for (int i = 0; i < count; i++) if (!ReadDouble(parts[i], out values[i])) throw new Exception("Invalid collision number.");
        return values;
    }
    public bool Ready { get { return Age <= .25 && P.Topology.Ready && Stamp == Identity(); } }
    public void WriteStatus(MyIni ini)
    {
        ini.Set(LinkSection, "SafetyReady", Ready); ini.Set(LinkSection, "SafetyHold", !Ready || Blocked);
        ini.Set(LinkSection, "SafetyAge", Age); ini.Set(LinkSection, "SafetyScale", Scale);
        ini.Set(LinkSection, "SafetyLimited", Limited); ini.Set(LinkSection, "SafetyFraction", Fraction);
        ini.Set(LinkSection, "SafetyReallocated", Reallocated); ini.Set(LinkSection, "SafetyRepairPasses", RepairPasses); ini.Set(LinkSection, "SafetyRepairBudget", RepairBudget); NavigationStatus(ini); ini.Set(LinkSection, "SafetyConstraints", Rows.Count); ini.Set(LinkSection, "SafetyConstraintLimit", ConstraintLimit);
        ini.Set(LinkSection, "SafetyReason", Ready ? Reason : "Waiting for fresh collision guidance.");
    }
    public double Preferred(int i) { return Ready && i < Bias.Length ? Bias[i] : 0; }
    public void Hold()
    {
        P.Topology.ZeroOwned(); if (P.LastQdot != null) Array.Clear(P.LastQdot, 0, P.LastQdot.Length);
        P.LastAchievedLinear = P.LastAchievedAngular = Vector3D.Zero;
        P.Say("COLLISION hold | " + (Ready ? Reason : "Waiting for fresh collision guidance."));
    }
    public bool NavActive, NavWait;
public string NavKey = "";
public int NavStage;
public double NavExtra, NavPeerRemaining;
public Vector3D NavPoint;
string NavigationKey() { return P.LocalPath == null || P.LocalPath.Done ? "" : P.LocalPath.Id + "/" + P.LocalPath.Completed; }
void WriteNavigation(MyIni ini)
{
    var path = P.LocalPath;
    bool allowed = P.Enabled && !P.GoingHome && path != null && !path.Done && !path.Linear && !path.Docking;
    ini.Set(LinkSection, "NavVersion", 1); ini.Set(LinkSection, "NavAllowed", allowed);
    if (!allowed) return;
    ini.Set(LinkSection, "NavKey", NavigationKey());
    WriteVector(ini, "NavHead", P.Topology.EndPosition); WriteVector(ini, "NavGoal", path.Current.Translation);
    ini.Set(LinkSection, "NavSpeed", Math.Min(path.Move, path.CurrentMove));
}
void ReadNavigation(MyIni ini)
{
    bool active = false, wait = false; string key = ""; int stage = 0; double extra = 0, peer = 0; Vector3D point = Vector3D.Zero;
    if (ini.ContainsKey(LinkSection, "NavVersion")) {
        if (ini.Get(LinkSection, "NavVersion").ToInt32() != 1 || !ini.Get(LinkSection, "NavActive").TryGetBoolean(out active)) throw new Exception("Invalid navigation reply.");
        if (active) {
            key = ini.Get(LinkSection, "NavKey").ToString(); stage = ini.Get(LinkSection, "NavStage").ToInt32();
            if (!ini.Get(LinkSection, "NavWait").TryGetBoolean(out wait) || !ini.Get(LinkSection, "NavExtra").TryGetDouble(out extra) || !ini.Get(LinkSection, "NavPeerRemaining").TryGetDouble(out peer) ||
                key.Length == 0 || key.Length > 80 || !Finite(extra) || !Finite(peer) || extra < 0 || extra > 100 || peer < 0 || peer > 1000 || stage < 1 || stage > 3) throw new Exception("Invalid navigation bounds.");
            var values = Numbers(ini.Get(LinkSection, "NavPoint").ToString(), 3); point = new Vector3D(values[0], values[1], values[2]);
            if (key != NavigationKey() || P.LocalPath.Linear || P.LocalPath.Docking) active = false;
            else if (Vector3D.Distance(point, P.LocalPath.Current.Translation) > 200) throw new Exception("Navigation point out of range.");
        }
    }
    NavActive = active; NavWait = wait; NavKey = key; NavStage = stage; NavExtra = extra; NavPoint = point; NavPeerRemaining = peer;
}
public void Navigate(ArmPath path, ref MatrixD demand)
{
    if (!Ready || Blocked || !NavActive || NavKey != NavigationKey() || path.Linear || path.Docking) return;
    if (path.NavBudgetKey != NavKey) { path.Budget += 30; path.NavBudgetKey = NavKey; path.NavBudgetDistance = 0; }
    if (NavExtra > path.NavBudgetDistance) { path.Budget += (NavExtra - path.NavBudgetDistance) / Math.Min(path.Move, path.CurrentMove); path.NavBudgetDistance = NavExtra; }
    demand.Translation = NavPoint;
}

    ini.Set(LinkSection, "NavPeerRemaining", active ? NavPeerRemaining : 0); ini.Set(LinkSection, "NavExtra", active ? NavExtra : 0);
}

// Experimental automatic-path reallocation. Uniform safety veto remains the
// fallback. Accept only a bounded candidate satisfying every received row.
double[,] RepairMetric, RepairDirections;
double[] RepairSeed, RepairCandidate, RepairChecked, RepairBest, RepairDual, RepairDiagonal;
public bool Reallocated, RepairBudget;
public int RepairPasses;
bool RepairWork() { return P.Runtime.MaxInstructionCount <= 0 || P.Runtime.CurrentInstructionCount < P.Runtime.MaxInstructionCount * .65; }
double RepairDot(int k, int n, double[] x) {
    if (k >= Rows.Count) { int i = (k - Rows.Count) / 2; return ((k - Rows.Count) % 2 == 0 ? 1 : -1) * x[i]; }
    double sum = 0; for (int i = 0; i < n; i++) sum += Rows[k][i] * P.Scale[i] * x[i]; return sum;
}
double RepairLower(int k, int n) {
    if (k < Rows.Count) return Rows[k][n];
    int i = (k - Rows.Count) / 2; return (k - Rows.Count) % 2 == 0 ? P.Lo[i] : -P.Hi[i];
}
bool RepairFeasible(double[] x, int n) {
    for (int i = 0; i < n; i++) if (!Finite(x[i]) || x[i] < P.Lo[i] - 1e-10 || x[i] > P.Hi[i] + 1e-10) return false;
    foreach (var row in Rows) { double dot = 0; for (int i = 0; i < n; i++) dot += row[i] * P.Scale[i] * x[i]; if (dot < row[n] - 1e-10) return false; }
    return true;
}
double RepairCost(double[] x, int n) {
    double cost = 0;
    for (int r = 0; r < 6; r++) { double e = -P.Task[r]; for (int i = 0; i < n; i++) e += P.B[r, i] * x[i]; cost += e * e; }
    for (int i = 0; i < n; i++) { double e = x[i] - P.Bias[i]; cost += SolverDamping * SolverDamping * e * e; }
    return cost;
}
public void Constrain(double[] rates) {
    Reallocated = RepairBudget = false; RepairPasses = 0; int n = rates.Length, m = Rows.Count + 2 * n;
    if (RepairCandidate == null || RepairCandidate.Length != n || RepairDual.Length != m) {
        RepairMetric = new double[n, n]; RepairDirections = new double[m, n];
        RepairSeed = new double[n]; RepairCandidate = new double[n]; RepairChecked = new double[n]; RepairBest = new double[n]; RepairDual = new double[m]; RepairDiagonal = new double[m];
    }
    for (int i = 0; i < n; i++) RepairSeed[i] = P.Scale[i] > 1e-12 ? rates[i] / P.Scale[i] : 0;
    Limit(rates);
    if (!P.ToolDriving || !Ready || Blocked || !Limited || Rows.Count == 0 || !P.LastSolverConverged || P.LastSolverBudgetLimited) return;
    if (!RepairWork()) { RepairBudget = true; return; }
    for (int i = 0; i < n; i++) { RepairCandidate[i] = RepairSeed[i]; RepairBest[i] = P.Scale[i] > 1e-12 ? rates[i] / P.Scale[i] : 0; }
    bool baselineFeasible = RepairFeasible(RepairBest, n); double baseline = baselineFeasible ? RepairCost(RepairBest, n) : double.MaxValue, best = baseline;
    P.BuildTaskMatrix(n, SolverDamping * SolverDamping, false);
    Array.Clear(P.Rhs, 0, 6);
    if (!Cholesky(P.Gram, P.Rhs, P.Chol, P.Temp, P.Solution)) return;
    // Woodbury: ridge * (B'B + ridge I)^-1. The common positive
    // scale cancels in each dual coordinate update.
    for (int col = 0; col < n; col++) {
        if (!RepairWork()) { RepairBudget = true; return; }
        for (int r = 0; r < 6; r++) { double v = P.B[r, col]; for (int j = 0; j < r; j++) v -= P.Chol[r, j] * P.Temp[j]; P.Temp[r] = v / P.Chol[r, r]; }
        for (int r = 5; r >= 0; r--) { double v = P.Temp[r]; for (int j = r + 1; j < 6; j++) v -= P.Chol[j, r] * P.Solution[j]; P.Solution[r] = v / P.Chol[r, r]; }
        for (int i = 0; i < n; i++) { double v = i == col ? 1 : 0; for (int r = 0; r < 6; r++) v -= P.B[r, i] * P.Solution[r]; RepairMetric[i, col] = P.Scale[i] > 1e-12 && P.Scale[col] > 1e-12 ? v : 0; }
    }
    for (int i = 0; i < n; i++) for (int j = 0; j < i; j++) RepairMetric[i, j] = RepairMetric[j, i] = (RepairMetric[i, j] + RepairMetric[j, i]) * .5;
    for (int k = 0; k < m; k++) {
        if (!RepairWork()) { RepairBudget = true; return; }
        RepairDual[k] = 0;
        for (int i = 0; i < n; i++) {
            double v = 0;
            if (k < Rows.Count) for (int j = 0; j < n; j++) v += RepairMetric[i, j] * Rows[k][j] * P.Scale[j];
            else v = RepairMetric[i, (k - Rows.Count) / 2] * ((k - Rows.Count) % 2 == 0 ? 1 : -1);
            RepairDirections[k, i] = v;
        }
        double diagonal = k < Rows.Count ? 0 : RepairDirections[k, (k - Rows.Count) / 2] * ((k - Rows.Count) % 2 == 0 ? 1 : -1);
        if (k < Rows.Count) for (int i = 0; i < n; i++) diagonal += Rows[k][i] * P.Scale[i] * RepairDirections[k, i];
        if (!Finite(diagonal) || diagonal < -1e-10) return; RepairDiagonal[k] = diagonal;
    }
    for (int pass = 0; pass < 12; pass++) {
        RepairPasses = pass + 1;
        for (int k = 0; k < m; k++) {
            if (!RepairWork()) { RepairBudget = true; return; }
            double unmet = RepairLower(k, n) - RepairDot(k, n, RepairCandidate);
            if (RepairDiagonal[k] <= 1e-14) { if (unmet > 1e-10) return; continue; }
            double next = Math.Max(0, RepairDual[k] + unmet / RepairDiagonal[k]), change = next - RepairDual[k]; RepairDual[k] = next;
            for (int i = 0; i < n; i++) RepairCandidate[i] += change * RepairDirections[k, i];
        }
        if (!RepairFeasible(RepairCandidate, n)) continue;
        for (int i = 0; i < n; i++) RepairChecked[i] = Clamp(RepairCandidate[i], P.Lo[i], P.Hi[i]);
        if (!RepairFeasible(RepairChecked, n)) continue;
        double cost = RepairCost(RepairChecked, n);
        if (Finite(cost) && cost < best - 1e-14) { best = cost; Array.Copy(RepairChecked, RepairBest, n); Reallocated = true; }
        if (Reallocated) break;
    }
    if (!Reallocated) return;
    for (int i = 0; i < n; i++) rates[i] = RepairBest[i] * P.Scale[i];
    // Retain the original strict row veto after numerical projection too.
    Limit(rates);
    for (int i = 0; i < n; i++) RepairChecked[i] = P.Scale[i] > 1e-12 ? rates[i] / P.Scale[i] : 0;
    if (!RepairFeasible(RepairChecked, n) || RepairCost(RepairChecked, n) >= baseline - 1e-14) {
        for (int i = 0; i < n; i++) rates[i] = RepairSeed[i] * P.Scale[i]; Limit(rates); Reallocated = false;
    } else Limited = true;
}

public void Limit(double[] rates)
    {
        Limited = false; Fraction = 1;
        if (!Ready || Blocked) { Array.Clear(rates, 0, rates.Length); Limited = true; Fraction = 0; return; }
        double alpha = P.GoingHome ? Scale : 1, minimum = 0;
        foreach (var row in Rows)
        {
            double dot = 0; for (int i = 0; i < rates.Length; i++) dot += row[i] * rates[i];
            double lower = row[rates.Length];
            if (dot > 1e-10) minimum = Math.Max(minimum, lower / dot);
            else if (dot < -1e-10) alpha = Math.Min(alpha, lower / dot);
            else if (lower > 1e-8) minimum = 2;
        }
        if (minimum > alpha + 1e-8 || minimum > 1 || alpha < 0) alpha = 0;
        alpha = Clamp(alpha, 0, 1);
        Fraction = alpha;
        if (alpha < 1) { Limited = true; for (int i = 0; i < rates.Length; i++) rates[i] *= alpha; }
    }
}
