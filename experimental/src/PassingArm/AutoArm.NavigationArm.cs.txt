public bool NavActive, NavWait;
public string NavKey = "";
public int NavStage;
public double NavExtra, NavPeerRemaining;
public Vector3D NavPoint;
string NavigationKey() { return P.LocalPath == null || P.LocalPath.Done ? "" : P.LocalPath.Id + "/" + P.LocalPath.Completed; }
void WriteNavigation(MyIni ini)
{
    var path = P.LocalPath;
    bool allowed = P.Enabled && !P.GoingHome && path != null && !path.Done && !path.Linear && !path.Docking;
    ini.Set(LinkSection, "NavVersion", 1); ini.Set(LinkSection, "NavAllowed", allowed);
    if (!allowed) return;
    ini.Set(LinkSection, "NavKey", NavigationKey());
    WriteVector(ini, "NavHead", P.Topology.EndPosition); WriteVector(ini, "NavGoal", path.Current.Translation);
    ini.Set(LinkSection, "NavSpeed", Math.Min(path.Move, path.CurrentMove));
}
void ReadNavigation(MyIni ini)
{
    bool active = false, wait = false; string key = ""; int stage = 0; double extra = 0, peer = 0; Vector3D point = Vector3D.Zero;
    if (ini.ContainsKey(LinkSection, "NavVersion")) {
        if (ini.Get(LinkSection, "NavVersion").ToInt32() != 1 || !ini.Get(LinkSection, "NavActive").TryGetBoolean(out active)) throw new Exception("Invalid navigation reply.");
        if (active) {
            key = ini.Get(LinkSection, "NavKey").ToString(); stage = ini.Get(LinkSection, "NavStage").ToInt32();
            if (!ini.Get(LinkSection, "NavWait").TryGetBoolean(out wait) || !ini.Get(LinkSection, "NavExtra").TryGetDouble(out extra) || !ini.Get(LinkSection, "NavPeerRemaining").TryGetDouble(out peer) ||
                key.Length == 0 || key.Length > 80 || !Finite(extra) || !Finite(peer) || extra < 0 || extra > 100 || peer < 0 || peer > 1000 || stage < 1 || stage > 3) throw new Exception("Invalid navigation bounds.");
            var values = Numbers(ini.Get(LinkSection, "NavPoint").ToString(), 3); point = new Vector3D(values[0], values[1], values[2]);
            if (key != NavigationKey() || P.LocalPath.Linear || P.LocalPath.Docking) active = false;
            else if (Vector3D.Distance(point, P.LocalPath.Current.Translation) > 200) throw new Exception("Navigation point out of range.");
        }
    }
    NavActive = active; NavWait = wait; NavKey = key; NavStage = stage; NavExtra = extra; NavPoint = point; NavPeerRemaining = peer;
}
public void Navigate(ArmPath path, ref MatrixD demand)
{
    if (!Ready || Blocked || !NavActive || NavKey != NavigationKey() || path.Linear || path.Docking) return;
    if (path.NavBudgetKey != NavKey) { path.Budget += 30; path.NavBudgetKey = NavKey; path.NavBudgetDistance = 0; }
    if (NavExtra > path.NavBudgetDistance) { path.Budget += (NavExtra - path.NavBudgetDistance) / Math.Min(path.Move, path.CurrentMove); path.NavBudgetDistance = NavExtra; }
    demand.Translation = NavPoint;
}
void NavigationStatus(MyIni ini)
{
    bool active = Ready && !Blocked && NavActive && NavKey == NavigationKey();
    ini.Set(LinkSection, "NavActive", active); ini.Set(LinkSection, "NavWait", active && NavWait);
    ini.Set(LinkSection, "NavKey", active ? NavKey : ""); ini.Set(LinkSection, "NavStage", active ? NavStage : 0);
    ini.Set(LinkSection, "NavRemaining", active ? Vector3D.Distance(P.Topology.EndPosition, NavPoint) : 0);
    ini.Set(LinkSection, "NavPeerRemaining", active ? NavPeerRemaining : 0); ini.Set(LinkSection, "NavExtra", active ? NavExtra : 0);
}
