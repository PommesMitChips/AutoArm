// Human-facing configuration; physical row identities live in PB Storage.
enum PilotMode { HEAD, HRZ, VRT }

sealed class AutoSettings
{
    public static readonly string[] Keys = { "HeadSpeed", "HeadTurnSpeed", "StepAngle", "JointSpeed", "PistonSpeed", "HomeJointSpeed", "HomePistonSpeed", OrientationToleranceKey, "JointAcceleration", "PistonAcceleration", "PositionCorrectionCap", "OrientationCorrectionCap", "TargetLead", "PositionIntegralGain", "OrientationIntegralGain", "PositionIntegralCap", "OrientationIntegralCap", "MouseSensitivity", "PositionDamping", "OrientationDamping" };
    public static readonly double[] Defaults = { .20, 5, 5, .50, .20, .50, .20, 3, 1, .40, .20, 5, 1.50, .50, .25, .20, 5, .025, 0, 0 };
    static readonly double[] Min = { 0, .01, .01, .01, .01, .01, .01, 0, .01, .01, 0, 0, .01, 0, 0, 0, 0, 0, 0, 0 };
    static readonly double[] Max = { 5, 30, 180, 5, 2, 5, 2, 30, 20, 10, 5, 30, 10, 5, 5, 5, 30, 10, 3, 3 };
    static readonly string[] Units = { "Head translation m/s", "Head rotation deg/s", "Toolbar step deg", "Rotor/hinge RPM cap", "Piston m/s cap", "Home RPM cap", "Home m/s cap", "Idle orientation band deg", "Rotor RPM/s cap", "Piston m/s^2 cap", "Correction m/s cap", "Correction deg/s cap", "Target lead m", "Bias gain 1/s^2; 0=off", "Angular bias gain 1/s^2; 0=off", "Bias m/s cap", "Bias deg/s cap", "Mouse scale; total rotation capped", "Velocity error damping; 0=off", "Angular velocity error damping; 0=off" };
    public readonly double[] V = (double[])Defaults.Clone();
    public bool Keyboard = true, Roll = true, Mouse;
    public PilotMode Mode = PilotMode.HEAD;

    public bool Read(MyIni ini, out string why)
    {
        why = "";
        if (ini.ContainsKey(ConfigSection, MovementModeKey) && !ReadMode(ini.Get(ConfigSection, MovementModeKey).ToString(), out Mode))
        { why = "MovementMode must be HEAD, HRZ or VRT."; return false; }
        for (int i = 0; i < Keys.Length; i++)
        {
            double v;
            if (!ini.ContainsKey(ConfigSection, Keys[i])) continue;
            if (!ini.Get(ConfigSection, Keys[i]).TryGetDouble(out v) || !Finite(v) || v < Min[i] || v > Max[i])
            { why = Keys[i] + " must be finite and in " + Min[i] + ".." + Max[i] + " (" + Units[i] + ")."; return false; }
            V[i] = v;
        }
        if (ini.ContainsKey(ConfigSection, KeyboardKey) && !ini.Get(ConfigSection, KeyboardKey).TryGetBoolean(out Keyboard))
        { why = "ReadKeyboard must be true or false."; return false; }
        if (ini.ContainsKey(ConfigSection, RollKey) && !ini.Get(ConfigSection, RollKey).TryGetBoolean(out Roll))
        { why = "ReadRoll must be true or false."; return false; }
        if (ini.ContainsKey(ConfigSection, MouseKey) && !ini.Get(ConfigSection, MouseKey).TryGetBoolean(out Mouse))
        { why = "ReadMouse must be true or false."; return false; }
        return true;
    }

    public void Write(MyIni ini)
    {
        ini.Set(ConfigSection, MovementModeKey, Mode.ToString());
        ini.SetComment(ConfigSection, MovementModeKey, "HEAD=head, HRZ=cockpit side, VRT=cockpit top/bottom. Reload applies defaults.");
        for (int i = 0; i < Keys.Length; i++)
        {
            ini.Set(ConfigSection, Keys[i], Number(V[i]));
            ini.SetComment(ConfigSection, Keys[i], Units[i] + "; " + Min[i] + ".." + Max[i]);
        }
        ini.Set(ConfigSection, KeyboardKey, Keyboard); ini.Set(ConfigSection, RollKey, Roll);
        ini.Set(ConfigSection, MouseKey, Mouse);
        ini.SetComment(ConfigSection, KeyboardKey, "Game movement input.");
        ini.SetComment(ConfigSection, RollKey, "Q/E: Head roll.");
        ini.SetComment(ConfigSection, MouseKey, "Optional view pitch/yaw.");
    }
    public static bool ReadMode(string text, out PilotMode mode)
    {
        mode = PilotMode.HEAD;
        switch (text.Trim().ToUpperInvariant())
        {
            case "HEAD": return true;
            case "HRZ": mode = PilotMode.HRZ; return true;
            case "VRT": mode = PilotMode.VRT; return true;
            default: return false;
        }
    }
    public static string Number(double v) { return v.ToString("R", System.Globalization.CultureInfo.InvariantCulture); }
}

// BEGIN STANDALONE CONFIG
bool ConfigDefaultsNeeded => string.IsNullOrWhiteSpace(Me.CustomData);
bool ReadConfig(string text, out MyIni ini, out string why)
{
    ini = new MyIni(); why = ""; MyIniParseResult error;
    if (!ini.TryParse(text ?? "", out error))
    { why = "Custom Data is not valid INI (" + error + "). Fix it or delete Custom Data."; return false; }
    if (string.IsNullOrWhiteSpace(text))
    { ini.Set(ArmSection, "Arm", ArmName); ini.Set(ArmSection, "Format", AutoConfig.Format); ini.Set(ConfigSection, "Actuators", ""); return true; }
    int format;
    if (!int.TryParse(ini.Get(ArmSection, "Format").ToString(), out format) || format != AutoConfig.Format)
    { why = "Custom Data Format must be " + AutoConfig.Format + ". Delete Custom Data."; return false; }
    if (ini.Get(ArmSection, "Arm").ToString() != ArmName)
    { why = "Custom Data belongs to a different arm. Delete Custom Data and regenerate it for " + ArmName + "."; return false; }
    // Linked raw fixtures mirror the host's separate public stage template.
    if (ini.Get("Modules", "ToolSwapPB").ToString().Trim().Length > 0) ini.Delete(ConfigSection, "Actuators");
    return true;
}
void WriteConfig(MyIni ini) { Me.CustomData = ini.ToString(); }
string ConfigText(MyIni ini) { return ini.ToString(); }
// END STANDALONE CONFIG

static class AutoConfig
{
    public const int Format = 6;
    const string Identity = "AutoArm Config";

    public static bool ReadCurrent(Program p, string text, out MyIni ini, out string why)
    {
        return p.ReadConfig(text, out ini, out why);
    }

    public static MyIni Saved(string text)
    {
        var ini = new MyIni(); MyIniParseResult error;
        if (!ini.TryParse(text ?? "", out error)) { ini.Clear(); ini.EndContent = text ?? ""; }
        return ini;
    }

    public static void EnsureDefaults(Program p)
    {
        if (!p.ConfigDefaultsNeeded) return;
        MyIni ini; string why;
        if (!ReadCurrent(p, "", out ini, out why)) throw new Exception(why);
        if (!new AutoSettings().Read(ini, out why)) throw new Exception(why);
        var defaults = new MyIni(); new AutoSettings().Write(defaults);
        if (p.Tools != null) p.Tools.WriteConfig(defaults);
        var keys = new List<MyIniKey>(); defaults.GetKeys(keys);
        foreach (var key in keys) if (!ini.ContainsKey(key.Section, key.Name)) ini.Set(key.Section, key.Name, defaults.Get(key).ToString());
        p.WriteConfig(ini);
    }

    public const string StageHeader = "Stage | Kind | Translation | Orientation";
    public static string GroupKind(AutoGroup group)
    {
        int piston = 0, rotor = 0, hinge = 0;
        foreach (var member in group.M)
            if (member.B is IMyPistonBase) piston++;
            else if ((member.B.BlockDefinition.SubtypeName ?? "").IndexOf("Hinge", StringComparison.OrdinalIgnoreCase) >= 0) hinge++;
            else rotor++;
        var kinds = new List<string>();
        if (piston > 0) kinds.Add("Piston x" + piston);
        if (rotor > 0) kinds.Add("Rotor x" + rotor);
        if (hinge > 0) kinds.Add("Hinge x" + hinge);
        return string.Join(" + ", kinds);
    }
    public static string Weights(List<AutoGroup> groups)
    {
        var text = new System.Text.StringBuilder(StageHeader); int stage = 0;
        foreach (var g in groups) text.Append('\n').Append((++stage).ToString("00")).Append(" | ").Append(GroupKind(g)).Append(" | ").Append(AutoSettings.Number(g.MoveW)).Append(" | ").Append(AutoSettings.Number(g.TurnW));
        return text.ToString();
    }
    public static bool ReadWeights(string table, List<AutoGroup> groups, out List<double[]> values, out string why)
    {
        values = new List<double[]>(); why = "";
        for (int i = 0; i < groups.Count; i++) values.Add(new double[] { 1, 1 });
        if (string.IsNullOrWhiteSpace(table)) return true;
        var seen = new HashSet<int>(); bool header = false;
        foreach (string raw in table.Split('\n'))
        {
            string line = raw.Trim(); if (line.Length == 0) continue;
            if (!header) { if (line != StageHeader) { why = "Actuators requires: " + StageHeader; return false; } header = true; continue; }
            string[] cells = line.Split('|'); int stage; double m, r;
            if (cells.Length != 4 || !int.TryParse(cells[0].Trim(), out stage) || stage < 1 || stage > groups.Count || !seen.Add(stage) ||
                cells[1].Trim() != GroupKind(groups[stage - 1]) || !ReadDouble(cells[2].Trim(), out m) || !ReadDouble(cells[3].Trim(), out r) || !Valid(m) || !Valid(r))
            { why = "Malformed or nonconforming actuator stage."; return false; }
            values[stage - 1] = new double[] { m, r };
        }
        if (seen.Count != groups.Count) { why = "Actuator stage count does not match configuration."; return false; }
        return true;
    }
    public static bool LoadOrGenerate(Program p, AutoTopology t, string customData,
        out string updated, out bool changed, out string why, bool captureToolWeights = false)
    {
        updated = customData ?? ""; changed = false; why = "";
        MyIni ini;
        if (!ReadCurrent(p, updated, out ini, out why)) throw new Exception(why);
        if (t == null || !captureToolWeights && (t.Groups.Count == 0 || t.Fingerprint.Length == 0))
            return Refuse(t, "Discover a valid arm before loading actuator settings.", out why);
        var settings = new AutoSettings();
        if (!settings.Read(ini, out why)) return Refuse(t, why, out why);
        var saved = Saved(p.Storage);
        if (saved.ContainsSection(Identity))
        {
            if (saved.Get(Identity, "Arm").ToString() != t.Arm || saved.Get(Identity, "PB").ToInt64() != p.Me.EntityId || saved.Get(Identity, "Format").ToInt32() != 1)
                return Refuse(t, "Actuator identity cache belongs to a different arm/PB or format.", out why);
            var names = new HashSet<string>(); var signatures = new HashSet<string>();
            foreach (string row in saved.Get(Identity, "Rows").ToString().Split('\n'))
            {
                if (row.Length == 0) continue;
                string[] cells = row.Split('|');
                if (cells.Length != 2 || cells[0].Trim().Length == 0 || !names.Add(cells[0]) || !Signature(cells[1]) || !signatures.Add(cells[1]))
                    return Refuse(t, "Actuator identity cache is corrupt.", out why);
            }
        }
        else if (saved.Get("AutoArm Tool Weights", "Rows").ToString().Length > 0 || !string.IsNullOrWhiteSpace(ini.Get(ConfigSection, "Actuators").ToString()))
            return Refuse(t, "Actuator identity cache is missing.", out why);
        List<double[]> values = null;
        bool explicitWeights = ini.ContainsKey(ConfigSection, "Actuators");
        if (explicitWeights && !ReadWeights(ini.Get(ConfigSection, "Actuators").ToString(), t.Groups, out values, out why)) return Refuse(t, why, out why);
        p.RememberToolWeights(false);
        if (values != null)
            for (int i = 0; i < t.Groups.Count; i++) p.ToolWeights[p.ToolGroupKey(t.Groups[i])] = values[i];
        settings.Write(ini); ini.DeleteSection("Instructions");
        if (p.Tools != null) p.Tools.WriteConfig(ini);
        if (!captureToolWeights)
        {
            foreach (var g in t.Groups)
            {
                double[] old;
                if (p.ToolWeights.TryGetValue(p.ToolGroupKey(g), out old)) { g.MoveW = old[0]; g.TurnW = old[1]; }
                else { g.MoveW = g.TurnW = 1; }
            }
            UpdateGroups(p);
            p.ApplySettings(settings); t.Ready = false; t.ZeroOwned(); why = "Configuration loaded; arm remains Off until On.";
        }
        else p.WriteToolWeights();
        if (explicitWeights || p.ConfigText(ini).Length > 0) ini.Set(ConfigSection, "Actuators", Weights(t.Groups));
        updated = p.ConfigText(ini); changed = updated != customData;
        p.WriteConfig(ini); return true;
    }

    // Keep startup stop identities and durable signed preferences together.
    public static void UpdateGroups(Program p)
    {
        var saved = Saved(p.Storage); var rows = new System.Text.StringBuilder(); int stage = 0;
        foreach (var g in p.Topology.Groups)
        { if (stage > 0) rows.Append('\n'); rows.Append((++stage).ToString("00")).Append('|').Append(g.Sig); }
        saved.Set(Identity, "Format", 1); saved.Set(Identity, "Arm", p.Topology.Arm); saved.Set(Identity, "PB", p.Me.EntityId); saved.Set(Identity, "Rows", rows.ToString());
        p.Storage = saved.ToString(); p.RememberToolWeights();
    }
    static bool Signature(string sig)
    {
        string[] parts = sig.Split(':');
        if (parts.Length < 2 || (parts[0] != "R" && parts[0] != "P")) return false;
        var ids = new HashSet<long>();
        for (int i = 1; i < parts.Length; i++)
        {
            string part = parts[i]; int at = part.IndexOf('@'), to = part.IndexOf('>'); long id, a, b;
            if (at < 2 || to <= at + 1 || (part[at - 1] != '+' && part[at - 1] != '-') ||
                !long.TryParse(part.Substring(0, at - 1), out id) || !long.TryParse(part.Substring(at + 1, to - at - 1), out a) ||
                !long.TryParse(part.Substring(to + 1), out b) || id == 0 || a == 0 || b == 0 || a == b || !ids.Add(id)) return false;
        }
        return true;
    }
    static bool Valid(double v) => !double.IsNaN(v) && !double.IsInfinity(v) && v >= 0 && v <= 10;
    static bool Refuse(AutoTopology t, string text, out string why)
    { why = text + " Custom Data and saved state were left untouched."; if (t != null) { t.Ready = false; t.ZeroOwned(); } return false; }
}
