// Optional safety overlay. Geometry stays in the shared Collision PB.
sealed class SafetyOverlay
{
    readonly Program P;
    public readonly ModuleLink Link;
    public double Age = 1e6, SendAge, Scale;
    public bool Limited;
    public bool Blocked = true;
    public double Fraction = 1;
    string Stamp = "", Reason = "Waiting for collision service.";
    double[] Bias = new double[0];
    readonly List<double[]> Rows = new List<double[]>();
    readonly Dictionary<long, double> Sent = new Dictionary<long, double>();
    double Clock;
    int ConstraintLimit = 8;
    public void Reset() { Age = 1e6; Sent.Clear(); SendAge = 1; }
    public SafetyOverlay(Program p, ModuleLink link) { P = p; Link = link; }
    double Cap(int i) { return P.Topology.Groups[i].Rotary ? (P.GoingHome ? P.HomeJointRpm : P.JointRpm) * Math.PI / 30 : P.GoingHome ? P.HomePistonMps : P.PistonMps; }
    string Identity() { return P.Topology.Fingerprint + "/" + P.Topology.Head.CubeGrid.EntityId + "/" + P.Tools.Contacts; }
    public void Tick(double elapsed, long revision)
    {
        if (Finite(elapsed) && elapsed > 0) { Clock += elapsed; Age += elapsed; SendAge += elapsed; }
        if (!P.Topology.Ready || P.Topology.Groups.Count == 0) { Age = 1e6; Stamp = ""; return; }
        string stamp = Identity();
        if (stamp != Stamp) { Stamp = stamp; Age = 1e6; Sent.Clear(); SendAge = 1; }
        if (SendAge < .1) return;
        SendAge = 0;
        var ini = Link.Packet("CHECK"); ini.Set(LinkSection, "Service", 1); ini.Set(LinkSection, "Generation", revision);
        ini.Set(LinkSection, "ConstraintLimit", 16);
        ini.Set(LinkSection, "Fingerprint", Stamp); ini.Set(LinkSection, "Groups", P.Topology.Fingerprint);
        ini.Set(LinkSection, "Hull", P.Topology.Base.CubeGrid.EntityId); ini.Set(LinkSection, "ToolGrid", P.Topology.Head.CubeGrid.EntityId);
        ini.Set(LinkSection, "Contacts", P.Tools.Contacts);
        var focus = P.Tools.Focus;
        var cell = focus == null ? P.Topology.End.Position : focus.Position;
        ini.Set(LinkSection, "Socket", focus != null);
        ini.Set(LinkSection, "TipGrid", P.Topology.End.CubeGrid.EntityId);
        ini.Set(LinkSection, "TipCell", cell.X + " " + cell.Y + " " + cell.Z);
        WriteVector(ini, "TipUp", P.Topology.EndPose.Up);
        var caps = new System.Text.StringBuilder(); var rates = new System.Text.StringBuilder();
        for (int i = 0; i < P.Topology.Groups.Count; i++)
        {
            if (i > 0) { caps.Append(' '); rates.Append(' '); }
            caps.Append(AutoSettings.Number(Cap(i)));
            rates.Append(AutoSettings.Number(P.LastQdot == null ? 0 : P.LastQdot[i]));
        }
        ini.Set(LinkSection, "Caps", caps.ToString()); ini.Set(LinkSection, "Rates", rates.ToString());
        if (Sent.Count >= 16) Sent.Clear(); Sent[Link.Sequence] = Clock;
        try { Link.Send(ini); } catch { Age = 1e6; Reason = "Collision service unavailable."; }
    }
    public void Read(MyIni ini)
    {
        long ack = ini.Get(LinkSection, "Ack").ToInt64(); double time;
        if (!Sent.TryGetValue(ack, out time) || Clock - time > .25 || ini.Get(LinkSection, "Fingerprint").ToString() != Stamp) return;
        int n = P.Topology.Groups.Count; double scale; bool blocked;
        if (!ini.Get(LinkSection, "Hold").TryGetBoolean(out blocked)) throw new Exception("Missing collision permission.");
        if (!ini.Get(LinkSection, "Scale").TryGetDouble(out scale) || !Finite(scale) || scale < 0 || scale > 1) throw new Exception("Invalid collision speed scale.");
        var bias = Numbers(ini.Get(LinkSection, "Bias").ToString(), n);
        for (int i = 0; i < n; i++)
            if (Math.Abs(bias[i]) > Cap(i) * .25 + 1e-9) throw new Exception("Collision preference exceeds cap.");
        var rows = new List<double[]>();
        int limit = 8;
        if (ini.ContainsKey(LinkSection, "ConstraintLimit") && (!ini.Get(LinkSection, "ConstraintLimit").TryGetInt32(out limit) || limit != 8 && limit != 16)) throw new Exception("Invalid collision constraint capacity.");
        foreach (string text in ini.Get(LinkSection, "Rows").ToString().Split('\n'))
        {
            if (text.Length == 0) continue;
            if (rows.Count >= limit) throw new Exception("Too many collision constraints.");
            var row = Numbers(text, n + 1);
            foreach (double value in row) if (Math.Abs(value) > 1e6) throw new Exception("Collision constraint out of range.");
            rows.Add(row);
        }
        // Commit only a complete, fresh, matching reply. Replay cannot extend expiry.
        Sent.Remove(ack); Bias = bias; Rows.Clear(); Rows.AddRange(rows); Scale = scale; Blocked = blocked;
        ConstraintLimit = limit;
        Reason = ini.Get(LinkSection, "Reason").ToString(); Age = Clock - time;
    }
    static double[] Numbers(string text, int count)
    {
        string[] parts = text.Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
        if (parts.Length != count) throw new Exception("Collision row size mismatch.");
        var values = new double[count];
        for (int i = 0; i < count; i++) if (!ReadDouble(parts[i], out values[i])) throw new Exception("Invalid collision number.");
        return values;
    }
    public bool Ready { get { return Age <= .25 && P.Topology.Ready && Stamp == Identity(); } }
    public void WriteStatus(MyIni ini)
    {
        ini.Set(LinkSection, "SafetyReady", Ready); ini.Set(LinkSection, "SafetyHold", !Ready || Blocked);
        ini.Set(LinkSection, "SafetyAge", Age); ini.Set(LinkSection, "SafetyScale", Scale);
        ini.Set(LinkSection, "SafetyLimited", Limited); ini.Set(LinkSection, "SafetyFraction", Fraction);
        ini.Set(LinkSection, "SafetyConstraints", Rows.Count); ini.Set(LinkSection, "SafetyConstraintLimit", ConstraintLimit);
        ini.Set(LinkSection, "SafetyReason", Ready ? Reason : "Waiting for fresh collision guidance.");
    }
    public double Preferred(int i) { return Ready && i < Bias.Length ? Bias[i] : 0; }
    public void Hold()
    {
        P.Topology.ZeroOwned(); if (P.LastQdot != null) Array.Clear(P.LastQdot, 0, P.LastQdot.Length);
        P.LastAchievedLinear = P.LastAchievedAngular = Vector3D.Zero;
        P.Say("COLLISION hold | " + (Ready ? Reason : "Waiting for fresh collision guidance."));
    }
    public void Limit(double[] rates)
    {
        Limited = false; Fraction = 1;
        if (!Ready || Blocked) { Array.Clear(rates, 0, rates.Length); Limited = true; Fraction = 0; return; }
        double alpha = P.GoingHome ? Scale : 1, minimum = 0;
        foreach (var row in Rows)
        {
            double dot = 0; for (int i = 0; i < rates.Length; i++) dot += row[i] * rates[i];
            double lower = row[rates.Length];
            if (dot > 1e-10) minimum = Math.Max(minimum, lower / dot);
            else if (dot < -1e-10) alpha = Math.Min(alpha, lower / dot);
            else if (lower > 1e-8) minimum = 2;
        }
        if (minimum > alpha + 1e-8 || minimum > 1 || alpha < 0) alpha = 0;
        alpha = Clamp(alpha, 0, 1);
        Fraction = alpha;
        if (alpha < 1) { Limited = true; for (int i = 0; i < rates.Length; i++) rates[i] *= alpha; }
    }
}
