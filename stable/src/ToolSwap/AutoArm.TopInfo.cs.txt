class ToolTopInfo
{
    public IMyCubeBlock Marker;
    public IMyMotorRotor Live;
    public long EntityId;
    public string Subtype;
    public MatrixD Local;
    public List<IMyTerminalBlock> Inventory;
    public bool Closed => Marker.Closed || (Live != null && Live.Closed);
    public IMyCubeGrid CubeGrid => Live == null ? Marker.CubeGrid : Live.CubeGrid;
    public MatrixD WorldMatrix => Live == null ? Local * Marker.WorldMatrix : Live.WorldMatrix;
    public Vector3I Position => Live == null ? CubeGrid.WorldToGridInteger(WorldMatrix.Translation) : Live.Position;
    public IMyMechanicalConnectionBlock Base
    {
        get
        {
            if (Live != null) return Live.Base;
            foreach (var b in Inventory)
            { var m = b as IMyMechanicalConnectionBlock; if (m != null && m.Top != null && (m.Top.EntityId == EntityId || EntityId == 0 && m.Top.CubeGrid.EntityId == CubeGrid.EntityId && m.Top.Position == Position)) return m; }
            return null;
        }
    }
    public static ToolTopInfo Capture(IMyCubeBlock marker, IMyMotorRotor top, List<IMyTerminalBlock> inventory)
    {
        var pose = top.WorldMatrix; pose.Translation = top.CubeGrid.GridIntegerToWorld(top.Position);
        return new ToolTopInfo { Marker = marker, Live = top, EntityId = top.EntityId, Subtype = top.BlockDefinition.SubtypeName, Local = pose * MatrixD.Invert(marker.WorldMatrix), Inventory = inventory };
    }
}
