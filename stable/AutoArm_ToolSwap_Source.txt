public Program()
{
    Runtime.UpdateFrequency = UpdateFrequency.Update10;
    IGC.UnicastListener.SetMessageCallback(WireTag);
    try
    {
        if (!Saved.TryParse(Storage)) { Saved.Clear(); Saved.EndContent = Storage; }
        if (Saved.Get("AutoArm Host", "PB").ToInt64() == Me.EntityId) { InputClock = Saved.Get("AutoArm Host", "InputClock").ToInt64(); InputOwner = Saved.Get("AutoArm Host", "InputOwner").ToInt64(); }
        LoadHost();
        string prior = Saved.Get("AutoArm Host", "Selected").ToString();
        if (Saved.Get("AutoArm Host", "PB").ToInt64() == Me.EntityId && Arms.ContainsKey(prior)) Selected = prior;
        if (Order.Count > 0) SelectArm(Selected.Length == 0 ? Order[0] : Selected, InputOwner == 0);
        Save();
    }
    catch (Exception e) { StopAll(e.Message); }
    Display();
}
public void Save()
{
    if (Order.Count == 0 && Configuration.Get(GlobalSection, "Format").ToInt32() != HostFormat) return;
    bool changed = Saved.Get("AutoArm Host", "PB").ToInt64() != Me.EntityId || Saved.Get("AutoArm Host", "Selected").ToString() != Selected || Saved.Get("AutoArm Host", "InputClock").ToInt64() != InputClock || Saved.Get("AutoArm Host", "InputOwner").ToInt64() != InputOwner;
    if (changed) { Saved.Set("AutoArm Host", "Format", 1); Saved.Set("AutoArm Host", "PB", Me.EntityId); Saved.Set("AutoArm Host", "Selected", Selected); Saved.Set("AutoArm Host", "InputClock", InputClock); Saved.Set("AutoArm Host", "InputOwner", InputOwner); }
    foreach (var pair in Arms) if (pair.Value.Storage != pair.Value.SavedData) { Saved.Set("State " + pair.Key, "Data", pair.Value.Storage); pair.Value.SavedData = pair.Value.Storage; changed = true; }
    if (!changed) return;
    string text = Saved.ToString(); if (text != Storage) Storage = text;
}

void LoadHost()
{
    var ini = new MyIni();
    if (string.IsNullOrWhiteSpace(Me.CustomData))
    {
        ini.Set(GlobalSection, "Format", HostFormat); ini.Set(GlobalSection, "SelectedArm", DefaultArm);
        var bases = new List<IMyMechanicalConnectionBlock>(); GridTerminalSystem.GetBlocksOfType(bases); var detected = new List<string>();
        foreach (var block in bases)
        {
            int split = block.CustomName.IndexOf(" - Base", StringComparison.Ordinal);
            if (split > 0 && !block.Closed && block.IsSameConstructAs(Me) && !detected.Contains(block.CustomName.Substring(0, split))) detected.Add(block.CustomName.Substring(0, split));
        }
        if (detected.Count == 0) detected.Add(DefaultArm); detected.Sort(StringComparer.Ordinal);
        {
            var blocks = new List<IMyTerminalBlock>(); GridTerminalSystem.GetBlocks(blocks);
            detected.RemoveAll(name => !blocks.Any(block => block.CustomName.StartsWith(name + " - Head ", StringComparison.Ordinal) && block.CustomName.Substring((name + " - Head ").Length).IndexOf(" - ", StringComparison.Ordinal) > 0));
            if (detected.Count == 0) detected.Add(DefaultArm);
        }
        foreach (string name in detected) ini.Set(name, "Tools.Enabled", "true");
        Me.CustomData = ini.ToString();
    }
    if (!ini.TryParse(Me.CustomData) || ini.Get(GlobalSection, "Format").ToInt32() != HostFormat)
        throw new Exception("Custom Data requires [global] Format=" + HostFormat + ". Delete Custom Data.");
    var names = new List<string>(); ini.GetSections(names); names.Remove(GlobalSection);
    if (names.Count > 8) throw new Exception("Configure at most 8 named arm sections.");
    foreach (string name in names) if (name.Trim() != name || name.Length == 0 || name.IndexOfAny(new char[] { ',', '|', '(', ')', ';', '\r', '\n' }) >= 0 || name.Equals(GlobalSection, StringComparison.OrdinalIgnoreCase)) throw new Exception("Invalid arm section name.");
    var removed = new List<string>(); foreach (string name in Order) if (!names.Contains(name)) removed.Add(name);
    foreach (string name in removed) { Arms[name].Stop("Arm removed from configuration."); Saved.Set("State " + name, "Data", Arms[name].Storage); Arms.Remove(name); }
    Configuration = ini; Order.Clear(); Order.AddRange(names);
    AutoLinks();
    foreach (string name in names) if (!Arms.ContainsKey(name))
    {
        string state = Saved.Get("AutoArm Host", "PB").ToInt64() == Me.EntityId ? Saved.Get("State " + name, "Data").ToString() : "";
        Arms.Add(name, new ArmCore(this, name, ConfigFor(name), state));
    }
    if (!Arms.ContainsKey(Selected))
    { string configured = ini.Get(GlobalSection, "SelectedArm").ToString(); Selected = Arms.ContainsKey(configured) ? configured : names.Count == 0 ? "" : names[0]; }
    HostError = "";
}
string Effective(string arm, string key)
{ return Configuration.Get(Configuration.ContainsKey(arm, key) ? arm : GlobalSection, key).ToString().Replace("{arm}", arm); }
void AutoLinks()
{
    var peers = new List<IMyProgrammableBlock>(); GridTerminalSystem.GetBlocksOfType(peers);
    foreach (string name in Order)
    {
        string key = "Link.ArmPB";
        if (Effective(name, key).Length > 0) continue;
        IMyProgrammableBlock found = null;
        foreach (var peer in peers)
        {
            if (peer == Me || peer.Closed || !peer.IsFunctional || !peer.IsSameConstructAs(Me)) continue;
            var data = new MyIni();
            if (!data.TryParse(peer.CustomData) || data.Get(GlobalSection, "Format").ToInt32() != (7) || !data.ContainsSection(name)) continue;
            if (found != null) throw new Exception("Multiple module PBs for " + name + ". Set " + key + ".");
            found = peer;
        }
        if (found != null) Configuration.Set(name, key, "@" + found.EntityId);
    }
}
MyIni ConfigFor(string name)
{
    var ini = new MyIni();
    var keys = new List<MyIniKey>(); Configuration.GetKeys(GlobalSection, keys);
    var own = new List<MyIniKey>(); Configuration.GetKeys(name, own); keys.AddRange(own);
    var seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
    foreach (var item in keys)
    {
        string key = item.Name; if (!seen.Add(key) || key.Equals("Format", StringComparison.OrdinalIgnoreCase) || key.Equals("SelectedArm", StringComparison.OrdinalIgnoreCase) || key.Equals("Structure", StringComparison.OrdinalIgnoreCase) || key.Equals("Actuators", StringComparison.OrdinalIgnoreCase)) continue;
        string section = "Config", field = key; int dot = key.IndexOf('.');
        if (dot > 0) { section = key.Substring(0, dot); field = key.Substring(dot + 1); }
        else if (key.Equals("ToolSwapPB", StringComparison.OrdinalIgnoreCase) || key.Equals("Peers", StringComparison.OrdinalIgnoreCase)) section = "Modules";
        ini.Set(section, field, Effective(name, key));
    }
    {
        if (!ini.ContainsKey("Link", "ArmPB")) ini.Set("Link", "ArmPB", name + " - Arm PB");
        if (!ini.ContainsKey("Tools", "Enabled")) ini.Set("Tools", "Enabled", true);
    }
    return ini;
}

void PublishConfig()
{
    // Generated values only fill missing keys. Preserve edits made while running.
    var live = new MyIni(); if (!live.TryParse(Me.CustomData) || live.Get(GlobalSection, "Format").ToInt32() != HostFormat) return;
    var keys = new List<MyIniKey>(); Configuration.GetKeys(keys);
    foreach (var key in keys) if (!live.ContainsKey(key.Section, key.Name)) live.Set(key.Section, key.Name, Configuration.Get(key).ToString());
    string text = live.ToString(); if (text != Me.CustomData) Me.CustomData = text;
}
void SelectArm(string name, bool claim = true, bool pair = false)
{
    if (!Arms.ContainsKey(name)) throw new Exception("Unknown arm: " + name);
    Selected = name;
    Save();
}


void Command(string raw)
{
    string text = raw.Trim(); if (text.Length == 0) return;
    HostError = "";
    if (text.StartsWith("Select ", StringComparison.OrdinalIgnoreCase)) { SelectArm(text.Substring(7).Trim(), true, true); return; }
    if (text.StartsWith("Release ", StringComparison.OrdinalIgnoreCase)) { {throw new Exception("Release is an Arm PB command."); } }
    if (text.Equals("StopAll", StringComparison.OrdinalIgnoreCase)) { StopAll("StopAll command."); return; }
    int end = text.IndexOfAny(new char[] { ' ', '(', ',' }); string first = (end < 0 ? text : text.Substring(0, end)).ToLowerInvariant();
    if (first == "on" || first == "reload" || first == "rescan")
    { try { LoadHost(); } catch (Exception e) { StopAll(e.Message); throw; } }
    string name = Selected, command = text; int open = text.IndexOf('(');
    if (open > 0)
    {
        if (!text.EndsWith(")", StringComparison.Ordinal)) throw new Exception("Command needs closing parenthesis.");
        string verb = text.Substring(0, open).Trim(); var values = new List<string>();
        foreach (string value in text.Substring(open + 1, text.Length - open - 2).Split(',')) if (value.Trim().Length > 0) values.Add(value.Trim());
        if (values.Count > 0 && Arms.ContainsKey(values[values.Count - 1])) { name = values[values.Count - 1]; values.RemoveAt(values.Count - 1); }
        else if (values.Count > Arity(verb)) throw new Exception("Unknown arm argument.");
        command = verb + (values.Count == 0 ? "" : " " + string.Join(" ", values));
    }
    else
    {
        int comma = text.LastIndexOf(',');
        if (comma >= 0) { name = text.Substring(comma + 1).Trim(); command = text.Substring(0, comma).Trim(); }
    }
    ArmCore target; if (!Arms.TryGetValue(name, out target)) throw new Exception("Unknown arm: " + name);
    string action = command.Split(' ')[0].ToLowerInvariant();
    if (action == "select") { int space = command.IndexOf(' '); string mode = space < 0 ? "" : command.Substring(space + 1).Trim().ToUpperInvariant(); if (mode.Length > 0 && mode != "HEAD" && mode != "HRZ" && mode != "VRT") throw new Exception("MovementMode expects HEAD, HRZ or VRT."); SelectArm(name, true, true); if (mode.Length > 0) target.Update("Mode " + mode, UpdateType.Terminal, 0); return; }
    if (action == "release") { {throw new Exception("Release is an Arm PB command."); } }
    if (action == "on" || action == "reload" || action == "rescan")
    { target.Config = ConfigFor(name); target.ConfigChanged = false; }
    target.Update(command, UpdateType.Terminal, 0);
    CaptureDefaults(target);
}
static int Arity(string verb)
{
    switch (verb.ToLowerInvariant())
    {
        case "on": case "off": case "stop": case "hold": case "check": case "info": case "joints": case "reload": case "rescan": case "sethome": case "gohome": case "toolscan": case "toolinfo": case "swapcancel": case "version": case "status": case "bindings": case "homeinfo": case "forward": case "back": case "left": case "right": case "up": case "down": case "yawleft": case "yawright": case "pitchup": case "pitchdown": case "rollleft": case "rollright": return 0;
        case "moveto": return 3;
        case "release": return 0;
        case "orientto": return 6;
        case "movetoposori": return 9;
        case "homespeed": case "step": return 2;
        default: return 1;
    }
}








void CaptureDefaults(ArmCore core)
{
    if (!core.ConfigChanged) return;
    core.ConfigChanged = false;
    var ini = core.Config;
    var keys = new List<MyIniKey>(); ini.GetKeys(keys);
    foreach (var key in keys)
    {
        if (key.Section == "AutoArm" || key.Name == "Actuators") continue;
        string field = key.Section == "Config" || key.Section == "Modules" ? key.Name : key.Section + "." + key.Name;
        if (!Configuration.ContainsKey(core.Name, field) && !Configuration.ContainsKey(GlobalSection, field))
            Configuration.Set(core.SharedSetting(field) ? GlobalSection : core.Name, field, ini.Get(key).ToString());
    }
    PublishConfig();
}
public void Main(string argument, UpdateType source)
{
    double elapsed = Runtime.TimeSinceLastRun.TotalSeconds;
    try
    {
        if (!string.IsNullOrWhiteSpace(argument) && !((source & UpdateType.IGC) != 0 && argument == WireTag))
            foreach (string raw in argument.Split(new char[] { ';', '\n' }, StringSplitOptions.RemoveEmptyEntries)) Command(raw);
    }
    catch (Exception e) { HostError = e.Message; }
    try
    {
        int read = 0;
        while (IGC.UnicastListener.HasPendingMessage && read++ < 16)
        {
            if (Runtime.MaxInstructionCount > 0 && Runtime.CurrentInstructionCount >= Runtime.MaxInstructionCount * .4) break;
            var message = IGC.UnicastListener.AcceptMessage(); var data = message.Data as string; var ini = new MyIni(); ArmCore core;
            if (message.Tag != WireTag || data == null || data.Length > 16000 || !ini.TryParse(data) || !Arms.TryGetValue(ini.Get("Link", "Arm").ToString(), out core)) continue;
            if (core.Inbox.Count < 32) core.Inbox.Enqueue(message);
        }
        int count = Order.Count;
        for (int i = 0; i < count; i++)
        {
            if (Runtime.MaxInstructionCount > 0 && Runtime.CurrentInstructionCount >= Runtime.MaxInstructionCount * .7) { StopAll("Shared PB instruction budget exhausted."); break; }
            var core = Arms[Order[(Next + i) % count]]; core.Update("", source, elapsed); CaptureDefaults(core);
        }
        if (count > 0) Next = (Next + 1) % count;
        Save();
    }
    catch (Exception e) { StopAll(e.Message); }
    finally
    {
        UpdateFrequency frequency = UpdateFrequency.Update10;
        foreach (var core in Arms.Values) if ((core.Frequency & UpdateFrequency.Update1) != 0) frequency = UpdateFrequency.Update1;
        Runtime.UpdateFrequency = frequency; Display();
    }
}
void StopAll(string why)
{ HostError = why; foreach (var core in Arms.Values) core.Stop(why); }
void Display()
{
    var lines = new List<string>(); lines.Add("AutoArm " + ControllerBuildVersion + " " + ("ToolSwap") + " | Selected: " + Selected);
    if (HostError.Length > 0) lines.Add(HostError);
    foreach (string name in Order) lines.Add(name + ": " + Arms[name].Summary);
    ArmCore current; if (Arms.TryGetValue(Selected, out current)) lines.Add(current.Report);
    Echo(string.Join("\n", lines));
}

void ConfigureLink()
{
    MyIni ini; string why; if (!AutoConfig.ReadCurrent(this, CustomData, out ini, out why)) throw new Exception(why);
    string name = ini.Get(LinkSection, "ArmPB").ToString(); var prior = Link.Peer;
    Link.Resolve(name);
    if (prior != null && prior != Link.Peer)
    {
        // Epochs and request acknowledgements belong to one owner PB.
        Link = new ModuleLink(this); Link.Resolve(name);
        Acquired = Initialised = RemoteReady = RemoteSettled = RemotePilot = Released = ResumeRequested = false;
        Ack = HelloRequest = StopRequest = BuildRequest = MoveRequest = ResumeRequest = TransitionRequest = SentAnchor = ReportedPath = 0;
        CompletedWaypoints = ResumeRetries = 0; ResumeSeconds = Clock = Heartbeat = 0;
        RemoteError = SentPath = TransitionKind = ""; Topology.Head = null; Topology.Groups.Clear();
    }
}
public void Main(string argument, UpdateType source)
{
    double elapsed = Elapsed;
    bool timed = (source & (UpdateType.Update1 | UpdateType.Update10 | UpdateType.Update100 | UpdateType.Once)) != 0;
    if (Finite(elapsed) && elapsed > 0) { Link.Age += elapsed; Clock += elapsed; Heartbeat += elapsed; if (ResumeRequested) ResumeSeconds += elapsed; }
    try
    {
        if (Link.Peer == null || argument.Trim().Equals("Reload", StringComparison.OrdinalIgnoreCase) || argument.Trim().Equals("On", StringComparison.OrdinalIgnoreCase)) ConfigureLink();
        Receive();
        if (!string.IsNullOrWhiteSpace(argument) && argument != LinkTag)
        {
            string command = argument.Trim(); string verb = command.Split(' ')[0].ToLowerInvariant();
            if (verb == "toolinfo" || verb == "info" || verb == "status") Tools.Handle("ToolInfo");
            else if (verb == "version") Say("AutoArm ToolSwap " + ControllerBuildVersion);
            else if (verb == "on" || verb == "reload" || verb == "tool" || verb == "toolswap" || verb == "park" || verb == "toolscan" || verb == "stop" || verb == "off" || verb == "swapcancel") Queue(command);
            else throw new Exception("Unknown ToolSwap command.");
        }
        if (PendingCommand.Length > 0)
        {
            if (!Acquired)
            {
                if (HelloRequest == 0)
                {
                    ConfigureLink(); var hello = Link.Packet("HELLO"); hello.Set(LinkSection, "Acquire", true);
                    string helloVerb = PendingCommand.Split(' ')[0].ToLowerInvariant(); hello.Set(LinkSection, "Changing", helloVerb == "tool" || helloVerb == "toolswap" || helloVerb == "park");
                    HelloRequest = Link.Sequence; Link.Send(hello);
                }
                return;
            }
            string command = PendingCommand, verb = command.Split(' ')[0].ToLowerInvariant();
            if (verb == "on" || verb == "reload")
            {
                string why; RefreshConfig(); if (!Tools.Load(out why) || !Tools.NamesForArm(out why)) throw new Exception(why);
                if (!Tools.Configured) throw new Exception("Tools.Enabled must be true in the ToolSwap PB.");
                MyIni ini; if (!AutoConfig.ReadCurrent(this, CustomData, out ini, out why)) throw new Exception(why);
                Tools.WriteConfig(ini); WriteConfig(ini); Initialised = true;
                if (verb == "on") Tools.StartOn();
            }
            else if (verb == "stop" || verb == "off" || verb == "swapcancel") { Tools.Cancel("Stopped; inspect, then On."); }
            else
            {
                if (!Initialised) throw new Exception("Run On to configure ToolSwap first.");
                Tools.Handle(command);
            }
            PendingCommand = ""; Released = false; Clock = 0;
        }
        if (Link.RemoteEpoch != 0 && Link.Age > .5)
        {
            Tools.Cancel("Arm PB communication timeout."); Acquired = false; PendingCommand = ""; return;
        }
        if (ResumeRequested && Acquired && ResumeSeconds >= .25)
        {
            if (++ResumeRetries > 3 || !Permit()) Tools.Cancel("Arm did not confirm tool handoff; On.");
            else EnableControl(true);
        }
        if (Tools.Busy && Acquired)
        {
            if (RemoteError.Length > 0) throw new Exception(RemoteError);
            if (Tools.Automatic && RemotePilot) { Tools.Cancel("Pilot canceled swap."); }
            else if (timed && Clock > .000001)
            { double dt = Clock; Clock = 0; if (dt > .5) throw new Exception("Tool update gap."); Tools.Tick(Math.Min(dt, .1), dt); }
        }
        if (!Tools.Busy && Acquired && !Released)
        {
            if (!Initialised || Tools.Recovery || Tools.Endpoint == null)
            { var abort = Link.Packet("ABORT"); abort.Set(LinkSection, "Reason", CancelReason); abort.Set(LinkSection, "ExplicitStop", ExplicitStop); Link.Send(abort); }
            else { var release = EquipmentPacket("RELEASE"); release.Set(LinkSection, "Permit", Permit()); Link.Send(release); }
            Released = true;
        }
        if (Link.RemoteEpoch != 0 && Heartbeat >= .1)
        { Heartbeat = 0; var ping = Link.Packet("PING"); ping.Set(LinkSection, "Permit", Permit()); Link.Send(ping); }
    }
    catch (RemoteWait) { }
    catch (Exception e)
    {
        PendingCommand = ""; Say("STOPPED: " + e.Message);
        try { Tools.Cancel(e.Message); } catch (RemoteWait) { PendingCommand = "Stop"; }
    }
    finally { Echo(DisplayText); }
}
void Queue(string command)
{
    string verb = command.Split(' ')[0].ToLowerInvariant();
    if (Tools.Busy && Tools.Automatic && verb != "stop" && verb != "off" && verb != "swapcancel") { Say("ToolSwap busy; Stop first."); return; }
    if (Released) { Acquired = false; HelloRequest = 0; }
    ExplicitStop = verb == "stop" || verb == "off" || verb == "swapcancel";
    PendingCommand = command; Released = false;
    if (!Acquired) HelloRequest = 0;
}
void Receive()
{
    int count = 0;
    while (Inbox.Count > 0 && count++ < 16)
    {
        if (Runtime.MaxInstructionCount > 0 && Runtime.CurrentInstructionCount >= Runtime.MaxInstructionCount * .5) break;
        var message = Inbox.Dequeue(); MyIni ini;
        if (!Link.Read(message, out ini)) continue;
        long epoch = ini.Get(LinkSection, "Epoch").ToInt64(), sequence = ini.Get(LinkSection, "Sequence").ToInt64();
        string op = ini.Get(LinkSection, "Operation").ToString();
        if (epoch < Link.RemoteEpoch || epoch == Link.RemoteEpoch && sequence <= Link.Received) continue;
        if (op == "COMMAND")
        {
            string incoming = ini.Get(LinkSection, "Command").ToString();
            if (incoming.Equals("ToolInfo", StringComparison.OrdinalIgnoreCase) && epoch == Link.RemoteEpoch && ini.Get(LinkSection, "Generation").ToInt64() == Link.Generation)
            { Link.Received = sequence; Link.Age = 0; Tools.Handle("ToolInfo"); var info = Link.Packet("INFO"); info.Set(LinkSection, "Text", DisplayText); Link.Send(info); continue; }
            if (epoch != Link.RemoteEpoch) { Link.Received = 0; Acquired = false; }
            Link.RemoteEpoch = epoch; Link.Generation = ini.Get(LinkSection, "Generation").ToInt64(); Link.Received = sequence; Link.Age = 0;
            RemoteStopped = ini.Get(LinkSection, "Stopped").ToBoolean(); MoveRequest = StopRequest = BuildRequest = HelloRequest = 0; RemoteError = "";
            TransitionRequest = 0; TransitionKind = ""; ResumeRequested = false;
            string command = ini.Get(LinkSection, "Command").ToString();
            if (command.Equals("ToolInfo", StringComparison.OrdinalIgnoreCase)) { Tools.Handle("ToolInfo"); var info = Link.Packet("INFO"); info.Set(LinkSection, "Text", DisplayText); Link.Send(info); continue; }
            if (command.Equals("Stop", StringComparison.OrdinalIgnoreCase))
            { ExplicitStop = !ini.Get(LinkSection, "RecoverBare").ToBoolean(); PendingCommand = ""; Acquired = false; Tools.Cancel("Arm stopped; inspect, then On."); continue; }
            if (Tools.Busy) Tools.Cancel("Arm requested new setup; inspect equipment.");
            Queue(command); Acquired = false; continue;
        }
        if (op != "STATE" || ini.Get(LinkSection, "Remote").ToInt64() != Link.Epoch) continue;
        if (Link.RemoteEpoch != 0 && epoch == Link.RemoteEpoch && ini.Get(LinkSection, "Generation").ToInt64() != Link.Generation) continue;
        Link.RemoteEpoch = epoch; Link.Generation = ini.Get(LinkSection, "Generation").ToInt64(); Link.Received = sequence; Link.Age = 0;
        Ack = ini.Get(LinkSection, "Ack").ToInt64(); RemoteReady = ini.Get(LinkSection, "Ready").ToBoolean(); RemoteStopped = ini.Get(LinkSection, "Stopped").ToBoolean();
        RemoteSettled = ini.Get(LinkSection, "Settled").ToBoolean(); RemotePilot = ini.Get(LinkSection, "Pilot").ToBoolean(); RemoteError = ini.Get(LinkSection, "Error").ToString();

class ToolRegionScan
{
    ArmCore P; IMyCubeBlock Marker; IMyCubeGrid Grid;
    long GridId; int Limit; string Axis, Fault;
    Queue<Vector3I> Pending = new Queue<Vector3I>();

    public HashSet<Vector3I> Hidden = new HashSet<Vector3I>();

    public ToolRegionScan(ArmCore p, IMyCubeBlock marker, int cellLimit, string mergeAxis)
    {
        P = p; Marker = marker; Limit = Math.Min(4096, Math.Max(1, cellLimit)); Axis = mergeAxis;
        if (marker == null) { Fault = "Missing Head/Mount."; return; }

    public bool Step(out bool done, out string why)
    {
        done = false; why = Fault;
        if (Fault != null) return false;
        if (Marker.CubeGrid.EntityId != GridId) return ToolGeometry.Fail(out why, Fault = "Head grid changed; retry.");
        for (int count = 0; Pending.Count > 0 && count < 128; count++)
        {
            if (P.Runtime.MaxInstructionCount > 0 && P.Runtime.CurrentInstructionCount >= P.Runtime.MaxInstructionCount * .5) return true;
            Vector3I cell = Pending.Dequeue(); var slim = Grid.GetCubeBlock(cell);
            if (slim == null && !Grid.CubeExists(cell)) return ToolGeometry.Fail(out why, Fault = "Head cells changed; retry.");
            if (slim == null) Hidden.Add(cell);
            var fat = slim == null ? null : slim.FatBlock;
            if (fat != null && Blocks.Add(fat.EntityId))
            {
                var merge = fat as IMyShipMergeBlock;
                if (merge != null) { Vector3I d; if (!ToolGeometry.MergeDirection(merge, Axis, out d, out why)) { Fault = why; return false; } Merges.Add(merge); }
            }
            foreach (var d in Dirs)
            {
                Vector3I next = cell + d; if (Cells.Contains(next)) continue;
                var neighbor = Grid.GetCubeBlock(next); if (neighbor == null && !Grid.CubeExists(next)) continue;
                var a = fat as IMyShipMergeBlock; var b = neighbor == null ? null : neighbor.FatBlock as IMyShipMergeBlock;
                if (a != null && b != null && a.EntityId != b.EntityId)
                {
                    Vector3I ad, bd;
                    if (!ToolGeometry.MergeDirection(a, Axis, out ad, out why) || !ToolGeometry.MergeDirection(b, Axis, out bd, out why)) { Fault = why; return false; }
                    if (ad == d && bd == -d && ToolGeometry.OnFace(a, cell, ad) && ToolGeometry.OnFace(b, next, bd)) continue;
                }
                if (Cells.Count >= Limit) return ToolGeometry.Fail(out why, Fault = "Head cell limit (" + Limit + ").");
                Cells.Add(next); Pending.Enqueue(next);
            }
        }
        done = Pending.Count == 0; return true;
    }
}

    public static bool MergeDirection(IMyShipMergeBlock b, string axis, out Vector3I d, out string why)
    {
        d = Vector3I.Zero; why = null; Vector3D n;

    public static bool OnFace(IMyShipMergeBlock b, Vector3I c, Vector3I d) => d.X > 0 ? c.X == b.Max.X : d.X < 0 ? c.X == b.Min.X : d.Y > 0 ? c.Y == b.Max.Y : d.Y < 0 ? c.Y == b.Min.Y : d.Z > 0 ? c.Z == b.Max.Z : c.Z == b.Min.Z;
    static bool GetFace(IMyShipMergeBlock b, string axis, out Face f, out string why)
    {
        f = null; Vector3I d; if (!Budget()) return Fail(out why, GeometryBudgetMessage); if (!MergeDirection(b, axis, out d, out why)) return false;
        Vector3I size = b.Max - b.Min + Vector3I.One;

    static Face TransformFace(Face a, MatrixD transform)
    {
        return new Face { Center = Vector3D.Transform(a.Center, transform), Normal = Vector3D.TransformNormal(a.Normal, transform), U = Vector3D.TransformNormal(a.U, transform), V = Vector3D.TransformNormal(a.V, transform), HalfU = a.HalfU, HalfV = a.HalfV };
    }
    static bool SameFace(Face a, Face b, double clearance)
    {
        if (Vector3D.Dot(a.Normal, b.Normal) > -.99999 || Vector3D.DistanceSquared(a.Center, b.Center + b.Normal * clearance) > .0004) return false;
        // Rectangles agree only when their cell axes and both extents agree;
        // quarter rolls may exchange the two axes.
        if (Math.Abs(Vector3D.Dot(a.U, b.U)) > .99999) return Math.Abs(a.HalfU - b.HalfU) <= .02 && Math.Abs(a.HalfV - b.HalfV) <= .02;
        return Math.Abs(Vector3D.Dot(a.U, b.V)) > .99999 && Math.Abs(a.HalfU - b.HalfV) <= .02 && Math.Abs(a.HalfV - b.HalfU) <= .02;
    }

        else if (d.Y != 0) min.Y = max.Y = d.Y > 0 ? max.Y : min.Y;
        else min.Z = max.Z = d.Z > 0 ? max.Z : min.Z;

        {
            if (!Budget()) return false;

            if (slim == null || slim.FatBlock == null || slim.FatBlock.EntityId != b.EntityId) return false;
        }
        return true;
    }

    }

                    if (!Budget()) { pairs = null; return Fail(out why, GeometryBudgetMessage); }
                    if (i == 0 && primaryStandID > 0 && j != anchor) continue;
                    if (used.Contains(rack[j].EntityId) || own[i].EntityId == rack[j].EntityId || !SameFace(moved, b[j], clearance)) continue;
                    var pair = new ToolPortPair { Head = own[i], Stand = rack[j] };
                    if ((rack[j].IsConnected || own[i].IsConnected) && !PairLocked(pair, axis)) continue;
                    if (match >= 0) { valid = false; break; } match = j;
                }
                if (match < 0) valid = false;
                else { used.Add(rack[match].EntityId); found.Add(new ToolPortPair { Head = own[i], Stand = rack[match] }); }
            }
            if (!valid) continue;
            best = cost; pose = trial; pairs = found;
            }
        }
        if (pairs == null) return Fail(out why, "No unique merge layout; check ports/mergeAxis.");
        return true;
    }
    public static bool TryTopPose(IMyMotorStator mount, ToolTopInfo top, out MatrixD pose, out string why, bool axial = false)
    {
        pose = MatrixD.Identity; MatrixD desiredBase;
        if (!TryAttachPose(mount, top, out desiredBase, out why)) return false;
        var delta = MatrixD.Invert(desiredBase) * mount.WorldMatrix;
        var desiredTop = top.WorldMatrix * delta;
        desiredTop.Translation = Vector3D.Transform(top.CubeGrid.GridIntegerToWorld(top.Position), delta);
        if (axial)
        {
            Vector3D a = top.WorldMatrix.Up, b = mount.WorldMatrix.Up;
            var cross = Vector3D.Cross(a, b);
            var rotation = cross.LengthSquared() > 1e-20 ? Unit(cross) * Math.Acos(Clamp(Vector3D.Dot(a, b), -1, 1)) : Vector3D.Dot(a, b) > 0 ? Vector3D.Zero : top.WorldMatrix.Forward * Math.PI;
            desiredTop = MatrixD.CreateWorld(desiredTop.Translation, Rotate(top.WorldMatrix.Forward, rotation), b);
        }
        pose = MatrixD.Invert(top.Local) * desiredTop;
        if (!Finite(pose.Translation) || !Finite(pose.Up) || !Finite(pose.Forward)) return Fail(out why, "Bad arm coupling frame.");
        return true;
    }
    public static bool TryAttachPose(IMyMotorStator mount, ToolTopInfo top, out MatrixD pose, out string why)
    {
        pose = MatrixD.Identity; why = null;
        if (mount == null || top == null) return Fail(out why, "Missing Mount/top.");
        string s = mount.BlockDefinition.SubtypeName, t = top.Subtype;
        int index = Array.IndexOf(Bases, s), headIndex = Array.IndexOf(Heads, t);
        if (index < 0) return Fail(out why, "Unsupported mount: " + s);
        if (headIndex < 0) return Fail(out why, "Incompatible factory head: " + t);
        Vector3D dummy = Dummies[index];
        double offset = index == 1 || index == 4 ? .045 : index == 3 ? .11 : 0;
        double lo = index == 3 ? -.02 : -.11, hi = .11;
        if ((index == 0 || index == 2) && top.CubeGrid.GridSizeEnum == MyCubeSize.Large) { lo = -.4; hi = .2; }
        double displacement = mount.Displacement;
        if (!Finite(displacement) || displacement < lo - .000001 || displacement > hi + .000001) return Fail(out why, "Attach would clamp displacement; use " + lo + " .. " + hi + ".");
        // Attach retains Angle and lock state. Preserve the phase for the rotor
        // instead of imposing a zero-angle head frame.
        double q = mount.Angle;
        if (!Finite(q)) return Fail(out why, "Bad mount angle.");
        double lower = mount.LowerLimitRad, upper = mount.UpperLimitRad;
        bool inLimits = false;
        for (int turn = -1; turn <= 1; turn++)
        {
            double candidate = q + turn * Math.PI * 2;
            if ((lower < -1e6 || candidate >= lower - .001) && (upper > 1e6 || candidate <= upper + .001)) inLimits = true;
        }
        if (!inLimits) return Fail(out why, "Angle outside mount limits.");
        Vector3D up = top.WorldMatrix.Up, forward = top.WorldMatrix.Forward;
        forward = forward * Math.Cos(q) + Vector3D.Cross(up, forward) * Math.Sin(q);
        pose = MatrixD.CreateWorld(Vector3D.Zero, forward, up);
        pose.Translation = top.CubeGrid.GridIntegerToWorld(top.Position) - Vector3D.TransformNormal(dummy + Vector3D.Up * (displacement - offset), pose);
        if (!Finite(pose.Translation) || !Finite(pose.Up) || !Finite(pose.Forward)) return Fail(out why, "Bad attach frame.");
        return true;
    }

        ScanIndex = 0; Scan = null; Parks.Clear(); Upstream.Clear(); UpstreamTops.Clear(); Clearing.Clear();
        foreach (var h in Profiles) { h.Heads.Clear(); h.Stands.Clear(); h.SafetyMerges.Clear(); }
        Last = verb == "toolscan" ? "Scan" : "Change";
        Go("Scan"); P.Frequency = UpdateFrequency.Update1; return true;
    }

    bool ScanStep(out bool done, out string why)
    {
        done = false; why = "";
        if (ScanIndex < Profiles.Count)
        {
            var h = Profiles[ScanIndex];
            IMyTerminalBlock current;
            if (!Exact(h.MarkerName, out current, out why) || current.EntityId != h.Marker.EntityId) { why = "Marker changed; Reload."; return false; }
            if (Scan == null) Scan = new ToolRegionScan(P, h.Marker, CellLimit, Axis);
            bool finished; if (!Scan.Step(out finished, out why)) { why = h.MarkerName + ": " + why; return false; }
            if (!finished) return true;
            if (h.Automatic && !DiscoverHardware(h, out why)) return false;
            if (!Scan.Blocks.Contains(h.Coupler.EntityId)) { why = "Tool Mount outside marker region."; return false; }
            foreach (var other in Profiles) if (other != h && Scan.Blocks.Contains(other.Marker.EntityId)) { why = "Tool regions overlap."; return false; }
            h.SafetyMerges.AddRange(Scan.Merges);
            if (ScanHead && !h.Automatic) h.Heads.AddRange(Scan.Merges);
            if (!MergeNames(h.HeadNames, h.Heads, out why) || !MergeNames(h.StandNames, h.Stands, out why)) return false;
            foreach (var merge in h.Heads) if (!Scan.Blocks.Contains(merge.EntityId)) { why = "Foreign head merge: " + merge.CustomName; return false; }
            if (ScanStand && !h.Automatic) Prefix(h.Prefix, h.Stands);
            foreach (var merge in h.Stands) if (h.Heads.Contains(merge)) { why = "Stand is on head: " + merge.CustomName; return false; }
            P.CaptureParkPose(h, Axis);
            if (h.Automatic) SaveHead(h);
            ScanIndex++; Scan = null; return true;
        }
        if (Tip == null)
        {
            if (ArmIndex < ArmOrder.Count)
            {
                var frame = GridFrame(ArmOrder[ArmIndex]);
                if (Scan == null) Scan = new ToolRegionScan(P, frame, CellLimit, Axis);
                bool finished; if (!Scan.Step(out finished, out why)) return false;
                if (!finished) return true;
                var candidate = ToolGeometry.FindTip(frame, Scan.Hidden, Inventory);
                if (candidate != null)
                { if (Found != null) { why = "Multiple loose arm rotor parts."; return false; } Found = candidate; }
                ArmIndex++; Scan = null; return true;
            }
            if (Found == null) { why = "No loose arm rotor part."; return false; }
            Tip = Found; SelectGrid(Tip.CubeGrid.EntityId);
        }
        if (!MergeNames(ParkNames, Parks, out why)) return false;
        if (ScanPark) Prefix(ArmName + " - ParkMerge ", Parks);
        foreach (var h in Profiles) foreach (var m in Parks) if (h.Heads.Contains(m)) { why = "Park is on head."; return false; }
        Scanned = true; done = true; return true;
    }
    void Go(string phase)
    {
        Phase = phase; Elapsed = StableSeconds = 0; Stable = 0; PhaseBudget = Timeout;
        if (phase == "Lock") SupportProven = false;
        if (phase == ApproachDockPhase || phase == "Dock" || phase == RetreatPhase || phase == ApproachTopPhase || phase == AlignTopPhase)
        {
            MatrixD target = phase == ApproachDockPhase ? Approach : phase == "Dock" ? Dock : phase == RetreatPhase ? Retreat : phase == ApproachTopPhase ? TopApproach : Attach;
            var anchor = phase == ApproachDockPhase || phase == "Dock" ? DockAnchor : phase == RetreatPhase ? RetreatAnchor : AttachAnchor;
            target *= anchor.WorldMatrix; var current = EndFrame;
            double trace = Vector3D.Dot(target.Right, current.WorldMatrix.Right) + Vector3D.Dot(target.Up, current.WorldMatrix.Up) + Vector3D.Dot(target.Forward, current.WorldMatrix.Forward);
            PhaseBudget += Vector3D.Distance(target.Translation, current.GetPosition()) / Move + Math.Acos(Math.Max(-1, Math.Min(1, (trace - 1) * .5))) * 180 / Math.PI / Turn;
        }
        if (phase != "Idle") P.Say("Tools " + phase + " OFF");
    }
    void Journal(string phase)
    {
        var s = AutoConfig.Saved(P.Storage); long relock = 0; foreach (var h in Profiles) if (h.Relock) relock = h.Coupler.EntityId; s.Set(SwapSection, "Relock", relock); s.Set(SwapSection, "PB", P.Me.EntityId); s.Set(SwapSection, "Mount", Mount == null ? 0 : Mount.EntityId);
        s.Set(SwapSection, "Phase", phase); s.Set(SwapSection, "Source", Source == null ? "" : Source.MarkerName); s.Set(SwapSection, "Destination", Destination == null ? "" : Destination.MarkerName); P.Storage = s.ToString();
    }

    {
        StartAfterScan = false;

    }

    {
        start |= StartAfterScan; StartAfterScan = false;

        Last = message;
        if (start) P.EnableControl(); else P.Say(message + " On required.");
    }

    bool Observe(bool condition, double seconds = 0)
    {
        if (!condition) { Stable = 0; StableSeconds = 0; return false; }
        if (++Stable > 1) StableSeconds += seconds;
        return seconds > 0 ? StableSeconds >= .2 : Stable >= 2;
    }
    bool TipOwned(out string why)
    {
        why = ""; var owner = Tip == null ? null : Tip.Base;
        if (owner == null) return true;
        foreach (var h in Profiles) if (h.Coupler != null && h.Coupler.EntityId == owner.EntityId && (Mounted(h) || ConnectionRefresh(h) || (h == Source && Phase == "Detach") || (h == Destination && Phase == "Attach"))) return true;
        why = "Unknown arm-tip owner."; return false;
    }
    bool ConnectionRefresh(Profile h)
    {
        if (Elapsed > 2 || Tip == null || Tip.EntityId == 0 ||
            !(h == Source && Phase == "Lock" && DockMoved && (!SupportProven || AllLocked()) || h == Destination && Phase == "Release")) return false;
        var top = h.Coupler.Top as IMyMotorRotor;
        return top == null || top.EntityId == Tip.EntityId && (top.Base == null || top.Base.EntityId == h.Coupler.EntityId);
    }
    bool Guard(out string why)
    {
        why = "";
        if (Mount == null || Mount.Closed || !Mount.IsFunctional || Mount.CustomName != MountName) { why = "Mount unavailable/renamed."; return false; }
        for (int n = 0; n < Upstream.Count; n++)
        {
            var b = Upstream[n]; if (b.Closed || !b.IsAttached || b.Top == null || b.Top.EntityId != UpstreamTops[n]) { why = "Upstream attachment changed."; return false; }
        }
        foreach (var h in Profiles) if (h.Marker == null || h.Marker.Closed || h.Marker.CustomName != h.MarkerName ||
            !(h.Automatic && Phase == "Scan" && h.Coupler == null) && (h.Coupler == null || h.Coupler.Closed || !h.Coupler.IsFunctional || !h.Automatic && h.Coupler.CustomName != h.CouplerName || h.Coupler.CubeGrid.EntityId == Reference.CubeGrid.EntityId || h.Coupler.CubeGrid.EntityId != h.Marker.CubeGrid.EntityId))
        { why = "Tool block changed."; return false; }
        foreach (var h in Profiles)
        {
            if (h.Automatic && Phase == "Scan" && h.Coupler == null) continue;
            foreach (var merge in h.SafetyMerges) if (merge.Closed || merge.CubeGrid.EntityId != h.Marker.CubeGrid.EntityId) { why = "Tool support body split."; return false; }
            if (h.Coupler.IsAttached && !Mounted(h) && !ConnectionRefresh(h)) { why = "Tool attached elsewhere."; return false; }
            if (h == Destination && !AxialPickup && (Phase == ApproachTopPhase || Phase == AlignTopPhase || Phase == "PrepareAttach") &&
                (!Finite(h.Coupler.Angle) || Math.Abs(Wrap(h.Coupler.Angle - h.Angle)) > .002))
            { why = h.Name + " mount phase changed during " + Phase + "; ToolScan."; return false; }
        }
        if (Reference.Closed) { why = "Arm anchor unavailable."; return false; }
        if (Tip != null && (Tip.Closed || Tip.CubeGrid.EntityId != Reference.CubeGrid.EntityId || !Reference.CubeGrid.CubeExists(Tip.Position)))
        { why = "Arm tip changed."; return false; }
        if (!TipOwned(out why)) return false;
        if (ParkedPairs != null)
            foreach (var pair in ParkedPairs) if (pair.Head.CubeGrid.EntityId != Source.Marker.CubeGrid.EntityId || !ToolGeometry.PairLocked(pair, Axis)) { why = "Parked support lost."; return false; }
        return true;
    }
    bool BeginChange(out string why)
    {
        why = "";
        foreach (var h in Profiles) if (Mounted(h)) Source = h;
        if (Tip == null) { why = "Rotor part not discovered."; return false; }
        AxialPickup = Tip.EntityId == 0;
        if (AxialPickup && Destination != null && !Unlimited(Destination.Coupler)) { why = "First pickup needs unlimited rotor angle limits."; return false; }
        if (!TipOwned(out why)) return false;
        foreach (var h in Profiles)
        {
            h.Angle = h.Coupler.Angle;
            if (h.Coupler.PendingAttachment || (h.Coupler.IsAttached && h != Source)) { why = "Other tool Mount attached/pending."; return false; }
        }
        if (Source == Destination && !ParkOnly) { Complete("Selected tool already mounted.", true); return true; }
        if (Source == null && ParkOnly) { Complete("Mount already bare; OFF."); return true; }
        int mountIndex = P.Topology.Groups.Count;
        for (int n = 0; n < P.Topology.Groups.Count; n++) foreach (var member in P.Topology.Groups[n].M) if (Source != null && member.B.EntityId == Source.Coupler.EntityId) mountIndex = n;
        for (int n = 0; n < mountIndex; n++) foreach (var member in P.Topology.Groups[n].M)
            if ((Source == null || member.B.EntityId != Source.Coupler.EntityId) && !Upstream.Contains(member.B)) { Upstream.Add(member.B); UpstreamTops.Add(member.B.Top == null ? 0 : member.B.Top.EntityId); }
        if (Destination != null && Destination != Source && !Connected(Destination.Heads)) { why = "Incoming tool must be parked/detached."; return false; }
        if (Destination != null && Destination != Source)
        {
            var before = Pairs; var previous = Supported; MatrixD pose;
            bool valid = FindSupport(Destination, out why) && ToolGeometry.TryTopPose(Destination.Coupler, Tip, out pose, out why, AxialPickup);
            Pairs = before; Supported = previous;
            if (!valid) return false;
        }
        if (Source == null) return BeginIncoming(out why);
        if (Vector3D.Dot(Source.Coupler.WorldMatrix.Up, Tip.WorldMatrix.Up) < -.95)
        { why = "Flipped mount; detach refused."; return false; }
        long anchor = 0;
        if (ParkOnly && ParkIndex > 0)
        {
            IMyTerminalBlock block;
            if (!Exact(ArmName + " - ParkMerge " + ParkIndex, out block, out why)) return false;
            var merge = block as IMyShipMergeBlock;
            if (merge == null || !Parks.Contains(merge)) { why = "Park port not configured."; return false; }
            anchor = merge.EntityId;
        }
        if (Connected(Source.SafetyMerges))
        {
            if (!FindSupport(Source, out why)) return false;
            if (anchor != 0 && Pairs[0].Stand.EntityId != anchor) { why = "Already merged to another park."; return false; }
            SetRetreat();
            P.StopMotion("Supported tool OFF"); Go("Lock"); return true;
        }
        var rack = ParkOnly ? Parks : Source.Stands;
        if (Source.Heads.Count == 0 || rack.Count == 0) { why = "Missing parking merges."; return false; }
        if (!P.ParkPose(Source, rack, anchor, out Dock, out Pairs) && !ToolGeometry.TryDockPoseAt(Source.Marker, Source.Heads, rack, Axis, 0, anchor, out Dock, out Pairs, out why)) return false;
        Vector3I direction; if (!ToolGeometry.MergeDirection(Pairs[0].Stand, Axis, out direction, out why)) return false;
        Vector3D outward = Vector3D.TransformNormal((Vector3D)direction, Pairs[0].Stand.CubeGrid.WorldMatrix);
        foreach (var pair in Pairs)
        {
            if (pair.Stand.CubeGrid.EntityId == Reference.CubeGrid.EntityId) { why = "Stand is on arm."; return false; }
            Vector3I other; if (!ToolGeometry.MergeDirection(pair.Stand, Axis, out other, out why)) return false;
            if (Vector3D.Dot(outward, Vector3D.TransformNormal((Vector3D)other, pair.Stand.CubeGrid.WorldMatrix)) < .99999)
            { why = "Parking faces must be parallel."; return false; }
        }
        Approach = Dock; Approach.Translation += outward * Clearance;
        DockAnchor = Pairs[0].Stand;
        var inverse = MatrixD.Invert(DockAnchor.WorldMatrix); Dock *= inverse; Approach *= inverse;
        Supported = Source;
        // Arm-side capture monitoring is acknowledged before magnets enable.
        foreach (var head in Source.Heads) head.Enabled = false;
        Go(ApproachDockPhase); return true;
    }
    bool AllLocked()
    {
        if (Supported == null || Pairs.Count == 0 || Pairs.Count != Supported.Heads.Count || Supported.Coupler.CubeGrid.EntityId != Supported.Marker.CubeGrid.EntityId) return false;
        foreach (var pair in Pairs) if (pair.Head.CubeGrid.EntityId != Supported.Marker.CubeGrid.EntityId || pair.Stand.CubeGrid.EntityId == Reference.CubeGrid.EntityId || !ToolGeometry.PairLocked(pair, Axis)) return false;
        foreach (var m in Supported.SafetyMerges) if (m.IsConnected)
        { bool covered = false; foreach (var pair in Pairs) if (pair.Head.EntityId == m.EntityId) covered = true; if (!covered) return false; }
        return true;
    }
    bool FindSupport(Profile h, out string why)
    {
        why = ""; Pairs = new List<ToolPortPair>(); Supported = h;
        foreach (var head in h.Heads)
        {
            IMyShipMergeBlock stand;
            if (!head.IsConnected || !ToolGeometry.TryLockedPartner(head, Axis, out stand)) { why = "Unproven support: " + head.CustomName; return false; }
            foreach (var tool in Profiles) if (tool.SafetyMerges.Contains(stand)) { why = "Support is on another tool."; return false; }
            Pairs.Add(new ToolPortPair { Head = head, Stand = stand });
        }
        if (AllLocked()) return true;
        why = "Unselected/unlocked support."; return false;
    }
    bool BeginIncoming(out string why)
    {
        why = "";
        if (Destination == null) { Complete("Tool parked; bare mount OFF."); return true; }
        if (!FindSupport(Destination, out why))
        return false;
        return TargetTop(out why);
    }
    bool TargetTop(out string why)
    {
        why = "";
        if (!ToolGeometry.TryTopPose(Destination.Coupler, Tip, out Attach, out why, AxialPickup)) return false;
        Destination.Angle = Destination.Coupler.Angle;
        TopApproach = Attach; TopApproach.Translation += Destination.Coupler.WorldMatrix.Up * Clearance;
        AttachAnchor = Destination.Marker;
        var inverse = MatrixD.Invert(AttachAnchor.WorldMatrix); Attach *= inverse; TopApproach *= inverse;
        Go(ApproachTopPhase); return true;
    }
    void SetRetreat()
    {
        RetreatAnchor = Source.Marker;
        Retreat = Reference.WorldMatrix; Retreat.Translation += Source.Coupler.WorldMatrix.Up * Clearance;
        Retreat *= MatrixD.Invert(RetreatAnchor.WorldMatrix);
    }
    bool MovePose(MatrixD pose, double dt, double obs, out bool settled, out string why)
    {
        settled = false; why = "";
        foreach (var h in Profiles) if (h.Coupler.PendingAttachment) { why = "Pending attachment; motion refused."; return false; }
        if (Source != null && Mounted(Source) && Connected(Source.SafetyMerges)) { why = "Mounted support; motion refused."; return false; }
        if (Phase == RetreatPhase || Phase == ApproachTopPhase || Phase == AlignTopPhase)
            foreach (var h in Profiles) if (h.Coupler.IsAttached || h.Coupler.Top != null) { why = "Unexpected attachment."; return false; }
        if ((Phase == ApproachDockPhase || Phase == "Dock") && !Mounted(Source)) { why = "Source attachment changed."; return false; }
        if (Phase == ApproachDockPhase || Phase == "Dock" || Phase == ApproachTopPhase || Phase == AlignTopPhase)
        {
            bool docking = Phase == ApproachDockPhase || Phase == "Dock";
            MatrixD first = docking ? Approach : TopApproach, second = docking ? Dock : Attach;
            if (docking) second.Translation += (Approach.Translation - Dock.Translation) * (Math.Min(.03, Clearance * .25) / Clearance);
            if (Focus != null) { first = Tip.Local * first; second = Tip.Local * second; }
            return P.ToolPathStep(first, second, docking ? DockAnchor : AttachAnchor,
                Phase == ApproachDockPhase || Phase == ApproachTopPhase ? 0 : 1, Move, Turn,
                docking ? PosTol : Math.Min(PosTol, .003), docking ? AngleTol : Math.Min(AngleTol, .2), out settled, out why);
        }
        if (Focus != null) pose = Tip.Local * pose;
        return P.ToolPoseStep(pose, dt, obs, Move, Turn, Phase == AlignTopPhase ? Math.Min(PosTol, .003) : PosTol,
            Phase == AlignTopPhase ? Math.Min(AngleTol, .2) : AngleTol, out settled, out why);
    }
    public void Tick(double dt, double observationDt)
    {
        if (!Busy || !(dt > 0) || !(observationDt > 0)) return;
        try
        {
            if (dt > 1 || observationDt > 1) { Cancel("Tool observation gap."); return; }
            Elapsed += observationDt; string why;
            if (Parking)
            {
                if (!DockMoved && P.CaptureReady) { Journal("CaptureApproach"); foreach (var pair in Pairs) pair.Head.Enabled = true; DockMoved = true; }
                if (Connected(Source.SafetyMerges)) { DockMoved = true; Go("Lock"); }
            }
            if (!Guard(out why)) { Cancel(why); return; }
            if (Elapsed > PhaseBudget) { Cancel("Tool phase timeout: " + Phase); return; }
            bool done, settled;
            if (Phase == ApproachDockPhase || Phase == "Dock" || Phase == RetreatPhase || Phase == ApproachTopPhase || Phase == AlignTopPhase)
            {
                if ((Phase == RetreatPhase || Phase == ApproachTopPhase || Phase == AlignTopPhase) && !AllLocked()) { Cancel("Support lost."); return; }
                MatrixD goal = Phase == ApproachDockPhase ? Approach : Phase == "Dock" ? Dock : Phase == RetreatPhase ? Retreat : Phase == ApproachTopPhase ? TopApproach : Attach;
                var anchor = Phase == ApproachDockPhase || Phase == "Dock" ? DockAnchor : Phase == RetreatPhase ? RetreatAnchor : AttachAnchor;
                if (anchor == null || anchor.Closed) { Cancel("Dock anchor lost."); return; }
                goal *= anchor.WorldMatrix;
                if (!MovePose(goal, dt, observationDt, out settled, out why)) { Cancel(why); return; }
                if (!Observe(settled)) return;
            }
            switch (Phase)
            {
                case "Scan":
                    if (!ScanStep(out done, out why)) { Cancel(why); return; }
                    if (!done) return;
                    foreach (var h in Profiles) if (h.Coupler.PendingAttachment)
                    {
                        if (h.Coupler.IsAttached || h.Coupler.Top != null || Last != "Scan" || PassiveScan) { Cancel("Explicit ToolScan required for pending Mount."); return; }
                        if (!Clearing.Contains(h.Coupler)) Clearing.Add(h.Coupler);
                    }
                    if (Clearing.Count > 0)
                    {
                        P.ExpectTransition("CancelPending");
                        Journal("ScanPending"); foreach (var coupler in Clearing) coupler.Detach(); Go("ScanPending"); return;
                    }
                    if (!P.RebuildToolTopology(out why)) { Cancel(why); return; }
                    Go("ScanSettle"); return;
                case "ScanPending":
                    P.StopMotion("Clear pending OFF"); bool clear = true;
                    foreach (var coupler in Clearing)
                    {
                        if (coupler.IsAttached || coupler.Top != null) { Cancel("Pending cancellation changed."); return; }
                        if (coupler.PendingAttachment) clear = false;
                    }
                    if (!Observe(clear)) return;
                    if (!P.RebuildToolTopology(out why)) { Cancel(why); return; } Go("ScanSettle"); return;
                case "ScanSettle":
                    if (!P.SettleToolTopology(out why)) return;
                    if (Last == "Scan")
                    {
                        foreach (var h in Profiles) if (h.Relock && !h.Coupler.PendingAttachment)
                        {
                            MatrixD pose;
                            bool bare = !h.Coupler.IsAttached && h.Coupler.Top == null;
                            if (bare || Mounted(h) && ToolGeometry.TryTopPose(h.Coupler, Tip, out pose, out why, true) && Vector3D.DistanceSquared(Reference.GetPosition(), pose.Translation) < .000025 && Vector3D.Dot(Reference.WorldMatrix.Up, pose.Up) > .9999939)
                            { h.Coupler.RotorLock = true; h.Relock = false; }
                        }
                        Complete("Scanned; OFF.");
                    }
                    else if (!BeginChange(out why)) Cancel(why); return;
                case ApproachDockPhase:
                    Go("Dock"); return;
                case "Dock":
                    P.StopMotion("Capture wait OFF"); return;
                case "Lock":
                    P.StopMotion("Support wait OFF");
                    bool locked = AllLocked();
                    if (SupportProven && !locked) { Cancel("Support lost."); return; }
                    SupportProven |= locked;
                    if (!Mounted(Source) || Source.Coupler.PendingAttachment)
                    { Observe(false); if (!ConnectionRefresh(Source)) Cancel("Source detached prematurely."); return; }
                    if (!Observe(locked, observationDt)) return;
                    P.CaptureParkPose(Source, Axis);
                    if (DockMoved) SetRetreat();
                    ParkedPairs = new List<ToolPortPair>(Pairs);
                    P.ExpectTransition("DetachMount");
                    Source.Angle = Source.Coupler.Angle; Journal("Detach"); Source.Coupler.Detach(); Go("Detach"); return;
                case "Detach":
                    P.StopMotion("Detach wait OFF");
                    if (!AllLocked()) { Cancel("Support lost."); return; }
                    if (!Observe(!Source.Coupler.IsAttached && !Source.Coupler.PendingAttachment && Source.Coupler.Top == null && Tip.Base == null, observationDt)) return;
                    Source.Angle = Source.Coupler.Angle;
                    if (!P.RebuildToolTopology(out why)) { Cancel(why); return; } Go("BareSettle"); return;
                case "BareSettle":
                    if (P.SettleToolTopology(out why)) Go(RetreatPhase); return;
                case RetreatPhase:
                    if (!BeginIncoming(out why)) Cancel(why); return;
                case ApproachTopPhase:
                    Go(AlignTopPhase); return;
                case AlignTopPhase:
                    Go("PrepareAttach"); return;
                case "PrepareAttach":
                    foreach (var h in Profiles) if (h.Coupler.IsAttached || h.Coupler.PendingAttachment || h.Coupler.Top != null) { Cancel("Arm tip no longer bare."); return; }
                    P.ExpectTransition("AttachMount"); if (AxialPickup) Destination.Relock = Destination.Coupler.RotorLock;
                    Journal("Attach"); if (Destination.Relock) Destination.Coupler.RotorLock = false; Destination.Coupler.Attach(); Go("Attach"); return;
                case "Attach":
                    P.StopMotion("Attach wait OFF");
                    if (!AllLocked()) { Cancel("Support lost."); return; }
                    if (Destination.Coupler.IsAttached && !Mounted(Destination)) { Cancel("Wrong arm tip attached."); return; }
                    MatrixD expected;
                    if (!ToolGeometry.TryTopPose(Destination.Coupler, Tip, out expected, out why, AxialPickup)) { Cancel(why); return; }
                    bool aligned = Vector3D.DistanceSquared(Reference.GetPosition(), expected.Translation) < .000025 &&
                        Vector3D.Dot(Reference.WorldMatrix.Up, expected.Up) > .9999939 && Vector3D.Dot(Reference.WorldMatrix.Forward, expected.Forward) > .9999939;
                    if (!Observe(Mounted(Destination) && !Destination.Coupler.PendingAttachment && aligned, observationDt)) return;
                    if (Destination.Relock) { Destination.Coupler.RotorLock = true; Destination.Relock = false; }
                    P.ExpectTransition("MergeRelease");
                    P.CaptureParkPose(Destination, Axis);
                    Journal("ReleaseSupport"); foreach (var pair in Pairs) pair.Head.Enabled = false; Go("Release"); return;
                case "Release":
                    P.StopMotion("Split wait OFF");
                    if (!Mounted(Destination) || Destination.Coupler.PendingAttachment)
                    { Observe(false); if (!ConnectionRefresh(Destination)) Cancel("Attachment lost after support release."); return; }
                    bool split = !Connected(Destination.SafetyMerges);
                    foreach (var pair in Pairs) if (pair.Head.Closed || pair.Stand.Closed || pair.Stand.IsConnected || Destination.Marker.CubeGrid.EntityId == pair.Stand.CubeGrid.EntityId) split = false;
                    if (!Observe(split, observationDt)) return;
                    if (!P.RebuildToolTopology(out why)) { Cancel(why); return; } Go("FullSettle"); return;
                case "FullSettle":
                    if (!Mounted(Destination) || Connected(Destination.SafetyMerges)) { Cancel("Attachment/support changed."); return; }
                    if (P.SettleToolTopology(out why)) Complete("Tool " + Destination.Name + " mounted.", true); return;
            }
        }
        catch (RemoteWait) { }
        catch (Exception e) { Cancel("Tools: " + e.Message); }
    }
    public void WriteEquipment(MyIni ini)
    {
        var rows = new List<string>();
        foreach (var h in Profiles)
        { var ports = new List<string>(); foreach (var m in h.SafetyMerges) ports.Add(m.EntityId.ToString()); rows.Add(h.Coupler.EntityId + "|" + h.Marker.EntityId + "|" + (h.Coupler.Top == null ? 0 : h.Coupler.Top.EntityId) + "|" + string.Join(",", ports)); }
        ini.Set(LinkSection, "Equipment", string.Join("\n", rows));
    }
    string Info()
    {
        var b = new System.Text.StringBuilder("Tools " + (EnabledTools ? EnabledKey : "disabled") + "; " + Phase + (Recovery ? "; recovery OFF" : "") + "\n" + Last);
        b.Append(Tip == null ? "\nNo arm rotor part found." : "\nArm rotor part detected.");
        for (int n = 0; n < Profiles.Count; n++) { var h = Profiles[n]; b.Append("\n").Append(n + 1).Append(" ").Append(h.Name).Append(Mounted(h) ? " mounted" : " detached").Append("; Mount ").Append(h.CouplerName).Append("; merges ").Append(h.Heads.Count).Append('/').Append(h.Stands.Count); }
        return b.ToString();
    }
}


// Tool hardware belongs to the occupied-cell region cut at facing merge pairs.
// Names select only the reference block; detected identities live on that block.
sealed partial class ToolSwapController
{
    bool AutoProfiles;
    const string HeadData = "AutoArm Tool";
    int HeadNumber(string name)
    {
        string prefix = ArmName + " - Head "; int number;
        if (!name.StartsWith(prefix, StringComparison.Ordinal)) return 0;
        string tail = name.Substring(prefix.Length); int split = tail.IndexOf(" - ", StringComparison.Ordinal);
        return split > 0 && split + 3 < tail.Length && int.TryParse(tail.Substring(0, split), out number) && number > 0 ? number : 0;
    }
    bool AutoHeads(List<string> heads, out string why)
    {
        why = ""; var names = new SortedDictionary<int, string>();
        foreach (var block in Inventory)
        {
            if (block.Closed || !block.IsSameConstructAs(P.Me)) continue;
            int number = HeadNumber(block.CustomName); if (number == 0) continue;
            if (names.ContainsKey(number)) { why = "More than one named block for " + ArmName + " - Head " + number + ". Keep that name only on its reference block."; return false; }
            names.Add(number, block.CustomName);
        }
        foreach (var pair in names) heads.Add(pair.Value);
        if (heads.Count == 0) { why = "Name one block per tool: " + ArmName + " - Head 1 - Drill."; return false; }
        return true;
    }
    MyIni ReadHead(Profile head)
    {
        var data = AutoConfig.Saved(head.Marker.CustomData);
        if (data.ContainsSection(HeadData) && data.Get(HeadData, "Format").ToInt32() != 1) throw new Exception(head.MarkerName + ": delete its AutoArm Tool section (unsupported Format).");
        return data;
    }
    void HeadHint(Profile head)
    {
        var data = ReadHead(head);
        if (data.Get(HeadData, "Arm").ToString() != ArmName) return;
        var mount = P.GridTerminalSystem.GetBlockWithId(data.Get(HeadData, "Mount").ToInt64()) as IMyMotorStator;
        if (mount != null && !mount.Closed && mount.CubeGrid.EntityId == head.Marker.CubeGrid.EntityId && ToolGeometry.Rotor(mount))
        { head.Coupler = mount; head.CouplerName = mount.CustomName; }
    }
    bool DiscoverHardware(Profile head, out string why)
    {
        why = "";
        var candidates = new List<IMyMotorStator>();
        foreach (var block in Inventory)
        {
            var rotor = block as IMyMotorStator;
            if (rotor == null || rotor.Closed || !ToolGeometry.Rotor(rotor) || !Scan.Blocks.Contains(rotor.EntityId)) continue;
            if (!rotor.IsAttached || rotor.Top != null && ArmGrids.ContainsKey(rotor.Top.CubeGrid.EntityId)) candidates.Add(rotor);
        }
        if (head.Coupler == null || !candidates.Contains(head.Coupler))
        {
            if (candidates.Count != 1) { why = head.MarkerName + ": need exactly one eligible rotor base inside the tool; found " + candidates.Count; return false; }
            head.Coupler = candidates[0];
        }
        head.CouplerName = head.Coupler.CustomName;
        foreach (var other in Profiles) if (other != head && other.Coupler == head.Coupler) { why = "Tool regions share a rotor base."; return false; }
        var data = ReadHead(head);
        if (ScanHead) foreach (var merge in Scan.Merges)
        {
            IMyShipMergeBlock stand;
            if (ToolGeometry.TryLockedPartner(merge, Axis, out stand) && !Scan.Blocks.Contains(stand.EntityId))
            { head.Heads.Add(merge); if (ScanStand && !head.Stands.Contains(stand)) head.Stands.Add(stand); }
        }
        // Mounted tools have no connected stand pairs. Validate stored ports
        // against the fresh tool region before reusing their identities.
        if (ScanHead && head.Heads.Count == 0 && data.Get(HeadData, "Arm").ToString() == ArmName)
            foreach (string token in data.Get(HeadData, "HeadMerges").ToString().Split(','))
            {
                long id; if (!long.TryParse(token, out id)) continue;
                var merge = P.GridTerminalSystem.GetBlockWithId(id) as IMyShipMergeBlock;
                if (merge == null || merge.Closed || !Scan.Blocks.Contains(id)) { why = "Stored head merge outside tool: " + head.MarkerName; return false; }
                head.Heads.Add(merge);
            }
        if (ScanStand && head.Stands.Count == 0 && data.Get(HeadData, "Arm").ToString() == ArmName)
            foreach (string token in data.Get(HeadData, "StandMerges").ToString().Split(','))
            {
                long id; if (!long.TryParse(token, out id)) continue;
                var merge = P.GridTerminalSystem.GetBlockWithId(id) as IMyShipMergeBlock;
                if (merge != null && !merge.Closed && !Scan.Blocks.Contains(id) && !head.Stands.Contains(merge)) head.Stands.Add(merge);
            }
        if (ScanHead && head.Heads.Count == 0 && !head.Coupler.IsAttached) { why = "Park " + head.MarkerName + " on facing connected merge blocks for automatic setup."; return false; }
        if (!RefreshTip(out why)) return false;
        var saved = AutoConfig.Saved(P.Storage);
        head.Relock = saved.Get(SwapSection, "PB").ToInt64() == P.Me.EntityId && saved.Get(SwapSection, "Relock").ToInt64() == head.Coupler.EntityId;
        return true;
    }
    static string PortIds(List<IMyShipMergeBlock> ports)
    { var ids = new List<string>(); foreach (var port in ports) ids.Add(port.EntityId.ToString()); return string.Join(",", ids); }
    void SaveHead(Profile head)
    {
        var data = ReadHead(head); data.Set(HeadData, "Format", 1); data.Set(HeadData, "Arm", ArmName);
        data.Set(HeadData, "Marker", head.Marker.EntityId); data.Set(HeadData, "Mount", head.Coupler.EntityId);
        data.Set(HeadData, "HeadMerges", PortIds(head.Heads)); data.Set(HeadData, "StandMerges", PortIds(head.Stands));
        string text = data.ToString(); if (text != head.Marker.CustomData) head.Marker.CustomData = text;
    }
}


// Automatically record the actual parked assembly pose, not an arbitrary
// quarter-turn alignment of visually symmetric merge blocks.
bool CaptureParkPose(ToolSwapController.Profile h, string axis)
{
    if (h.Heads.Count == 0) return false;
    var pairs = new List<ToolPortPair>();
    foreach (var head in h.Heads)
    {
        IMyShipMergeBlock stand;
        if (!head.IsConnected || !ToolGeometry.TryLockedPartner(head, axis, out stand)) return false;
        var pair = new ToolPortPair { Head = head, Stand = stand }; if (!ToolGeometry.PairLocked(pair, axis)) return false;
        pairs.Add(pair);
    }
    var data = new MyIni(); var anchor = pairs[0].Stand;
    data.Set(LinkSection, "Marker", h.Marker.EntityId); data.Set(LinkSection, "Anchor", anchor.EntityId);
    WriteFrame(data, "Dock", h.Marker.WorldMatrix * MatrixD.Invert(anchor.WorldMatrix));
    var rows = new List<string>();
    for (int i = 0; i < pairs.Count; i++)
    {
        var pair = pairs[i]; rows.Add(pair.Head.EntityId + " " + pair.Stand.EntityId);
        WriteFrame(data, "Head" + i, pair.Head.WorldMatrix * MatrixD.Invert(h.Marker.WorldMatrix));
        WriteFrame(data, "Stand" + i, pair.Stand.WorldMatrix * MatrixD.Invert(anchor.WorldMatrix));
    }
    data.Set(LinkSection, "Pairs", string.Join("\n", rows));
    var saved = AutoConfig.Saved(Storage); const string section = "AutoArm Park Poses";
    saved.Set(section, "PB", Me.EntityId); saved.Set(section, "Arm", ArmName);
    saved.Set(section, h.Coupler.EntityId.ToString(), data.ToString()); Storage = saved.ToString();
    var markerData = AutoConfig.Saved(h.Marker.CustomData);
    markerData.Set("AutoArm Tool", "Format", 1); markerData.Set("AutoArm Tool", "Arm", ArmName);
    markerData.Set("AutoArm Tool", "ParkPose", data.ToString()); h.Marker.CustomData = markerData.ToString(); return true;
}
bool ParkPose(ToolSwapController.Profile h, List<IMyShipMergeBlock> rack, long selected, out MatrixD pose, out List<ToolPortPair> pairs)
{
    pose = MatrixD.Identity; pairs = new List<ToolPortPair>(); const string section = "AutoArm Park Poses";
    var saved = AutoConfig.Saved(Storage);
    var markerData = AutoConfig.Saved(h.Marker.CustomData);
    string text = markerData.Get("AutoArm Tool", "Format").ToInt32() == 1 && markerData.Get("AutoArm Tool", "Arm").ToString() == ArmName ? markerData.Get("AutoArm Tool", "ParkPose").ToString() : "";
    if (text.Length == 0 && saved.Get(section, "PB").ToInt64() == Me.EntityId && saved.Get(section, "Arm").ToString() == ArmName) text = saved.Get(section, h.Coupler.EntityId.ToString()).ToString();
    if (text.Length == 0) return false;
    var data = new MyIni(); if (!data.TryParse(text)) throw new Exception("Invalid saved park pose; park tool and ToolScan.");
    if (data.Get(LinkSection, "Marker").ToInt64() != h.Marker.EntityId) return false;
    var anchor = GridTerminalSystem.GetBlockWithId(data.Get(LinkSection, "Anchor").ToInt64()) as IMyShipMergeBlock;
    if (anchor == null || anchor.Closed || selected != 0 && selected != anchor.EntityId) return false;
    string[] rows = data.Get(LinkSection, "Pairs").ToString().Split('\n'); if (rows.Length != h.Heads.Count || rows.Length > 16) return false;
    for (int i = 0; i < rows.Length; i++)
    {
        string[] cells = rows[i].Split(' '); long headId, standId;
        if (cells.Length != 2 || !long.TryParse(cells[0], out headId) || !long.TryParse(cells[1], out standId)) throw new Exception("Invalid saved park ports; park tool and ToolScan.");
        var head = GridTerminalSystem.GetBlockWithId(headId) as IMyShipMergeBlock;
        var stand = GridTerminalSystem.GetBlockWithId(standId) as IMyShipMergeBlock;
        if (head == null || stand == null || !h.Heads.Contains(head) || !rack.Contains(stand) || head.Closed || stand.Closed) return false;
        if (!SameParkFrame(head.WorldMatrix * MatrixD.Invert(h.Marker.WorldMatrix), LinkFrame(data, "Head" + i)) ||
            !SameParkFrame(stand.WorldMatrix * MatrixD.Invert(anchor.WorldMatrix), LinkFrame(data, "Stand" + i))) return false;
        pairs.Add(new ToolPortPair { Head = head, Stand = stand });
    }
    pose = LinkFrame(data, "Dock") * anchor.WorldMatrix; return true;
}
static bool SameParkFrame(MatrixD a, MatrixD b)
{ return Vector3D.DistanceSquared(a.Translation, b.Translation) < .000001 && Vector3D.Dot(a.Forward, b.Forward) > .9999995 && Vector3D.Dot(a.Up, b.Up) > .9999995; }


static Vector3D Rotate(Vector3D value, Vector3D rotation)
{
    double angle = rotation.Length();
    if (angle < 1e-12) return value;
    Vector3D axis = rotation / angle;
    return value * Math.Cos(angle) + Vector3D.Cross(axis, value) * Math.Sin(angle) +
        axis * Vector3D.Dot(axis, value) * (1 - Math.Cos(angle));
}

static bool Orthonormalise(ref Vector3D f, ref Vector3D u)
{
    double fn = f.Length();
    if (!Finite(fn) || fn < 1e-10) return false;
    f /= fn;
    u -= f * Vector3D.Dot(u, f);
    double un = u.Length();
    if (!Finite(un) || un < 1e-10) return false;
    u /= un;
    return true;
}

static Vector3D Unit(Vector3D value)
{
    double n = value.Length();
    if (!Finite(n) || n < 1e-12) throw new Exception("Invalid zero/non-finite vector.");
    return value / n;
}





static double Clamp(double value, double lo, double hi) => Math.Max(lo, Math.Min(hi, value));

static double Wrap(double angle) => angle - Math.Floor((angle + Math.PI) / (2 * Math.PI)) * (2 * Math.PI);

static bool Finite(double value) => !double.IsNaN(value) && !double.IsInfinity(value);

static bool Finite(Vector3D value) => Finite(value.X) && Finite(value.Y) && Finite(value.Z);

static bool Unlimited(IMyMotorStator r)
{
    return r.LowerLimitRad < -1000000 && r.UpperLimitRad > 1000000;
}


static bool ReadDouble(string text, out double value) => double.TryParse(text, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out value) && Finite(value);


// Versioned unicast transport. Epochs survive recompilation; generations revoke
// outstanding commands on Stop. Messages never grant direct actuator access.
const string LinkTag = "AutoArm/5";
const string LinkSection = "Link";

sealed class ModuleLink
{
    readonly ArmCore P;
    string ArmName => P.ArmName;
    public IMyProgrammableBlock Peer;
    public readonly long Epoch;
    public long RemoteEpoch, Generation, Sequence, Received;
    public double Age;
    bool Persisted;
    public ModuleLink(ArmCore p)
    {
        P = p;
        var ini = LinkSaved(p.Storage); long epoch = ini.Get("AutoArm Link", "Epoch").ToInt64();
        if (epoch < 0 || epoch == long.MaxValue) throw new Exception("Invalid link epoch.");
        Epoch = epoch + 1;
        p.IGC.UnicastListener.SetMessageCallback(LinkTag);
    }
    public void Resolve(string name)
    {
        Peer = null;
        long id;
        if (name.StartsWith("@", StringComparison.Ordinal) && long.TryParse(name.Substring(1), out id))
        {
            Peer = P.GridTerminalSystem.GetBlockWithId(id) as IMyProgrammableBlock;
            if (Peer == null || Peer == P.Me || Peer.Closed || !Peer.IsFunctional || !Peer.IsSameConstructAs(P.Me)) throw new Exception("Module PB unavailable.");
            return;
        }
        var blocks = new List<IMyProgrammableBlock>(); P.GridTerminalSystem.GetBlocksOfType(blocks);
        foreach (var b in blocks) if (b.CustomName == name && !b.Closed && b.IsSameConstructAs(P.Me))
        { if (Peer != null) throw new Exception("Duplicate PB: " + name); Peer = b; }
        if (Peer == null || Peer == P.Me || !Peer.IsFunctional) throw new Exception("Missing/invalid PB: " + name);
    }
    public MyIni Packet(string operation)
    {
        var ini = new MyIni(); ini.Set(LinkSection, "Version", 5); ini.Set(LinkSection, "Arm", ArmName);
        ini.Set(LinkSection, "Epoch", Epoch); ini.Set(LinkSection, "Remote", RemoteEpoch);
        ini.Set(LinkSection, "Generation", Generation); ini.Set(LinkSection, "Sequence", ++Sequence);
        ini.Set(LinkSection, "Operation", operation); return ini;
    }
    public void Send(MyIni ini)
    {
        if (Peer == null || Peer.Closed || !Peer.IsFunctional || !Peer.IsSameConstructAs(P.Me)) throw new Exception("Module PB unavailable.");
        if (!Persisted) { var saved = LinkSaved(P.Storage); saved.Set("AutoArm Link", "Epoch", Epoch); P.Storage = saved.ToString(); Persisted = true; }
        string text = ini.ToString(); if (text.Length > 16000) throw new Exception("Module packet too large.");
        P.IGC.SendUnicastMessage(Peer.EntityId, LinkTag, text);
    }
    public bool Read(MyIGCMessage message, out MyIni ini)
    {
        ini = new MyIni(); var text = message.Data as string;
        return Peer != null && !Peer.Closed && Peer.IsFunctional && Peer.IsSameConstructAs(P.Me) && message.Source == Peer.EntityId && message.Tag == LinkTag &&
            text != null && text.Length <= 16000 && ini.TryParse(text) && ini.Get(LinkSection, "Version").ToInt32() == 5 && ini.Get(LinkSection, "Arm").ToString() == ArmName &&
            ini.Get(LinkSection, "Epoch").ToInt64() > 0 && ini.Get(LinkSection, "Sequence").ToInt64() > 0;
    }
}
static MyIni LinkSaved(string text)
{
    var ini = new MyIni(); if (!ini.TryParse(text ?? "")) { ini.Clear(); ini.EndContent = text ?? ""; } return ini;
}
static void WriteVector(MyIni ini, string key, Vector3D value)
{ ini.Set(LinkSection, key, value.X.ToString("R", System.Globalization.CultureInfo.InvariantCulture) + " " + value.Y.ToString("R", System.Globalization.CultureInfo.InvariantCulture) + " " + value.Z.ToString("R", System.Globalization.CultureInfo.InvariantCulture)); }
static Vector3D LinkVector(MyIni ini, string key)
{
    string[] cells = ini.Get(LinkSection, key).ToString().Split(' '); double x, y, z;
    if (cells.Length != 3 || !ReadDouble(cells[0], out x) || !ReadDouble(cells[1], out y) || !ReadDouble(cells[2], out z)) throw new Exception("Invalid module vector: " + key);
    return new Vector3D(x, y, z);
}
static void WriteFrame(MyIni ini, string key, MatrixD frame)
{ WriteVector(ini, key + "P", frame.Translation); WriteVector(ini, key + "F", frame.Forward); WriteVector(ini, key + "U", frame.Up); }
static MatrixD LinkFrame(MyIni ini, string key)
{
    var p = LinkVector(ini, key + "P"); var f = LinkVector(ini, key + "F"); var u = LinkVector(ini, key + "U");
    if (Math.Abs(f.LengthSquared() - 1) > .001 || Math.Abs(u.LengthSquared() - 1) > .001 || Math.Abs(Vector3D.Dot(f, u)) > .001 || !Orthonormalise(ref f, ref u)) throw new Exception("Invalid module frame.");
    return MatrixD.CreateWorld(p, f, u);
}

static string PathRow(MatrixD frame)
{
    var values = new double[] { frame.Translation.X, frame.Translation.Y, frame.Translation.Z, frame.Forward.X, frame.Forward.Y, frame.Forward.Z, frame.Up.X, frame.Up.Y, frame.Up.Z };
    var row = new List<string>(); foreach (double v in values) row.Add(AutoSettings.Number(v)); return string.Join(" ", row);
}
