// Startup recovery runs before host configuration and again for each new core.
// Saved identities include parked tools which current discovery may not reach.
public static class SavedDrives
{
    public static void Stop(MyIni state, string arm, IMyProgrammableBlock me, IMyGridTerminalSystem terminal)
    {
        const string section = "AutoArm Config";
        if (state.Get(section, "Format").ToInt32() != 1 || state.Get(section, "PB").ToInt64() != me.EntityId || state.Get(section, "Arm").ToString() != arm) return;
        var ids = new HashSet<long>();
        foreach (string row in (state.Get(section, "Rows").ToString() + "\n" + state.Get("AutoArm Tool Weights", "Rows").ToString()).Split('\n'))
        {
            string[] cells = row.Split('|'); string key = cells.Length == 2 ? cells[1] : cells.Length == 3 ? cells[0] : "";
            foreach (string part in key.Split(':'))
            {
                string token = part.Split('@')[0]; long id;
                if (token.Length < 2 || (token[token.Length - 1] != '+' && token[token.Length - 1] != '-') || !long.TryParse(token.Substring(0, token.Length - 1), out id) || !ids.Add(id)) continue;
                try
                {
                    var block = terminal.GetBlockWithId(id);
                    if (block == null || block.Closed || !block.IsSameConstructAs(me)) continue;
                    var piston = block as IMyPistonBase; var rotor = block as IMyMotorStator;
                    if (piston != null) piston.Velocity = 0; else if (rotor != null && !(rotor is IMyMotorSuspension)) rotor.TargetVelocityRad = 0;
                }
                catch { }
            }
        }
    }
}
