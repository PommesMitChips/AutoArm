public Program()
{
    Runtime.UpdateFrequency = UpdateFrequency.Update10;
    IGC.UnicastListener.SetMessageCallback(WireTag);
    try
    {
        if (!Saved.TryParse(Storage)) { Saved.Clear(); Saved.EndContent = Storage; }
        if (Saved.Get("AutoArm Host", "PB").ToInt64() == Me.EntityId) { InputClock = Saved.Get("AutoArm Host", "InputClock").ToInt64(); InputOwner = Saved.Get("AutoArm Host", "InputOwner").ToInt64(); }
        if (!ToolHost) { Selection = IGC.RegisterBroadcastListener(SelectionTag); if (Selection != null) Selection.SetMessageCallback(WireTag); }
        if (!ToolHost) { PilotListener = IGC.RegisterBroadcastListener(PilotTag); if (PilotListener != null) PilotListener.SetMessageCallback(WireTag); Pairing = Saved.Get("AutoArm Host", "PB").ToInt64() == Me.EntityId && Saved.Get("AutoArm Host", "Pairing").ToBoolean(); }
        if (!ToolHost) StopSaved();
        LoadHost();
        if (!ToolHost && Saved.Get("AutoArm Host", "PB").ToInt64() == Me.EntityId) foreach (string name in Order) {
            long seat = Saved.Get("Pilot " + name, "Seat").ToInt64(), clock = Saved.Get("Pilot " + name, "Clock").ToInt64();
            if (seat != 0 && clock > 0) Pilots[name] = new PilotClaim { Seat = seat, Clock = clock, PB = Me.EntityId, Arm = name };
        }
        string prior = Saved.Get("AutoArm Host", "Selected").ToString();
        if (Saved.Get("AutoArm Host", "PB").ToInt64() == Me.EntityId && Arms.ContainsKey(prior)) Selected = prior;
        if (Order.Count > 0) SelectArm(Selected.Length == 0 ? Order[0] : Selected, InputOwner == 0);
        Save();
    }
    catch (Exception e) { StopAll(e.Message); }
    Display();
}
public void Save()
{
    if (Order.Count == 0 && Configuration.Get(GlobalSection, "Format").ToInt32() != HostFormat) return;
    bool changed = Saved.Get("AutoArm Host", "PB").ToInt64() != Me.EntityId || Saved.Get("AutoArm Host", "Selected").ToString() != Selected || Saved.Get("AutoArm Host", "InputClock").ToInt64() != InputClock || Saved.Get("AutoArm Host", "InputOwner").ToInt64() != InputOwner;
    if (changed) { Saved.Set("AutoArm Host", "Format", 1); Saved.Set("AutoArm Host", "PB", Me.EntityId); Saved.Set("AutoArm Host", "Selected", Selected); Saved.Set("AutoArm Host", "InputClock", InputClock); Saved.Set("AutoArm Host", "InputOwner", InputOwner); }
    if (!ToolHost) {
        if (Saved.Get("AutoArm Host", "Pairing").ToBoolean() != Pairing) { Saved.Set("AutoArm Host", "Pairing", Pairing); changed = true; }
        foreach (string name in Order) {
            PilotClaim pilot; long seat = Pilots.TryGetValue(name, out pilot) ? pilot.Seat : 0, clock = pilot == null ? 0 : pilot.Clock;
            if (Saved.Get("Pilot " + name, "Seat").ToInt64() != seat || Saved.Get("Pilot " + name, "Clock").ToInt64() != clock) { Saved.Set("Pilot " + name, "Seat", seat); Saved.Set("Pilot " + name, "Clock", clock); changed = true; }
        }
    }
    foreach (var pair in Arms) if (pair.Value.Storage != pair.Value.SavedData) { Saved.Set("State " + pair.Key, "Data", pair.Value.Storage); pair.Value.SavedData = pair.Value.Storage; changed = true; }
    if (!changed) return;
    string text = Saved.ToString(); if (text != Storage) Storage = text;
}
void StopSaved()
{
    var sections = new List<string>(); Saved.GetSections(sections);
    foreach (string section in sections)
    {
        if (!section.StartsWith("State ", StringComparison.Ordinal)) continue;
        var state = new MyIni(); if (!state.TryParse(Saved.Get(section, "Data").ToString())) continue;
        ArmCore.SavedDrives.Stop(state, section.Substring(6), Me, GridTerminalSystem);
    }
}
void LoadHost()
{
    var ini = new MyIni();
    if (string.IsNullOrWhiteSpace(Me.CustomData))
    {
        ini.Set(GlobalSection, "Format", HostFormat); ini.Set(GlobalSection, "SelectedArm", DefaultArm);
        var bases = new List<IMyMechanicalConnectionBlock>(); GridTerminalSystem.GetBlocksOfType(bases); var detected = new List<string>();
        foreach (var block in bases)
        {
            int split = block.CustomName.IndexOf(" - Base", StringComparison.Ordinal);
            if (split > 0 && !block.Closed && block.IsSameConstructAs(Me) && !detected.Contains(block.CustomName.Substring(0, split))) detected.Add(block.CustomName.Substring(0, split));
        }
        if (detected.Count == 0) detected.Add(DefaultArm); detected.Sort(StringComparer.Ordinal);
        if (ToolHost)
        {
            var blocks = new List<IMyTerminalBlock>(); GridTerminalSystem.GetBlocks(blocks);
            detected.RemoveAll(name => !blocks.Any(block => block.CustomName.StartsWith(name + " - Head ", StringComparison.Ordinal) && block.CustomName.Substring((name + " - Head ").Length).IndexOf(" - ", StringComparison.Ordinal) > 0));
            if (detected.Count == 0) detected.Add(DefaultArm);
        }
        foreach (string name in detected) ini.Set(name, ToolHost ? "Tools.Enabled" : "ToolSwapPB", ToolHost ? "true" : "");
        Me.CustomData = ini.ToString();
    }
    if (!ini.TryParse(Me.CustomData) || ini.Get(GlobalSection, "Format").ToInt32() != HostFormat)
        throw new Exception("Custom Data requires [global] Format=" + HostFormat + ". Delete Custom Data.");
    var names = new List<string>(); ini.GetSections(names); names.Remove(GlobalSection);
    if (names.Count > 8) throw new Exception("Configure at most 8 named arm sections.");
    foreach (string name in names) if (name.Trim() != name || name.Length == 0 || name.IndexOfAny(new char[] { ',', '|', '(', ')', ';', '\r', '\n' }) >= 0 || name.Equals(GlobalSection, StringComparison.OrdinalIgnoreCase)) throw new Exception("Invalid arm section name.");
    if (!ToolHost)
    {
        var peers = new List<IMyProgrammableBlock>(); GridTerminalSystem.GetBlocksOfType(peers);
        foreach (var peer in peers)
        {
            if (peer == Me || peer.Closed || !peer.IsSameConstructAs(Me)) continue;
            var data = new MyIni(); if (!data.TryParse(peer.CustomData) || data.Get(GlobalSection, "Format").ToInt32() != 7) continue;
            foreach (string name in names) if (data.ContainsSection(name)) throw new Exception("Arm already assigned to another PB: " + name);
        }
    }
    var removed = new List<string>(); foreach (string name in Order) if (!names.Contains(name)) removed.Add(name);
    foreach (string name in removed) { Arms[name].Stop("Arm removed from configuration."); Saved.Set("State " + name, "Data", Arms[name].Storage); Arms.Remove(name); if (!ToolHost) Pilots.Remove(name); }
    Configuration = ini; Order.Clear(); Order.AddRange(names);
    AutoLinks();
    foreach (string name in names) if (!Arms.ContainsKey(name))
    {
        string state = Saved.Get("AutoArm Host", "PB").ToInt64() == Me.EntityId ? Saved.Get("State " + name, "Data").ToString() : "";
        Arms.Add(name, new ArmCore(this, name, ConfigFor(name), state));
    }
    if (!Arms.ContainsKey(Selected))
    { string configured = ini.Get(GlobalSection, "SelectedArm").ToString(); Selected = Arms.ContainsKey(configured) ? configured : names.Count == 0 ? "" : names[0]; }
    if (!ToolHost) UpdatePilots();
    HostError = "";
}

void AutoLinks()
{
    var peers = new List<IMyProgrammableBlock>(); GridTerminalSystem.GetBlocksOfType(peers);
    foreach (string name in Order)
    {
        string key = ToolHost ? "Link.ArmPB" : "ToolSwapPB";
        if (Effective(name, key).Length > 0) continue;
        IMyProgrammableBlock found = null;
        foreach (var peer in peers)
        {
            if (peer == Me || peer.Closed || !peer.IsFunctional || !peer.IsSameConstructAs(Me)) continue;
            var data = new MyIni();
            if (!data.TryParse(peer.CustomData) || data.Get(GlobalSection, "Format").ToInt32() != (ToolHost ? 7 : 2) || !data.ContainsSection(name)) continue;
            if (!ToolHost && !data.Get(data.ContainsKey(name, "Tools.Enabled") ? name : GlobalSection, "Tools.Enabled").ToBoolean(true)) continue;
            if (found != null) throw new Exception("Multiple module PBs for " + name + ". Set " + key + ".");
            found = peer;
        }
        if (found != null) Configuration.Set(name, key, "@" + found.EntityId);
    }
}
MyIni ConfigFor(string name)
{
    var ini = new MyIni();
    var keys = new List<MyIniKey>(); Configuration.GetKeys(GlobalSection, keys);
    var own = new List<MyIniKey>(); Configuration.GetKeys(name, own); keys.AddRange(own);
    var seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
    foreach (var item in keys)
    {
        string key = item.Name; if (!seen.Add(key) || key.Equals("Format", StringComparison.OrdinalIgnoreCase) || key.Equals("SelectedArm", StringComparison.OrdinalIgnoreCase) || key.Equals("Structure", StringComparison.OrdinalIgnoreCase) || key.Equals("Actuators", StringComparison.OrdinalIgnoreCase)) continue;
        string section = "Config", field = key; int dot = key.IndexOf('.');
        if (dot > 0) { section = key.Substring(0, dot); field = key.Substring(dot + 1); }
        else if (key.Equals("ToolSwapPB", StringComparison.OrdinalIgnoreCase) || key.Equals("Peers", StringComparison.OrdinalIgnoreCase)) section = "Modules";
        ini.Set(section, field, Effective(name, key));
    }
    if (ToolHost)
    {
        if (!ini.ContainsKey("Link", "ArmPB")) ini.Set("Link", "ArmPB", name + " - Arm PB");
        if (!ini.ContainsKey("Tools", "Enabled")) ini.Set("Tools", "Enabled", true);
    }
    return ini;
}
bool Layout(ArmCore core, out string why)
{
    why = ""; string name = core.Name, table = Effective(name, "Actuators"), expected = Effective(name, "Structure");
    bool shared = !Configuration.ContainsKey(name, "Actuators") && Configuration.ContainsKey(GlobalSection, "Actuators") && table.Trim().Length > 0;
    if (shared && Configuration.ContainsKey(name, "Structure")) { why = "A local Structure requires local Actuators; clear it to inherit the global template."; return false; }
    if (shared && expected.Length == 0)
    {
        ArmCore prototype;
        if (!Arms.TryGetValue(Order[0], out prototype) || !prototype.HasLayout) { why = "Waiting for global structure prototype: " + Order[0]; return false; }
        expected = prototype.Shape; Configuration.Set(GlobalSection, "Structure", expected);
    }
    string shape = core.Shape;
    if (expected.Length > 0 && expected != shape) { why = name + " does not match configured joint kinds/parallel layout."; return false; }
    if (!core.ApplyWeights(table, out why)) return false;
    foreach (var joint in core.Joints())
    {
        ArmCore owner;
        if (Owners.TryGetValue(joint.EntityId, out owner) && owner != core && Arms.ContainsKey(owner.Name))
        { owner.Stop("Actuator also claimed by " + name); why = "Actuator shared with " + owner.Name + ": " + joint.CustomName; return false; }
    }
    var release = new List<long>(); foreach (var pair in Owners) if (pair.Value == core) release.Add(pair.Key);
    foreach (long id in release) Owners.Remove(id);
    foreach (var joint in core.Joints()) Owners[joint.EntityId] = core;
    if (!shared && table.Trim().Length == 0) Configuration.Set(name, "Actuators", core.WeightTable);
    PublishConfig(); return true;
}
void PublishConfig()
{
    // Generated values only fill missing keys. Preserve edits made while running.
    var live = new MyIni(); if (!live.TryParse(Me.CustomData) || live.Get(GlobalSection, "Format").ToInt32() != HostFormat) return;

    foreach (var key in keys) if (!live.ContainsKey(key.Section, key.Name)) live.Set(key.Section, key.Name, Configuration.Get(key).ToString());
    string text = live.ToString(); if (text != Me.CustomData) Me.CustomData = text;
}

void Command(string raw)
{
    string text = raw.Trim(); if (text.Length == 0) return;
    HostError = "";
    if (!ToolHost)
    {
        if(text.StartsWith("Claim ",StringComparison.OrdinalIgnoreCase)){ClaimArm(text.Substring(6).Trim());return;}
        if(text.StartsWith("Unclaim ",StringComparison.OrdinalIgnoreCase)){UnclaimArm(text.Substring(8).Trim());return;}
    }
    if (text.StartsWith("Select ", StringComparison.OrdinalIgnoreCase)) { SelectArm(text.Substring(7).Trim(), true, true); return; }
    if (text.StartsWith("Release ", StringComparison.OrdinalIgnoreCase)) { if (ToolHost) throw new Exception("Release is an Arm PB command."); else { ReleasePilot(text.Substring(8).Trim()); return; } }
    if (text.Equals("StopAll", StringComparison.OrdinalIgnoreCase)) { StopAll("StopAll command."); return; }
    int end = text.IndexOfAny(new char[] { ' ', '(', ',' }); string first = (end < 0 ? text : text.Substring(0, end)).ToLowerInvariant();
    if (first == "on" || first == "reload" || first == "rescan")
    { try { LoadHost(); if (!ToolHost && InputOwner == 0) SelectArm(Selected); } catch (Exception e) { StopAll(e.Message); throw; } }
    string name = Selected, command = text; int open = text.IndexOf('(');
    if (open > 0)
    {
        if (!text.EndsWith(")", StringComparison.Ordinal)) throw new Exception("Command needs closing parenthesis.");
        string verb = text.Substring(0, open).Trim(); var values = new List<string>();
        foreach (string value in text.Substring(open + 1, text.Length - open - 2).Split(',')) if (value.Trim().Length > 0) values.Add(value.Trim());
        if (values.Count > 0 && Arms.ContainsKey(values[values.Count - 1])) { name = values[values.Count - 1]; values.RemoveAt(values.Count - 1); }
        else if (values.Count > Arity(verb)) throw new Exception("Unknown arm argument.");
        command = verb + (values.Count == 0 ? "" : " " + string.Join(" ", values));
    }
    else
    {
        int comma = text.LastIndexOf(',');
        if (comma >= 0) { name = text.Substring(comma + 1).Trim(); command = text.Substring(0, comma).Trim(); }
    }
    ArmCore target; if (!Arms.TryGetValue(name, out target)) throw new Exception("Unknown arm: " + name);
    string action = command.Split(' ')[0].ToLowerInvariant();
    if (action == "select") { int space = command.IndexOf(' '); string mode = space < 0 ? "" : command.Substring(space + 1).Trim().ToUpperInvariant(); if (mode.Length > 0 && mode != "HEAD" && mode != "HRZ" && mode != "VRT") throw new Exception("MovementMode expects HEAD, HRZ or VRT."); SelectArm(name, true, true); if (mode.Length > 0) target.Update("Mode " + mode, UpdateType.Terminal, 0); return; }
    if (action == "release") { if (ToolHost) throw new Exception("Release is an Arm PB command."); else { ReleasePilot(name); return; } }
    if (action == "on" || action == "reload" || action == "rescan")
    { target.Config = ConfigFor(name); target.ConfigChanged = false; }
    target.Update(command, UpdateType.Terminal, 0);
    CaptureDefaults(target);
}
static int Arity(string verb)
{
    switch (verb.ToLowerInvariant())
    {
        case "on": case "off": case "stop": case "hold": case "check": case "info": case "joints": case "reload": case "rescan": case "sethome": case "gohome": case "toolscan": case "toolinfo": case "swapcancel": case "version": case "status": case "bindings": case "homeinfo": case "forward": case "back": case "left": case "right": case "up": case "down": case "yawleft": case "yawright": case "pitchup": case "pitchdown": case "rollleft": case "rollright": return 0;
        case "moveto": return 3;
        case "release": return 0;
        case "orientto": return 6;
        case "movetoposori": return 9;
        case "homespeed": case "step": return 2;
        default: return 1;
    }
}
void AnnounceInput()
{
    if (Selection != null) IGC.SendBroadcastMessage(SelectionTag, InputClock + "|" + Selected, TransmissionDistance.CurrentConstruct);
    if (PilotListener != null) { var text = new System.Text.StringBuilder(InputClock.ToString()); foreach (var pilot in Pilots.Values) text.Append('|').Append(pilot.Arm).Append('@').Append(pilot.Seat).Append('@').Append(pilot.Clock); IGC.SendBroadcastMessage(PilotTag, text.ToString(), TransmissionDistance.CurrentConstruct); }
}

bool SeatAvailableFor(string name, long seat)
{
    foreach (var pilot in Pilots.Values) if (pilot.Seat == seat && pilot.Arm != name) return false;
    PilotClaim remote; return !RemotePilots.TryGetValue(seat, out remote) || remote.Age > .5;
}
void UpdatePilots()
{ foreach (var core in Arms.Values) core.Select(Pilots.ContainsKey(core.Name) || !Pairing && core.Name == Selected && InputOwner == Me.EntityId); }
void ReleasePilot(string name)
{
    if (!Arms.ContainsKey(name)) throw new Exception("Unknown arm: " + name);
    Pairing = true; Pilots.Remove(name); InputClock++; UpdatePilots(); AnnounceInput(); Save(); InputNote = name + ": cockpit released; arm holds. Select pairs again.";
}
void PairPilot(string name)
{
    var seats = new List<IMyShipController>(); GridTerminalSystem.GetBlocksOfType(seats, c => !c.Closed && c.IsSameConstructAs(Me) && c.IsUnderControl);
    if (seats.Count > 1) throw new Exception("More than one cockpit is occupied. Please ensure exactly one cockpit is occupied.");
    if (seats.Count == 0) throw new Exception("No cockpit is occupied. Please ensure exactly one cockpit is occupied.");
    var found = seats[0]; Pairing = true;
    var remove = new List<string>(); foreach (var pilot in Pilots.Values) if (pilot.Seat == found.EntityId) remove.Add(pilot.Arm); foreach (string old in remove) { Pilots.Remove(old); Arms[old].Select(false); } Arms[name].Select(false);
    RemotePilots.Remove(found.EntityId); InputClock++; Pilots[name] = new PilotClaim { Arm = name, Seat = found.EntityId, Clock = InputClock, PB = Me.EntityId };
    InputNote = name + " paired to " + found.CustomName + " (" + found.EntityId + ").";
}
void ReadPilotClaims(double elapsed)
{
    foreach (var pilot in RemotePilots.Values) if (elapsed > 0 && !double.IsNaN(elapsed) && !double.IsInfinity(elapsed)) pilot.Age += elapsed;
    if (PilotListener == null) return; int count = 0;
    while (PilotListener.HasPendingMessage && count++ < 8) {
        var message = PilotListener.AcceptMessage(); var text = message.Data as string; var peer = GridTerminalSystem.GetBlockWithId(message.Source) as IMyProgrammableBlock;
        if (message.Tag != PilotTag || message.Source == Me.EntityId || text == null || text.Length > 2048 || peer == null || peer.Closed || !peer.IsFunctional || !peer.IsSameConstructAs(Me)) continue;
        var config = new MyIni(); if (!config.TryParse(peer.CustomData) || config.Get(GlobalSection, "Format").ToInt32() != 7) continue;
        string[] rows = text.Split('|'); long snapshot, previous;
        if (rows.Length > 9 || !long.TryParse(rows[0], out snapshot) || snapshot <= 0 || PilotSnapshots.TryGetValue(message.Source, out previous) && snapshot < previous) continue;
        var incoming = new List<PilotClaim>(); var seen = new HashSet<long>(); var names = new HashSet<string>(); bool valid = true;
        for (int i = 1; i < rows.Length; i++) {
            int last = rows[i].LastIndexOf('@'), middle = last > 0 ? rows[i].LastIndexOf('@', last - 1) : -1; long seat, clock;
            string arm = middle < 0 ? "" : rows[i].Substring(0, middle);
            if (middle <= 0 || !long.TryParse(rows[i].Substring(middle + 1, last - middle - 1), out seat) || !long.TryParse(rows[i].Substring(last + 1), out clock) || seat == 0 || clock <= 0 || clock > snapshot || !seen.Add(seat) || !names.Add(arm) || !config.ContainsSection(arm)) { valid = false; break; }
            var controller = GridTerminalSystem.GetBlockWithId(seat) as IMyShipController;
            if (controller == null || controller.Closed || !controller.IsSameConstructAs(Me)) { valid = false; break; }
            incoming.Add(new PilotClaim { PB = message.Source, Seat = seat, Clock = clock, Arm = arm });
        }
        if (!valid) continue; PilotSnapshots[message.Source] = snapshot; InputClock = Math.Max(InputClock, snapshot);
        var erase = new List<long>(); foreach (var pilot in RemotePilots.Values) if (pilot.PB == message.Source) erase.Add(pilot.Seat); foreach (long seat in erase) RemotePilots.Remove(seat);
        foreach (var claim in incoming) {
            PilotClaim remote; if (RemotePilots.TryGetValue(claim.Seat, out remote) && (remote.Clock > claim.Clock || remote.Clock == claim.Clock && remote.PB > claim.PB)) continue;
            string lost = ""; bool keep = false; foreach (var local in Pilots.Values) if (local.Seat == claim.Seat) { if (local.Clock > claim.Clock || local.Clock == claim.Clock && Me.EntityId > claim.PB) keep = true; else lost = local.Arm; break; }
            if (keep) continue; if (lost.Length > 0) { Pilots.Remove(lost); InputNote = lost + ": cockpit claimed by another Arm PB."; }
            RemotePilots[claim.Seat] = claim;
        }
        UpdatePilots();
    }
}
void ReceiveSelection(double elapsed)
{
    if (Selection == null) return;
    int read = 0;
    while (Selection.HasPendingMessage && read++ < 8)
    {
        var message = Selection.AcceptMessage(); string text = message.Data as string; long clock;
        if (message.Tag != SelectionTag || message.Source == Me.EntityId || text == null || text.Length > 512) continue;
        string[] cells = text.Split('|');
        if (cells.Length != 2 || !long.TryParse(cells[0], out clock) || clock <= 0 || clock < InputClock || clock == InputClock && message.Source <= InputOwner) continue;
        var peer = GridTerminalSystem.GetBlockWithId(message.Source) as IMyProgrammableBlock; var config = new MyIni();
        if (peer == null || peer.Closed || !peer.IsFunctional || !peer.IsSameConstructAs(Me) || !config.TryParse(peer.CustomData) || config.Get(GlobalSection, "Format").ToInt32() != 7 || !config.ContainsSection(cells[1])) continue;
        InputClock = clock; InputOwner = message.Source;
        UpdatePilots();
    }
    ReadPilotClaims(elapsed);
    if (elapsed > 0 && !double.IsInfinity(elapsed) && !double.IsNaN(elapsed)) InputAge += elapsed;
    if (InputAge >= .1) { InputAge = 0; if (InputOwner == Me.EntityId || Pilots.Count > 0) AnnounceInput(); }
}
void CaptureDefaults(ArmCore core)
{
    if (!core.ConfigChanged) return;
    core.ConfigChanged = false;
    var ini = core.Config;
    var keys = new List<MyIniKey>(); ini.GetKeys(keys);
    foreach (var key in keys)
    {
        if (key.Section == "AutoArm" || key.Name == "Actuators") continue;
        string field = key.Section == "Config" || key.Section == "Modules" ? key.Name : key.Section + "." + key.Name;
        if (!Configuration.ContainsKey(core.Name, field) && !Configuration.ContainsKey(GlobalSection, field))
            Configuration.Set(core.SharedSetting(field) ? GlobalSection : core.Name, field, ini.Get(key).ToString());
    }
    PublishConfig();
}
public void Main(string argument, UpdateType source)
{
    double elapsed = Runtime.TimeSinceLastRun.TotalSeconds;
    try { if (!ToolHost) ReceiveSelection(elapsed); } catch (Exception e) { StopAll(e.Message); }
    try
    {
        if (!string.IsNullOrWhiteSpace(argument) && !((source & UpdateType.IGC) != 0 && argument == WireTag))
            foreach (string raw in argument.Split(new char[] { ';', '\n' }, StringSplitOptions.RemoveEmptyEntries)) Command(raw);
    }
    catch (Exception e) { HostError = e.Message; }
    try
    {
        int read = 0;
        while (IGC.UnicastListener.HasPendingMessage && read++ < 16)
        {
            if (Runtime.MaxInstructionCount > 0 && Runtime.CurrentInstructionCount >= Runtime.MaxInstructionCount * .4) break;
            var message = IGC.UnicastListener.AcceptMessage(); var data = message.Data as string; var ini = new MyIni(); ArmCore core;
            if (message.Tag != WireTag || data == null || data.Length > 16000 || !ini.TryParse(data) || !Arms.TryGetValue(ini.Get("Link", "Arm").ToString(), out core)) continue;
            if (core.Inbox.Count < 32) core.Inbox.Enqueue(message);
        }
        int count = Order.Count;
        for (int i = 0; i < count; i++)
        {
            if (Runtime.MaxInstructionCount > 0 && Runtime.CurrentInstructionCount >= Runtime.MaxInstructionCount * .7) { StopAll("Shared PB instruction budget exhausted."); break; }
            var core = Arms[Order[(Next + i) % count]]; core.Update("", source, elapsed); CaptureDefaults(core);
        }
        if (count > 0) Next = (Next + 1) % count;
        Save();
    }
    catch (Exception e) { StopAll(e.Message); }
    finally
    {
        UpdateFrequency frequency = UpdateFrequency.Update10;
        foreach (var core in Arms.Values) if ((core.Frequency & UpdateFrequency.Update1) != 0) frequency = UpdateFrequency.Update1;
        Runtime.UpdateFrequency = frequency; Display();
    }
}
void StopAll(string why)
{ HostError = why; foreach (var core in Arms.Values) core.Stop(why); }
void Display()
{
    var lines = new List<string>(); lines.Add("AutoArm " + ControllerBuildVersion + " " + (ToolHost ? "ToolSwap" : "Arm") + " | Selected: " + Selected);
    if (!ToolHost && !Pairing && InputOwner != Me.EntityId) lines.Add("Cockpit input inactive here. Run Select " + Selected + ".");
    if (!ToolHost && InputNote.Length > 0) lines.Add(InputNote);
    if (HostError.Length > 0) lines.Add(HostError);
    foreach (string name in Order) lines.Add(name + ": " + Arms[name].Summary);
    ArmCore current; if (Arms.TryGetValue(Selected, out current)) lines.Add(current.Report);
    Echo(string.Join("\n", lines));
}
