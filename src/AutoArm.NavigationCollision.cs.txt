// Experimental shared passing lanes. Existing full-arm inequalities remain
// authoritative. Only non-linear automatic paths advertise consent to detours.
sealed class PassingLane {
    public string Key, Peer;
    public int Stage, Trials;
    public bool Planned;
    public double Extra;
    public readonly List<Vector3D> Points = new List<Vector3D>();
}
void ReadNavigation(Arm arm, MyIni ini, Arm previous)
{
    if (!ini.ContainsKey(Section, "NavVersion")) return;
    if (ini.Get(Section, "NavVersion").ToInt32() != 1 || !ini.Get(Section, "NavAllowed").TryGetBoolean(out arm.NavAllowed)) throw new Exception("Invalid navigation consent.");
    if (!arm.NavAllowed) return;
    arm.NavKey = ini.Get(Section, "NavKey").ToString(); var head = Numbers(ini.Get(Section, "NavHead").ToString(), 3); var goal = Numbers(ini.Get(Section, "NavGoal").ToString(), 3);
    arm.NavHead = new Vector3D(head[0], head[1], head[2]); arm.NavGoal = new Vector3D(goal[0], goal[1], goal[2]);
    if (arm.NavKey.Length == 0 || arm.NavKey.Length > 80 || !ini.Get(Section, "NavSpeed").TryGetDouble(out arm.NavSpeed) || !Finite(arm.NavSpeed) || arm.NavSpeed <= 0 || arm.NavSpeed > 5 || Vector3D.Distance(arm.NavHead, arm.NavGoal) > 1000) throw new Exception("Invalid navigation demand.");
    if (previous.Epoch == arm.Epoch && previous.Fingerprint == arm.Fingerprint && previous.NavKey == arm.NavKey) arm.Lane = previous.Lane;
}
double HeadExtent(Arm arm, Vector3D axis)
{
    GridShape grid; if (!Scene.TryGetValue(arm.Tool, out grid) || !grid.Ready) return 1000;
    double reach = 0; var axes = Axes(grid.Grid.WorldMatrix);
    foreach (var box in grid.Shapes) reach = Math.Max(reach, Math.Abs(Vector3D.Dot(Vector3D.Transform(box.Centre, grid.Grid.WorldMatrix) - arm.NavHead, axis)) + Radius(box, axes, axis));
    return reach;
}
Vector3D PassingSide(Arm arm, Vector3D direction)
{
    // Project translation preferences through the angular null space. This
    // avoids choosing a depth lane an orientation-locked planar arm cannot use.
    int n = arm.Groups.Count; var linear = new Vector3D[n]; var angular = new Vector3D[n];
    var a = new Vector3D(1e-8, 0, 0); var b = new Vector3D(0, 1e-8, 0); var c = new Vector3D(0, 0, 1e-8);
    for (int i = 0; i < n; i++) {
        var g = arm.Groups[i]; var m = g.Members[0]; var axis = m.Block.WorldMatrix.Up * ((m.Block.CubeGrid.EntityId == m.From ? 1 : -1) * m.Sign);
        angular[i] = g.Rotary ? -axis * arm.Caps[i] : Vector3D.Zero;
        linear[i] = g.Rotary ? Vector3D.Cross(angular[i], arm.NavHead - m.Block.GetPosition()) : axis * arm.Caps[i];
        var w = angular[i]; a += w * w.X; b += w * w.Y; c += w * w.Z;
    }
    double determinant = Vector3D.Dot(a, Vector3D.Cross(b, c)), best = 0; Vector3D chosen = Vector3D.Zero;
    foreach (var axis in Axes(Scene[arm.Hull].Grid.WorldMatrix)) {
        var side = axis - direction * Vector3D.Dot(axis, direction); if (side.LengthSquared() < .001) continue; side.Normalize();
        Vector3D rhs = Vector3D.Zero;
        for (int i = 0; i < n; i++) rhs += angular[i] * Vector3D.Dot(linear[i], side);
        var multiplier = determinant > 1e-28 ? new Vector3D(Vector3D.Dot(rhs, Vector3D.Cross(b, c)), Vector3D.Dot(a, Vector3D.Cross(rhs, c)), Vector3D.Dot(a, Vector3D.Cross(b, rhs))) / determinant : Vector3D.Zero;
        double score = 0;
        for (int i = 0; i < n; i++) { double v = Vector3D.Dot(linear[i], side) - Vector3D.Dot(angular[i], multiplier); score += v * v; }
        if (score > best) { best = score; chosen = side; }
    }
    return best > 1e-7 ? chosen : Vector3D.Zero;
}
bool ClearHeadStep(Arm arm, Vector3D target)
{
    var body = Scene[arm.Tool]; var delta = Vector3D.TransformNormal(target - arm.NavHead, MatrixD.Transpose(body.Grid.WorldMatrix.GetOrientation()));
    foreach (var box in body.Shapes) {
        var sweep = new Shape { Centre = box.Centre + delta * .5, Half = box.Half + Abs(delta) * .5 + new Vector3D(.02), Tag = box.Tag };
        foreach (var obstacle in Scene.Values) {
            if (arm.Bodies.ContainsKey(obstacle.Grid.EntityId)) continue;
            if (!Budget()) return false;
            try {
                foreach (var other in Candidates(body, sweep, obstacle, Clearance)) {
                    if (!Budget()) return false;
                    Vector3D normal; if (Separation(body, sweep, obstacle, other, out normal) < Clearance) return false;
                }
            } catch (Exception e) { if (e.Message.IndexOf("budget", StringComparison.OrdinalIgnoreCase) < 0) throw; return false; }
        }
    }
    return true;
}
void PassingGuidance(Arm arm, MyIni response)
{
    response.Set(Section, "NavVersion", 1); response.Set(Section, "NavActive", false);
    if (!arm.NavAllowed || response.Get(Section, "Hold").ToBoolean() || !Scene.ContainsKey(arm.Hull) || !Scene.ContainsKey(arm.Tool)) return;
    if (Vector3D.Distance(arm.NavGoal, arm.NavHead) < .05) return;
    var basis = Scene[arm.Hull].Grid.WorldMatrix; var inverse = MatrixD.Invert(basis); var lane = arm.Lane;
    if (lane == null) {
        var delta = arm.NavGoal - arm.NavHead; double length = delta.Length(); if (length < .05) return; var direction = delta / length;
        Arm peer = null; double nearest = double.MaxValue;
        foreach (var other in Arms.Values) {
            if (other == arm || !other.NavAllowed || other.Hull != arm.Hull || Clock - other.SeenAt > .25 || string.CompareOrdinal(other.PB.EntityId + "/" + other.Name, arm.PB.EntityId + "/" + arm.Name) >= 0) continue;
            var difference = other.NavHead - arm.NavHead; double along = Vector3D.Dot(difference, direction);
            if (along <= 0 || along >= length || difference.Length() > 18) continue;
            var offset = difference - direction * along; var normal = offset.Length() > .01 ? offset / offset.Length() : basis.Up;
            if (offset.Length() <= HeadExtent(arm, normal) + HeadExtent(other, normal) + Influence && along < nearest) { nearest = along; peer = other; }
        }
        if (peer == null) return;
        lane = arm.Lane = new PassingLane { Key = arm.NavKey, Peer = peer.PB.EntityId + "/" + peer.Name };
    }
    Arm partner; if (!Arms.TryGetValue(lane.Peer, out partner) || Clock - partner.SeenAt > .25) { response.Set(Section, "Hold", true); response.Set(Section, "Reason", "Passing peer expired; hold."); return; }
    if (!lane.Planned && lane.Trials < 6 && Budget()) {
        var direction = arm.NavGoal - arm.NavHead; direction.Normalize(); var side = PassingSide(arm, direction);
        double width = HeadExtent(arm, side) + HeadExtent(partner, side) + Clearance + .5;
        int trial = lane.Trials++; double sign = trial % 2 == 0 ? -1 : 1, factor = 1 + trial / 2 * .5;
        var shift = side * (sign * width * factor); var first = arm.NavHead + shift;
        if (side.LengthSquared() > .9 && shift.Length() < 40 && ClearHeadStep(arm, first)) {
            lane.Points.Add(Vector3D.Transform(first, inverse)); lane.Points.Add(Vector3D.Transform(arm.NavGoal + shift, inverse)); lane.Points.Add(Vector3D.Transform(arm.NavGoal, inverse));
            lane.Extra = shift.Length() * 2; lane.Planned = true;
        }
    }
    if (lane.Planned && lane.Stage < 2 && Vector3D.Distance(arm.NavHead, Vector3D.Transform(lane.Points[lane.Stage], basis)) < .08) lane.Stage++;
    var wanted = lane.Planned ? Vector3D.Transform(lane.Points[lane.Stage], basis) : arm.NavHead;
    bool wait = !lane.Planned || !ClearHeadStep(arm, wanted);
    response.Set(Section, "NavActive", true); response.Set(Section, "NavKey", arm.NavKey); response.Set(Section, "NavStage", lane.Stage + 1);
    response.Set(Section, "NavPoint", Row(new double[] { wait ? arm.NavHead.X : wanted.X, wait ? arm.NavHead.Y : wanted.Y, wait ? arm.NavHead.Z : wanted.Z }));
    response.Set(Section, "NavWait", wait); response.Set(Section, "NavExtra", lane.Extra);
    response.Set(Section, "NavPeerRemaining", Vector3D.Distance(partner.NavHead, partner.NavGoal));
    response.Set(Section, "Reason", wait ? "Passing lane waiting; full collision guard remains active." : "Passing lane " + (lane.Stage + 1) + "/3; full collision guard remains active.");
}
