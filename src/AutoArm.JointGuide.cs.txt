// Prototype only: a Plan peer may send native joint waypoints alongside poses.
double[] GuidedRates, GuidedLo, GuidedHi;
void ReadJointGuide(ArmPath path, MyIni ini)
{
    string text = ini.Get(LinkSection, "Joints").ToString();
    if (text.Length == 0) return;
    if (Services == null || Services.Safety == null || !Services.Safety.Ready || Services.Safety.Blocked || path.Docking || path.Lines != null)
        throw new Exception("Joint routes require ready Safety and ordinary travel.");
    if (ini.Get(LinkSection, "PlanFingerprint").ToString() != PlanIdentity()) throw new Exception("Planning model changed.");
    var rows = text.Split('|'); int count = 0; foreach (var g in Topology.Groups) count += g.M.Count;
    if (rows.Length != path.Waypoints.Count) throw new Exception("Joint/pose waypoint count differs.");
    path.Joints = new double[rows.Length][];
    for (int k = 0; k < rows.Length; k++)
    {
        var words = rows[k].Trim().Split(new char[] { ' ' }, StringSplitOptions.RemoveEmptyEntries);
        if (words.Length != count) throw new Exception("Joint row size differs.");
        var values = path.Joints[k] = new double[count]; int i = 0;
        foreach (var g in Topology.Groups) foreach (var m in g.M)
        {
            double value, resolved;
            if (!ReadDouble(words[i], out value) || !ResolveHome(m, value, out resolved) || Math.Abs(resolved - value) > .00001 && !(g.Rotary && Unlimited((IMyMotorStator)m.B)))
                throw new Exception("Joint route exceeds native limits.");
            values[i++] = value;
        }
    }
    int start = 0; foreach (var g in Topology.Groups) foreach (var m in g.M)
    { double e = MemberPosition(m) - path.Joints[0][start++]; if (g.Rotary && Unlimited((IMyMotorStator)m.B)) e = Wrap(e); if (Math.Abs(e) > .0005) throw new Exception("Joint route start moved; plan again."); }
    for (int k = 1; k < rows.Length; k++) { int column = 0; foreach (var g in Topology.Groups) { double first = double.NaN; foreach (var m in g.M) { double d = (path.Joints[k][column] - path.Joints[0][column++]) * m.Sign; if (!Finite(first)) first = d; else if (Math.Abs(d - first) > .000001) throw new Exception("Parallel route targets disagree."); } } }
    path.JointStamp = PlanIdentity();
}
string PlanIdentity()
{ return Topology.Fingerprint + "/" + Topology.End.EntityId + "/" + JointRpm + "/" + PistonMps; }
void WritePlanningModel(MyIni ini)
{
    if (!ConfigValid || !Topology.Ready || !Enabled || Tools.Busy || LocalPath != null || GoingHome || HasPilotInput())
        throw new Exception("Planning needs an idle ON arm and neutral input.");
    ini.Set(LinkSection, "PlanFingerprint", PlanIdentity()); ini.Set(LinkSection, "Groups", Topology.Fingerprint);
    ini.Set(LinkSection, "Hull", Topology.Base.CubeGrid.EntityId); ini.Set(LinkSection, "Base", Topology.Base.EntityId);
    ini.Set(LinkSection, "ToolGrid", Topology.End.CubeGrid.EntityId); ini.Set(LinkSection, "TipGrid", Topology.End.CubeGrid.EntityId);
    ini.Set(LinkSection, "Socket", Tools.Focus != null); var cell = Tools.Focus == null ? Topology.End.Position : Tools.Focus.Position;
    ini.Set(LinkSection, "TipCell", cell.X + " " + cell.Y + " " + cell.Z); WriteVector(ini, "TipUp", Topology.EndPose.Up);
    var positions = new System.Text.StringBuilder(); var caps = new System.Text.StringBuilder();
    int frozen = 0, stage = 0; foreach (var g in Topology.Groups)
    {
        if (g.MoveW + g.TurnW == 0) frozen |= 1 << stage; stage++;
        if (caps.Length > 0) caps.Append(' '); caps.Append(AutoSettings.Number(g.Rotary ? JointRpm * Math.PI / 30 : PistonMps));
        foreach (var m in g.M) { if (positions.Length > 0) positions.Append(' '); positions.Append(AutoSettings.Number(MemberPosition(m))); }
    }
    ini.Set(LinkSection, "Positions", positions.ToString()); ini.Set(LinkSection, "Caps", caps.ToString());
    ini.Set(LinkSection, "Frozen", frozen);
}
bool StepJointGuide(ArmPath path, double dt, out bool settled, out string why)
{
    settled = false; why = "";
    if (path.JointStamp != PlanIdentity()) { why = "Joint route model changed."; return false; }
    var safety = Services == null ? null : Services.Safety;
    if (safety == null) { why = "Joint route lost Safety."; return false; }
    if (!safety.Ready || safety.Blocked) { safety.Hold(); return true; }
    int i = 0, index = 0; bool joints = true;
    var rates = new double[Topology.Groups.Count]; var lower = new double[rates.Length]; var upper = new double[rates.Length];
    foreach (var g in Topology.Groups)
    {
        lower[index] = -1e6; upper[index] = 1e6;
        foreach (var m in g.M) {
        double q = MemberPosition(m), target = path.Joints[path.Completed][i];
        if (g.Rotary && Unlimited((IMyMotorStator)m.B)) q += Math.Round(((path.Completed > 0 ? path.Joints[path.Completed - 1][i] : target) - q) / (2 * Math.PI)) * (2 * Math.PI);
        if (path.Completed > 0 && (q < Math.Min(path.Joints[path.Completed - 1][i], target) - .0005 || q > Math.Max(path.Joints[path.Completed - 1][i], target) + .0005))
        { why = "Joint left checked segment; stop and replan."; return false; }
        double previous = path.Completed > 0 ? path.Joints[path.Completed - 1][i] : target;
        double accel = g.Rotary ? JointAccelerationRpmPerSec * Math.PI / 30 : PistonAccelerationMps2;
        double a = -Math.Sqrt(2 * accel * Math.Max(0, q - Math.Min(previous,target))) * m.Sign;
        double b = Math.Sqrt(2 * accel * Math.Max(0, Math.Max(previous,target) - q)) * m.Sign;
        lower[index] = Math.Max(lower[index], Math.Min(a,b)); upper[index] = Math.Min(upper[index], Math.Max(a,b));
        rates[index] += (target - q) * m.Sign * 2;
        joints &= Math.Abs(q - target) <= .0005; i++;
        }
        rates[index++] /= g.M.Count;
    }
    bool headSettled;
    GuidedRates = rates; GuidedLo = lower; GuidedHi = upper;
    try { if (!ToolPoseStep(path.Current, dt, dt, path.CurrentMove, path.CurrentTurn, path.PositionTolerance * .1, path.AngleTolerance * .1, out headSettled, out why)) return false; }
    finally { GuidedRates = GuidedLo = GuidedHi = null; }
    if (!Enabled) { why = LastFault; return false; }
    var pose = Topology.EndPose; var goal = path.Current;
    LastPositionError = PathRemainingDistance = Vector3D.Distance(pose.Translation, goal.Translation);
    LastOrientationErrorDeg = RotationError(pose.Forward, pose.Up, goal.Forward, goal.Up).Length() * 180 / Math.PI;
    settled = joints && LastPositionError <= path.PositionTolerance && LastOrientationErrorDeg <= path.AngleTolerance;
    return true;
}
