bool ReadConfig(string text, out MyIni ini, out string why)
{
    ini = new MyIni(); why = "";
    if (string.IsNullOrWhiteSpace(text))
    {
        ini.Set(ArmSection, "Arm", ArmName); ini.Set(ArmSection, "Format", 1);
        ini.Set(LinkSection, "ArmPB", ArmName + " - Arm PB"); ini.Set(ToolsSection, EnabledKey, true);
        WriteConfig(ini); return true;
    }
    if (!ini.TryParse(text)) { why = "Invalid Custom Data INI."; return false; }
    if (ini.Get(ArmSection, "Format").ToInt32() != 1 || ini.Get(ArmSection, "Arm").ToString() != ArmName)
    { why = "ToolSwap Custom Data requires Format 1 and this arm name. Delete Custom Data."; return false; }
    return true;
}
void WriteConfig(MyIni ini) { Me.CustomData = ini.ToString(); }
string ConfigText(MyIni ini) { return ini.ToString(); }

public Program()
{
    Runtime.UpdateFrequency = UpdateFrequency.Update1;
    Link = new ModuleLink(this); Tools = new ToolSwapController(this);
    try { ConfigureLink(); } catch (Exception e) { PendingCommand = ""; Say(e.Message); }
    Echo(DisplayText);
}
public void Save() { }
void ConfigureLink()
{
    MyIni ini; string why; if (!AutoConfig.ReadCurrent(this, Me.CustomData, out ini, out why)) throw new Exception(why);
    Link.Resolve(ini.Get(LinkSection, "ArmPB").ToString());
}
public void Main(string argument, UpdateType source)
{
    double elapsed = Runtime.TimeSinceLastRun.TotalSeconds;
    bool timed = (source & (UpdateType.Update1 | UpdateType.Update10 | UpdateType.Update100 | UpdateType.Once)) != 0;
    if (Finite(elapsed) && elapsed > 0) { Link.Age += elapsed; Clock += elapsed; Heartbeat += elapsed; if (ResumeRequested) ResumeSeconds += elapsed; }
    try
    {
        if (Link.Peer == null) ConfigureLink();
        Receive();
        if (!string.IsNullOrWhiteSpace(argument) && argument != LinkTag)
        {
            string command = argument.Trim(); string verb = command.Split(' ')[0].ToLowerInvariant();
            if (verb == "toolinfo" || verb == "info" || verb == "status") Tools.Handle("ToolInfo");
            else if (verb == "version") Say("AutoArm ToolSwap " + ControllerBuildVersion);
            else if (verb == "on" || verb == "reload" || verb == "tool" || verb == "toolswap" || verb == "park" || verb == "toolscan" || verb == "stop" || verb == "off" || verb == "swapcancel") Queue(command);
            else throw new Exception("Unknown ToolSwap command.");
        }
        if (PendingCommand.Length > 0)
        {
            if (!Acquired)
            {
                if (HelloRequest == 0)
                {
                    ConfigureLink(); var hello = Link.Packet("HELLO"); hello.Set(LinkSection, "Acquire", true);
                    string helloVerb = PendingCommand.Split(' ')[0].ToLowerInvariant(); hello.Set(LinkSection, "Changing", helloVerb == "tool" || helloVerb == "toolswap" || helloVerb == "park");
                    HelloRequest = Link.Sequence; Link.Send(hello);
                }
                return;
            }
            string command = PendingCommand, verb = command.Split(' ')[0].ToLowerInvariant();
            if (verb == "on" || verb == "reload")
            {
                string why; if (!Tools.Load(out why)) throw new Exception(why);
                if (!Tools.Configured) throw new Exception("Tools.Enabled must be true in the ToolSwap PB.");
                MyIni ini; if (!AutoConfig.ReadCurrent(this, Me.CustomData, out ini, out why)) throw new Exception(why);
                Tools.WriteConfig(ini); WriteConfig(ini); Initialised = true;
                if (verb == "on") Tools.StartOn();
            }
            else if (verb == "stop" || verb == "off" || verb == "swapcancel") { Tools.Cancel("Stopped; inspect, then On."); }
            else
            {
                if (!Initialised) throw new Exception("Run On to configure ToolSwap first.");
                Tools.Handle(command);
            }
            PendingCommand = ""; Released = false; Clock = 0;
        }
        if (Link.RemoteEpoch != 0 && Link.Age > .5)
        {
            Tools.Cancel("Arm PB communication timeout."); Acquired = false; PendingCommand = ""; return;
        }
        if (ResumeRequested && Acquired && ResumeSeconds >= .25)
        {
            if (++ResumeRetries > 3 || !Permit()) Tools.Cancel("Arm did not confirm tool handoff; On.");
            else EnableControl(true);
        }
        if (Tools.Busy && Acquired)
        {
            if (RemoteError.Length > 0) throw new Exception(RemoteError);
            if (Tools.Automatic && (RemotePilot || HasPilotInput())) { Tools.Cancel("Pilot canceled swap."); }
            else if (timed && Clock > .000001)
            { double dt = Clock; Clock = 0; if (dt > .5) throw new Exception("Tool update gap."); Tools.Tick(Math.Min(dt, .1), dt); }
        }
        if (!Tools.Busy && Acquired && !Released)
        {
            if (!Initialised || Tools.Recovery || Tools.Endpoint == null)
            { var abort = Link.Packet("ABORT"); abort.Set(LinkSection, "Reason", CancelReason); abort.Set(LinkSection, "ExplicitStop", ExplicitStop); Link.Send(abort); }
            else { var release = EquipmentPacket("RELEASE"); release.Set(LinkSection, "Permit", Permit()); Link.Send(release); }
            Released = true;
        }
        if (Link.RemoteEpoch != 0 && Heartbeat >= .1)
        { Heartbeat = 0; var ping = Link.Packet("PING"); ping.Set(LinkSection, "Permit", Permit()); Link.Send(ping); }
    }
    catch (RemoteWait) { }
    catch (Exception e)
    {
        PendingCommand = ""; Say("STOPPED: " + e.Message);
        try { Tools.Cancel(e.Message); } catch (RemoteWait) { PendingCommand = "Stop"; }
    }
    finally { Echo(DisplayText); }
}
void Queue(string command)
{
    string verb = command.Split(' ')[0].ToLowerInvariant();
    if (Tools.Busy && Tools.Automatic && verb != "stop" && verb != "off" && verb != "swapcancel") { Say("ToolSwap busy; Stop first."); return; }
    if (Released) { Acquired = false; HelloRequest = 0; }
    ExplicitStop = verb == "stop" || verb == "off" || verb == "swapcancel";
    PendingCommand = command; Released = false;
    if (!Acquired) HelloRequest = 0;
}
void Receive()
{
    int count = 0;
    while (IGC.UnicastListener.HasPendingMessage && count++ < 16)
    {
        if (Runtime.MaxInstructionCount > 0 && Runtime.CurrentInstructionCount >= Runtime.MaxInstructionCount * .5) break;
        var message = IGC.UnicastListener.AcceptMessage(); MyIni ini;
        if (!Link.Read(message, out ini)) continue;
        long epoch = ini.Get(LinkSection, "Epoch").ToInt64(), sequence = ini.Get(LinkSection, "Sequence").ToInt64();
        string op = ini.Get(LinkSection, "Operation").ToString();
        if (epoch < Link.RemoteEpoch || epoch == Link.RemoteEpoch && sequence <= Link.Received) continue;
        if (op == "COMMAND")
        {
            string incoming = ini.Get(LinkSection, "Command").ToString();
            if (incoming.Equals("ToolInfo", StringComparison.OrdinalIgnoreCase) && epoch == Link.RemoteEpoch && ini.Get(LinkSection, "Generation").ToInt64() == Link.Generation)
            { Link.Received = sequence; Link.Age = 0; Tools.Handle("ToolInfo"); var info = Link.Packet("INFO"); info.Set(LinkSection, "Text", DisplayText); Link.Send(info); continue; }
            if (epoch != Link.RemoteEpoch) { Link.Received = 0; Acquired = false; }
            Link.RemoteEpoch = epoch; Link.Generation = ini.Get(LinkSection, "Generation").ToInt64(); Link.Received = sequence; Link.Age = 0;
            RemoteStopped = ini.Get(LinkSection, "Stopped").ToBoolean(); MoveRequest = StopRequest = BuildRequest = HelloRequest = 0; RemoteError = "";
            TransitionRequest = 0; TransitionKind = ""; ResumeRequested = false;
            string command = ini.Get(LinkSection, "Command").ToString();
            if (command.Equals("ToolInfo", StringComparison.OrdinalIgnoreCase)) { Tools.Handle("ToolInfo"); var info = Link.Packet("INFO"); info.Set(LinkSection, "Text", DisplayText); Link.Send(info); continue; }
            if (command.Equals("Stop", StringComparison.OrdinalIgnoreCase))
            { ExplicitStop = !ini.Get(LinkSection, "RecoverBare").ToBoolean(); PendingCommand = ""; Acquired = false; Tools.Cancel("Arm stopped; inspect, then On."); continue; }
            if (Tools.Busy) Tools.Cancel("Arm requested new setup; inspect equipment.");
            Queue(command); Acquired = false; continue;
        }
        if (op != "STATE" || ini.Get(LinkSection, "Remote").ToInt64() != Link.Epoch) continue;
        if (Link.RemoteEpoch != 0 && epoch == Link.RemoteEpoch && ini.Get(LinkSection, "Generation").ToInt64() != Link.Generation) continue;
        Link.RemoteEpoch = epoch; Link.Generation = ini.Get(LinkSection, "Generation").ToInt64(); Link.Received = sequence; Link.Age = 0;
        Ack = ini.Get(LinkSection, "Ack").ToInt64(); RemoteReady = ini.Get(LinkSection, "Ready").ToBoolean(); RemoteStopped = ini.Get(LinkSection, "Stopped").ToBoolean();
        RemoteSettled = ini.Get(LinkSection, "Settled").ToBoolean(); RemotePilot = ini.Get(LinkSection, "Pilot").ToBoolean(); RemoteError = ini.Get(LinkSection, "Error").ToString();
        PilotSelected = ini.Get(LinkSection, "PilotSelected").ToBoolean(true);
        ReportedPath = ini.Get(LinkSection, "PathId").ToInt64(); CompletedWaypoints = ini.Get(LinkSection, "Completed").ToInt32();
        ReadKeyboard = ini.Get(LinkSection, "Keyboard").ToBoolean(true); ReadMouse = ini.Get(LinkSection, "Mouse").ToBoolean(); ReadRoll = ini.Get(LinkSection, "Roll").ToBoolean(true);
        DefaultMove = ini.Get(LinkSection, "DefaultMove").ToDouble(.2); DefaultTurn = ini.Get(LinkSection, "DefaultTurn").ToDouble(5);
        Acquired = ini.Get(LinkSection, "Lease").ToBoolean();
        if (ResumeRequested && Ack >= ResumeRequest && !Acquired && (ini.Get(LinkSection, "Enabled").ToBoolean() || RemoteError.Length > 0))
        { ResumeRequested = false; Say(RemoteError.Length > 0 ? "Arm resume failed: " + RemoteError : "Tool ready; arm ON."); }
        if (HelloRequest != 0 && Ack >= HelloRequest && !Acquired) HelloRequest = 0;
        if (MoveRequest != 0 && ini.Get(LinkSection, "MoveAck").ToInt64() != MoveRequest) RemoteSettled = false;
        Topology.Head = GridTerminalSystem.GetBlockWithId(ini.Get(LinkSection, "Head").ToInt64()); Topology.Groups.Clear();
        string rows = ini.Get(LinkSection, "Groups").ToString();
        if (rows.Length > 4096) throw new Exception("Bad remote group list.");
        if (rows.Length > 0) foreach (var row in rows.Split('\n'))
        {
            if (Topology.Groups.Count >= 24) throw new Exception("Remote group limit."); var group = new AutoGroup();
            foreach (var cell in row.Split(','))
            { long id; if (!long.TryParse(cell, out id) || group.M.Count >= 48) throw new Exception("Bad remote actuator."); var b = GridTerminalSystem.GetBlockWithId(id) as IMyMechanicalConnectionBlock; if (b == null || !b.IsSameConstructAs(Me)) throw new Exception("Remote actuator lost."); group.M.Add(new AutoMember { B = b }); }
            Topology.Groups.Add(group);
        }
    }
}
bool Permit() { string why; return Initialised && Tools.AllowManual(out why); }
bool CaptureReady => MoveRequest != 0 && ReportedPath == MoveRequest && Ack >= MoveRequest;
void Say(string text) { DisplayText = text; }
void StopMotion(string reason)
{
    if (RemoteStopped && (StopRequest == 0 || Ack >= StopRequest)) { StopRequest = 0; MoveRequest = 0; return; }
    if (StopRequest == 0) { var stop = Link.Packet("STOP"); stop.Set(LinkSection, "ExplicitStop", ExplicitStop); StopRequest = Link.Sequence; Link.Send(stop); }
    throw new RemoteWait();
}
void CancelMotion(string reason)
{ CancelReason = reason; ResumeRequested = false; try { StopMotion(reason); } catch (RemoteWait) { } catch { } }
void ExpectTransition(string kind)
{
    StopMotion("");
    if (TransitionKind != kind || TransitionRequest == 0)
    { var packet = Link.Packet("TRANSITION"); packet.Set(LinkSection, "Kind", kind); TransitionKind = kind; TransitionRequest = Link.Sequence; Link.Send(packet); throw new RemoteWait(); }
    if (Ack < TransitionRequest || !RemoteStopped) throw new RemoteWait();
}
bool RebuildToolTopology(out string why)
{
    StopMotion(""); var build = EquipmentPacket("BUILD"); BuildRequest = Link.Sequence;
    TransitionRequest = 0; TransitionKind = "";
    RemoteReady = false; RemoteSettled = false; Link.Send(build); why = ""; return true;
}
bool SettleToolTopology(out string why)
{
    why = "Waiting for arm topology.";
    if (RemoteError.Length > 0) throw new Exception(RemoteError);
    return BuildRequest != 0 && Ack >= BuildRequest && RemoteReady;
}
bool ToolPoseStep(MatrixD world, double dt, double observationDt, double moveCap, double turnCap, double positionTolerance, double angleTolerance, out bool settled, out string why)
{
    return SubmitPath(PathRow(world), 0, 0, moveCap, turnCap, positionTolerance, angleTolerance, false, out settled, out why);
}
bool ToolPathStep(MatrixD first, MatrixD second, IMyCubeBlock anchor, int waypoint, double moveCap, double turnCap, double positionTolerance, double angleTolerance, out bool settled, out string why)
{ return SubmitPath(PathRow(first) + "|" + PathRow(second), anchor.EntityId, waypoint, moveCap, turnCap, positionTolerance, angleTolerance, true, out settled, out why); }
bool SubmitPath(string rows, long anchor, int waypoint, double moveCap, double turnCap, double positionTolerance, double angleTolerance, bool travel, out bool settled, out string why)
{
    settled = false; why = ""; if (RemoteError.Length > 0) { why = RemoteError; return false; }
    if (MoveRequest != 0 && rows == SentPath && anchor == SentAnchor)
    { settled = ReportedPath == MoveRequest && CompletedWaypoints > waypoint && Link.Age < .2 && (waypoint == 0 && rows.Contains("|") || RemoteSettled && RemoteStopped); return true; }
    var packet = Link.Packet("PATH"); packet.Set(LinkSection, "Waypoints", rows); packet.Set(LinkSection, "Anchor", anchor); packet.Set(LinkSection, "Move", moveCap); packet.Set(LinkSection, "Turn", turnCap);
    packet.Set(LinkSection, "PosTol", positionTolerance); packet.Set(LinkSection, "AngleTol", angleTolerance);
    packet.Set(LinkSection, "Docking", true);
    packet.Set(LinkSection, "Capture", Tools.Parking);
    packet.Set(LinkSection, "Moves", travel ? "0|" + AutoSettings.Number(moveCap) : AutoSettings.Number(moveCap));
    packet.Set(LinkSection, "Turns", travel ? "0|" + AutoSettings.Number(turnCap) : AutoSettings.Number(turnCap));
    packet.Set(LinkSection, "Lines", travel ? "false|true" : "true");
    MoveRequest = Link.Sequence; SentPath = rows; SentAnchor = anchor; RemoteStopped = RemoteSettled = false; Link.Send(packet); return true;
}
void EnableControl(bool retry = false)
{
    if (!retry) ResumeRetries = 0;
    ResumeSeconds = 0;
    var resume = EquipmentPacket("RESUME"); resume.Set(LinkSection, "Permit", Permit()); ResumeRequest = Link.Sequence; Link.Send(resume); Released = ResumeRequested = true;
    Say("Tool ready; waiting for arm ON confirmation.");
}
MyIni EquipmentPacket(string operation)
{
    var ini = Link.Packet(operation); var endpoint = Tools.Endpoint; var frame = Tools.EndFrame;
    if (endpoint == null || frame == null) throw new Exception("Tool endpoint unavailable.");
    var inventory = new List<IMyTerminalBlock>(); GridTerminalSystem.GetBlocks(inventory); IMyTerminalBlock owner = null;
    foreach (var b in inventory)
    { var joint = b as IMyMechanicalConnectionBlock; if (b.EntityId == frame.EntityId || joint != null && joint.Top != null && joint.Top.EntityId == frame.EntityId) { owner = b; break; } }
    if (owner == null) throw new Exception("Tool endpoint anchor unavailable.");
    ini.Set(LinkSection, "Endpoint", endpoint.EntityId); ini.Set(LinkSection, "AnchorOwner", owner.EntityId); ini.Set(LinkSection, "Anchor", frame.EntityId); ini.Set(LinkSection, "Ignore", Tools.IgnoredMount);
    var focus = Tools.Focus; ini.Set(LinkSection, "Focus", focus != null);
    if (focus != null) { ini.Set(LinkSection, "Part", focus.EntityId); WriteFrame(ini, "Local", focus.Local); }
    ini.Set(LinkSection, "ExplicitStop", ExplicitStop); Tools.WriteEquipment(ini); return ini;
}
bool HasPilotInput()
{
    if (!PilotSelected) return false;
    if (ControllerPoll-- <= 0) { GridTerminalSystem.GetBlocksOfType(Controllers); ControllerPoll = 9; }
    foreach (var c in Controllers) if (!c.Closed && c.IsUnderControl && c.IsSameConstructAs(Me) && (ReadKeyboard && c.MoveIndicator.LengthSquared() > 1e-8 || ReadMouse && c.RotationIndicator.LengthSquared() > 1e-8 || ReadRoll && Math.Abs(c.RollIndicator) > 1e-4)) return true;
    return false;
}
