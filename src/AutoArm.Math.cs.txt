static Vector3D Rotate(Vector3D value, Vector3D rotation)
{
    double angle = rotation.Length();
    if (angle < 1e-12) return value;
    Vector3D axis = rotation / angle;
    return value * Math.Cos(angle) + Vector3D.Cross(axis, value) * Math.Sin(angle) +
        axis * Vector3D.Dot(axis, value) * (1 - Math.Cos(angle));
}

static bool Orthonormalise(ref Vector3D f, ref Vector3D u)
{
    double fn = f.Length();
    if (!Finite(fn) || fn < 1e-10) return false;
    f /= fn;
    u -= f * Vector3D.Dot(u, f);
    double un = u.Length();
    if (!Finite(un) || un < 1e-10) return false;
    u /= un;
    return true;
}

static Vector3D Unit(Vector3D value)
{
    double n = value.Length();
    if (!Finite(n) || n < 1e-12) throw new Exception("Invalid zero/non-finite vector.");
    return value / n;
}

static Vector3D ClampLength(Vector3D value, double cap)
{
    double n = value.Length();
    if (!Finite(n)) return new Vector3D(double.NaN);
    return n > cap && n > 1e-12 ? value * (cap / n) : value;
}

static Vector3D InFrame(Vector3D value, MatrixD frame) => new Vector3D(Vector3D.Dot(value, frame.Forward), Vector3D.Dot(value, frame.Left), Vector3D.Dot(value, frame.Up));

static double Clamp(double value, double lo, double hi) => Math.Max(lo, Math.Min(hi, value));

static double Wrap(double angle) => angle - Math.Floor((angle + Math.PI) / (2 * Math.PI)) * (2 * Math.PI);

static bool Finite(double value) => !double.IsNaN(value) && !double.IsInfinity(value);

static bool Finite(Vector3D value) => Finite(value.X) && Finite(value.Y) && Finite(value.Z);

static bool Unlimited(IMyMotorStator r)
{
    return r.LowerLimitRad < -1000000 && r.UpperLimitRad > 1000000;
}


static bool ReadDouble(string text, out double value) => double.TryParse(text, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out value) && Finite(value);
