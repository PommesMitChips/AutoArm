// Optional direct peers. The arm remains the only joint writer; planners submit
// complete paths, observers read state, safety peers may stop every controller.
sealed class ArmServices
{
    sealed class Client
    {
        public ModuleLink Link;
        public string Role, State = "Idle", Error = "";
        public ArmPath Path;
        public long Seen;
        public double ReplyAge;
    }
    readonly Program P;
    readonly Dictionary<long, Client> Peers = new Dictionary<long, Client>();
    Client Owner;
    string Config = "";
    long Revision = 1;
    public ArmServices(Program p) { P = p; }
    public void Load(MyIni ini)
    {
        var peers = new Dictionary<long, Client>();
        string text = ini.Get("Modules", "Peers").ToString();
        foreach (string row in text.Split('\n'))
        {
            if (row.Trim().Length == 0) continue;
            string[] cells = row.Split('|');
            if (cells.Length != 2 || peers.Count >= 16) throw new Exception("Modules.Peers needs PB name | Observe, Plan or Stop (max 16).");
            string name = cells[0].Trim(), role = cells[1].Trim();
            if (role != "Observe" && role != "Plan" && role != "Stop" || name == ini.Get("Modules", "ToolSwapPB").ToString().Trim()) throw new Exception("Invalid/duplicate module role.");
            var link = new ModuleLink(P); link.Resolve(name);
            if (peers.ContainsKey(link.Peer.EntityId)) throw new Exception("Duplicate module PB.");
            peers.Add(link.Peer.EntityId, new Client { Link = link, Role = role });
        }
        Revoke("Configuration reloaded."); Peers.Clear();
        foreach (var pair in peers) Peers.Add(pair.Key, pair.Value);
        Config = text;
    }
    public void WriteConfig(MyIni ini) { if (Config.Length > 0) ini.Set("Modules", "Peers", Config); }
    public void Revoke(string reason)
    {
        Revision++;
        if (Owner != null) { Owner.State = "Canceled"; Owner.Error = reason; Owner = null; }
    }
    public void Tick(double elapsed)
    {
        if (Owner == null) return;
        var client = Owner;
        if (Finite(elapsed) && elapsed > 0) client.Link.Age += elapsed;
        if (Finite(elapsed) && elapsed > 0) client.ReplyAge += elapsed;
        var peer = client.Link.Peer;
        if (peer.Closed || !peer.IsFunctional || !peer.IsSameConstructAs(P.Me) || client.Link.Age > .5 || client.ReplyAge > .5)
        { P.Fault("Planner communication timeout/unavailable."); return; }
        if (P.LocalPath == null)
        {
            client.State = client.Path.Done && P.Enabled ? "Complete" : "Canceled";
            Owner = null; Revision++;
        }
    }
    public bool Receive(MyIGCMessage message)
    {
        Client client;
        if (!Peers.TryGetValue(message.Source, out client)) return false;
        MyIni ini; var link = client.Link;
        if (!link.Read(message, out ini) || ini.Get(LinkSection, "Service").ToInt32() != 1) return true;
        long epoch = ini.Get(LinkSection, "Epoch").ToInt64(), seq = ini.Get(LinkSection, "Sequence").ToInt64();
        string op = ini.Get(LinkSection, "Operation").ToString();
        long remote = ini.Get(LinkSection, "Remote").ToInt64();
        if (op == "HELLO")
        {
            if (epoch < link.RemoteEpoch || remote != 0 && remote != link.Epoch) return true;
            if (epoch != link.RemoteEpoch)
            {
                if (Owner == client) P.StopMotion("Planner restarted.");
                link.RemoteEpoch = epoch; link.Received = 0;
                client.Seen = 0;
            }
        }
        else if (epoch != link.RemoteEpoch || remote != link.Epoch ||
            op != "STATUS" && op != "PING" && op != "STOP" && ini.Get(LinkSection, "Generation").ToInt64() != Revision) return true;
        if (seq <= link.Received) return true;
        long seen = ini.Get(LinkSection, "Seen").ToInt64();
        if (op != "HELLO" && (seen < 0 || seen > link.Sequence)) return true;
        if (op != "HELLO" && seen > client.Seen) { client.Seen = seen; client.ReplyAge = 0; }
        link.Received = seq; link.Age = 0; client.Error = "";
        try
        {
            if (op == "HELLO" || op == "STATUS" || op == "PING") { }
            else if (op == "STOP")
            {
                if (client.Role != "Stop" && Owner != client) throw new Exception("Module has no stop authority.");
                P.Tools.InhibitRecovery();
                if (P.Tools.Busy) P.Tools.Cancel("Module emergency stop.");
                P.StopMotion("Module stop: " + link.Peer.CustomName); P.Say(P.Status);
            }
            else if (op == "PATH")
            {
                string why;
                if (client.Role != "Plan") throw new Exception("Module is read only.");
                if (seen == 0) throw new Exception("Acknowledge the arm STATE before PATH.");
                if (Owner != null || P.Tools.Busy || P.LocalPath != null || P.GoingHome) throw new Exception("Arm motion already owned.");
                if (!P.Enabled || !P.ConfigValid || !P.Topology.Ready || !P.Tools.AllowManual(out why)) throw new Exception("Arm not ready; run On.");
                var path = P.ReadPath(ini.Get(LinkSection, "Waypoints").ToString(), ini.Get(LinkSection, "Anchor").ToInt64(), seq, P.MoveMps, P.TurnDeg, .005, .2);
                P.ConfigurePathStages(path, ini);
                Vector3D p, f, u;
                if (!P.ReadPose(out p, out f, out u)) throw new Exception("Arm focus unavailable.");
                // No state changes until every waypoint and policy has passed.
                P.StopMotion(""); P.LocalPath = path; P.CapturePose(p, f, u); P.Enabled = true;
                P.Runtime.UpdateFrequency = UpdateFrequency.Update1;
                client.Path = path; client.State = "Moving"; Owner = client;
                client.ReplyAge = 0;
                P.Say("Module path: " + link.Peer.CustomName);
            }
            else throw new Exception("Unsupported service operation.");
        }
        catch (Exception e) { client.Error = e.Message; }
        Reply(client, seq); return true;
    }
    void Reply(Client client, long ack)
    {
        var link = client.Link; link.Generation = Revision;
        var ini = link.Packet("STATE"); ini.Set(LinkSection, "Service", 1); ini.Set(LinkSection, "Ack", ack);
        ini.Set(LinkSection, "Role", client.Role); ini.Set(LinkSection, "Owner", Owner == null ? 0 : Owner.Link.Peer.EntityId);
        ini.Set(LinkSection, "Ready", P.ConfigValid && P.Topology.Ready); ini.Set(LinkSection, "Enabled", P.Enabled);
        ini.Set(LinkSection, "Busy", P.Tools.Busy || P.LocalPath != null || P.GoingHome);
        ini.Set(LinkSection, "State", client.State); ini.Set(LinkSection, "Error", client.Error);
        ini.Set(LinkSection, "PathId", client.Path == null ? 0 : client.Path.Id);
        ini.Set(LinkSection, "Completed", client.Path == null ? 0 : client.Path.Completed);
        ini.Set(LinkSection, "Count", client.Path == null ? 0 : client.Path.Waypoints.Count);
        ini.Set(LinkSection, "DefaultMove", P.MoveMps); ini.Set(LinkSection, "DefaultTurn", P.TurnDeg);
        if (P.Topology.Ready) WriteFrame(ini, "Pose", P.Topology.EndPose);
        // Failure to reply to an idle observer must not fault a healthy arm.
        try { link.Send(ini); } catch { if (Owner == client) P.Fault("Planner reply failed."); }
    }
}
